### app.js

```js
// Sign up — the page script.

var FIELDS = ['email', 'password', 'confirm', 'terms'];

var RULES = {
  email: {
    check: function (form) {
      var v = form.elements.email.value.trim();
      return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v);
    },
    message: 'Enter a valid email address.'
  },
  password: {
    check: function (form) {
      var v = form.elements.password.value;
      return v.length >= 8 && /\d/.test(v);
    },
    message: 'Use at least 8 characters, including a number.'
  },
  confirm: {
    check: function (form) {
      var v = form.elements.confirm.value;
      return v.length > 0 && v === form.elements.password.value;
    },
    message: 'Passwords do not match.'
  },
  terms: {
    check: function (form) {
      return form.elements.terms.checked;
    },
    message: 'Accept the terms to continue.'
  }
};

// Fields whose error has been shown at least once. Once true, the field
// re-validates on every input so the message clears as soon as it is fixed.
var shown = {};

function validateField(form, name) {
  var field = form.elements[name];
  var errorEl = form.querySelector('#' + name + '-error');
  var ok = RULES[name].check(form);
  field.setAttribute('aria-invalid', ok ? 'false' : 'true');
  if (errorEl) errorEl.textContent = ok ? '' : RULES[name].message;
  return ok;
}

function touch(form, name) {
  var ok = validateField(form, name);
  if (!ok) shown[name] = true;
  return ok;
}

vf.attach('#signup', {
  delegates: [
    {
      selector: '#email, #password, #confirm',
      eventType: 'focusout',
      onEvent: function (e) {
        touch(e.sender.$node, e.target.name);
      }
    },
    {
      selector: '#email, #password, #confirm',
      eventType: 'input',
      onEvent: function (e) {
        var name = e.target.name;
        if (shown[name]) validateField(e.sender.$node, name);
      }
    },
    {
      selector: '#terms',
      eventType: 'change',
      onEvent: function (e) {
        touch(e.sender.$node, 'terms');
      }
    }
  ],
  events: [
    {
      eventType: 'submit',
      onEvent: function (e) {
        e.event.preventDefault();
        var form = e.sender.$node;
        var firstInvalid = null;

        for (var i = 0; i < FIELDS.length; i++) {
          var name = FIELDS[i];
          var ok = touch(form, name);
          if (!ok && !firstInvalid) firstInvalid = form.elements[name];
        }

        if (firstInvalid) {
          firstInvalid.focus();
          return;
        }

        var email = form.elements.email.value.trim();
        form.setAttribute('hidden', '');

        var done = document.querySelector('[data-ref="done"]');
        if (done) {
          done.hidden = false;
          done.textContent = 'Welcome, ' + email + '!';
        }
      }
    }
  ]
});
```

### REPORT.md

```markdown
# 가입 폼 검증 — 보고

## 바꾼 파일
- JS: `app.js` — 검증 로직 전체 추가
- HTML: `index.html` — 에러 메시지 요소(`email-error`/`password-error`/`confirm-error`/`terms-error`), `aria-describedby`, `data-ref="done"` 요소 추가. 기존 폼 필드·구조·문구는 그대로 유지
- CSS: `style.css` — `.field__error`, `.signup__done` 규칙 추가(전부 `var(--vf-*)` 토큰만 사용)

## 가정
1. 에러 메시지 요소와 `data-ref="done"` 요소는 원래 마크업에 없었으므로 과제 요구("필드마다 `id="<필드 id>-error"`인 메시지 요소", "`data-ref="done"`인 요소")를 만족시키기 위해 새로 추가했습니다. "폼 마크업은 유지"는 기존 필드 구조·속성·문구를 보존하라는 뜻으로 해석했고, 규칙이 요구하는 요소 추가는 이 범위 밖이라고 보았습니다.
2. `blur`는 버블링하지 않아 위임 셀렉터로 잡을 수 없으므로 텍스트 필드 포커스 아웃 검증에는 `focusout`(버블링됨)을 썼습니다. 동작은 blur와 동일합니다.
3. "한 번 오류를 보인 필드는 입력할 때마다 다시 검증"을 "검증 결과가 틀렸던 적이 있는 필드"로 해석해, `shown[name]`을 검증이 **실패했을 때만** true로 표시하고, 이후 input마다 재검증하도록 했습니다.
4. `.field__error` 색상에 `--vf-color-danger` 토큰을 썼습니다. 킷의 llms.ko.txt는 토큰 예시를 `--vf-color-primary`, `--vf-color-text-muted`, `--vf-space-3`, `--vf-chart-1..8` **등**으로만 들고 전체 목록을 주지 않아 `--vf-color-danger`의 존재를 킷 파일만으로 확인하지 못했습니다. `VERIFY: --vf-color-danger` — 이 토큰이 없으면 브라우저가 정의되지 않은 커스텀 속성으로 처리해 `color` 선언이 무시되고 상속색이 쓰이므로 콘솔 오류는 나지 않지만 에러 텍스트가 강조색으로 보이지 않을 수 있습니다.
5. `confirm` 규칙은 비밀번호와 "공백 트림 없이" 원문 그대로 비교했습니다. 표에는 이메일만 "앞뒤 공백을 뺀 값" 조건이 명시되어 있어 비밀번호류는 트림하지 않는 것이 맞다고 판단했습니다.
6. 제출 시 유효한 필드도 `touch()`로 다시 검증해 `aria-invalid="false"`를 명시적으로 채웁니다(요구사항에 어긋나지 않음).

## 확인 목록 (프롬프트 킷 기준)
- [x] 새 마크업은 `vf.html`을 쓰지 않았습니다 — 정적 마크업만 추가했고 동적 삽입은 `textContent`(자동 이스케이프, XSS 위험 없음)만 사용해 `vf.html`이 필요 없었습니다. 새 이벤트는 모두 `data-*`/`id` 셀렉터의 `delegates`/`events`에 걸었습니다(CSS 클래스로 고르지 않음)
- [x] 새 텍스트("Enter a valid email address." 등)는 과제가 지정한 고정 영문 문구를 그대로 썼습니다(로케일 파일이 없는 정적 퍼블리싱 프로젝트라 메시지 키 구조가 없음 — 과제 규칙 6 "화면 문구는 과제에 적힌 영어 문구를 그대로 씁니다"를 우선했습니다)
- [x] 새 타이머·리스너·구독 없음(단발성 `vf.attach` delegates만 사용, 페이지 이동/destroy 로직이 없는 단일 정적 페이지라 `onDestroy` 해제 대상 없음)
- [x] JS에 디자인 값(색·글꼴·간격) 없음. CSS는 토큰만 사용(가정 4의 `--vf-color-danger` 존재 여부만 미확인)
- [ ] 콘솔 에러·경고 0개 — **확인 못 함**(브라우저에서 직접 실행해 보지 못했습니다). 코드 검토 기준: `vf.attach`에 전달한 옵션은 킷 문서의 `delegates`/`events`/셀렉터 형식을 따랐고, `on*` 인라인 속성이나 `innerHTML` 문자열 삽입을 쓰지 않았으며, CSP에 걸리는 인라인 스크립트/스타일도 없습니다.

## 동작 요약
- 텍스트 필드: `focusout`에서 최초 검증, 실패하면 이후 `input`마다 재검증
- 체크박스: `change`에서 검증
- 제출: 4개 필드를 표 순서(email → password → confirm → terms)로 모두 검증하고, 실패한 첫 필드로 포커스 이동 후 종료. 전부 통과하면 폼에 `hidden` 속성을 달고 `[data-ref="done"]` 요소에 `Welcome, <trim된 이메일>!`을 텍스트로 표시
- 비밀번호 값은 `form.elements.password.value`/`.confirm.value`로만 읽고 어디에도 로그·표시하지 않음(제출 성공 메시지는 이메일만 사용)
```
