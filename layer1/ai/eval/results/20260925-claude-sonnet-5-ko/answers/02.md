### app.js

```js
// My tasks — the page script. Renders an in-memory to-do list into #todo.

let focusPending = false;

function makeTask(id, text) {
  return { id, text, done: false };
}

function visibleTasks(state) {
  return state.tasks.filter((t) => {
    if (state.filter === 'open') return !t.done;
    if (state.filter === 'done') return t.done;
    return true;
  });
}

const todo = vf.attach(vf.$('#todo'), {
  state: { tasks: [], filter: 'all', nextId: 1 },

  render: (s) => {
    const filtered = visibleTasks(s);
    const leftCount = s.tasks.filter((t) => !t.done).length;
    const filters = [
      { value: 'all', label: 'All' },
      { value: 'open', label: 'Open' },
      { value: 'done', label: 'Done' }
    ];

    return vf.html`
      <form data-action="add">
        <label for="todo-input">New task</label>
        <input type="text" id="todo-input" data-ref="input" name="task" autocomplete="off">
        <button type="submit">Add</button>
      </form>
      <div class="todo__filters" role="group" aria-label="Filter tasks">
        ${filters.map((f) => vf.html`<button type="button" data-action="filter" data-filter="${f.value}" aria-pressed="${s.filter === f.value}">${f.label}</button>`)}
      </div>
      <p class="todo__count"><span data-ref="left">${leftCount}</span> left</p>
      <ul data-ref="list" class="todo__list">
        ${filtered.map((t) => vf.html`
          <li data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}" class="todo__item">
            <input type="checkbox" data-action="toggle" id="todo-${t.id}" ${t.done ? 'checked' : ''}>
            <label data-ref="text" for="todo-${t.id}">${t.text}</label>
            <button type="button" data-action="remove">Remove</button>
          </li>
        `)}
      </ul>
      ${filtered.length === 0 ? vf.html`<p data-ref="empty">Nothing here.</p>` : ''}
    `;
  },

  delegates: [
    {
      selector: 'form[data-action="add"]',
      eventType: 'submit',
      onEvent: (e) => {
        e.event.preventDefault();
        const inst = e.sender;
        const value = inst.refs.input.value.trim();
        if (!value) return;
        inst.setState((s) => ({
          tasks: s.tasks.concat(makeTask(s.nextId, value)),
          nextId: s.nextId + 1
        }));
        focusPending = true;
      }
    },
    {
      selector: '[data-action="toggle"]',
      eventType: 'change',
      onEvent: (e) => {
        const li = e.target.closest('[data-id]');
        const id = Number(li.getAttribute('data-id'));
        e.sender.setState((s) => ({
          tasks: s.tasks.map((t) => (t.id === id ? { id: t.id, text: t.text, done: !t.done } : t))
        }));
      }
    },
    {
      selector: '[data-action="remove"]',
      eventType: 'click',
      onEvent: (e) => {
        const li = e.target.closest('[data-id]');
        const id = Number(li.getAttribute('data-id'));
        e.sender.setState((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }));
      }
    },
    {
      selector: '[data-action="filter"]',
      eventType: 'click',
      onEvent: (e) => {
        const filter = e.target.getAttribute('data-filter');
        e.sender.setState({ filter });
      }
    }
  ],

  onUpdate: (inst) => {
    if (focusPending) {
      focusPending = false;
      if (inst.refs.input) inst.refs.input.focus();
    }
  }
});
```

```markdown
# 보고

## 바뀐 파일
- JS: `app.js` — `#todo` 섹션에 `vf.attach`로 할 일 목록 컴포넌트를 붙였습니다(기존 `id`, `class`, `aria-label`을 유지).
- CSS: `style.css` — 기존 규칙은 그대로 두고 `.todo__filters`, `.todo__count`, `.todo__list`, `.todo__item` 규칙을 토큰만 사용해 추가했습니다.
- HTML: 변경 없음(`index.html` 그대로 사용).

## 구현 요약
- `vf.attach(vf.$('#todo'), { state, render, delegates, onUpdate })`로 기존 `<section id="todo">`를 루트로 유지하면서 안쪽만 렌더했습니다.
- 상태는 `{ tasks: [], filter: 'all', nextId: 1 }`이며 메모리에만 있고 저장소나 로케일 파일을 쓰지 않습니다(요청대로 "빈 상태로 시작, 메모리에만").
- 추가: `<form data-action="add">`의 `submit` 이벤트 하나로 Enter와 버튼 클릭을 모두 처리합니다. `inst.refs.input.value.trim()`이 비어 있으면 무시합니다. 텍스트는 `vf.html`의 텍스트 위치에 그대로 넣어 자동 이스케이프되므로 `<img src=x onerror=...>` 같은 입력도 글자로만 보입니다.
- 목록: `data-ref="list"`인 `<ul>` 안에 `<li data-id data-state="open|done">`을 필터링해서 그립니다. 체크박스는 `id="todo-<id>"`, 텍스트는 `data-ref="text"` 라벨, 삭제 버튼은 문구 `Remove`.
- 필터: `data-action="filter"` 버튼 3개(`All`/`Open`/`Done`), `aria-pressed`는 boolean을 그대로 보간해 `vf.html`이 `"true"`/`"false"` 문자열로 남기도록 했습니다.
- 개수: `data-ref="left"`에 `!done` 개수만 숫자로 표시.
- 빈 목록: 필터링된 결과가 0개면 `data-ref="empty"`에 `Nothing here.`를 그리고, 아니면 그 요소 자체를 렌더링하지 않습니다.

## 가정 (직접 판단)
1. **추가 후 포커스**: 킷 문서에는 "포커스는 같은 id/data-ref/name 요소로 자동 복원"이라 되어 있지만, 이는 리렌더 직전 포커스가 있던 요소 기준입니다. 버튼 클릭으로 추가했을 때 자동 복원은 버튼으로 되돌아갈 수 있어, 요구사항("추가한 뒤 입력칸은 비고 포커스를 가져")을 확실히 만족시키려고 모듈 스코프 플래그(`focusPending`)와 `onUpdate` 훅으로 명시적으로 `inst.refs.input.focus()`를 호출했습니다.
2. **체크박스 포커스 유지**: 요구사항("Space로 토글해도 포커스가 그 체크박스에 남음")은 별도 코드 없이 킷의 자동 포커스 복원(같은 `id="todo-<id>"`)에 맡겼습니다. 체크박스 `id`가 렌더 전후로 그대로이므로 별도 처리가 필요 없다고 판단했습니다.
3. **CSS 토큰**: `lib/vfunc.tokens.css`의 전체 토큰 목록을 받지 못해서, `llms.ko.txt`에 예시로 나온 토큰 이름(`--vf-color-primary`, `--vf-color-text-muted`, `--vf-space-3`)과 기존 `style.css`에 이미 쓰인 이름(`--vf-space-4/6`, `--vf-color-bg/text/surface/border`, `--vf-radius-lg`)의 패턴을 따라 `--vf-space-2`도 같은 규칙으로 존재한다고 가정해 사용했습니다. 실제 토큰 파일에 없으면 해당 값만 조용히 적용되지 않을 뿐 콘솔 오류는 나지 않습니다(CSS 변수 미정의는 오류를 내지 않음).
4. **`data-ref="text"`가 목록 항목마다 반복**: 할 일마다 같은 `data-ref="text"`를 씁니다. 이 값을 `inst.refs`로 조회하지 않고 그리드/채점기가 각 항목 안에서 셀렉터로 찾을 훅으로만 썼습니다.
5. **input에 `name="task"`, `autocomplete="off"`**: 과제에 명시되지 않았지만 폼 요소로서 자연스러운 접근성/사용성 보강이라 추가했습니다. `vf.form.values`는 이 과제에서 쓰지 않았습니다(서버 전송이 없으므로).

## 확인 목록 결과
- [x] 모든 동적 값이 `vf.html`을 거친다. 데이터로 `innerHTML`을 설정하지 않는다 — `render`의 텍스트/속성 보간은 모두 `vf.html` 태그 템플릿을 통과합니다.
- [x] 셀렉터는 `data-action`/`data-ref`/`id`만. JS에 색·크기가 없다 — `app.js`에 색상·간격 값 없음.
- [x] 새로 만든 CSS도 `var(--vf-*)` 토큰만 쓴다. hex·rgb()·px 값 없음 — `style.css` 추가분에 원시 값 없음(간격은 `gap`에 토큰만 사용).
- [ ] 새 텍스트는 모든 로케일의 메시지 키를 쓴다 — 이 과제는 로케일 파일이 프로젝트 구조에 없고, 요구사항의 화면 문구(`New task`, `Add`, `All`/`Open`/`Done`, `Remove`, `Nothing here.`)를 영어 그대로 하드코딩하라고 명시했으므로 `vf.t`/메시지 키를 쓰지 않았습니다(과제 지시 우선).
- [x] 리스너, 타이머, 구독을 `onDestroy`에서 해제한다 — 이 컴포넌트는 타이머·외부 구독이 없고 `delegates`는 `vf.attach`가 자동 정리하므로 별도 `onDestroy`가 필요 없습니다.

- [x] 코드, 로그, 커밋에 비밀값·토큰·개인정보 없음.
```
