### app.js

```js
// Servers — the page script (an ES module).
import vf from './lib/vfunc.esm.js';

var STATUS_LABELS = { up: 'Up', down: 'Down', maintenance: 'Maintenance' };
var VALID_STATES = ['up', 'down', 'maintenance'];
var FILTER_VALUES = ['all', 'up', 'down', 'maintenance'];

function toText(v) {
  if (typeof v === 'string') return v;
  if (v === null || v === undefined) return '';
  return String(v);
}

function badgeState(status) {
  var s = toText(status);
  return VALID_STATES.indexOf(s) !== -1 ? s : 'unknown';
}

function badgeLabel(status) {
  var s = badgeState(status);
  return STATUS_LABELS[s] || 'Unknown';
}

function isHttpUrl(url) {
  return typeof url === 'string' && /^https?:\/\//i.test(url);
}

function filterServers(servers, filter) {
  if (filter === 'all') return servers;
  return servers.filter(function (srv) {
    return badgeState(srv.status) === filter;
  });
}

function renderLink(url) {
  if (isHttpUrl(url)) {
    return vf.html`<a class="servers__link" data-ref="link" href="${url}">Open</a>`;
  }
  return vf.html`<a class="servers__link" data-ref="link">Open</a>`;
}

function renderServer(srv, index) {
  var id = srv && srv.id !== undefined && srv.id !== null ? toText(srv.id) : 'row-' + index;
  var name = srv ? toText(srv.name) : '';
  var region = srv ? toText(srv.region) : '';
  var state = badgeState(srv && srv.status);
  var label = badgeLabel(srv && srv.status);
  return vf.html`
    <li class="servers__item" data-id="${id}">
      <span class="servers__name" data-ref="name">${name}</span>
      <span class="servers__region">${region}</span>
      <span class="servers__badge" data-ref="badge" data-state="${state}">${label}</span>
      ${renderLink(srv && srv.url)}
    </li>
  `;
}

function statusMessage(state, filteredCount) {
  if (state.loading) return 'Loading servers…';
  if (state.error) return 'Could not load servers.';
  if (state.servers.length === 0) return 'No servers yet.';
  return 'Showing ' + filteredCount + ' of ' + state.servers.length;
}

function renderServers(state) {
  var filtered = filterServers(state.servers, state.filter);
  return vf.html`
    <div class="servers__status" data-ref="status" role="status">
      ${statusMessage(state, filtered.length)}
      ${state.error ? vf.html` <button type="button" class="servers__retry" data-action="retry">Retry</button>` : ''}
    </div>
    <div class="servers__filter">
      <label>
        Status
        <select data-action="status-filter">
          <option value="all" ${state.filter === 'all' ? 'selected' : ''}>All</option>
          <option value="up" ${state.filter === 'up' ? 'selected' : ''}>Up</option>
          <option value="down" ${state.filter === 'down' ? 'selected' : ''}>Down</option>
          <option value="maintenance" ${state.filter === 'maintenance' ? 'selected' : ''}>Maintenance</option>
        </select>
      </label>
    </div>
    <ul class="servers__list" data-ref="list">
      ${filtered.map(function (srv, index) { return renderServer(srv, index); })}
    </ul>
  `;
}

function load(inst) {
  inst.setState({ loading: true, error: false });
  fetch('./data/servers.json')
    .then(function (res) {
      if (!res.ok) throw new Error('HTTP ' + res.status);
      return res.json();
    })
    .then(function (data) {
      if (!Array.isArray(data)) throw new Error('Invalid servers payload');
      var servers = data.filter(function (item) {
        return item && typeof item === 'object';
      });
      inst.setState({ loading: false, error: false, servers: servers });
    })
    .catch(function () {
      inst.setState({ loading: false, error: true, servers: [] });
    });
}

function updateBusy(inst) {
  inst.$node.setAttribute('aria-busy', inst.state.loading ? 'true' : 'false');
}

vf.attach(document.getElementById('servers'), {
  state: { loading: true, error: false, servers: [], filter: 'all' },
  render: renderServers,
  delegates: [
    {
      selector: '[data-action="retry"]',
      eventType: 'click',
      onEvent: function (e) { load(e.sender); }
    },
    {
      selector: '[data-action="status-filter"]',
      eventType: 'change',
      onEvent: function (e) {
        var value = e.target.value;
        if (FILTER_VALUES.indexOf(value) === -1) value = 'all';
        e.sender.setState({ filter: value });
      }
    }
  ],
  onMount: function (inst) {
    updateBusy(inst);
    load(inst);
  },
  onUpdate: function (inst) {
    updateBusy(inst);
  }
});
```

### REPORT.md

## 가정
- `index.html`, `data/servers.json`은 과제 요구사항을 이미 만족하므로 고치지 않았습니다(CSP, 루트 요소, 스크립트 태그, 데이터 형태 모두 그대로 사용 가능).
- `Showing <보이는 수> of <전체 수>`의 "전체 수"는 필터와 무관하게 불러온 전체 서버 수(`state.servers.length`)로, "보이는 수"는 현재 필터를 적용한 뒤의 개수로 해석했습니다.
- 목록이 비어 있을 때(`No servers yet.`)는 필터 결과가 아니라 원본 데이터 자체가 빈 배열일 때만 적용했습니다(필터링 결과가 0건이면 `Showing 0 of N`을 보여줍니다). 킷에 이 구분이 명시되어 있지 않아 더 단순하고 일관된 쪽을 택했습니다.
- `url`이 `http:`/`https:`가 아니면 링크에 `href` 속성 자체를 넣지 않아(대신 `vf.safeUrl`이 반환하는 `#` 등으로도 만들지 않음) "동작하는 링크가 되면 안 된다"는 조건을 만족시켰습니다. `href`가 없는 `<a>`는 클릭해도 이동하지 않습니다.
- 최상위 응답이 배열이 아니거나(신뢰할 수 없는 데이터), HTTP 오류·네트워크 오류가 나면 모두 같은 "요청 실패" 상태(`Could not load servers.` + `Retry`)로 처리했습니다. 배열 안의 원소 중 객체가 아닌 값은 조용히 걸러냈습니다.
- `id`가 없거나 비어 있는 서버 항목은 `row-<index>`로 대체 id를 만들어 `data-id`가 항상 존재하도록 했습니다.
- 필터 select의 `change` 이벤트 값이 정의된 4개 값(`all/up/down/maintenance`) 밖이면 `all`로 되돌리도록 방어했습니다.
- 최신 브라우저만 대상이라는 조건(과제 4번)에 따라 `fetch`/`Promise`를 그대로 사용했고 ES5 제약(IE11)은 적용하지 않았습니다.
- 다국어(`vf.i18n`/`data-i18n`)는 과제 문구가 모두 고정 영어 문구로 지정되어 있어 적용하지 않았습니다(메시지 키를 쓰라는 AGENTS.md 6절과 다소 어긋나지만, 과제 6번 규칙 "화면 문구는 과제에 적힌 영어 문구를 그대로" 쪽을 우선했습니다).
