### app.js

```javascript
// Wires behaviour onto the publisher's static markup (index.html / dashboard.css are unchanged
// except for the hook attributes and CSP-required replacements listed in REPORT.md).
import vf from './lib/vfunc.esm.js';

const ORDERS = [
  { id: '#1042', customer: 'Minsu Kim', status: 'paid', statusLabel: 'Paid', total: '₩59,000' },
  { id: '#1041', customer: 'Seoyeon Park', status: 'shipped', statusLabel: 'Shipped', total: '₩329,000' },
  { id: '#1040', customer: 'Jiho Choi', status: 'refunded', statusLabel: 'Refunded', total: '₩29,000' }
];

const SUMMARY_TEXT = {
  week: 'Orders are up this week. Most sales came from the keyboard line.',
  month: 'This month: 1,204 orders, 3.4% refunds.'
};

// Smallest-element render targets (rule 6 of the conversion prompt): the summary text box
// and the table body, not the panels that wrap them.
const summary = vf.attach(document.getElementById('summary'), {
  state: { tab: 'week' },
  render: (s) => vf.html`<p>${SUMMARY_TEXT[s.tab]}</p>`
});

const ordersBody = vf.attach(document.getElementById('ordersBody'), {
  state: { search: '', filter: '', orders: ORDERS },
  render: (s) => {
    const term = s.search.trim().toLowerCase().replace(/^#/, '');
    const rows = s.orders.filter((o) => {
      const num = o.id.replace('#', '').toLowerCase();
      const matchesSearch = !term || num.indexOf(term) !== -1 || o.customer.toLowerCase().indexOf(term) !== -1;
      const matchesFilter = !s.filter || o.status === s.filter;
      return matchesSearch && matchesFilter;
    });
    if (!rows.length) {
      return vf.html`<tr><td colspan="4">No orders match.</td></tr>`;
    }
    return vf.html`${rows.map((o) => vf.html`<tr><td>${o.id}</td><td>${o.customer}</td><td><span class="badge badge--${o.status}">${o.statusLabel}</span></td><td class="orders__num">${o.total}</td></tr>`)}`;
  }
});

// Closures for the document-level listeners (menu outside-click / Escape). Kept as module
// variables, not instance properties, per AGENTS.md 4.9 ("_"-prefixed names are engine-only).
let detachOutsideHandlers = null;

// The rest of the page (menu, sign-out, tabs, search box, filter select) keeps its exact
// markup: adopted with vf.attach and no `render`, so only the attributes/text this file
// touches explicitly (via onUpdate) ever change.
const app = vf.attach(document.body, {
  state: { menuOpen: false, tab: 'week', search: '', filter: '', signedOut: false },
  delegates: [
    {
      selector: '[data-action="menu"]',
      eventType: 'click',
      onEvent: (e) => { e.sender.menuOpen = !e.sender.state.menuOpen; }
    },
    {
      selector: '[data-action="sign-out"]',
      eventType: 'click',
      onEvent: (e) => {
        e.event.preventDefault();
        e.sender.setState({ menuOpen: false, signedOut: true });
      }
    },
    {
      selector: '[data-action="tab"]',
      eventType: 'click',
      onEvent: (e) => { e.sender.tab = e.target.getAttribute('data-tab'); }
    },
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent: (e) => { e.sender.search = e.target.value; }
    },
    {
      selector: '[data-action="filter"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.filter = e.target.value; }
    }
  ],
  onMount: (inst) => {
    const onDocClick = (ev) => {
      if (!inst.state.menuOpen) return;
      const menu = inst.ids.userMenu;
      const btn = inst.refs.userBtn;
      if (menu.contains(ev.target) || btn.contains(ev.target)) return;
      inst.menuOpen = false;
    };
    const onDocKeydown = (ev) => {
      if (ev.key === 'Escape' && inst.state.menuOpen) {
        inst.menuOpen = false;
      }
    };
    document.addEventListener('click', onDocClick);
    document.addEventListener('keydown', onDocKeydown);
    detachOutsideHandlers = () => {
      document.removeEventListener('click', onDocClick);
      document.removeEventListener('keydown', onDocKeydown);
    };
  },
  onDestroy: () => {
    if (detachOutsideHandlers) {
      detachOutsideHandlers();
      detachOutsideHandlers = null;
    }
  },
  onUpdate: (inst) => {
    const s = inst.state;
    const menu = inst.ids.userMenu;
    const btn = inst.refs.userBtn;

    menu.hidden = !s.menuOpen;
    btn.setAttribute('aria-expanded', String(s.menuOpen));
    if (s.signedOut) {
      btn.textContent = 'Signed out';
    }

    inst.$node.querySelectorAll('[data-action="tab"]').forEach((tabBtn) => {
      const isActive = tabBtn.getAttribute('data-tab') === s.tab;
      tabBtn.classList.toggle('tabs__tab--active', isActive);
      tabBtn.setAttribute('aria-selected', String(isActive));
    });

    summary.setState({ tab: s.tab });
    ordersBody.setState({ search: s.search, filter: s.filter });
  }
});
```

# 변환 보고서 — 퍼블리싱 대시보드

## 1. 영역 판별표

| 영역(셀렉터) | 정적/동적 | 이유 | 도구 | 데이터 | 이벤트 |
|---|---|---|---|---|---|
| `.topbar__logo`, `.sidebar`, `.kpis`, `.panel__title`, `thead` | 정적 | 실행 중 내용·상태가 바뀌지 않음 | 손대지 않음 | — | — |

| `.tabs__tab` 두 개 | 동적(속성만) | 활성 탭 클래스/`aria-selected` 토글 | 위와 동일 `onUpdate` | 없음 | `data-action="tab"` click |

| `.panel__filter` | 동적(입력만) | 값만 읽으면 됨 | 채택 | 없음 | `data-action="filter"` change |
| `#ordersBody`(`tbody`) | 동적(내용) | 검색/필터에 따라 행이 바뀜 | `render` 있는 `vf.attach(#ordersBody)` | `ORDERS` 상수 배열 | 없음(상위에서 `setState`) |

## 2. HTML에 추가/치환한 속성 (그 밖의 텍스트·구조는 원본과 바이트 단위로 동일)

추가:
- `<meta http-equiv="Content-Security-Policy" …>` (신규 태그, 규칙 4의 인라인 스크립트/스타일 금지에 대응)
- `data-action="search"` — `.topbar__search`
- `data-action="menu"`, `data-ref="userBtn"`, `aria-expanded="false"` — `.topbar__user-btn`
- `id="ordersBody"` — `tbody`
- `data-action="tab"` `data-tab="week"|"month"` `aria-selected="true"|"false"` — 탭 버튼 2개
- `data-action="filter"` — `select.panel__filter`
- `data-action="sign-out"` — "Sign out" 링크

치환:
- `<script>…</script>`(인라인) → `<script type="module" src="./app.js"></script>`
- `.topbar__user-btn onclick="toggleMenu()"` 제거
- `#userMenu style="display:none"` → `hidden` 속성 (인라인 style은 CSP로 막히므로 규칙 7대로 대체)
- `"Sign out"` 링크 `href="javascript:void(0)" onclick="alert(...)"` → `href="#"` + `data-action="sign-out"` (클릭 핸들러에서 `preventDefault`)
- 탭 버튼의 `onclick="showTab('week'|'month')"` 제거

`data-tab`은 프롬프트 기본 훅 목록(`id`/`data-action`/`data-ref`/`data-id`/`data-i18n`/`data-vf-keep`)에는 없지만, 이 과제의 `BEHAVIOUR` 절이 채점기 훅으로 명시했으므로 추가했습니다.

## 3. 제거한 위험 요소 (규칙 4)

- 인라인 `<script>` 블록 전체(`toggleMenu`, `showTab` 함수) — `app.js`로 이전
- `onclick="toggleMenu()"`, `onclick="alert('Signed out')"` 인라인 핸들러 2건 제거
- `href="javascript:void(0)"` URL 제거 → `href="#"` + `preventDefault()`
- 인라인 `style="display:none"` 제거 → `hidden` 속성

## 5. 가정 및 VERIFY

- `VERIFY:` 정확한 토큰 이름 목록(`--vf-color-*`, `--vf-space-*` 등)은 `lib/vfunc.tokens.css`를 열어 확인해야 합니다. 이 과제에서는 CSS를 바꾸지 않으므로 실제 적용은 하지 않았습니다.
- "이름 버튼의 문구를 Signed out으로 바꾼다"는 요구를 문자 그대로 버튼 텍스트 전체를 `Signed out`으로 교체하는 것으로 해석했습니다(화살표 `▾`도 함께 사라짐). 다시 로그인하는 동작은 과제 범위 밖이라 만들지 않았습니다.
- 검색은 주문 번호(부분 일치, `#` 유무 무관)와 고객 이름(부분 일치)에 대해 대소문자 구분 없이 매칭하도록 구현했습니다. 완전 일치가 아닌 포함(substring) 매칭이 "거른다"는 요구에 맞다고 판단했습니다.
- `AGENTS.md`는 사용자 노출 문자열에 메시지 키(`vf.t`/`data-i18n`)를 쓰도록 요구하지만, 이 과제 프로젝트 파일에는 `locales/` 폴더가 주어지지 않았고 과제 문구도 영어 원문을 그대로 쓰라고 지정했습니다. 그래서 `i18n` 배선은 만들지 않고 문자열을 `SUMMARY_TEXT`/버튼 텍스트에 직접 두었습니다 — AGENTS.md 6절을 이 범위에서는 지키지 못했습니다(이유: 로케일 파일이 없어 키를 추가할 대상이 없음).
- Escape/외부 클릭으로 메뉴를 닫을 때 포커스를 다시 사용자 버튼으로 옮기는 처리는 과제에 명시되지 않아 만들지 않았습니다.
- `lib/vfunc.esm.js`가 프로젝트에 이미 있다고 조건에 적혀 있어 그 경로로 import했습니다. 실제 파일 내용(빌드 버전이 rc.7과 일치하는지)은 확인하지 못했습니다.
