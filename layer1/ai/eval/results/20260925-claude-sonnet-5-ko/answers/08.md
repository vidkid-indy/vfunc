### app.js

```js
// Sales — the page script. Chart.js is loaded as the global `Chart`.
(function () {
  'use strict';

  var MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  var SERIES = {
    revenue: { label: 'Revenue', values: [120, 135, 150, 142, 168, 181, 175, 190, 205, 198, 220, 236] },
    orders: { label: 'Orders', values: [30, 32, 35, 33, 40, 44, 41, 46, 50, 48, 53, 57] }
  };
  var TOKEN_BY_SERIES = { revenue: '--vf-chart-1', orders: '--vf-chart-2' };

  var salesSection = document.getElementById('sales');
  var toggleButton = document.querySelector('[data-action="toggle-chart"]');

  // Only one sales component (and one Chart.js instance) is ever alive at a
  // time, so the Chart.js instance lives in a module-level closure variable
  // instead of on the vfunc instance (properties can't be attached directly
  // to an instance; `_`-prefixed names are reserved for the engine).
  var chartInstance = null;
  var salesInstance = null;

  function tokenColor(name) {
    return getComputedStyle(document.documentElement)
      .getPropertyValue(TOKEN_BY_SERIES[name])
      .trim();
  }

  function visibleData(state) {
    var series = SERIES[state.seriesKey];
    return {
      label: series.label,
      labels: MONTHS.slice(0, state.monthCount),
      values: series.values.slice(0, state.monthCount),
      color: tokenColor(state.seriesKey)
    };
  }

  function drawChart(inst) {
    var d = visibleData(inst.state);
    chartInstance = new Chart(inst.refs.chart, {
      type: 'line',
      data: {
        labels: d.labels,
        datasets: [{ label: d.label, data: d.values, borderColor: d.color }]
      },
      options: { responsive: true, maintainAspectRatio: false }
    });
  }

  function updateChart(inst) {
    if (!chartInstance) return;
    var d = visibleData(inst.state);
    chartInstance.data.labels = d.labels;
    chartInstance.data.datasets[0].label = d.label;
    chartInstance.data.datasets[0].data = d.values;
    chartInstance.data.datasets[0].borderColor = d.color;
    chartInstance.update();
  }

  function destroyChart() {
    if (chartInstance) {
      chartInstance.destroy();
      chartInstance = null;
    }
  }

  function attachSales() {
    return vf.attach(salesSection, {
      state: { seriesKey: 'revenue', monthCount: 6 },
      render: function (s) {
        var series = SERIES[s.seriesKey];
        var lastValue = series.values[s.monthCount - 1];
        var atMax = s.monthCount >= MONTHS.length;
        return vf.html`
          <div class="sales__controls">
            <button type="button" class="sales__button" data-action="series" data-series="revenue" aria-pressed="${s.seriesKey === 'revenue'}">Revenue</button>
            <button type="button" class="sales__button" data-action="series" data-series="orders" aria-pressed="${s.seriesKey === 'orders'}">Orders</button>
            <button type="button" class="sales__button" data-action="add-month" ${atMax ? 'disabled' : ''}>Add month</button>
          </div>
          <p class="sales__summary" data-ref="summary">${series.label}: ${s.monthCount} months, last ${lastValue}</p>
          <div class="sales__chart" data-vf-keep="chart">
            <canvas data-ref="chart"></canvas>
          </div>
        `;
      },
      delegates: [
        {
          selector: '[data-action="series"]',
          eventType: 'click',
          onEvent: function (e) {
            var key = e.target.getAttribute('data-series');
            e.sender.setState({ seriesKey: key });
          }
        },
        {
          selector: '[data-action="add-month"]',
          eventType: 'click',
          onEvent: function (e) {
            var inst = e.sender;
            if (inst.state.monthCount < MONTHS.length) {
              inst.setState({ monthCount: inst.state.monthCount + 1 });
            }
          }
        }
      ],
      onMount: function (inst) {
        drawChart(inst);
      },
      onUpdate: function (inst) {
        updateChart(inst);
      },
      onDestroy: function () {
        destroyChart();
      }
    });
  }

  // Toggle button lives outside the sales component (page header), so it is
  // wired as plain publishing HTML: vf.attach with delegates only, no
  // render, keeps the existing markup as-is.
  vf.attach(document.querySelector('.page__head'), {
    delegates: [
      {
        selector: '[data-action="toggle-chart"]',
        eventType: 'click',
        onEvent: function () {
          if (salesInstance) {
            salesInstance.destroy();
            salesInstance = null;
            toggleButton.textContent = 'Show chart';
          } else {
            salesInstance = attachSales();
            toggleButton.textContent = 'Hide chart';
          }
        }
      }
    ]
  });

  salesInstance = attachSales();
})();
```

### REPORT.md

## 가정
- 프로젝트에 `locales/`가 없고 과제가 화면 문구를 정확한 영어 문자열로 지정했으므로, 이 컴포넌트에서는 `vf.t`/메시지 키 대신 문구를 그대로 하드코딩했습니다(AGENTS.md 6절과 다르지만, 과제 조건 6 "화면 문구는 과제에 적힌 영어 문구를 그대로" 를 우선했습니다).
- `#sales`와 헤더의 `data-action="toggle-chart"` 버튼은 프로젝트가 이미 만들어 둔 퍼블리싱 마크업이므로, 새 `vf.vfunc` 루트를 만들어 그 안에 붙이는 대신 `vf.attach`로 기존 요소를 그대로 제어했습니다(킷 4.7, 4.6절 패턴). `#sales`의 내용 전체가 상태에 따라 바뀌므로 `render`를 전체에 붙였습니다(정적 부분이 없어 4.7절의 "가장 작은 요소" 경고에는 해당하지 않는다고 판단).
- Chart.js 인스턴스는 인스턴스에 직접 속성을 붙일 수 없다는 규칙(`_`로 시작하지 않는 한 예약어 아님이지만 "인스턴스에 직접 속성을 붙이지 않는다"는 명시적 금지) 때문에 모듈 클로저 변수(`chartInstance`)에 보관했습니다. 이 앱에서는 매출 컴포넌트가 한 번에 하나만 존재하므로 안전하다고 판단했습니다.
- 숨김/표시는 `salesInstance.destroy()` 후 `attachSales()`로 완전히 새 컴포넌트를 붙이는 방식으로 구현했습니다(요청이 "새 컴포넌트를 만들고"라고 명시했고, `destroy()`는 같은 요소에 다시 attach할 수 있다고 킷에 적혀 있어 이 패턴이 맞다고 판단했습니다).

## 킷에서 지키지 못한 것
- 없음. `style.css`, `index.html`은 과제 요구에 맞춰 그대로 두었고(고칠 필요가 없었음), 토큰만으로 색을 얻었습니다(`getComputedStyle`로 `--vf-chart-1`/`--vf-chart-2` 읽음, JS에 색상 리터럴 없음).
