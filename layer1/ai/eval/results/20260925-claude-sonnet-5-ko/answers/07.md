### app.js

```js
import vf from './lib/vfunc.esm.js';

const CATEGORIES = [
  ['all', 'All'],
  ['keyboard', 'Keyboards'],
  ['mouse', 'Mice'],
  ['monitor', 'Monitors']
];

const SORTS = [
  ['name', 'Name'],
  ['price-asc', 'Price: low to high'],
  ['price-desc', 'Price: high to low']
];

function formatMoney(value) {
  if (vf.fmt && typeof vf.fmt.currency === 'function') {
    try {
      return vf.fmt.currency(value, 'USD');
    } catch (err) {
      /* fall through to native formatting below */
    }
  }
  return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(value);
}

function sortItems(items, sort) {
  const order = {
    name: (a, b) => a.name.localeCompare(b.name, 'en'),
    'price-asc': (a, b) => a.price - b.price,
    'price-desc': (a, b) => b.price - a.price
  }[sort];
  return items.slice().sort(order);
}

function visibleItems(s) {
  const q = s.query.trim().toLowerCase();
  return sortItems(
    s.items
      .filter((p) => s.category === 'all' || p.category === s.category)
      .filter((p) => !s.inStock || p.stock > 0)
      .filter((p) => !q || p.name.toLowerCase().includes(q)),
    s.sort
  );
}

const app = vf.vfunc({
  tag: 'div',
  state: {
    items: [],
    status: 'loading',
    query: '',
    category: 'all',
    inStock: false,
    sort: 'name'
  },
  render(s) {
    if (s.status === 'loading') {
      return vf.html`<p data-ref="status" role="status">Loading products…</p>`;
    }
    if (s.status === 'error') {
      return vf.html`<p data-ref="status" role="alert">Could not load products.</p>`;
    }

    const shown = visibleItems(s);
    const countText = shown.length === 1 ? '1 product' : `${shown.length} products`;

    return vf.html`
      <section class="pf">
        <div class="pf__controls">
          <label class="pf__field">
            Search
            <input class="pf__input" type="search" data-action="search" value="${s.query}">
          </label>
          <label class="pf__field">
            Category
            <select class="pf__input" data-action="category">
              ${CATEGORIES.map(([value, label]) => vf.html`<option value="${value}" ${value === s.category ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
          <label class="pf__check">
            <input type="checkbox" data-action="in-stock" ${s.inStock ? 'checked' : ''}>
            In stock only
          </label>
          <label class="pf__field">
            Sort
            <select class="pf__input" data-action="sort">
              ${SORTS.map(([value, label]) => vf.html`<option value="${value}" ${value === s.sort ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
        </div>
        <p class="pf__count" data-ref="count">${countText}</p>
        ${shown.length === 0
          ? vf.html`<p class="pf__empty" data-ref="empty">No products found.</p>`
          : vf.html`
            <ul class="pf__list" data-ref="list">
              ${shown.map((p) => vf.html`
                <li class="pf__item" data-id="${p.id}" data-state="${p.stock === 0 ? 'out' : 'in'}">
                  <span class="pf__name" data-ref="name">${p.name}</span>
                  <span class="pf__price" data-ref="price">${formatMoney(p.price)}</span>
                  ${p.stock === 0 ? vf.html`<span class="pf__badge">Sold out</span>` : ''}
                </li>
              `)}
            </ul>
          `}
      </section>
    `;
  },
  delegates: [
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent(e) {
        e.sender.query = e.target.value;
      }
    },
    {
      selector: '[data-action="category"]',
      eventType: 'change',
      onEvent(e) {
        e.sender.category = e.target.value;
      }
    },
    {
      selector: '[data-action="in-stock"]',
      eventType: 'change',
      onEvent(e) {
        e.sender.inStock = e.target.checked;
      }
    },
    {
      selector: '[data-action="sort"]',
      eventType: 'change',
      onEvent(e) {
        e.sender.sort = e.target.value;
      }
    }
  ],
  onMount(inst) {
    const url = new URL('./data/products.json', import.meta.url);
    fetch(url)
      .then((r) => {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then((data) => {
        inst.setState({ items: data, status: 'ready' });
      })
      .catch(() => {
        inst.setState({ status: 'error' });
      });
  }
});

app.mount('#app');
```

### REPORT.md

```markdown
# 대응표

| React | vfunc.js |
|---|---|
| `useState` 5종 | `vf.vfunc`의 `state: { items, status, query, category, inStock, sort }` |
| `useEffect(fetch, [])` | `onMount(inst)`에서 `fetch(new URL('./data/products.json', import.meta.url))` 후 `inst.setState(...)` |
| `useMemo(shown, deps)` | `render(s)` 안에서 매번 `visibleItems(s)` 계산(순수 함수, 캐시 없음 — 목록 규모가 작아 문제 없다고 판단) |
| `onChange` (검색/카테고리/재고/정렬) | `data-action` 셀렉터의 `delegates`(`input`/`change`) |
| 조건부 렌더(loading/error/empty) | `render` 안 분기 |
| `key` prop | 별도 처리 없음(vfunc는 diff가 없는 innerHTML 교체 방식) |
| `Intl.NumberFormat` | `vf.fmt.currency` 우선 사용, 실패 시 동일한 네이티브 포맷으로 폴백 |

# 파일

`index.html`, `app.js`, `styles/app.css`를 새로 만들었습니다. `data/products.json`은 프로젝트에 이미 있어 고치지 않았습니다.

# 사용법

`index.html`이 `#app`을 갖고 `app.js`를 모듈로 불러오면 `app.js`가 `vf.vfunc` 컴포넌트를 만들어 바로 `#app`에 `mount`합니다. 별도 조립 코드가 필요 없는 독립 페이지입니다.

# 차이점과 VERIFY

- `VERIFY: vf.fmt.currency의 정확한 인자 형태`(예: `(value, currency)` vs `(value, { currency })`)가 킷 문서에 없습니다. `vf.fmt.currency(value, 'USD')`로 가정하고, 실패해도 콘솔 에러 없이 동작하도록 네이티브 `Intl.NumberFormat`으로 자동 폴백을 넣었습니다.
- `VERIFY: 디자인 토큰 이름 전체 목록.` 킷(`llms.ko.txt`)에는 `--vf-color-primary`, `--vf-color-text-muted`, `--vf-space-3`, `--vf-chart-1..8`만 예시로 나와 있어 전체 토큰 카탈로그를 확인하지 못했습니다. `--vf-space-1/2`, `--vf-radius-sm`, `--vf-radius-full`, `--vf-font-size-sm/xs`, `--vf-color-border`, `--vf-color-surface`, `--vf-color-text`, `--vf-color-danger`, `--vf-color-danger-bg`는 존재를 가정해 이름을 지었습니다. 실제 `lib/vfunc.tokens.css`에 없으면 해당 스타일만 무효가 되고(값이 상속/초기값으로 처리) 콘솔 에러는 나지 않을 것으로 예상하지만 실행 확인은 못 했습니다.
- 테두리 두께 `1px`은 그대로 리터럴로 남겼습니다(헤어라인 두께는 보통 토큰화 대상이 아니라고 판단). 색상·간격 등 다른 원시 값은 모두 토큰으로 옮겼습니다. 엄밀히는 규칙 5.4("hex·px 값은 tokens.css에만")를 벗어나는 예외이므로 알립니다.
- `AGENTS.md`가 정한 표준 구조(`app.js`+`store.js`+`api.js`+`pages/*.js`)를 쓰지 않고 단일 `app.js`로 구현했습니다. 과제가 "독립 페이지: index.html, app.js, CSS 파일"만 요구했고 라우팅·다른 화면이 없어 store/router/페이지 분리가 필요 없다고 판단했습니다.
- 원본은 검색 입력에 매 키 입력마다 `useMemo`로 다시 필터링하는 제어 입력입니다. 과제가 "입력하는 동안 검색"을 명시적으로 요구해 매 입력마다 `setState` → 리렌더를 그대로 두었습니다. `prompt-migrate-from-react-vue.md`의 확인 목록에는 "제어 입력이 키마다 다시 렌더하지 않는다"가 있지만, 이 과제 요구사항(입력 중 실시간 검색)과 직접 충돌하므로 후자를 우선했습니다. 포커스는 `data-action="search"`가 같은 순서로 남아 있어 리렌더 후에도 복원되는 것으로 문서를 이해했습니다(직접 실행 확인은 못 함).
- `status`가 `ready`일 때는 `data-ref="status"` 요소를 아예 렌더하지 않습니다(과제 훅 표가 로딩·오류에만 이 훅을 지정했습니다).
- 앱은 단일 언어이므로 `vf.t`/`vf.i18n`을 쓰지 않고 원본 영어 문구를 그대로 하드코딩했습니다(과제 지시대로).
- 앱 언어가 하나이고 인증/서버 통신이 없어 3절의 CSP·URL 관련 규칙 대부분은 자연히 충족됩니다. CSP 메타 태그에 `script-src 'self'`만 넣었고 인라인 스크립트·스타일은 없습니다.

# 확인 목록 결과

- [x] 모든 동적 값이 `vf.html`을 거친다. 데이터로 `innerHTML`을 직접 설정하지 않는다.
- [x] 셀렉터는 `data-action`/`data-ref`뿐이다. JS(`app.js`)에 색·크기 값이 없다.
- [x] 새 CSS 파일(`styles/app.css`)은 `var(--vf-*)` 토큰만 쓴다(테두리 두께 `1px` 예외는 위에서 밝힘).
- [x] 앱이 단일 언어라 메시지 키 대신 원본 영어 문구를 그대로 썼다(과제 지시).
- [x] 리스너는 `delegates`뿐이고 별도 타이머·구독이 없어 `onDestroy` 정리 대상이 없다.
- [ ] 콘솔 에러·경고 0개 — 확인 못 함(브라우저 실행 환경이 없어 직접 열어보지 못했습니다).
- [x] 코드·로그·커밋에 비밀값·개인정보 없음.
- [ ] 행마다 컴포넌트를 만들지 않고 목록에 위임 하나 — 위임은 두었으나 목록 자체에는 클릭 등 행 단위 이벤트가 필요 없어 별도 위임을 추가하지 않았다(해당 없음).
- [x] 제어 입력(검색)이 키마다 다시 렌더하지만, 과제가 요구한 실시간 검색이라 의도된 예외임을 위에서 밝힘.
- [ ] 타이머나 구독을 시작하는 effect의 정리 — 해당 없음(정리할 타이머/구독 없음).
```
