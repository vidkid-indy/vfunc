### app.js

```js
// Team board — the page script.
import vf from './lib/vfunc.esm.js';
import { store, setTheme, setLang, toggleTask, removeTask } from './store.js';

// Preferences: theme and language.
const prefs = vf.attach('#prefs', {
  render: () => {
    const p = store.get('prefs');
    return vf.html`
      <p class="prefs__row">Theme: <span data-ref="theme">${p.theme}</span>
        <button class="prefs__button" type="button" data-action="theme">Switch theme</button></p>
      <p class="prefs__row">Language: <span data-ref="lang">${p.lang}</span>
        <button class="prefs__button" type="button" data-action="lang" data-lang="en">English</button>
        <button class="prefs__button" type="button" data-action="lang" data-lang="ko">한국어</button></p>`;
  },
  delegates: [
    { selector: '[data-action="theme"]', eventType: 'click', onEvent: () => setTheme(store.get('prefs').theme === 'dark' ? 'light' : 'dark') },
    { selector: '[data-action="lang"]', eventType: 'click', onEvent: (e) => setLang(e.target.getAttribute('data-lang')) }
  ]
});

// Notice: how many tasks are open.
const notice = vf.attach('#notice', {
  render: () => {
    const open = store.get('tasks').filter((t) => !t.done).length;
    return vf.html`<p class="notice__text">Open tasks: ${open}</p>`;
  }
});

// Details: a disclosure.
vf.attach('#details-panel', {
  state: { open: false },
  delegates: [{
    selector: '[data-action="details"]',
    eventType: 'click',
    onEvent: (e) => {
      const open = !e.sender.state.open;
      e.sender.state.open = open;
      e.sender.ids.details.hidden = !open;
      e.target.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
  }],
  onMount: (inst) => inst.$node.querySelector('[data-action="details"]').setAttribute('aria-expanded', 'false')
});

// Tasks, open ones first.
const tasks = vf.attach('#tasks', {
  render: () => vf.html`
    <h2 class="panel__title">Tasks</h2>
    <ul class="tasks">${store.get('tasks').slice().sort((a, b) => a.done - b.done).map((t) => vf.html`
      <li class="task" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
        <input type="checkbox" id="toggle-${t.id}" data-action="toggle" aria-label="Done" ${t.done ? 'checked' : ''}>
        <span class="task__title">${t.title}</span>
        <button class="task__delete" type="button" data-action="remove">Remove</button>
      </li>`)}</ul>`,
  delegates: [
    { selector: '[data-action="toggle"]', eventType: 'change', onEvent: (e) => toggleTask(e.target.closest('[data-id]').getAttribute('data-id')) },
    { selector: '[data-action="remove"]', eventType: 'click', onEvent: (e) => removeTask(e.target.closest('[data-id]').getAttribute('data-id')) }
  ]
});

// Comments.
const comments = [];
vf.attach('#comments', {
  delegates: [{
    selector: '[data-action="comment"]',
    eventType: 'submit',
    onEvent: (e) => {
      e.event.preventDefault();
      const text = e.sender.refs.comment.value.trim();
      if (!text) return;
      comments.push(text);
      e.sender.refs['comment-list'].innerHTML = comments.map((c) => '<li class="comments__item">' + vf.esc(c) + '</li>').join('');
      e.sender.refs.comment.value = '';
    }
  }]
});

// Footer: how often the clock has updated.
const footer = vf.attach('#footer', {
  state: { updates: 0 },
  render: (s) => vf.html`Clock updates: <span data-ref="updates">${s.updates}</span>`
});

// Clock, with a close button.
let clockTimer = null;
const clock = vf.vfunc({
  state: { now: new Date() },
  render: (s) => vf.html`
    <p class="clock">Time: <span data-ref="time">${vf.fmt.date(s.now, { timeStyle: 'medium' })}</span>
      <button class="clock__close" type="button" data-action="close-clock">Close clock</button></p>`,
  delegates: [{ selector: '[data-action="close-clock"]', eventType: 'click', onEvent: (e) => e.sender.destroy() }],
  onMount: (inst) => {
    clockTimer = setInterval(() => {
      inst.setState({ now: new Date() });
      footer.setState({ updates: footer.state.updates + 1 });
    }, 250);
  },
  onDestroy: () => clearInterval(clockTimer)
});
clock.mount('#clock-slot');

store.subscribe(() => {
  prefs.refresh();
  notice.refresh();
  tasks.refresh();
});
```

### store.js

```js
// Shared state and the only functions that change it.
import vf from './lib/vfunc.esm.js';

export const store = vf.store({
  prefs: { theme: 'light', lang: 'en' },
  tasks: [
    { id: 't1', title: 'Write the release notes', done: false },
    { id: 't2', title: 'Review the login page', done: false },
    { id: 't3', title: 'Update the icons', done: true }
  ]
});

export function setTheme(theme) {
  store.set((s) => ({ prefs: { theme: theme, lang: s.prefs.lang } }));
}

export function setLang(lang) {
  store.set((s) => ({ prefs: { theme: s.prefs.theme, lang: lang } }));
}

export function toggleTask(id) {
  store.set((s) => ({ tasks: s.tasks.map((t) => (t.id === id ? { id: t.id, title: t.title, done: !t.done } : t)) }));
}

export function removeTask(id) {
  store.set((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }));
}
```

```markdown
# 버그 일곱 개 — 원인과 수정

## 1. `#notice` 요소가 두 개가 됨
- **원인**: `render`가 `<section id="notice">…</section>` 자신을 통째로 반환했다. `vf.attach`의 `render`는 대상 요소의 **안쪽만** 돌려줘야 하는데, 같은 `id="notice"`를 가진 `<section>`을 다시 만들어 안에 중첩시켰다(콘솔 경고와 정확히 일치).
- **수정**: `render`가 `<p class="notice__text">Open tasks: ${open}</p>`만 반환하도록 줄였다. `id`, `class`, `aria-live`는 이미 `index.html`의 `#notice`가 가지고 있으므로 그대로 유지된다.
- **확인 방법**: 페이지를 열어 콘솔에 `attach: render returned the target element itself` 경고가 없는지, `document.querySelectorAll('#notice').length === 1`인지, 할 일을 체크하면 숫자가 바뀌는지 확인. (실행은 못 해봤음 — "확인 못 함")

## 2. Details 버튼의 `aria-expanded=""`
- **원인**: `onMount`와 클릭 핸들러 모두 닫힌 상태를 빈 문자열 `''`로 썼다. `aria-*` 속성은 boolean 문자열(`"true"`/`"false"`)이어야 한다.
- **수정**: `onMount`의 초기값과 토글 핸들러의 `else` 분기를 `'false'`로 바꿨다.
- **확인 방법**: 첫 로드 시 `aria-expanded="false"`, 클릭 시 `"true"`/`"false"`로 토글되는지, `#details`가 닫혀 있을 때 `hidden`인지 확인. (확인 못 함)

## 3. 언어를 바꾸면 테마가 light로 돌아감
- **원인**: `setLang`이 `store.set({ prefs: { lang: lang } })`을 호출했다. `store.set`은 최상위 키만 얕은 병합하므로 `prefs` 키 전체가 `{ lang }`으로 **교체**되어 `theme`가 사라진다(`setTheme`는 이미 `s.prefs.lang`을 보존하는 함수형 패턴을 썼지만 `setLang`은 그러지 않았다).
- **수정**: `setLang`도 `setTheme`와 같은 함수형 패턴으로 바꿔 `s.prefs.theme`를 보존한다.
- **확인 방법**: 테마를 dark로 바꾼 뒤 언어를 한국어로 바꿔도 "Theme: dark"가 유지되는지 확인. (확인 못 함)

## 4. Close clock 이후에도 "Clock updates" 숫자가 계속 오름
- **원인**: `onMount`는 타이머를 `inst._timer`에 저장했는데 `onDestroy`는 `inst.timer`(밑줄 없음)를 지웠다 — 이름이 달라 `clearInterval`이 실제 타이머를 지우지 못했다. 또한 `_`로 시작하는 인스턴스 속성은 엔진 몫이라 애초에 인스턴스에 붙이면 안 된다(AGENTS.md 4-9, 리스너·타이머는 클로저/모듈 변수에 두라고 명시).
- **수정**: 인스턴스 속성 대신 모듈 스코프 클로저 변수 `clockTimer`를 만들어 `onMount`에서 채우고 `onDestroy`에서 `clearInterval(clockTimer)`로 지운다.
- **확인 방법**: "Close clock" 클릭 후 몇 초 기다려 `footer`의 "Clock updates" 숫자가 더 이상 늘지 않는지 확인. (확인 못 함)

## 5. Remove 버튼이 동작하지 않음
- **원인**: delegate 셀렉터가 클래스 `.task__remove`를 가리키는데, 디자이너가 마크업 클래스를 `task__delete`로 바꿨다. 애초에 동작 훅은 클래스가 아니라 `data-action`에 걸어야 한다(AGENTS.md 4-1, "CSS 클래스로 고르지 않습니다"). 버튼에는 이미 `data-action="remove"`가 있다.
- **수정**: delegate 셀렉터를 `[data-action="remove"]`로 바꿨다. 마크업(클래스 포함)은 손대지 않았다.
- **확인 방법**: Remove 클릭 시 해당 할 일이 목록에서 사라지는지 확인. (확인 못 함)

## 6. 댓글 XSS (`<b>hi</b>`가 굵게 보임)
- **원인**: 댓글 텍스트를 이스케이프 없이 문자열 이어붙이기로 `innerHTML`에 넣었다(`'<li…>' + c + '</li>'`). 사용자 입력이 마크업으로 해석되어 굵게 보이고, `<script>` 등도 실행될 수 있다.
- **수정**: 붙이기 전에 `vf.esc(c)`로 댓글 텍스트를 이스케이프한다(킷이 제공하는 이스케이프 헬퍼). 이제 `<b>hi</b>`는 `&lt;b&gt;hi&lt;/b&gt;`로 들어가 화면에는 그대로 글자로 보인다.
- **확인 방법**: 댓글창에 `<b>hi</b>`를 입력해 Post 후, 목록에 굵지 않은 텍스트 `<b>hi</b>`로 표시되는지, `<script>` 삽입 시 실행되지 않는지 확인. (확인 못 함)

## 7. Space로 체크 후 포커스가 다른 할 일로 이동
- **원인**: 체크박스에 `id`/`data-ref`가 없었다. `toggle`을 누르면 목록이 열린 항목 우선으로 다시 정렬되어 같은 위치의 요소가 바뀌는데, 엔진은 `id`/`data-ref`/`name`이 없으면 "같은 `data-action`의 같은 순서 요소"로 포커스를 되돌린다. 정렬로 순서가 바뀌었으니 엉뚱한 할 일의 체크박스로 포커스가 옮겨간 것이다.
- **수정**: 체크박스에 `id="toggle-${t.id}"`를 추가해 항목별로 안정된 식별자를 줬다. `data-id`, `data-action="toggle"` 등 기존 훅은 그대로 두었다.
- **확인 방법**: 열린 할 일 하나를 Space로 체크(→ 목록 끝으로 이동)한 뒤 포커스가 여전히 그 할 일의 체크박스(새 위치)에 있는지 확인. (확인 못 함)

## 가정
- `store.set`의 병합은 최상위 키만 얕게 병합한다는 llms.ko.txt 설명(`얕은 병합: 다른 키는 유지`)을 근거로 3번 원인을 판단했다.
- 댓글 XSS 수정은 `vf.html`로 리스트 전체를 다시 그리는 대신, 문서에 명시된 `vf.esc` 헬퍼로 최소 변경했다. `#comments` 패널 전체에 `render`를 붙이면 폼까지 다시 그려지는 부작용이 생기므로(흔한 실수 항목) 피했다.
- 킷에서 지키지 못한 것은 없다고 판단했다. `id`, `data-action`, `data-ref`, `data-id`, 화면 문구는 전부 그대로 두었고, 새로 추가한 속성은 체크박스의 `id`뿐이다(과제에서 "고치는 데 새 속성이 필요하면 추가" 허용).
```
