### app.js

```javascript
import vf from './lib/vfunc.esm.js';
import { store } from './store.js';
import homePage from './pages/home.js';
import serversPage from './pages/servers.js';
import serverDetailPage from './pages/serverDetail.js';
import notFoundPage from './pages/notFound.js';

const LOCALE_KEY = 'statusboard.locale';

function detectLocale() {
  try {
    const saved = window.localStorage.getItem(LOCALE_KEY);
    if (saved === 'en' || saved === 'ko') return saved;
  } catch (err) {
    // localStorage may be unavailable (private mode, storage disabled)
  }
  const nav = (navigator.language || navigator.userLanguage || 'en').toLowerCase();
  return nav.indexOf('ko') === 0 ? 'ko' : 'en';
}

async function main() {
  let currentLocale = detectLocale();
  let currentPage = null;

  const shell = vf.attach('#app', {
    delegates: [
      {
        selector: '[data-action="locale"]',
        eventType: 'click',
        onEvent: (e) => {
          const locale = e.target.dataset.locale;
          if (!locale || locale === currentLocale) return;
          currentLocale = locale;
          try {
            window.localStorage.setItem(LOCALE_KEY, locale);
          } catch (err) {
            // ignore storage failures
          }
          vf.i18n.set(locale);
        }
      }
    ]
  });

  function updateLocaleButtons() {
    shell.ids['locale-en'].setAttribute('aria-pressed', currentLocale === 'en' ? 'true' : 'false');
    shell.ids['locale-ko'].setAttribute('aria-pressed', currentLocale === 'ko' ? 'true' : 'false');
    document.documentElement.lang = currentLocale;
  }

  function updateNav(path) {
    const onServers = path === '/servers' || path.indexOf('/servers/') === 0;
    const onHome = path === '/';
    shell.ids['nav-home'].setAttribute('aria-current', onHome ? 'page' : 'false');
    shell.ids['nav-servers'].setAttribute('aria-current', onServers ? 'page' : 'false');
  }

  function updateFavCount() {
    shell.refs['fav-count'].textContent = String(store.get('favorites').length);
  }

  await vf.i18n.setup({
    locales: ['en', 'ko'],
    fallback: 'en',
    load: (l) => fetch(new URL('./locales/' + l + '.json', import.meta.url)).then((r) => r.json())
  });

  vf.i18n.subscribe(() => {
    vf.i18n.apply();
    updateLocaleButtons();
  });

  vf.i18n.set(currentLocale);
  vf.i18n.apply();
  updateLocaleButtons();

  store.subscribe(updateFavCount);
  updateFavCount();

  function mountPage(page) {
    if (currentPage) {
      currentPage.destroy();
      currentPage = null;
    }
    currentPage = page;
    currentPage.mount('#view');
  }

  const router = vf.router({
    mode: 'hash',
    routes: {
      '/': (ctx) => mountPage(homePage(ctx, router)),
      '/servers': (ctx) => mountPage(serversPage(ctx, router)),
      '/servers/:id': (ctx) => mountPage(serverDetailPage(ctx, router))
    },
    notFound: (ctx) => mountPage(notFoundPage(ctx, router)),
    onChange: (ctx) => updateNav(ctx.path),
    linkSelector: 'a[data-link]',
    focus: 'main'
  });

  router.start();
}

main();
```

### store.js

```javascript
import vf from './lib/vfunc.esm.js';

export const store = vf.store({ favorites: [] });

export function addFavorite(id) {
  store.set((state) => {
    if (state.favorites.indexOf(id) !== -1) return {};
    return { favorites: state.favorites.concat(id) };
  });
}

export function removeFavorite(id) {
  store.set((state) => ({ favorites: state.favorites.filter((fid) => fid !== id) }));
}

export function toggleFavorite(id) {
  if (store.get('favorites').indexOf(id) !== -1) {
    removeFavorite(id);
  } else {
    addFavorite(id);
  }
}

export function isFavorite(id) {
  return store.get('favorites').indexOf(id) !== -1;
}
```

### api.js

```javascript
export function fetchServers() {
  return fetch(new URL('./data/servers.json', import.meta.url)).then((res) => {
    if (!res.ok) throw new Error('Failed to load servers (' + res.status + ')');
    return res.json();
  });
}

export function fetchServer(id) {
  return fetchServers().then((servers) => {
    const found = servers.filter((s) => s.id === id)[0];
    return found || null;
  });
}
```

```javascript
import vf from '../lib/vfunc.esm.js';
import { fetchServers } from '../api.js';

export default function serversPage(ctx, router) {
  let offI18n = null;

  return vf.vfunc({
    tag: 'section',
    state: { status: 'loading', servers: [] },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p role="status">${vf.t('servers.loading')}</p>
        `;
      }
      if (s.status === 'error') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p role="alert">${vf.t('servers.error')}</p>
        `;
      }
      if (s.servers.length === 0) {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p>${vf.t('servers.empty')}</p>
        `;
      }
      return vf.html`
        <h1>${vf.t('servers.title')}</h1>
        <ul class="server-list">
          ${s.servers.map((srv) => vf.html`
            <li class="server-list__item" data-id="${srv.id}">
              <a data-link href="${router.href('/servers/' + srv.id)}">${srv.name}</a>
            </li>
          `)}
        </ul>
      `;
    },
    onMount: (inst) => {
      offI18n = vf.i18n.subscribe(() => inst.refresh());
      fetchServers()
        .then((servers) => inst.setState({ status: 'ready', servers: servers }))
        .catch(() => inst.setState({ status: 'error', servers: [] }));
    },
    onDestroy: () => {
      if (offI18n) offI18n();
    }
  });
}
```

```javascript
import vf from '../lib/vfunc.esm.js';
import { fetchServer } from '../api.js';
import { store, toggleFavorite, isFavorite } from '../store.js';

export default function serverDetailPage(ctx) {
  const id = ctx.params.id;
  let offI18n = null;
  let offStore = null;

  return vf.vfunc({
    tag: 'section',
    state: { status: 'loading', server: null },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`
          <h1>${vf.t('servers.loading')}</h1>
        `;
      }
      if (s.status === 'error' || !s.server) {
        return vf.html`
          <h1>${vf.t('server.missing')}</h1>
        `;
      }
      const fav = isFavorite(s.server.id);
      return vf.html`
        <h1>${s.server.name}</h1>
        <dl class="server-detail">
          <dt>${vf.t('nav.servers')}</dt>
          <dd>${s.server.region}</dd>
        </dl>
        <p data-state="${s.server.status}">${s.server.status}</p>
        <button type="button" data-action="favorite" aria-pressed="${fav}">
          ${fav ? vf.t('server.favorite.remove') : vf.t('server.favorite.add')}
        </button>
      `;
    },
    delegates: [
      {
        selector: '[data-action="favorite"]',
        eventType: 'click',
        onEvent: (e) => {
          const srv = e.sender.state.server;
          if (srv) toggleFavorite(srv.id);
        }
      }
    ],
    onMount: (inst) => {
      offI18n = vf.i18n.subscribe(() => inst.refresh());
      offStore = store.subscribe(() => inst.refresh());
      fetchServer(id)
        .then((server) => inst.setState({ status: server ? 'ready' : 'error', server: server }))
        .catch(() => inst.setState({ status: 'error', server: null }));
    },
    onDestroy: () => {
      if (offI18n) offI18n();
      if (offStore) offStore();
    }
  });
}
```

### REPORT.md

```markdown
# Report — 06-spa-scaffold

## Assumptions

- **Locale persistence**: the kit's `vf.i18n.setup({ persist: true })` is mentioned but not documented in detail (no storage key name, no note on whether it also does initial browser-language detection). To meet the requirement exactly ("survives a reload" + "without a saved choice, follow the browser language"), I implemented persistence and detection myself with a dedicated `localStorage` key (`statusboard.locale`) in `app.js`, read before `vf.i18n.setup()` and written on every `data-action="locale"` click, then applied with `vf.i18n.set(locale)`. I did not pass `persist: true` to avoid two competing/unknown persistence mechanisms.
- **`router({ focus: 'main' })`**: llms.txt lists this option without explaining its matching rule. I assumed it focuses the element matching the CSS selector `'main'` after each navigation, so `<main id="view" tabindex="-1">` (an actual `<main>` element) receives focus. `tabindex="-1"` was added so it is programmatically focusable.
- **Locale button labels (`English` / `한국어`)**: these are proper names of the languages themselves and are not meant to be translated with the UI language. To still satisfy "no hard-coded visible text; both locale files have every key", I added `locale.en` / `locale.ko` keys with the *same* value in both `en.json` and `ko.json`, and wired them with `data-i18n` so the text is still key-driven even though it never changes.
- **Server-detail loading heading**: the task table only defines the `<h1>` for the ready state (server name) and the missing state (`server.missing`); it does not define one for the brief loading window between navigation and the (local, static) `fetch` of `data/servers.json` resolving. I used `servers.loading` as a reasonable placeholder heading for that transient state; it disappears once data loads.
- **Favorite toggle**: implemented as shared state in `store.js` (`vf.store`), changed only through `addFavorite` / `removeFavorite` / `toggleFavorite`, matching "shared state changes only through store.js functions". It is not persisted to storage, so it resets on reload but survives navigation and locale switches, as required.
- No `components/*.js` files were created: the header/nav/locale/favorites UI is static markup in `index.html` kept alive with `vf.attach('#app', { delegates })` (no `render`), which the kit explicitly recommends for "published HTML that only needs behaviour". No other reusable widget was needed for this scope.
- `vf.ext.update` / `version.json` intentionally omitted, as instructed by the task (update plugin not available in this evaluation).

## Not followed / could not verify

- Nothing from `AGENTS.md` sections 3–8 was knowingly broken.
- **Not verified**: I could not run the app in a browser in this one-shot evaluation, so behavior (routing, focus movement, i18n switching, favorites, CSP compliance in practice) is based on careful reading of the kit docs, not execution.

## Screen list / state map (folded in per output rules)

| Route | Page file | Purpose | Data | Actions |
|---|---|---|---|---|
| `#/` | `pages/home.js` | Landing/intro | none | none |
| `#/servers` | `pages/servers.js` | List servers | `data/servers.json` via `api.fetchServers` | navigate to detail |
| `#/servers/:id` | `pages/serverDetail.js` | One server + favorite toggle | `api.fetchServer(id)` + `store.favorites` | `data-action="favorite"` |
| anything else | `pages/notFound.js` | 404 | none | none |

Shared state: `favorites: string[]` in `store.js` (`addFavorite`, `removeFavorite`, `toggleFavorite`, `isFavorite`). Local state: each page's own `loading/error/ready` status and fetched data.

## Checklist result

- [x] Every route has a page, including not found
- [x] Data pages show loading, error and empty states (`servers.js`; `serverDetail.js` shows loading/missing)
- [x] Shared state changes only through `store.js` functions
- [x] No hard-coded visible text; both locale files have every key (including `locale.en`/`locale.ko` for the language-name buttons)
- [x] No colors or sizes in JS; CSS uses tokens only (`var(--vf-*)`, no raw hex/px in JS or CSS)
- [ ] No console errors or warnings when you click through every screen — **not verified** (no browser execution available in this evaluation)
```
