### app.js

```js
import vf from './lib/vfunc.esm.js';

// Orders kept exactly as the publisher's markup had them (no server for this sample).
const ORDERS = [
  { id: '1042', customer: 'Minsu Kim', status: 'paid', statusLabel: 'Paid', total: '₩59,000' },
  { id: '1041', customer: 'Seoyeon Park', status: 'shipped', statusLabel: 'Shipped', total: '₩329,000' },
  { id: '1040', customer: 'Jiho Choi', status: 'refunded', statusLabel: 'Refunded', total: '₩29,000' }
];

// Summary texts kept exactly as the publisher's inline script had them.
const SUMMARY_TEXT = {
  week: 'Orders are up this week. Most sales came from the keyboard line.',
  month: 'This month: 1,204 orders, 3.4% refunds.'
};

function matchesOrder(order, search, status) {
  if (status && order.status !== status) return false;
  const needle = search.trim().toLowerCase();
  if (!needle) return true;
  const bare = needle.charAt(0) === '#' ? needle.slice(1) : needle;
  if (order.id.toLowerCase().indexOf(bare) !== -1) return true;
  if (order.customer.toLowerCase().indexOf(needle) !== -1) return true;
  return false;
}

// --- Recent orders table: attach to the published <tbody>, rows are parsed as its content. ---
const ordersTable = vf.attach('#ordersBody', {
  state: { search: '', status: '' },
  render: (s) => {
    const rows = ORDERS.filter((o) => matchesOrder(o, s.search, s.status));
    if (rows.length === 0) {
      return vf.html`<tr><td colspan="4">No orders match.</td></tr>`;
    }
    return rows.map((o) => vf.html`
      <tr>
        <td>#${o.id}</td>
        <td>${o.customer}</td>
        <td><span class="badge badge--${o.status}">${o.statusLabel}</span></td>
        <td class="orders__num">${o.total}</td>
      </tr>
    `);
  }
});

// --- Search box: adopt only (no render), so typed focus and caret stay in the input. ---
vf.attach('#searchInput', {
  events: [
    { eventType: 'input', onEvent: (e) => {
      ordersTable.setState({ search: e.target.value });
    } }
  ]
});

// --- Status filter select: adopt only. ---
vf.attach('#statusFilter', {
  events: [
    { eventType: 'change', onEvent: (e) => {
      ordersTable.setState({ status: e.target.value });
    } }
  ]
});

// --- Summary tabs + panel body: replaced together since both depend on the active tab. ---
vf.attach('#summaryPanel', {
  state: { tab: 'week' },
  render: (s) => vf.html`
    <div class="panel__head">
      <h2 class="panel__title">Summary</h2>
      <div class="tabs">
        <button class="tabs__tab${s.tab === 'week' ? ' tabs__tab--active' : ''}" type="button" data-action="tab" data-tab="week" aria-selected="${s.tab === 'week'}">This week</button>
        <button class="tabs__tab${s.tab === 'month' ? ' tabs__tab--active' : ''}" type="button" data-action="tab" data-tab="month" aria-selected="${s.tab === 'month'}">This month</button>
      </div>
    </div>
    <div class="panel__body" id="summary">
      <p>${SUMMARY_TEXT[s.tab]}</p>
    </div>
  `,
  delegates: [
    { selector: '[data-action="tab"]', eventType: 'click', onEvent: (e) => {
      e.sender.setState({ tab: e.target.dataset.tab });
    } }
  ]
});

// --- User menu: adopt only, direct attribute/text updates (no state-driven render needed). ---
function isMenuOpen(inst) {
  return inst.refs.menuBtn.getAttribute('aria-expanded') === 'true';
}
function openMenu(inst) {
  inst.refs.menuBtn.setAttribute('aria-expanded', 'true');
  inst.refs.menu.hidden = false;
}
function closeMenu(inst) {
  inst.refs.menuBtn.setAttribute('aria-expanded', 'false');
  inst.refs.menu.hidden = true;
}

// Outside-click/Escape listeners are kept in module-level closures, never as
// custom properties on the instance (reserved for the engine).
let onDocClick = null;
let onDocKeydown = null;

vf.attach('#userMenuContainer', {
  delegates: [
    { selector: '[data-action="menu"]', eventType: 'click', onEvent: (e) => {
      e.event.stopPropagation();
      if (isMenuOpen(e.sender)) closeMenu(e.sender); else openMenu(e.sender);
    } },
    { selector: '[data-action="sign-out"]', eventType: 'click', onEvent: (e) => {
      e.event.preventDefault();
      closeMenu(e.sender);
      e.sender.refs.menuBtn.textContent = 'Signed out';
    } }
  ],
  onMount: (inst) => {
    onDocClick = (ev) => {
      if (!inst.$node.contains(ev.target)) closeMenu(inst);
    };
    onDocKeydown = (ev) => {
      if (ev.key === 'Escape') closeMenu(inst);
    };
    document.addEventListener('click', onDocClick);
    document.addEventListener('keydown', onDocKeydown);
  },
  onDestroy: () => {
    document.removeEventListener('click', onDocClick);
    document.removeEventListener('keydown', onDocKeydown);
  }
});
```

# Conversion report — dashboard → vfunc.js

`dashboard.css` is not included above because it stays byte-for-byte unchanged (task requirement 7); nothing in it was touched.

## 1. Area table

| Area (selector) | Static / dynamic | Why | Tool | Data | Events |
|---|---|---|---|---|---|
| `.topbar__logo`, `.sidebar`, `.kpis`, `.footer`, panel titles, table `<thead>` | Static | Content never changes at runtime | none | — | — |
| `#userMenuContainer` (`.topbar__user`) | Dynamic | Menu open/closed state, sign-out text change | `vf.attach` (adopt, no `render`) | — | click on `[data-action="menu"]`, `[data-action="sign-out"]`; document click/Escape for outside-close |
| `#summaryPanel` (`.panel` wrapping tabs + `#summary`) | Dynamic | Active tab changes both the tab buttons' classes/`aria-selected` and the `#summary` text | `vf.attach` with `render` | `SUMMARY_TEXT` map (from the publisher's inline script) | click on `[data-action="tab"]` |
| `#searchInput` (`.topbar__search`) | Dynamic (drives table) | Typed text filters orders | `vf.attach` (adopt, no `render`) — keeps focus/caret | — | `input` |
| `#statusFilter` (`.panel__filter`) | Dynamic (drives table) | Selected status filters orders | `vf.attach` (adopt, no `render`) | — | `change` |
| `#ordersBody` (`<tbody>`) | Dynamic | Rows filtered by search + status, or the "No orders match." row | `vf.attach` with `render`, attached to the `<tbody>` itself so rows parse correctly | `ORDERS` array (from the publisher's static rows) | — (driven by the two filter components calling `ordersTable.setState`) |

- `.topbar__search` input: `+id="searchInput"`, `+data-action="search"`.
- `.topbar__user` div: `+id="userMenuContainer"`.
- `.topbar__user-btn` button: `-onclick="toggleMenu()"`, `+data-action="menu"`, `+data-ref="menuBtn"`, `+aria-expanded="false"`.
- `#userMenu` ul: `-style="display:none"` → `+hidden`, `+data-ref="menu"`.
- "Sign out" link: `-href="javascript:void(0)"` → `href="#"`, `-onclick="alert('Signed out')"`, `+data-action="sign-out"`.

No publisher class was renamed, moved, or used as a JS selector; `badge--*` and `tabs__tab--active` stay purely visual.

## 3. Removed unsafe items (rule 4)

- `onclick="toggleMenu()"` and `onclick="alert('Signed out')"` inline handlers — removed, replaced by `delegates`.
- `href="javascript:void(0)"` — replaced with `href="#"` plus `e.event.preventDefault()` in the click handler.
- `style="display:none"` — replaced with the `hidden` attribute, toggled from JS instead of inline style.
- The publisher's `<script>` block (`toggleMenu`, `showTab`) — removed; its behaviour and exact text content moved into `app.js` (`SUMMARY_TEXT`).

## 4. Escaping review

Every dynamic value (`o.id`, `o.customer`, `o.statusLabel`, `o.total`, `s.tab`, `aria-selected` booleans) is written through `vf.html` tagged templates. No string concatenation into HTML and no `innerHTML` assignment from data anywhere in `app.js`. `vf.unsafeHtml` is not used.

`dashboard.css` keeps its raw values as instructed.
