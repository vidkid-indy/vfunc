### app.js

```javascript
import vf from './lib/vfunc.esm.js';

const CATEGORIES = [
  ['all', 'All'],
  ['keyboard', 'Keyboards'],
  ['mouse', 'Mice'],
  ['monitor', 'Monitors']
];
const SORTS = [
  ['name', 'Name'],
  ['price-asc', 'Price: low to high'],
  ['price-desc', 'Price: high to low']
];
const money = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' });

function computeShown(s) {
  const q = s.query.trim().toLowerCase();
  const order = {
    name: (a, b) => a.name.localeCompare(b.name, 'en'),
    'price-asc': (a, b) => a.price - b.price,
    'price-desc': (a, b) => b.price - a.price
  }[s.sort];
  return s.items
    .filter((p) => s.category === 'all' || p.category === s.category)
    .filter((p) => !s.inStock || p.stock > 0)
    .filter((p) => !q || p.name.toLowerCase().includes(q))
    .sort(order);
}

function render(s) {
  if (s.status === 'loading') {
    return vf.html`<p class="pf__note" role="status" data-ref="status">Loading products…</p>`;
  }
  if (s.status === 'error') {
    return vf.html`<p class="pf__note pf__note--error" role="alert" data-ref="status">Could not load products.</p>`;
  }

  const shown = computeShown(s);
  const count = shown.length;

  return vf.html`
    <section class="pf">
      <div class="pf__controls">
        <label class="pf__field">Search
          <input class="pf__input" type="search" name="query" data-action="search" value="${s.query}">
        </label>
        <label class="pf__field">Category
          <select class="pf__input" name="category" data-action="category">
            ${CATEGORIES.map(([value, label]) => vf.html`<option value="${value}"${s.category === value ? ' selected' : ''}>${label}</option>`)}
          </select>
        </label>
        <label class="pf__check">
          <input type="checkbox" name="inStock" data-action="in-stock"${s.inStock ? ' checked' : ''}>
          In stock only
        </label>
        <label class="pf__field">Sort
          <select class="pf__input" name="sort" data-action="sort">
            ${SORTS.map(([value, label]) => vf.html`<option value="${value}"${s.sort === value ? ' selected' : ''}>${label}</option>`)}
          </select>
        </label>
      </div>
      <p class="pf__count" data-ref="count">${count === 1 ? '1 product' : `${count} products`}</p>
      ${count === 0
        ? vf.html`<p class="pf__empty" data-ref="empty">No products found.</p>`
        : vf.html`<ul class="pf__list" data-ref="list">
            ${shown.map((p) => vf.html`
              <li class="pf__item" data-id="${p.id}" data-state="${p.stock === 0 ? 'out' : 'in'}">
                <span class="pf__name" data-ref="name">${p.name}</span>
                <span class="pf__price" data-ref="price">${money.format(p.price)}</span>
                ${p.stock === 0 ? vf.html`<span class="pf__badge">Sold out</span>` : ''}
              </li>
            `)}
          </ul>`
      }
    </section>
  `;
}

let alive = false;

vf.attach('#app', {
  state: {
    status: 'loading',
    items: [],
    query: '',
    category: 'all',
    inStock: false,
    sort: 'name'
  },
  render,
  delegates: [
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent: (e) => { e.sender.setState({ query: e.target.value }); }
    },
    {
      selector: '[data-action="category"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ category: e.target.value }); }
    },
    {
      selector: '[data-action="in-stock"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ inStock: e.target.checked }); }
    },
    {
      selector: '[data-action="sort"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ sort: e.target.value }); }
    }
  ],
  onMount: (inst) => {
    alive = true;
    const url = new URL('./data/products.json', import.meta.url);
    fetch(url)
      .then((r) => {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then((data) => {
        if (alive) inst.setState({ items: data, status: 'ready' });
      })
      .catch(() => {
        if (alive) inst.setState({ status: 'error' });
      });
  },
  onDestroy: () => {
    alive = false;
  }
});
```

### REPORT.md

```markdown
# Report — 07-react-port

## Assumptions

1. **Single page, no router/store scaffold.** The task asks to rewrite `ProductFilter` as "a page of its own" (`index.html`, `app.js`, a CSS file) and the app has only one screen and one language. I did not add `store.js`, `api.js`, `pages/*.js` or a router from the `AGENTS.md` full-app structure, since there is only one screen and no navigation — that structure would be dead weight for a single component. All logic lives in `app.js`, mounted with `vf.attach('#app', { state, render, delegates, onMount, onDestroy })`.
2. **No i18n setup**, as the task explicitly says to keep the English texts of the source and there is one language — matches AGENTS.md §4.8 ("create components that use `vf.t` after `vf.i18n.setup()` resolves") by simply not using `vf.t` at all.
3. **Focus while typing**: the search input has no `id`/`data-ref` (the hooks table only requires `data-action="search"`), so I added a `name="query"` attribute (plus `name` on the other two selects/checkbox for consistency) so the engine's refresh-time focus/caret restoration matches by `name` rather than relying only on the "same `data-action` at the same position" fallback. This satisfies "keeps the focus while typing" without changing the required hook.
4. **Data URL resolution**: fetched with `new URL('./data/products.json', import.meta.url)` per the kit's "Common mistakes" note, so it still resolves correctly if the app is later served from a version folder.
5. **Fetch cancellation**: used a closure variable (`let alive`) in `app.js` module scope instead of adding a custom property to the instance (the kit forbids adding instance properties); `onDestroy` flips it to `false` so a late response after unmount is ignored. Since this page never unmounts on its own (no router), this is defensive but harmless.
6. **Money formatting**: kept the source's own `Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' })` — this is plain browser API, not a vfunc concern, and reproduces `$49.00` exactly.
7. **Empty state**: like the source, the `<ul data-ref="list">` is omitted entirely (not just hidden) when there are zero results, and `<p data-ref="empty">` is shown instead — matches "absent or hidden when there are products" / the React source's conditional rendering.
8. Design tokens: all colors, spacing, radii and font sizes in `styles/product-filter.css` come from `--vf-*` tokens (`lib/vfunc.tokens.css`); no raw hex/px values remain from the source CSS.

## Kit items I could not follow / not verified

- I could not run the page in a browser from this environment, so **console warnings/errors, cross-browser behavior (Chromium/Firefox/WebKit), and the grader's automated checks are not verified** — I relied on the kit's documented `vf.html`/`vf.attach`/`setState` semantics and mirrored the source's logic 1:1.
- `vf.i18n` / message keys: intentionally not used (see assumption 2) — not a broken rule, just out of scope per the task's own instruction.

## Checklist (prompt-migrate-from-react-vue.md)

- [x] No per-row components: one delegated listener set on the root handles all rows via `data-id`/`data-action` selectors (no listener is created per `<li>`).
- [x] Controlled inputs do not lose focus/caret: search input re-renders on every keystroke (needed for live filtering) but keeps focus via the `name` attribute + same-position fallback described in llms.txt.
- [x] No timers/subscriptions started, so nothing needed beyond the fetch-cancellation guard, which is released in `onDestroy`.
- [x] All markup built with `vf.html`; no `innerHTML` from data, no string concatenation into HTML.
- [ ] No console errors or warnings — **not verified** (no browser available in this environment).
```
