### app.js

```js
// My tasks — the page script. The to-do list lives in #todo.

let nextId = 1;
let focusInputNext = false;

function renderItem(item) {
  const rowId = 'todo-' + item.id;
  return vf.html`<li data-id="${item.id}" data-state="${item.done ? 'done' : 'open'}" class="todo__item">
    <input type="checkbox" data-action="toggle" id="${rowId}" ${item.done ? 'checked' : ''} class="todo__checkbox">
    <label for="${rowId}" data-ref="text" class="todo__text">${item.text}</label>
    <button type="button" data-action="remove" class="todo__remove-btn">Remove</button>
  </li>`;
}

vf.attach('#todo', {
  state: { items: [], filter: 'all' },

  render(state) {
    const filtered = state.items.filter((item) => {
      if (state.filter === 'open') return !item.done;
      if (state.filter === 'done') return item.done;
      return true;
    });
    const leftCount = state.items.filter((item) => !item.done).length;

    return vf.html`
      <form data-action="add" class="todo__form">
        <label for="todo-input" class="todo__label">New task</label>
        <input type="text" id="todo-input" data-ref="input" name="task" class="todo__input" autocomplete="off">
        <button type="submit" class="todo__add-btn">Add</button>
      </form>
      <div class="todo__filters" role="group" aria-label="Filter tasks">
        <button type="button" data-action="filter" data-filter="all" aria-pressed="${state.filter === 'all'}" class="todo__filter-btn">All</button>
        <button type="button" data-action="filter" data-filter="open" aria-pressed="${state.filter === 'open'}" class="todo__filter-btn">Open</button>
        <button type="button" data-action="filter" data-filter="done" aria-pressed="${state.filter === 'done'}" class="todo__filter-btn">Done</button>
      </div>
      <ul data-ref="list" class="todo__list">${filtered.map(renderItem)}</ul>
      <p class="todo__status"><span data-ref="left">${leftCount}</span> left</p>
      <p data-ref="empty" class="todo__empty" ${filtered.length === 0 ? '' : 'hidden'}>Nothing here.</p>
    `;
  },

  delegates: [
    {
      selector: '[data-action="add"]',
      eventType: 'submit',
      onEvent(e) {
        e.event.preventDefault();
        const input = e.sender.refs.input;
        const text = input.value.trim();
        if (!text) return;
        e.sender.setState((state) => ({
          items: state.items.concat({ id: nextId++, text: text, done: false })
        }));
        focusInputNext = true;
      }
    },
    {
      selector: '[data-action="toggle"]',
      eventType: 'change',
      onEvent(e) {
        const row = e.target.closest('[data-id]');
        const id = Number(row.getAttribute('data-id'));
        e.sender.setState((state) => ({
          items: state.items.map((item) => (item.id === id ? { ...item, done: !item.done } : item))
        }));
      }
    },
    {
      selector: '[data-action="remove"]',
      eventType: 'click',
      onEvent(e) {
        const row = e.target.closest('[data-id]');
        const id = Number(row.getAttribute('data-id'));
        e.sender.setState((state) => ({
          items: state.items.filter((item) => item.id !== id)
        }));
      }
    },
    {
      selector: '[data-action="filter"]',
      eventType: 'click',
      onEvent(e) {
        const filter = e.target.getAttribute('data-filter');
        e.sender.setState({ filter: filter });
      }
    }
  ],

  onUpdate(inst) {
    if (focusInputNext) {
      focusInputNext = false;
      if (inst.refs.input) inst.refs.input.focus();
    }
  }
});
```

### REPORT.md

## Assumptions

- `index.html` needed no changes: the CSP meta tag, script tags and `<section id="todo">` already match what the component needs, so it is not repeated in the output (per output rule 1, it stays as given).
- Used `vf.attach('#todo', { render, delegates, state, onUpdate })` rather than `vf.vfunc` + `mount`, since `#todo` already exists in the page markup and `vf.attach` is the documented way to take over published HTML while keeping its attributes.
- "the number of open tasks" for `data-ref="left"` is computed from all tasks (not just the currently filtered ones), since the spec says "open tasks", not "visible tasks".
- "When the current filter shows no task" for `data-ref="empty"` is evaluated against the filtered list (what filter is currently selected), not the total unfiltered list.
- The checkbox is wrapped with a `<label for="todo-<id>">` that also serves as `data-ref="text"`, so the visible task text doubles as the checkbox's accessible label. This isn't required by the spec but is a reasonable default for accessibility and doesn't conflict with any hook name required.
- Task ids are generated with an incrementing in-memory counter (`nextId`) kept in a module-level closure variable (not as a custom instance property, per AGENTS.md rule 4.9).
- After adding a task, focus is moved to the input explicitly in `onUpdate` via a closure flag (`focusInputNext`), because relying only on the engine's "same id/ref" focus restoration would leave focus on the Add button when the button (not Enter) was used to submit.
- No locale files were touched: the task did not ask for i18n and the kit's message-key rule (AGENTS.md §6) is about "user-visible text" in a project that already uses `vf.i18n`; this project has no locale setup at all (no `vf.i18n.setup()` call, no `locales/` folder), so hardcoded English strings (`New task`, `Add`, `All`, `Open`, `Done`, `Remove`, `left`, `Nothing here.`) were kept as-is, matching the existing hardcoded English in `index.html` (`My tasks`, `To-do list`).
- Kept `style.css` filename and its existing 4 base rules unchanged, only appending todo-specific rules, all through `var(--vf-*)` tokens (design separation rule 5, no colors/sizes in JS).
