### app.js

```js
// Servers — the page script (an ES module).
import vf from './lib/vfunc.esm.js';
import { fetchServers } from './api.js';

const STATUS_LABELS = { up: 'Up', down: 'Down', maintenance: 'Maintenance' };

function badgeInfo(status) {
  const label = STATUS_LABELS[status];
  return label ? { state: status, label } : { state: 'unknown', label: 'Unknown' };
}

function safeHref(url) {
  return /^https?:\/\//i.test(String(url || '')) ? url : '#';
}

function filterServers(servers, filter) {
  if (filter === 'all') return servers;
  return servers.filter((s) => s.status === filter);
}

function renderStatus(state) {
  if (state.status === 'loading') {
    return vf.html`Loading servers…`;
  }
  if (state.status === 'error') {
    return vf.html`Could not load servers. <button type="button" class="servers__retry" data-action="retry">Retry</button>`;
  }
  const total = state.servers.length;
  if (total === 0) {
    return vf.html`No servers yet.`;
  }
  const shown = filterServers(state.servers, state.filter).length;
  return vf.html`Showing ${shown} of ${total}`;
}

function renderRow(server) {
  const badge = badgeInfo(server.status);
  const href = safeHref(server.url);
  return vf.html`
    <li class="servers__item" data-id="${server.id}">
      <span class="servers__name" data-ref="name">${server.name}</span>
      <span class="servers__region">${server.region}</span>
      <span class="servers__badge" data-ref="badge" data-state="${badge.state}">${badge.label}</span>
      <a class="servers__link" data-ref="link" href="${href}">Open</a>
    </li>
  `;
}

function render(state) {
  const rows = state.status === 'ready' ? filterServers(state.servers, state.filter).map(renderRow) : [];
  return vf.html`
    <div class="servers__status" data-ref="status" role="status">${renderStatus(state)}</div>
    <label class="servers__filter">
      Status
      <select data-action="status-filter">
        <option value="all" ${state.filter === 'all' ? 'selected' : ''}>All</option>
        <option value="up" ${state.filter === 'up' ? 'selected' : ''}>Up</option>
        <option value="down" ${state.filter === 'down' ? 'selected' : ''}>Down</option>
        <option value="maintenance" ${state.filter === 'maintenance' ? 'selected' : ''}>Maintenance</option>
      </select>
    </label>
    <ul class="servers__list" data-ref="list">${rows}</ul>
  `;
}

function updateAria(inst) {
  inst.$node.setAttribute('aria-busy', inst.state.status === 'loading' ? 'true' : 'false');
}

async function load(inst) {
  inst.setState({ status: 'loading' });
  try {
    const servers = await fetchServers();
    inst.setState({ status: 'ready', servers });
  } catch (err) {
    inst.setState({ status: 'error', servers: [] });
  }
}

vf.attach('#servers', {
  state: { status: 'loading', servers: [], filter: 'all' },
  render,
  delegates: [
    {
      selector: '[data-action="retry"]',
      eventType: 'click',
      onEvent: (e) => { load(e.sender); }
    },
    {
      selector: '[data-action="status-filter"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ filter: e.target.value }); }
    }
  ],
  onMount: (inst) => { updateAria(inst); load(inst); },
  onUpdate: (inst) => { updateAria(inst); }
});
```

### api.js

```js
// Servers — every server call lives here (AGENTS.md §2).
const serversUrl = new URL('./data/servers.json', import.meta.url);

export async function fetchServers() {
  const res = await fetch(serversUrl);
  if (!res.ok) {
    throw new Error('HTTP ' + res.status);
  }
  const data = await res.json();
  return Array.isArray(data) ? data : [];
}
```

### REPORT.md

```markdown
## Assumptions

1. **"No servers yet." vs. "Showing X of Y".** Read "the list is empty" as the *unfiltered* server list being empty (no data at all). If the filter narrows a non-empty list down to zero matching rows, that is shown as `Showing 0 of <total>`, not `No servers yet.`, since the underlying list is not empty.
2. **Non-`http(s)` URLs.** The link's `href` is set to `#` (a non-navigating same-page anchor) when `server.url` is not `http:`/`https:`, so it "does not become a working link" while still rendering a visible `<a data-ref="link">Open</a>` element as the spec requires (it never omits the element).
3. **`vf.html`'s built-in `vf.safeUrl` on `href` is not enough by itself**, because it also allows `mailto:`/`tel:` through as working links; the task only wants `http`/`https` to work, so I added an explicit `/^https?:\/\//i` check before letting the value reach `href`.
4. **File split.** Added `api.js` for the one server call (`fetchServers`), following `AGENTS.md` §2 ("every server call" lives there), even though the rest of the app is a single screen with no router/store/pages split — a full SPA scaffold seemed like more structure than a one-screen task needs.
5. **Retry reuses the same `load()` used for the first fetch**, so clicking it immediately goes back to the `loading` state (`aria-busy="true"`, `Loading servers…`) before the new request settles.
6. **Design separation:** added `servers__*` classes (block__element) purely for styling, kept all behaviour on `data-action`/`data-ref`/`data-id`, and state coloring (`up`/`down`/`maintenance`/`unknown`) is done in CSS off `data-state`, never in JS. All colors/spacing come from `--vf-*` tokens already in `lib/vfunc.tokens.css`; no new tokens were needed.
7. Did not add new locale files / `vf.t` message keys: the task gave fixed English strings ("Loading servers…", "Could not load servers.", "Retry", "No servers yet.", "Showing…", "All/Up/Down/Maintenance", "Open", "Up/Down/Maintenance/Unknown") as literal texts the grader checks for, and the kit files given contain no `locales/` setup to wire into for this task, so I left them as plain strings in `render()` rather than inventing an i18n setup not asked for.
```
