### app.js

```js
// Team board — the page script.
import vf from './lib/vfunc.esm.js';
import { store, setTheme, setLang, toggleTask, removeTask } from './store.js';

// Preferences: theme and language.
const prefs = vf.attach('#prefs', {
  render: () => {
    const p = store.get('prefs');
    return vf.html`
      <p class="prefs__row">Theme: <span data-ref="theme">${p.theme}</span>
        <button class="prefs__button" type="button" data-action="theme">Switch theme</button></p>
      <p class="prefs__row">Language: <span data-ref="lang">${p.lang}</span>
        <button class="prefs__button" type="button" data-action="lang" data-lang="en">English</button>
        <button class="prefs__button" type="button" data-action="lang" data-lang="ko">한국어</button></p>`;
  },
  delegates: [
    { selector: '[data-action="theme"]', eventType: 'click', onEvent: () => setTheme(store.get('prefs').theme === 'dark' ? 'light' : 'dark') },
    { selector: '[data-action="lang"]', eventType: 'click', onEvent: (e) => setLang(e.target.getAttribute('data-lang')) }
  ]
});

// Notice: how many tasks are open.
// Bug 1 fix: render only the inside of #notice (vf.attach keeps the existing
// element as the root); returning a second <section id="notice"> wrapped the
// target inside itself and produced a duplicate id after a refresh.
const notice = vf.attach('#notice', {
  render: () => {
    const open = store.get('tasks').filter((t) => !t.done).length;
    return vf.html`<p class="notice__text">Open tasks: ${open}</p>`;
  }
});

// Details: a disclosure.
// Bug 2 fix: aria-expanded must be the string "false" (not "") when closed.
vf.attach('#details-panel', {
  state: { open: false },
  delegates: [{
    selector: '[data-action="details"]',
    eventType: 'click',
    onEvent: (e) => {
      const open = !e.sender.state.open;
      e.sender.state.open = open;
      e.sender.ids.details.hidden = !open;
      e.target.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
  }],
  onMount: (inst) => inst.$node.querySelector('[data-action="details"]').setAttribute('aria-expanded', 'false')
});

// Tasks, open ones first.
const tasks = vf.attach('#tasks', {
  render: () => vf.html`
    <h2 class="panel__title">Tasks</h2>
    <ul class="tasks">${store.get('tasks').slice().sort((a, b) => a.done - b.done).map((t) => vf.html`
      <li class="task" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
        <input type="checkbox" id="task-${t.id}-toggle" data-action="toggle" aria-label="Done" ${t.done ? 'checked' : ''}>
        <span class="task__title">${t.title}</span>
        <button class="task__delete" type="button" data-action="remove">Remove</button>
      </li>`)}</ul>`,
  delegates: [
    { selector: '[data-action="toggle"]', eventType: 'change', onEvent: (e) => toggleTask(e.target.closest('[data-id]').getAttribute('data-id')) },
    { selector: '[data-action="remove"]', eventType: 'click', onEvent: (e) => removeTask(e.target.closest('[data-id]').getAttribute('data-id')) }
  ]
});
// Bug 5 fix: the delegate selector targeted the old CSS class ".task__remove";
// the designer renamed the class to "task__delete", so the click was never
// matched. Selectors must use the behaviour hook data-action="remove" anyway
// (never a class), which also survives future class renames.
// Bug 7 fix: each checkbox now has a per-row id ("task-<id>-toggle") so the
// engine's focus restore (by id first) keeps focus on the same task's
// checkbox after the list re-sorts, instead of falling back to "same
// data-action at the same position".

// Comments.
const comments = [];
vf.attach('#comments', {
  delegates: [{
    selector: '[data-action="comment"]',
    eventType: 'submit',
    onEvent: (e) => {
      e.event.preventDefault();
      const text = e.sender.refs.comment.value.trim();
      if (!text) return;
      comments.push(text);
      // Bug 6 fix: build the list with vf.html instead of concatenating raw
      // text into an HTML string. vf.html escapes each comment in text
      // position, so "<b>hi</b>" shows as text instead of running as markup.
      e.sender.refs['comment-list'].innerHTML =
        String(vf.html`${comments.map((c) => vf.html`<li class="comments__item">${c}</li>`)}`);
      e.sender.refs.comment.value = '';
    }
  }]
});

// Footer: how often the clock has updated.
const footer = vf.attach('#footer', {
  state: { updates: 0 },
  render: (s) => vf.html`Clock updates: <span data-ref="updates">${s.updates}</span>`
});

// Clock, with a close button.
// Bug 4 fix: the timer was stored as inst._timer (set) but read back as
// inst.timer (a different, always-undefined property) in onDestroy, so
// clearInterval never ran and the interval kept firing after destroy.
// Instance properties starting with "_" are reserved for the engine anyway;
// keep the timer id in a module-level variable instead.
let clockTimer = null;
const clock = vf.vfunc({
  state: { now: new Date() },
  render: (s) => vf.html`
    <p class="clock">Time: <span data-ref="time">${vf.fmt.date(s.now, { timeStyle: 'medium' })}</span>
      <button class="clock__close" type="button" data-action="close-clock">Close clock</button></p>`,
  delegates: [{ selector: '[data-action="close-clock"]', eventType: 'click', onEvent: (e) => e.sender.destroy() }],
  onMount: (inst) => {
    clockTimer = setInterval(() => {
      inst.setState({ now: new Date() });
      footer.setState({ updates: footer.state.updates + 1 });
    }, 250);
  },
  onDestroy: () => {
    clearInterval(clockTimer);
    clockTimer = null;
  }
});
clock.mount('#clock-slot');

store.subscribe(() => {
  prefs.refresh();
  notice.refresh();
  tasks.refresh();
});
```

### store.js

```js
// Shared state and the only functions that change it.
import vf from './lib/vfunc.esm.js';

export const store = vf.store({
  prefs: { theme: 'light', lang: 'en' },
  tasks: [
    { id: 't1', title: 'Write the release notes', done: false },
    { id: 't2', title: 'Review the login page', done: false },
    { id: 't3', title: 'Update the icons', done: true }
  ]
});

export function setTheme(theme) {
  store.set((s) => ({ prefs: { theme: theme, lang: s.prefs.lang } }));
}

// Bug 3 fix: store.set shallow-merges only top-level keys, so
// `{ prefs: { lang } }` replaced the whole prefs object and dropped theme.
// Merge the current theme in, the same way setTheme already merges lang.
export function setLang(lang) {
  store.set((s) => ({ prefs: { theme: s.prefs.theme, lang: lang } }));
}

export function toggleTask(id) {
  store.set((s) => ({ tasks: s.tasks.map((t) => (t.id === id ? { id: t.id, title: t.title, done: !t.done } : t)) }));
}

export function removeTask(id) {
  store.set((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }));
}
```

### REPORT.md

## Summary

Fixed all seven reported bugs in `app.js` and `store.js`, keeping every hook (`id`, `data-action`, `data-ref`, `data-id`) and visible text unchanged. `index.html` and `style.css` needed no changes.

## Bug-by-bug

1. **Duplicate `#notice`** — `vf.attach('#notice', { render })` must return only the *inside* of the element it attaches to; the render function was returning a whole `<section id="notice">…</section>`, which vfunc detected (per the console warning) and which produced a nested duplicate-id element after a refresh. Fixed by rendering only the `<p class="notice__text">…</p>` content.
2. **`aria-expanded=""`** — Both the initial `onMount` and the click handler set `aria-expanded` to `''` when closed instead of the string `"false"`. Both now set `'false'`.
3. **Language switch resets theme** — `setLang` replaced the whole `prefs` object (`store.set({ prefs: { lang } })`), losing `theme` because `store.set` only shallow-merges top-level keys. Fixed by merging the current theme in, mirroring `setTheme`.
4. **Clock keeps counting after close** — `onMount` stored the timer id on `inst._timer`, but `onDestroy` read `inst.timer` (no underscore) — a different, always-`undefined` property — so `clearInterval` was a no-op. Also, per the kit rules, custom names starting with `_` shouldn't be added to an instance at all. Moved the timer id to a module-level `clockTimer` variable used by both `onMount` and `onDestroy`.
5. **Remove button does nothing** — The delegate selector was `.task__remove`, a CSS class the designer had since renamed to `task__delete`; delegated selectors must use `data-action` anyway (never a class), so it was changed to `[data-action="remove"]`, which was already present as a hook on the button and untouched by the rename.
6. **Comment XSS** — The comment list was built by string-concatenating raw comment text into an `innerHTML` string, so `<b>hi</b>` (and any `<script>`) would run as markup. Rebuilt the list with nested `vf.html` calls, which escape each comment in text position, then assigned the result's string form to `innerHTML`.
7. **Focus jumps after Space** — Task checkboxes had no `id`/`name`, so after the list re-sorted (open tasks first) the engine's focus-restore fell back to "same `data-action` at the same position," landing on whatever checkbox ended up in the old slot. Added a stable per-row `id="task-<id>-toggle"` (a new attribute, as the task allows when a fix needs one) so focus restores to the same task by id regardless of its new position.

## Assumptions

- The result of `vf.html\`…\`` has a usable `toString()` (consistent with the kit's statement that "vf.html results … inserted as markup" when nested, and with `vf.vfunc` instances having `toString()` for outerHTML). Wrapped it in `String(...)` explicitly before assigning to `innerHTML` in the comments handler, rather than relying on implicit coercion, to keep the intent obvious.
- "`#notice` … follows the tasks" (expected-behaviour bullet) is read as describing the existing DOM order in `index.html` (notice section is already present once, tasks render separately), not as a request to physically move the `<section id="notice">` below `#tasks` in the markup — the task said to keep hooks and structure as given except where a fix requires a change, and no bug report asked for reordering.
- Kept `tasks`/`prefs`/`notice` refreshed via the existing `store.subscribe` block; no change was needed there since none of the seven bugs originate in it.
