### app.js

```js
// Sales — the page script. Chart.js is loaded as the global `Chart`.

var MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
var SERIES = {
  revenue: { label: 'Revenue', values: [120, 135, 150, 142, 168, 181, 175, 190, 205, 198, 220, 236] },
  orders: { label: 'Orders', values: [30, 32, 35, 33, 40, 44, 41, 46, 50, 48, 53, 57] }
};

function chartColor(seriesKey) {
  var varName = seriesKey === 'revenue' ? '--vf-chart-1' : '--vf-chart-2';
  return getComputedStyle(document.documentElement).getPropertyValue(varName).trim();
}

function shownValues(state) {
  return SERIES[state.series].values.slice(0, state.count);
}

function createSalesComponent() {
  var chart = null;

  return vf.vfunc({
    tag: 'div',
    state: { series: 'revenue', count: 6 },

    render: function (s) {
      var label = SERIES[s.series].label;
      var values = shownValues(s);
      var last = values[values.length - 1];
      return vf.html`
        <div class="sales__controls">
          <button type="button" class="sales__button" data-action="series" data-series="revenue" aria-pressed="${s.series === 'revenue'}">Revenue</button>
          <button type="button" class="sales__button" data-action="series" data-series="orders" aria-pressed="${s.series === 'orders'}">Orders</button>
          <button type="button" class="sales__button" data-action="add-month" ${s.count >= 12 ? 'disabled' : ''}>Add month</button>
        </div>
        <p class="sales__summary" data-ref="summary">${label}: ${values.length} months, last ${last}</p>
        <div class="sales__chart-wrap" data-vf-keep="chart"><canvas data-ref="chart"></canvas></div>
      `;
    },

    delegates: [
      {
        selector: '[data-action="series"]', eventType: 'click',
        onEvent: function (e) {
          var next = e.target.getAttribute('data-series');
          e.sender.setState({ series: next });
        }
      },
      {
        selector: '[data-action="add-month"]', eventType: 'click',
        onEvent: function (e) {
          e.sender.setState(function (st) {
            return { count: Math.min(12, st.count + 1) };
          });
        }
      }
    ],

    onMount: function (i) {
      chart = new Chart(i.refs.chart, {
        type: 'line',
        data: {
          labels: MONTHS.slice(0, i.state.count),
          datasets: [{
            label: SERIES[i.state.series].label,
            data: shownValues(i.state),
            borderColor: chartColor(i.state.series)
          }]
        },
        options: { responsive: true, maintainAspectRatio: true }
      });
    },

    onUpdate: function (i) {
      if (!chart) { return; }
      chart.data.labels = MONTHS.slice(0, i.state.count);
      chart.data.datasets[0].label = SERIES[i.state.series].label;
      chart.data.datasets[0].data = shownValues(i.state);
      chart.data.datasets[0].borderColor = chartColor(i.state.series);
      chart.update();
    },

    onDestroy: function () {
      if (chart) {
        chart.destroy();
        chart = null;
      }
    }
  });
}

var salesTarget = document.getElementById('sales');
var sales = null;

function mountSales() {
  sales = createSalesComponent();
  sales.mount(salesTarget);
}

mountSales();

vf.attach('#page-head', {
  delegates: [
    {
      selector: '[data-action="toggle-chart"]', eventType: 'click',
      onEvent: function (e) {
        if (sales) {
          sales.destroy();
          sales = null;
          e.target.textContent = 'Show chart';
        } else {
          mountSales();
          e.target.textContent = 'Hide chart';
        }
      }
    }
  ]
});
```

### REPORT.md

## Assumptions

- `AGENTS.md` §9 asks to list files and wait for approval "for more than a small fix", and the top conditions say to decide myself and record the assumption instead of waiting — I proceeded directly to implementation.
- Added `id="page-head"` to the `<header>` in `index.html`. The toggle button lives outside `<section id="sales">`, so it needs its own attach root; per the rules, event/ref selectors must use `data-action`/`data-ref`/`id`, not classes, so I added an `id` rather than binding on `.page__head`. No other markup, text or existing attributes/classes were changed.
- The sales component is created fresh (`createSalesComponent()` + `mount()`) each time it is shown, so "clicking again creates a new component in its initial state (revenue, 6 months)" is satisfied automatically — no extra reset code needed.
- The Chart.js instance and its canvas live inside `data-vf-keep="chart"`, and only `chart.data`/`chart.update()` are touched in `onUpdate`, so the same canvas element and the same `Chart` instance persist for the component's whole life, per the spec and the kit's "third-party widgets" rule (AGENTS.md §4.6).
- Chart line color is read from the CSS custom property via `getComputedStyle(document.documentElement)`, matching AGENTS.md §5.3's explicit exception for libraries that need a color value in JS. No other colors/spacing were hardcoded in JS.
- No message-key/i18n system was requested or referenced by the task, and the kit's i18n section is optional infrastructure not present in this minimal project, so plain English literals (`'Revenue'`, `'Orders'`, `'Add month'`) were kept as given text rather than introducing a locales setup that the task didn't ask for and that has no place to live in this file set.
- `options: { responsive: true, maintainAspectRatio: true }` was added to Chart.js config as a reasonable default; not specified by the task, no tokens/colors involved.
