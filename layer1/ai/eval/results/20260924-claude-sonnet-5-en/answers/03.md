### app.js

```js
import vf from './lib/vfunc.esm.js';

// Static sample data (no server in this task): keep the three published orders as they are.
const ORDERS = [
  { id: '1042', customer: 'Minsu Kim', status: 'paid', statusLabel: 'Paid', total: '₩59,000' },
  { id: '1041', customer: 'Seoyeon Park', status: 'shipped', statusLabel: 'Shipped', total: '₩329,000' },
  { id: '1040', customer: 'Jiho Choi', status: 'refunded', statusLabel: 'Refunded', total: '₩29,000' }
];

// Texts from the publisher's original showTab() script.
const SUMMARY_TEXT = {
  week: 'Orders are up this week. Most sales came from the keyboard line.',
  month: 'This month: 1,204 orders, 3.4% refunds.'
};

// Combined search + status filter state (module-scope: there is no router/navigation on this page).
const filters = { query: '', status: '' };

function orderMatches(order) {
  if (filters.status && order.status !== filters.status) return false;
  const q = filters.query.trim().toLowerCase();
  if (!q) return true;
  const num = q.charAt(0) === '#' ? q.slice(1) : q;
  return order.id.toLowerCase().indexOf(num) !== -1 || order.customer.toLowerCase().indexOf(q) !== -1;
}

// Recent orders table body: filtered rows, or the "no match" row.
const ordersBody = vf.attach('#ordersBody', {
  render: () => {
    const rows = ORDERS.filter(orderMatches);
    if (!rows.length) {
      return vf.html`<tr><td colspan="4">No orders match.</td></tr>`;
    }
    return vf.html`${rows.map((o) => vf.html`<tr id="order-${o.id}"><td>#${o.id}</td><td>${o.customer}</td><td><span class="badge badge--${o.status}">${o.statusLabel}</span></td><td class="orders__num">${o.total}</td></tr>`)}`;
  }
});

// Summary panel: text depends only on the active tab.
const summary = vf.attach('#summary', {
  state: { tab: 'week' },
  render: (s) => vf.html`<p>${SUMMARY_TEXT[s.tab]}</p>`
});

function closeMenu() {
  const menu = document.getElementById('userMenu');
  const btn = document.getElementById('userBtn');
  if (menu) menu.hidden = true;
  if (btn) btn.setAttribute('aria-expanded', 'false');
}

function openMenu() {
  const menu = document.getElementById('userMenu');
  const btn = document.getElementById('userBtn');
  if (menu) menu.hidden = false;
  if (btn) btn.setAttribute('aria-expanded', 'true');
}

function onDocClick(e) {
  const menu = document.getElementById('userMenu');
  const btn = document.getElementById('userBtn');
  if (!menu || menu.hidden) return;
  if (menu.contains(e.target) || (btn && btn.contains(e.target))) return;
  closeMenu();
}

function onDocKeydown(e) {
  if (e.key === 'Escape') closeMenu();
}

// Whole-page behaviour: adopt <body> (no render) so all published markup and input values stay untouched.
vf.attach('body', {
  delegates: [
    {
      selector: '[data-action="menu"]',
      eventType: 'click',
      onEvent: (e) => {
        const expanded = e.target.getAttribute('aria-expanded') === 'true';
        if (expanded) { closeMenu(); } else { openMenu(); }
      }
    },
    {
      selector: '[data-action="sign-out"]',
      eventType: 'click',
      onEvent: (e) => {
        e.event.preventDefault();
        closeMenu();
        const btn = document.getElementById('userBtn');
        if (btn) btn.textContent = 'Signed out';
      }
    },
    {
      selector: '[data-action="tab"]',
      eventType: 'click',
      onEvent: (e) => {
        const tab = e.target.getAttribute('data-tab');
        vf.$$('[data-action="tab"]').forEach((el) => {
          const active = el.getAttribute('data-tab') === tab;
          el.classList.toggle('tabs__tab--active', active);
          el.setAttribute('aria-selected', active ? 'true' : 'false');
        });
        summary.tab = tab;
      }
    },
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent: (e) => {
        filters.query = e.target.value;
        ordersBody.refresh();
      }
    },
    {
      selector: '[data-action="filter"]',
      eventType: 'change',
      onEvent: (e) => {
        filters.status = e.target.value;
        ordersBody.refresh();
      }
    }
  ],
  onMount: () => {
    document.addEventListener('click', onDocClick);
    document.addEventListener('keydown', onDocKeydown);
  },
  onDestroy: () => {
    document.removeEventListener('click', onDocClick);
    document.removeEventListener('keydown', onDocKeydown);
  }
});
```

```markdown
# Conversion report — dashboard.html

## Area table

| Area (selector) | Static / dynamic | Why | Tool | Data | Events |
|---|---|---|---|---|---|
| `.topbar__logo`, `.sidebar`, `.kpis`, table `<thead>`, `.footer` | Static | Content never changes | none | — | — |
| `.topbar__user` (button + `#userMenu`) | Dynamic (state only) | Open/closed + signed-out text | `vf.attach('body', …)` delegates, direct DOM attribute/hidden updates | — | click on `[data-action="menu"]`, `[data-action="sign-out"]`; document click/Escape to close |
| `.tabs` + `#summary` | Dynamic (markup depends on tab) | Text swaps per tab | `vf.attach('#summary', { render })` + delegate on `[data-action="tab"]` | `SUMMARY_TEXT` map (from the publisher's script) | click |
| `.topbar__search` | Dynamic (drives filtering) | Filters table as you type | `vf.attach('body', …)` delegate, no render on the input itself | — | input |
| `.panel__filter` | Dynamic (drives filtering) | Status filter | same body delegate | — | change |
| `<tbody id="ordersBody">` | Dynamic (markup depends on filters) | Rows/“No orders match.” | `vf.attach('#ordersBody', { render })` | `ORDERS` (kept exactly as published) | — |

## HTML changes (added attributes / replacements only)

- `<head>`: added CSP `<meta http-equiv="Content-Security-Policy" …>` (was missing).
- `<input class="topbar__search" …>`: added `data-action="search"`.
- `<button class="topbar__user-btn" …>`: removed `onclick="toggleMenu()"`; added `id="userBtn"`, `data-action="menu"`, `aria-expanded="false"`.
- `<ul class="topbar__menu" id="userMenu" …>`: removed `style="display:none"`; added `hidden`.
- `<a href="javascript:void(0)" onclick="alert('Signed out')">Sign out</a>`: removed `onclick` and the `javascript:` URL; replaced `href="javascript:void(0)"` with `href="#"` (default prevented in JS); added `data-action="sign-out"`.
- Both `.tabs__tab` buttons: removed `onclick="showTab(...)"`; added `data-action="tab"`, `data-tab="week"|"month"`, `aria-selected="true"|"false"`. The active button keeps `tabs__tab--active` (rule 7, for looks; `aria-selected` is the real selector target).
- `<select class="panel__filter">`: added `data-action="filter"`.
- `<tbody>`: added `id="ordersBody"`.
- Bottom `<script>…</script>` (inline `toggleMenu`/`showTab`) removed entirely; replaced with `<script type="module" src="./app.js"></script>`.

## Removed unsafe items (rule 4)

- Inline `<script>` block with `toggleMenu()` / `showTab()` — moved into `app.js` as delegated handlers, since inline scripts are blocked by the CSP.
- `onclick="toggleMenu()"` and `onclick="alert('Signed out')"` inline handlers — replaced by `delegates` in `app.js`.
- `href="javascript:void(0)"` — replaced with `href="#"` + `e.event.preventDefault()`.
- `style="display:none"` on `#userMenu` — inline style is blocked by the CSP; replaced with the `hidden` attribute (per the kit's "common mistakes" note).

## Escaping review

- Order rows and the “No orders match.” row are all built with `vf.html` in `app.js` (tagged template); no string concatenation into HTML, no `innerHTML` from data.
- Summary text is built with `vf.html` from a fixed lookup map (`SUMMARY_TEXT`), not user input.
- `vf.unsafeHtml` is not used anywhere.
- The one non-`vf.html` write is `btn.textContent = 'Signed out'`, a fixed literal string, not data — `textContent` is inherently safe.

## Suggested token replacements (optional, CSS not changed)

`dashboard.css` is kept byte-for-byte as required, so this is a suggestion list only:
- `#1e3a8a` (topbar bg) → `var(--vf-color-primary)` or a dedicated brand token; `#ffffff` on the topbar → `var(--vf-color-on-primary)`.
- `#1e293b` (body text), `#64748b` (muted text) → `var(--vf-color-text)`, `var(--vf-color-text-muted)`.
- `#f4f6fb` (page bg), `#ffffff` (cards) → `var(--vf-color-bg)`, `var(--vf-color-surface)`.
- `#e2e8f0` / `#cbd5e1` borders → `var(--vf-color-border)`.
- `#1d4ed8` (active tab/link), `#eff6ff` (active sidebar bg) → `var(--vf-color-primary)`, `var(--vf-color-primary-soft)`.
- `#047857`/`#ecfdf5`, `#b91c1c`/`#fef2f2` (badges/deltas) → `var(--vf-color-success)`/`-soft`, `var(--vf-color-danger)`/`-soft`.
- `border-radius: 6px/8px/10px` → `var(--vf-radius-sm|md|lg)`; `box-shadow` values → `var(--vf-shadow-1)`.

## Assumptions

1. No server/API in this task, so `api.js` was not created; `ORDERS` and `SUMMARY_TEXT` live as constants in `app.js`, matching the instruction to keep the three published orders and KPI values as they are.
2. "Sign out" only changes the button text to `Signed out` and closes the menu — no real request, no route change, no re-enabling of the button (the task says no request/alert, and gives no further behaviour for it).
3. Search matches are substring matches (not exact), case-insensitive, on the order id (with or without a leading `#`) and the customer name — this satisfies "filters … by order number (`1041` or `#1041`) or customer name, ignoring case" without over-specifying exact-match semantics the task didn't ask for.
4. The "no match" row is a single `<td colspan="4">No orders match.</td>` inside one `<tr>`, since the task only specifies the row's text.
5. `#userMenu`'s `Profile`/`Settings` links were left as inert `href="#"` placeholders (no behaviour was requested for them in `BEHAVIOUR`).
6. `vf.attach('body', …)` is used for page-wide delegation (menu, sign-out, tabs, search, filter) since none of that requires re-rendering the `<body>`'s own markup — this keeps the smallest-tool rule and guarantees the search input never loses focus (its container is never replaced, and `#ordersBody`/`#summary` are attached and refreshed independently).
7. Outside-click and Escape-to-close are implemented as native `document` listeners registered in `onMount` and removed in `onDestroy` of the body-level `vf.attach`, since vfunc's `delegates` are scoped to inside the attached root and don't cover "click outside".

- [x] No selector uses a publisher class — all delegates/ids use `data-action="…"` or `id`; `tabs__tab--active` is toggled for looks only, never selected on.
- [x] Every dynamic value is inside `vf.html` — order rows, no-match row, summary text.
- [x] No inline scripts, inline handlers or `javascript:` URLs remain; CSP meta tag present without `'unsafe-inline'` in `script-src`.
- [x] Forms/inputs keep their typed values — the search input's container (`<body>`) is adopted without `render`, so the input node itself is never touched by vfunc, only read via the delegate's `e.target.value`.
- [x] Timers, listeners and third-party widgets are released in `onDestroy` — the two `document` listeners are added in `onMount` and removed in `onDestroy` of the body attach. No timers or third-party widgets are used.
```
