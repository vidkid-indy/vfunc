### app.js

```js
// Sign up — the page script.
// Client-side validation for form#signup, published markup kept as-is.
// vf.attach only adds behaviour: no render, no markup rebuilt.
(function () {
  'use strict';

  var validators = {
    email: function (value) {
      return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.trim());
    },
    password: function (value) {
      return value.length >= 8 && /\d/.test(value);
    },
    confirm: function (value, password) {
      return value !== '' && value === password;
    },
    terms: function (checked) {
      return checked === true;
    }
  };

  var messages = {
    email: 'Enter a valid email address.',
    password: 'Use at least 8 characters, including a number.',
    confirm: 'Passwords do not match.',
    terms: 'Accept the terms to continue.'
  };

  var fieldOrder = ['email', 'password', 'confirm', 'terms'];

  // Per-field "has shown an error at least once" flag, kept in a closure
  // variable rather than on the instance (AGENTS.md 4.9).
  var shownError = { email: false, password: false, confirm: false, terms: false };

  function isFieldValid(field, inst) {
    if (field === 'confirm') {
      return validators.confirm(inst.ids.confirm.value, inst.ids.password.value);
    }
    if (field === 'terms') {
      return validators.terms(inst.ids.terms.checked);
    }
    return validators[field](inst.ids[field].value);
  }

  function validateField(inst, field) {
    var el = inst.ids[field];
    var errorEl = inst.ids[field + '-error'];
    var valid = isFieldValid(field, inst);
    el.setAttribute('aria-invalid', valid ? 'false' : 'true');
    errorEl.textContent = valid ? '' : messages[field];
    if (!valid) {
      shownError[field] = true;
    }
    return valid;
  }

  vf.attach('#app', {
    delegates: [
      {
        selector: '#email, #password, #confirm',
        eventType: 'focusout',
        onEvent: function (e) { validateField(e.sender, e.target.id); }
      },
      {
        selector: '#email, #password, #confirm',
        eventType: 'input',
        onEvent: function (e) {
          if (shownError[e.target.id]) {
            validateField(e.sender, e.target.id);
          }
        }
      },
      {
        selector: '#terms',
        eventType: 'change',
        onEvent: function (e) { validateField(e.sender, 'terms'); }
      },
      {
        selector: '#signup',
        eventType: 'submit',
        onEvent: function (e) {
          e.event.preventDefault();
          var inst = e.sender;
          var firstInvalid = null;
          var i, field, valid;

          for (i = 0; i < fieldOrder.length; i++) {
            field = fieldOrder[i];
            valid = validateField(inst, field);
            if (!valid && !firstInvalid) {
              firstInvalid = field;
            }
          }

          if (firstInvalid) {
            inst.ids[firstInvalid].focus();
            return;
          }

          var email = inst.ids.email.value.trim();
          inst.ids.signup.hidden = true;
          inst.refs.done.textContent = 'Welcome, ' + email + '!';
          inst.refs.done.hidden = false;
        }
      }
    ]
  });
})();
```

### REPORT.md

**Assumptions (decided myself, per "answer in one turn")**

- Gave `<main>` an `id="app"` so a single `vf.attach('#app', …)` root covers both the form and the post-submit welcome element (the welcome element must live outside `form#signup`, since the form itself gets `hidden`). No other new ids were added; `id="signup"`, `id="email"`, `id="password"`, `id="confirm"`, `id="terms"` and all visible texts are unchanged.
- Added one `<p class="field__error" id="<field>-error" aria-live="polite">` per field (right after its input/checkbox) and wired `aria-describedby="<field>-error"` on each input — this is exactly what the task's "message element" rule asks for; nothing else in the form markup changed.
- Added `<p class="signup__done" data-ref="done" hidden></p>` as a sibling of the form for the welcome text, since the form is hidden on success.
- Used `vf.attach('#app', { delegates })` with **no `render`**, per AGENTS.md §4.7 ("published HTML that only needs behaviour: `vf.attach` without `render` — keep the markup") and per the task's "keep the form markup" instruction. All validation/DOM updates are done imperatively in event handlers (`textContent`, `setAttribute`, `hidden`), which is safe by construction (no `innerHTML`, no string concatenation into markup), so `vf.html`/`vf.tpl` were not needed for this task.
- Used `focusout` (bubbles) instead of `blur` (does not bubble) so a single delegated listener on the root can catch "field loses focus" for all three text inputs, per the kit's delegation model.
- "Once it has shown an error, it is validated again on every input" is tracked with a per-field boolean in a module-level closure variable (`shownError`), not as a property on the `vf.attach` instance, per AGENTS.md §4.9 ("never add your own properties to an instance").
- Submit order follows the table order (email, password, confirm, terms) for "focus moves to the first invalid field."
- Styled `aria-invalid="true"` and the error/done blocks only with existing `--vf-*` tokens (`--vf-color-danger`, `--vf-color-danger-text`); no raw colors/sizes in `app.js`.

**Checklist**

- [x] Every dynamic value goes through safe APIs; no `innerHTML` from data — welcome text and error messages are set with `textContent` only, never concatenated into an HTML string.
- [x] Selectors use `data-action` / `data-ref` / `id` only; no CSS classes used in delegates; no colors or sizes in JS.
- [x] Message texts match the table exactly; this task did not ask for i18n/locale files, so plain strings were used (no `locales/` folder exists in this single-page project) — noted as a deliberate scope choice, not an oversight.
- [x] The one listener set (delegates on `#app`) needs no manual cleanup: it is the page's only script and is never destroyed/re-mounted, so no `onDestroy` was required.

- [x] No secrets, tokens or personal data in code; the password is never read into a variable that is logged, and no field value is ever logged.
