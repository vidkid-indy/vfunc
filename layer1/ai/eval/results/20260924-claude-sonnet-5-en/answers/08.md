### app.js

```js
// Sales — the page script. Chart.js is loaded as the global `Chart`.

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const SERIES = {
  revenue: { label: 'Revenue', values: [120, 135, 150, 142, 168, 181, 175, 190, 205, 198, 220, 236] },
  orders: { label: 'Orders', values: [30, 32, 35, 33, 40, 44, 41, 46, 50, 48, 53, 57] }
};

function seriesColor(key) {
  const name = key === 'revenue' ? '--vf-chart-1' : '--vf-chart-2';
  return getComputedStyle(document.documentElement).getPropertyValue(name).trim();
}

function shownValues(state) {
  return SERIES[state.series].values.slice(0, state.monthsShown);
}

function createSales() {
  let chart = null;

  return vf.attach('#sales', {
    state: { series: 'revenue', monthsShown: 6 },
    render: (s) => {
      const series = SERIES[s.series];
      const values = shownValues(s);
      const last = values[values.length - 1];
      const atMax = s.monthsShown >= MONTHS.length;
      return vf.html`
        <div class="sales__controls">
          <button type="button" class="sales__button" data-action="series" data-series="revenue" aria-pressed="${s.series === 'revenue'}">Revenue</button>
          <button type="button" class="sales__button" data-action="series" data-series="orders" aria-pressed="${s.series === 'orders'}">Orders</button>
          <button type="button" class="sales__button" data-action="add-month" ${atMax ? 'disabled' : ''}>Add month</button>
        </div>
        <p class="sales__summary" data-ref="summary">${series.label}: ${values.length} months, last ${last}</p>
        <div class="sales__chart-wrap" data-vf-keep="chart">
          <canvas data-ref="chart"></canvas>
        </div>
      `;
    },
    delegates: [
      {
        selector: '[data-action="series"]', eventType: 'click',
        onEvent: (e) => {
          const key = e.target.getAttribute('data-series');
          if (key && key !== e.sender.state.series) {
            e.sender.setState({ series: key });
          }
        }
      },
      {
        selector: '[data-action="add-month"]', eventType: 'click',
        onEvent: (e) => {
          const s = e.sender.state;
          if (s.monthsShown < MONTHS.length) {
            e.sender.setState({ monthsShown: s.monthsShown + 1 });
          }
        }
      }
    ],
    onMount: (inst) => {
      const s = inst.state;
      const ctx = inst.refs.chart.getContext('2d');
      chart = new Chart(ctx, {
        type: 'line',
        data: {
          labels: MONTHS.slice(0, s.monthsShown),
          datasets: [{
            label: SERIES[s.series].label,
            data: shownValues(s),
            borderColor: seriesColor(s.series)
          }]
        },
        options: { responsive: true }
      });
    },
    onUpdate: (inst) => {
      if (!chart) return;
      const s = inst.state;
      chart.data.labels = MONTHS.slice(0, s.monthsShown);
      chart.data.datasets[0].data = shownValues(s);
      chart.data.datasets[0].label = SERIES[s.series].label;
      chart.data.datasets[0].borderColor = seriesColor(s.series);
      chart.update();
    },
    onDestroy: () => {
      if (chart) {
        chart.destroy();
        chart = null;
      }
    }
  });
}

let salesInst = createSales();

vf.attach(document.body, {
  delegates: [
    {
      selector: '[data-action="toggle-chart"]', eventType: 'click',
      onEvent: (e) => {
        const sales = document.getElementById('sales');
        if (salesInst) {
          salesInst.destroy();
          salesInst = null;
          sales.replaceChildren();
          e.target.textContent = 'Show chart';
        } else {
          salesInst = createSales();
          e.target.textContent = 'Hide chart';
        }
      }
    }
  ]
});
```
