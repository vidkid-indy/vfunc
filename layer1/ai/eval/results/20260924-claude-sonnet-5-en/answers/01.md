### app.js

```javascript
import vf from './lib/vfunc.esm.js';

function greetingText(name) {
  const trimmed = name.trim();
  return trimmed ? `Hello, ${trimmed}!` : 'Hello, stranger!';
}

const app = vf.vfunc({
  state: { count: 0, name: '' },

  render: (s) => vf.html`
    <section class="widget widget__greeting">
      <label for="name">Your name</label>
      <input type="text" id="name" value="${s.name}" autocomplete="off">
      <p data-ref="greeting">${greetingText(s.name)}</p>
    </section>
    <section class="widget widget__counter">
      <p data-ref="count" data-state="${s.count === 0 ? 'zero' : 'positive'}">${s.count}</p>
      <button type="button" data-action="inc">+1</button>
      <button type="button" data-action="dec" ${s.count === 0 ? 'disabled' : ''}>&minus;1</button>
      <button type="button" data-action="reset">Reset</button>
    </section>
  `,

  events: [
    {
      id: 'name',
      eventType: 'input',
      onEvent: (e) => {
        const inst = e.sender;
        // Store the value directly (no setState/refresh) so typing never
        // re-renders the input: focus, caret and characters stay intact.
        inst.state.name = e.target.value;
        // Update the greeting node's text directly; textContent never
        // parses markup, so a name like "<b>Kim</b>" shows as plain text.
        inst.refs.greeting.textContent = greetingText(inst.state.name);
      }
    }
  ],

  delegates: [
    {
      selector: '[data-action="inc"]',
      eventType: 'click',
      onEvent: (e) => { e.sender.count = e.sender.state.count + 1; }
    },
    {
      selector: '[data-action="dec"]',
      eventType: 'click',
      onEvent: (e) => {
        const inst = e.sender;
        if (inst.state.count > 0) inst.count = inst.state.count - 1;
      }
    },
    {
      selector: '[data-action="reset"]',
      eventType: 'click',
      onEvent: (e) => { e.sender.count = 0; }
    }
  ]
});

app.mount('#app');
```

- Used the ES module build (`lib/vfunc.esm.js`, `import vf from './lib/vfunc.esm.js'`) with a single `app.js`, since the task allows any approach and the kit documents both the script global and ESM paths.
