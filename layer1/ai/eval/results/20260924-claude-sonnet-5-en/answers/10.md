### app.js

```js
// Team board — the page script.
import vf from './lib/vfunc.esm.js';
import { store, setTheme, setLang, toggleTask, removeTask } from './store.js';

// Preferences: theme and language.
const prefs = vf.attach('#prefs', {
  render: () => {
    const p = store.get('prefs');
    return vf.html`
      <p class="prefs__row">Theme: <span data-ref="theme">${p.theme}</span>
        <button class="prefs__button" type="button" data-action="theme">Switch theme</button></p>
      <p class="prefs__row">Language: <span data-ref="lang">${p.lang}</span>
        <button class="prefs__button" type="button" data-action="lang" data-lang="en">English</button>
        <button class="prefs__button" type="button" data-action="lang" data-lang="ko">한국어</button></p>`;
  },
  delegates: [
    { selector: '[data-action="theme"]', eventType: 'click', onEvent: () => setTheme(store.get('prefs').theme === 'dark' ? 'light' : 'dark') },
    { selector: '[data-action="lang"]', eventType: 'click', onEvent: (e) => setLang(e.target.getAttribute('data-lang')) }
  ]
});

// Notice: how many tasks are open.
// Fix (bug 1): render only the inside of #notice (it is already the attach
// target and already carries id="notice"). Returning a whole <section id="notice">
// made attach nest a duplicate element with the same id.
const notice = vf.attach('#notice', {
  render: () => {
    const open = store.get('tasks').filter((t) => !t.done).length;
    return vf.html`<p class="notice__text">Open tasks: ${open}</p>`;
  }
});

// Details: a disclosure.
vf.attach('#details-panel', {
  state: { open: false },
  delegates: [{
    selector: '[data-action="details"]',
    eventType: 'click',
    onEvent: (e) => {
      const open = !e.sender.state.open;
      e.sender.state.open = open;
      e.sender.ids.details.hidden = !open;
      // Fix (bug 2): aria-expanded must always be "true"/"false", never "".
      e.target.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
  }],
  onMount: (inst) => inst.$node.querySelector('[data-action="details"]').setAttribute('aria-expanded', 'false')
});

// Tasks.
const tasks = vf.attach('#tasks', {
  render: () => vf.html`
    <h2 class="panel__title">Tasks</h2>
    <ul class="tasks">${store.get('tasks').map((t) => vf.html`
      <li class="task" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
        <input type="checkbox" id="toggle-${t.id}" data-action="toggle" aria-label="Done" ${t.done ? 'checked' : ''}>
        <span class="task__title">${t.title}</span>
        <button class="task__delete" type="button" data-action="remove">Remove</button>
      </li>`)}</ul>`,
  delegates: [
    // Fix (bug 7): give each checkbox an id so vfunc's focus restore on
    // refresh can find the same element again after Space toggles it.
    { selector: '[data-action="toggle"]', eventType: 'change', onEvent: (e) => toggleTask(e.target.closest('[data-id]').getAttribute('data-id')) },
    // Fix (bug 5): select by data-action, not by a CSS class (the designer
    // renamed task__remove to task__delete, which broke the old selector).
    { selector: '[data-action="remove"]', eventType: 'click', onEvent: (e) => removeTask(e.target.closest('[data-id]').getAttribute('data-id')) }
  ]
});

// Comments.
// Fix (bug 6): build the list with vf.html (state-driven render) instead of
// concatenating raw text into innerHTML, so a comment like "<b>hi</b>" is
// shown as text, not parsed as markup.
vf.attach('#comments', {
  state: { comments: [] },
  render: (s) => vf.html`
    <form class="comments__form" data-action="comment">
      <label class="comments__label" for="comment-text">Comment</label>
      <input class="comments__input" id="comment-text" data-ref="comment" autocomplete="off">
      <button class="comments__button" type="submit">Post</button>
    </form>
    <ul class="comments__list" data-ref="comment-list">${s.comments.map((c) => vf.html`<li class="comments__item">${c}</li>`)}</ul>`,
  delegates: [{
    selector: '[data-action="comment"]',
    eventType: 'submit',
    onEvent: (e) => {
      e.event.preventDefault();
      const text = e.sender.refs.comment.value.trim();
      if (!text) return;
      e.sender.setState({ comments: e.sender.state.comments.concat(text) });
    }
  }]
});

// Footer: how often the clock has updated.
const footer = vf.attach('#footer', {
  state: { updates: 0 },
  render: (s) => vf.html`Clock updates: <span data-ref="updates">${s.updates}</span>`
});

// Clock, with a close button.
const clock = vf.vfunc({
  state: { now: new Date() },
  render: (s) => vf.html`
    <p class="clock">Time: <span data-ref="time">${vf.fmt.date(s.now, { timeStyle: 'medium' })}</span>
      <button class="clock__close" type="button" data-action="close-clock">Close clock</button></p>`,
  delegates: [{ selector: '[data-action="close-clock"]', eventType: 'click', onEvent: (e) => e.sender.destroy() }],
  onMount: (inst) => {
    inst._timer = setInterval(() => {
      inst.setState({ now: new Date() });
      footer.setState({ updates: footer.state.updates + 1 });
    }, 250);
  },
  // Fix (bug 4): the timer was stored on inst._timer but cleared from
  // inst.timer (undefined), so destroy() never stopped the interval.
  onDestroy: (inst) => clearInterval(inst._timer)
});
clock.mount('#clock-slot');

store.subscribe(() => {
  prefs.refresh();
  notice.refresh();
  tasks.refresh();
});
```

### store.js

```js
// Shared state and the only functions that change it.
import vf from './lib/vfunc.esm.js';

export const store = vf.store({
  prefs: { theme: 'light', lang: 'en' },
  tasks: [
    { id: 't1', title: 'Write the release notes', done: false },
    { id: 't2', title: 'Review the login page', done: false },
    { id: 't3', title: 'Update the icons', done: true }
  ]
});

export function setTheme(theme) {
  store.set((s) => ({ prefs: { theme: theme, lang: s.prefs.lang } }));
}

// Fix (bug 3): setLang replaced the whole prefs object with only { lang },
// dropping theme back to its default. Keep the current theme, like setTheme does.
export function setLang(lang) {
  store.set((s) => ({ prefs: { theme: s.prefs.theme, lang: lang } }));
}

export function toggleTask(id) {
  store.set((s) => ({ tasks: s.tasks.map((t) => (t.id === id ? { id: t.id, title: t.title, done: !t.done } : t)) }));
}

export function removeTask(id) {
  store.set((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }));
}
```

### REPORT.md

```markdown
# Report — Fix seven bugs

## Assumptions
- "The notice box ... follows the tasks" (expected behaviour, bug 1) is read as a DOM-order requirement, not just a description of the content. I moved `<section id="notice">` in `index.html` to after `<section id="tasks">` (it previously sat between `#prefs` and `#details-panel`). No hooks, ids or visible text changed — only the position of the section.
- Kept `store.set` semantics as documented (shallow merge at the top level of the patch); the fix for bug 3 follows the same pattern already used correctly in `setTheme`.
- For bug 6 I switched the comment list from manual `innerHTML` string concatenation to a small `state`-driven `vf.html` render (fewest-lines fix while staying inside the kit's security rule "build dynamic markup only with vf.html/vf.tpl; never set innerHTML from data"). The comment input has no `value` attribute in the template, so it is naturally empty after each re-render (same end result as the original manual `.value = ''`).
- For bug 7 I added a per-row `id="toggle-<task id>"` to the checkbox (the kit explicitly recommends this: "give per-row buttons and checkboxes an id so focus survives a refresh"). No existing hook, class or text was removed.

## Per-bug fix
1. `notice`'s `render` returned a whole `<section id="notice">…</section>`; since `#notice` is itself the attach target, this nested a duplicate `id="notice"` element inside it (exactly the console warning quoted in the report). Fixed by returning only the inside (`<p class="notice__text">…</p>`), and moved the section to follow `#tasks` in `index.html`.
2. The click handler and `onMount` set `aria-expanded` to `''` when closed instead of `'false'`. Fixed both to always set an explicit `'true'`/`'false'` string.
3. `setLang` replaced the whole `prefs` object with `{ lang }` only, dropping `theme` back to the store's default merge behaviour (shallow merge only recreates the keys given in the patch). Fixed to carry the current `theme` forward, mirroring `setTheme`.
4. The clock stored its interval id on `inst._timer` in `onMount` but read `inst.timer` (undefined) in `onDestroy`, so `clearInterval` was a no-op. Fixed to read `inst._timer`.
5. The remove delegate selected `.task__remove`, a class name the designer had already renamed to `task__delete`; class selectors are also against the kit's rule ("never select by CSS class"). Fixed to select `[data-action="remove"]`, which was already present in the markup and unchanged.
6. Comments were inserted with `'<li class="comments__item">' + c + '</li>'`, letting a comment's raw HTML run. Fixed by rendering the list through `vf.html`, which escapes `c` by position (text node), so `<b>hi</b>` now shows as literal text.
7. Checkboxes had no `id`, so vfunc's refresh-time focus restore could not reliably find "the same" checkbox after the full task list re-render triggered by `toggleTask` → `store.subscribe` → `tasks.refresh()`. Fixed by giving each checkbox `id="toggle-<task id>"`.

## Checklist
- [x] Every dynamic value goes through `vf.html` / `vf.tpl`; no `innerHTML` from data — comments now render via `vf.html`; nothing else in the file builds markup from strings.
- [x] Selectors use `data-action` / `data-ref` / `id` only; no colors or sizes in JS — replaced the one class selector (`.task__remove`) with `[data-action="remove"]`.
- [x] New text uses message keys in every locale; dates and numbers use `vf.fmt` — no new user-visible text was introduced; `vf.fmt.date` usage for the clock is unchanged.
- [x] Listeners, timers and subscriptions are released in `onDestroy` — the clock's timer is now actually cleared (`inst._timer`).
- [ ] The page opens with no console errors or warnings — not verified (no browser available in this environment); the specific warning quoted in bug 1 ("render returned the target element itself") is addressed by no longer returning a wrapping `<section id="notice">` from that render.
- [x] No secrets, tokens or personal data in code, logs or commits — none added.

All fixes are traced to their exact reported symptom and cross-checked against the kit's API descriptions (`vf.attach`, `vf.html`, `store.set`, `onDestroy`, focus-restore-by-id) instead of being verified live.
