### app.js

```js
import vf from './lib/vfunc.esm.js';

const CATEGORIES = [
  ['all', 'All'],
  ['keyboard', 'Keyboards'],
  ['mouse', 'Mice'],
  ['monitor', 'Monitors']
];
const SORTS = [
  ['name', 'Name'],
  ['price-asc', 'Price: low to high'],
  ['price-desc', 'Price: high to low']
];
// USD price formatting: kept as Intl.NumberFormat (same as the source) because
// the app has no i18n setup (single language) and vf.fmt.currency's locale
// behaviour without vf.i18n.setup() is not documented in the kit. VERIFY: whether
// vf.fmt.currency(n, 'USD') would format identically without an active locale.
const money = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' });

function shownProducts(s) {
  const q = s.query.trim().toLowerCase();
  const order = {
    name: (a, b) => a.name.localeCompare(b.name, 'en'),
    'price-asc': (a, b) => a.price - b.price,
    'price-desc': (a, b) => b.price - a.price
  }[s.sort];
  return s.items
    .filter((p) => s.category === 'all' || p.category === s.category)
    .filter((p) => !s.inStock || p.stock > 0)
    .filter((p) => !q || p.name.toLowerCase().includes(q))
    .sort(order);
}

let alive = true;

const app = vf.vfunc({
  tag: 'div',
  state: {
    items: [],
    status: 'loading',
    query: '',
    category: 'all',
    inStock: false,
    sort: 'name'
  },
  render: (s) => {
    if (s.status === 'loading') {
      return vf.html`<p class="pf__note" data-ref="status" role="status">Loading products&hellip;</p>`;
    }
    if (s.status === 'error') {
      return vf.html`<p class="pf__note pf__note--error" data-ref="status" role="alert">Could not load products.</p>`;
    }

    const shown = shownProducts(s);

    return vf.html`
      <section class="pf">
        <div class="pf__controls">
          <label class="pf__field">
            Search
            <input class="pf__input" type="search" data-action="search" value="${s.query}">
          </label>
          <label class="pf__field">
            Category
            <select class="pf__input" data-action="category">
              ${CATEGORIES.map(([value, label]) => vf.html`<option value="${value}" ${value === s.category ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
          <label class="pf__check">
            <input type="checkbox" data-action="in-stock" ${s.inStock ? 'checked' : ''}>
            In stock only
          </label>
          <label class="pf__field">
            Sort
            <select class="pf__input" data-action="sort">
              ${SORTS.map(([value, label]) => vf.html`<option value="${value}" ${value === s.sort ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
        </div>
        <p class="pf__count" data-ref="count">${shown.length === 1 ? '1 product' : shown.length + ' products'}</p>
        ${shown.length === 0
          ? vf.html`<p class="pf__empty" data-ref="empty">No products found.</p>`
          : vf.html`<ul class="pf__list" data-ref="list">
              ${shown.map((p) => vf.html`
                <li class="pf__item" data-id="${p.id}" data-state="${p.stock === 0 ? 'out' : 'in'}">
                  <span class="pf__name" data-ref="name">${p.name}</span>
                  <span class="pf__price" data-ref="price">${money.format(p.price)}</span>
                  ${p.stock === 0 ? vf.html`<span class="pf__badge">Sold out</span>` : ''}
                </li>
              `)}
            </ul>`}
      </section>
    `;
  },
  delegates: [
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent: (e) => { e.sender.setState({ query: e.target.value }); }
    },
    {
      selector: '[data-action="category"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ category: e.target.value }); }
    },
    {
      selector: '[data-action="in-stock"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ inStock: e.target.checked }); }
    },
    {
      selector: '[data-action="sort"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ sort: e.target.value }); }
    }
  ],
  onMount: function () {
    const inst = this;
    fetch(new URL('./data/products.json', import.meta.url))
      .then((r) => {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then((data) => {
        if (alive) inst.setState({ items: data, status: 'ready' });
      })
      .catch(() => {
        if (alive) inst.setState({ status: 'error' });
      });
  },
  onDestroy: function () {
    alive = false;
  }
});

app.mount('#app');
```

### REPORT.md

```markdown
# Report — 07 Port a React component

## Assumptions
- Single-language app (English only, as required): no `vf.i18n.setup()`, texts hardcoded exactly as in `ProductFilter.jsx`.
- Kept the component as one `vf.vfunc` mounted directly to `#app` (no router/store) since the task is a single page, not a full SPA — the `AGENTS.md` structure table is for multi-page apps and this task only asked for `index.html` + `app.js` + a CSS file.
- Loading/error states render *only* the `data-ref="status"` paragraph (no controls, no list), matching the source's early `return`. When `status === 'ready'`, no `status` element is emitted at all (absent, not hidden).
- The empty-results message (`data-ref="empty"`) is only emitted when the filtered list is empty; the `<ul data-ref="list">` is only emitted when there are results (mirrors the source's conditional render, and satisfies "absent or hidden when there are products").
- Live search filtering re-renders the whole root on every keystroke (`e.sender.setState({ query: ... })`), which is required for "search while typing" to actually filter the visible list; I relied on the engine's documented focus/caret restoration on refresh to satisfy "keeps the focus while typing" rather than deferring the render to blur/submit (which the general `AGENTS.md`/migration-prompt advice for controlled inputs would otherwise suggest — flagged below).
- USD formatting kept as `new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' })` (same as the source) instead of `vf.fmt.currency`, since the kit doesn't document `vf.fmt.currency`'s output without an active `vf.i18n.setup()` locale, and exact `$49.00` formatting is a grader hook. Marked `VERIFY:` in `app.js`.
- `data-ref="name"` / `data-ref="price"` are repeated on every list row (one per product), not unique per root — used purely as query hooks for the grader, not read back through `inst.refs` (which would only resolve to a single element). This matches how the task's hook table describes them ("in each product … `data-ref="name"`, …").
- CSS raw values mapped to the nearest existing `--vf-*` tokens (spacing → `--vf-space-1..3`, radius → `--vf-radius-sm`/`-full`, muted/disabled/danger colors → the matching semantic tokens, font sizes → `--vf-font-size-xs`/`-sm`). Exact pixel values from `ProductFilter.css` are not reproduced 1:1 since the task asked for tokens instead of raw values.
- Data file `data/products.json` was left unchanged (task said it already exists in the project); it was echoed back only because the output rules ask for the complete content of every file referenced — no edits were made to it.
- CSP meta tag uses `'self'` only for `script-src`/`style-src`/`connect-src` (no inline, no `unsafe-inline`), per section 3.5/3.6 of `AGENTS.md`.
```
