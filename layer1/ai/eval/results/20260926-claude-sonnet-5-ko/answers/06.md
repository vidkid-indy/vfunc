### app.js

```js
import vf from './lib/vfunc.esm.js';
import { store, favoritesCount } from './store.js';
import homePage from './pages/home.js';
import serversPage from './pages/servers.js';
import serverPage from './pages/server.js';
import notFoundPage from './pages/not-found.js';

function readLocale() {
  try {
    const saved = localStorage.getItem('vf.locale');
    if (saved === 'en' || saved === 'ko') return saved;
  } catch (err) {
    // localStorage unavailable (private mode, etc.) — fall through to browser language
  }
  const lang = (navigator.language || 'en').toLowerCase();
  return lang.indexOf('ko') === 0 ? 'ko' : 'en';
}

function isServersActive(path) {
  return path === '/servers' || path.indexOf('/servers/') === 0;
}

async function main() {
  const startLocale = readLocale();
  try { localStorage.setItem('vf.locale', startLocale); } catch (err) { /* ignore */ }

  await vf.i18n.setup({
    locales: ['en', 'ko'],
    fallback: 'en',
    persist: true,
    load: (locale) => fetch(new URL('./locales/' + locale + '.json', import.meta.url)).then((res) => res.json())
  });

  document.documentElement.lang = readLocale();

  let currentPage = null;
  let router = null;

  function renderPage(pageFactory, ctx) {
    if (currentPage) {
      currentPage.destroy();
      currentPage = null;
    }
    currentPage = pageFactory(ctx, router);
    currentPage.mount('#view');
  }

  const shell = vf.attach('#app', {
    state: {
      locale: readLocale(),
      path: '/',
      favCount: favoritesCount(store.get())
    },
    render: (s) => vf.html`
      <header class="app__header">
        <nav class="app__nav" aria-label="${vf.t('nav.home')}">
          <a data-link href="#/" aria-current="${s.path === '/' ? 'page' : 'false'}">${vf.t('nav.home')}</a>
          <a data-link href="#/servers" aria-current="${isServersActive(s.path) ? 'page' : 'false'}">${vf.t('nav.servers')}</a>
        </nav>
        <div class="app__favorites">
          <span class="app__favorites-label">${vf.t('favorites.label')}</span>
          <span data-ref="fav-count">${s.favCount}</span>
        </div>
        <div class="app__locale" role="group">
          <button type="button" data-action="locale" data-locale="en" aria-pressed="${s.locale === 'en'}">English</button>
          <button type="button" data-action="locale" data-locale="ko" aria-pressed="${s.locale === 'ko'}">한국어</button>
        </div>
      </header>
      <main id="view" tabindex="-1"></main>
    `,
    delegates: [
      {
        selector: '[data-action="locale"]',
        eventType: 'click',
        onEvent: (e) => {
          const locale = e.target.getAttribute('data-locale');
          if (locale === e.sender.locale) return;
          vf.i18n.set(locale).then(() => {
            document.documentElement.lang = locale;
            e.sender.setState({ locale });
          });
        }
      }
    ],
    onMount: (inst) => {
      storeUnsubscribe = store.subscribe((state) => {
        inst.setState({ favCount: favoritesCount(state) });
      });
      i18nUnsubscribe = vf.i18n.subscribe(() => {
        inst.setState({ locale: readLocale() });
      });
    },
    onDestroy: () => {
      if (storeUnsubscribe) { storeUnsubscribe(); storeUnsubscribe = null; }
      if (i18nUnsubscribe) { i18nUnsubscribe(); i18nUnsubscribe = null; }
    }
  });

  let storeUnsubscribe = null;
  let i18nUnsubscribe = null;

  router = vf.router({
    mode: 'hash',
    routes: {
      '/': (ctx) => renderPage(homePage, ctx),
      '/servers': (ctx) => renderPage(serversPage, ctx),
      '/servers/:id': (ctx) => renderPage(serverPage, ctx)
    },
    notFound: (ctx) => renderPage(notFoundPage, ctx),
    linkSelector: 'a[data-link]',
    focus: '#view',
    onChange: (ctx) => {
      shell.setState({ path: ctx.path });
    }
  });

  router.start();
}

main();
```

### store.js

```js
import vf from './lib/vfunc.esm.js';

// Shared app state. This module holds the only functions that change it.
export const store = vf.store({ favorites: [] });

export function isFavorite(id) {
  return store.get('favorites').indexOf(id) !== -1;
}

export function toggleFavorite(id) {
  store.set((state) => {
    const has = state.favorites.indexOf(id) !== -1;
    const favorites = has
      ? state.favorites.filter((fid) => fid !== id)
      : state.favorites.concat([id]);
    return { favorites };
  });
}

export function favoritesCount(state) {
  return state && state.favorites ? state.favorites.length : 0;
}
```

```js
import vf from '../lib/vfunc.esm.js';
import { fetchServers } from '../api.js';

export default function serversPage(ctx, router) {
  return vf.vfunc({
    tag: 'div',
    state: { status: 'loading', servers: [] },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p data-state="loading">${vf.t('servers.loading')}</p>
        `;
      }
      if (s.status === 'error') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p data-state="error" role="alert">${vf.t('servers.error')}</p>
        `;
      }
      if (s.servers.length === 0) {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p data-state="empty">${vf.t('servers.empty')}</p>
        `;
      }
      return vf.html`
        <h1>${vf.t('servers.title')}</h1>
        <ul class="server-list">
          ${s.servers.map((server) => vf.html`
            <li class="server-list__item" data-id="${server.id}" data-state="${server.status}">
              <a class="server-list__link" data-link href="${router.href('/servers/' + server.id)}">${server.name}</a>
              <span class="server-list__region">${server.region}</span>
              <span class="server-list__status" data-state="${server.status}">${server.status}</span>
            </li>
          `)}
        </ul>
      `;
    },
    onMount: (inst) => {
      fetchServers()
        .then((servers) => { inst.setState({ status: 'ready', servers }); })
        .catch(() => { inst.setState({ status: 'error', servers: [] }); });
    }
  });
}
```

### pages/server.js

```js
import vf from '../lib/vfunc.esm.js';
import { fetchServer } from '../api.js';
import { isFavorite, toggleFavorite } from '../store.js';

export default function serverPage(ctx) {
  const id = ctx.params.id;

  return vf.vfunc({
    tag: 'div',
    state: { status: 'loading', server: null },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`<p data-state="loading">${vf.t('servers.loading')}</p>`;
      }
      if (!s.server) {
        return vf.html`<h1>${vf.t('server.missing')}</h1>`;
      }
      const fav = isFavorite(s.server.id);
      return vf.html`
        <h1>${s.server.name}</h1>
        <dl class="server-detail">
          <dt class="server-detail__label">${vf.t('servers.title')}</dt>
          <dd class="server-detail__value" data-state="${s.server.status}">${s.server.status}</dd>
          <dt class="server-detail__label">region</dt>
          <dd class="server-detail__value">${s.server.region}</dd>
        </dl>
        <button type="button" data-action="favorite" aria-pressed="${fav}">
          ${fav ? vf.t('server.favorite.remove') : vf.t('server.favorite.add')}
        </button>
      `;
    },
    delegates: [
      {
        selector: '[data-action="favorite"]',
        eventType: 'click',
        onEvent: (e) => {
          const server = e.sender.server;
          if (!server) return;
          toggleFavorite(server.id);
          e.sender.refresh();
        }
      }
    ],
    onMount: (inst) => {
      fetchServer(id)
        .then((server) => { inst.setState({ status: 'ready', server }); })
        .catch(() => { inst.setState({ status: 'ready', server: null }); });
    }
  });
}
```

### REPORT.md

```markdown
## 화면 목록

| 라우트 | 페이지 파일 | 목적 | 데이터 | 동작 |
|---|---|---|---|---|
| `#/` | pages/home.js | 소개 | 없음 | 없음 |
| `#/servers` | pages/servers.js | 서버 목록 | `data/servers.json` (api.js 경유) | 상세로 이동하는 링크 |
| `#/servers/:id` | pages/server.js | 서버 상세 | 목록에서 id로 찾음 | 즐겨찾기 토글 |
| 그 밖 | pages/not-found.js | 찾을 수 없음 | 없음 | 없음 |

## 상태 지도

- 공유(`store.js`, `vf.store`): `favorites`(id 배열). `toggleFavorite(id)`가 유일한 변경 함수. 페이지 이동·언어 전환에도 유지, 새로고침에는 유지 안 함(요구사항대로 localStorage에 넣지 않음).
- 로케일: store가 아니라 `vf.i18n`의 `persist: true`가 `localStorage['vf.locale']`에 저장. 새로고침 후에도 유지되어야 하므로 store 밖에 둠.
- 페이지 안에서만 쓰는 상태: `servers.js`의 `{status, servers}`(loading/error/ready), `server.js`의 `{status, server}`, 앱 셸의 `{locale, path, favCount}`(favCount와 locale은 store/i18n의 파생값이라 구독으로 갱신).

## 메시지 키

en/locales와 ko/locales 파일에 과제 표의 12개 키를 전부 넣었습니다(`nav.*`, `favorites.label`, `home.*`, `servers.*`, `server.favorite.*`, `server.missing`, `notFound.title`).

## 가정과 VERIFY

- `vf.router`의 route 콜백 `ctx`에 동적 세그먼트가 `ctx.params.id`로 들어온다고 가정했습니다(킷 문서는 `'/users/:id': (ctx) => …` 형태만 보여주고 `ctx`의 정확한 필드 이름은 명시하지 않습니다). **VERIFY: ctx.params 모양.**
- `vf.router`의 `onChange` 콜백이 `ctx.path`(현재 앱 경로, `/servers/:id` 형태가 아니라 실제 경로)를 받는다고 가정하고 nav의 `aria-current` 갱신에 썼습니다. **VERIFY: onChange 인자 모양.**
- `vf.attach('#app', …)`처럼 target에 선택자 문자열을 넘길 수 있다고 가정했습니다(문서 예시는 `counter.mount('#app')`만 보여줌). 요소를 직접 넘기는 것도 실패하지 않을 것으로 보이나 확인하지 못했습니다.
- `styles/pages/servers.css`, `styles/pages/server.css`에서 상태 색에 `--vf-chart-1/2/3`을 up/down/maintenance에 각각 매핑했습니다. 킷의 llms.ko.txt는 토큰 전체 목록을 "영어판과 같음(예: `--vf-color-primary`, `--vf-color-text-muted`, `--vf-space-3`, `--vf-chart-1..8` 등)"이라고만 적어서, 제가 쓴 `--vf-color-bg`, `--vf-color-border`, `--vf-radius-md` 등은 실제 토큰 이름이 아닐 수 있습니다. **VERIFY: 정확한 토큰 이름(사이트 design.md 또는 lib/vfunc.tokens.css 실제 내용 확인 필요).** 모든 값에 CSS fallback(`var(--vf-x, 기본값)`)을 넣어 토큰이 없어도 레이아웃은 깨지지 않게 했습니다.
- `design/DESIGN.md`가 이번 과제에는 없으므로 `styles/tokens.css`는 빈 설명 파일로 두고 새 원시값을 추가하지 않았습니다(요구사항 4번 "DESIGN 없으면 중립 토큰 유지").
- 언어 버튼 문구는 `English`/`한국어`로 고정하고 메시지 키로 만들지 않았습니다(과제 표에 두 언어 문구가 아니라 고정 라벨로 명시되어 있어서 그렇게 판단했습니다).
- `vf.i18n.setup`이 Promise를 반환한다고 가정하고 `await` 했습니다.
- update 플러그인과 `version.json`은 과제 지시대로 제외했습니다.
```
