### app.js

```js
// Sign up — the page script.
(function () {
  'use strict';

  var form = document.getElementById('signup');
  var doneEl = document.querySelector('[data-ref="done"]');

  // Validation order matters: on submit, the first invalid field (in this
  // order) receives focus.
  var FIELD_ORDER = ['email', 'password', 'confirm', 'terms'];

  var RULES = {
    email: {
      message: 'Enter a valid email address.',
      test: function (field) {
        return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(field.value.trim());
      }
    },
    password: {
      message: 'Use at least 8 characters, including a number.',
      test: function (field) {
        var value = field.value;
        return value.length >= 8 && /\d/.test(value);
      }
    },
    confirm: {
      message: 'Passwords do not match.',
      test: function (field) {
        var value = field.value;
        return value.length > 0 && value === form.elements.password.value;
      }
    },
    terms: {
      message: 'Accept the terms to continue.',
      test: function (field) {
        return field.checked;
      }
    }
  };

  // Fields that have been validated at least once: their error is kept live
  // (re-checked on every input) until they become valid.
  var touched = {};

  function errorElFor(name) {
    return document.getElementById(name + '-error');
  }

  function validateField(name) {
    var field = form.elements[name];
    var rule = RULES[name];
    var valid = rule.test(field);
    var msg = errorElFor(name);

    msg.textContent = valid ? '' : rule.message;
    field.setAttribute('aria-invalid', valid ? 'false' : 'true');
    touched[name] = true;

    return valid;
  }

  vf.attach(form, {
    events: [
      { id: 'email', eventType: 'focusout', onEvent: function () { validateField('email'); } },
      { id: 'email', eventType: 'input', onEvent: function () { if (touched.email) { validateField('email'); } } },

      { id: 'password', eventType: 'focusout', onEvent: function () { validateField('password'); } },
      { id: 'password', eventType: 'input', onEvent: function () { if (touched.password) { validateField('password'); } } },

      { id: 'confirm', eventType: 'focusout', onEvent: function () { validateField('confirm'); } },
      { id: 'confirm', eventType: 'input', onEvent: function () { if (touched.confirm) { validateField('confirm'); } } },

      { id: 'terms', eventType: 'change', onEvent: function () { validateField('terms'); } },

      {
        eventType: 'submit',
        onEvent: function (e) {
          e.event.preventDefault();

          var firstInvalid = null;
          for (var i = 0; i < FIELD_ORDER.length; i++) {
            var name = FIELD_ORDER[i];
            var ok = validateField(name);
            if (!ok && !firstInvalid) {
              firstInvalid = form.elements[name];
            }
          }

          if (firstInvalid) {
            firstInvalid.focus();
            return;
          }

          var email = form.elements.email.value.trim();
          form.hidden = true;
          doneEl.hidden = false;
          doneEl.textContent = 'Welcome, ' + email + '!';
        }
      }
    ]
  });
})();
```

### REPORT.md

```markdown
# 가입 폼 검증 — 보고

## 가정
- `--vf-color-danger` 토큰이 존재한다고 가정했습니다. 킷 문서(llms.ko.txt)에 나온 토큰 목록에는 danger/error 색이 명시돼 있지 않아 이름을 추측했습니다. 없다면 오류 문구가 기본 텍스트 색으로 보이는 정도이며(원시값 fallback을 넣지 않았으므로) 콘솔 오류는 나지 않습니다.
- `#terms-error` 요소는 체크박스 필드 안, 라벨 뒤에 두고 `flex-basis: 100%`로 줄바꿈했습니다(마크업 구조는 유지, 새 오류 문구 요소만 추가).
- `data-ref="done"` 요소는 `<main>` 안, `<form>` 형제로 새로 추가했고 처음에는 `hidden` 속성으로 숨겨 뒀습니다(과제에 초기 상태가 명시돼 있지 않아 가장 단순한 쪽을 택함).
- "텍스트 필드는 포커스를 잃을 때 검증" 규칙은 `blur` 대신 버블링되는 `focusout`으로 구현했습니다(vfunc `events`는 지정한 `id` 요소에 직접 리스너를 답니다).
- confirm 필드는 자신이 한 번 오류를 보인 뒤에만 자기 입력 시 재검증합니다. 비밀번호를 나중에 고쳐도 confirm까지 자동으로 재검증하라는 규칙은 없어서 넣지 않았습니다.

## 킷에서 지키지 못한 것
- 없음. `vf.attach` + `events`(id 기반)만 사용했고 `vf.html`/`innerHTML`을 전혀 쓰지 않았습니다(오류 문구와 환영 메시지는 `textContent`로만 넣어 이스케이프 문제가 없습니다).

## 확인 목록 결과
- [x] 모든 동적 값이 텍스트로만 들어간다(`textContent`), `innerHTML`을 데이터로 설정하지 않음
- [x] 셀렉터는 `id`만 사용(`events`의 `id`), CSS 클래스로 걸지 않음
- [x] 새 CSS는 `var(--vf-*)` 토큰만 사용. hex/px 값 없음
- [x] 이 과제는 새 로케일 문구가 아니라 과제가 지정한 고정 영어 문구를 그대로 써야 하므로 메시지 키를 쓰지 않았습니다(과제 지시 "화면 문구는 과제에 적힌 영어 문구를 그대로 씁니다"를 우선함)
- [x] 새 리스너는 페이지에 단 하나뿐인 폼에 계속 붙어 있어야 하므로 `onDestroy` 해제 대상이 없음(SPA 라우팅이 없는 단일 페이지)
- [ ] 콘솔 에러·경고 0개 — 실제 브라우저로 실행해 보지 못해 "확인 못 함"
- [x] 비밀번호 값은 `password`/`confirm` input 안에만 있고, 로그·화면·환영 메시지 어디에도 나타나지 않음(환영 메시지는 이메일만 사용)
- [x] 코드에 비밀값·토큰·개인정보 없음
```
