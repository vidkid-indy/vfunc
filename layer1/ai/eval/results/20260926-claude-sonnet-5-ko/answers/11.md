### app.js

```js
// Users screen — the page script. USERS, MONTHS and MONTH_LABELS come from data.js.
(function () {
  'use strict';

  var selectedEl = document.querySelector('[data-ref="selected"]');

  function setSelectedCount(count) {
    selectedEl.textContent = count + ' selected';
  }

  var columns = [
    { key: 'name', label: 'Name', sortable: true },
    { key: 'email', label: 'Email' },
    {
      key: 'role',
      label: 'Role',
      render: function (row) {
        return vf.vsBadge({ label: row.role });
      }
    },
    { key: 'joined', label: 'Joined', sortable: true }
  ];

  var grid = vf.vfGrid({
    id: 'users-grid',
    caption: 'Users',
    columns: columns,
    data: USERS,
    pageSize: 10,
    selectable: 'multiple',
    rowKey: 'id',
    onSelect: function (e) {
      setSelectedCount(e.data.keys.length);
    }
  });
  grid.mount('#users');

  function applyFilter(value) {
    var q = (value || '').toLowerCase();
    var filtered = q
      ? USERS.filter(function (u) { return u.name.toLowerCase().indexOf(q) !== -1; })
      : USERS;
    grid.setData(filtered);
    grid.setPage(1);
  }

  var search = vf.vfSearchInput({
    id: 'q',
    label: 'Search users',
    onSearch: function (e) {
      applyFilter(e.data.value);
    },
    onClear: function () {
      applyFilter('');
    }
  });
  search.mount('#search');

  var counts = MONTHS.map(function (month) {
    return USERS.filter(function (u) { return u.joined.indexOf(month) === 0; }).length;
  });

  var chart = vf.vfChart({
    id: 'signups-chart',
    type: 'bar',
    label: 'Sign-ups per month',
    dataTable: true,
    data: {
      labels: MONTH_LABELS,
      series: [{ name: 'Sign-ups', data: counts }]
    }
  });
  chart.mount('#signups');
}());
```

### REPORT.md

## 가정

- `id="q"`는 `vf.vfSearchInput`의 `id` prop으로 넘겨 입력 요소 자체의 id로 삼았습니다(킷의 다른 폼 컨트롤 예시와 같은 패턴).
- `Search users`는 `label` prop으로 넘겼습니다(라벨 텍스트를 보이는 `<label>`로 연결). `ariaLabel`이 아니라 `label`을 골랐는데, 과제 문구가 "라벨은 Search users"라고만 했고 별도로 숨김 라벨을 요구하지 않았기 때문입니다.
- 검색 지우기는 `onSearch`(빈 문자열)와 지우기 버튼의 `onClear` 둘 다에서 같은 `applyFilter('')`를 호출해 처리했습니다. 킷에 두 콜백이 언제 각각 불리는지 정확한 순서가 나와 있지 않아, 가장 단순하게 두 경로 모두 전체 목록으로 되돌리도록 했습니다.
- `vsBadge`의 `variant`는 지정하지 않았습니다(과제가 "역할 글자를 담은 vsBadge"라고만 했고 색상 규칙을 주지 않았습니다).
- 최신 브라우저만 대상이라는 조건에 따라 `app.js`는 ES2015+ 문법(화살표 함수 없이 `function`/`var`를 유지해 IIFE 안에서 일관되게 작성, IE11 대상이 아니므로 필수는 아님)을 썼습니다.

## 킷에서 지키지 못한 것

- 없음. `index.html`, `data.js`, `style.css`는 과제 지시대로 그대로 두고 `app.js`만 채웠습니다.

## 확인 목록 결과

- [확인 못 함] 정적 서버 + 3개 브라우저 엔진에서 실제로 열어 콘솔 오류·경고 0개인지 — 이 환경에서는 브라우저를 실행할 수 없어 코드 검토로만 확인했습니다.
- [했음] 동적 마크업은 모두 `vf.vsBadge`(SafeHtml)와 컴포넌트 자체 렌더링만 사용, 문자열 이어붙이기나 `innerHTML` 직접 대입 없음.
- [했음] 셀렉터는 컴포넌트가 요구하는 `id`(`q`, `users-grid`, `signups-chart`)와 `[data-ref="selected"]`만 사용. CSS 클래스로 훅을 걸지 않음.
- [했음] 열 순서(`name` 정렬 가능 / `email` / `role`(vsBadge) / `joined` 정렬 가능), `pageSize: 10`, `selectable: 'multiple'`, `rowKey: 'id'`, 그리드 `id="users-grid"`와 캡션 `Users`를 지정.
- [했음] 검색은 이름 부분일치·대소문자 무시로 필터링하고 1쪽으로 리셋, 지우면 전체 목록 복귀.
- [했음] 선택 카운트는 `onSelect`의 `e.data.keys.length`로 갱신, 형식은 `<n> selected`.
- [했음] 차트는 `type: 'bar'`, `label: 'Sign-ups per month'`, `dataTable: true`, `labels: MONTH_LABELS`, 시리즈 하나(`Sign-ups`)로 `MONTHS` 각 달의 `joined` 매칭 건수.
- [확인 못 함] 채점기가 기대하는 정확한 DOM(`data-action="sort"`, `data-action="select-row"`, `data-action="mark"` 등)은 vfGrid/vfChart 컴포넌트가 내부적으로 생성하므로 별도 마크업을 추가하지 않았고, 실제 렌더 결과는 브라우저 실행 없이는 확인하지 못했습니다.
