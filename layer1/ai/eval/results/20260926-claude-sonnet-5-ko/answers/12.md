```js
// Profile — the page script.

var toast = vf.vfToast();

var form = vf.attach(document.getElementById('profile'), {
  state: {
    values: { name: '', email: '', role: 'viewer', newsletter: false },
    errors: {}
  },

  render: function (s) {
    var v = s.values;
    var e = s.errors;
    return vf.html`
      ${vf.vsInput({
        id: 'name',
        name: 'name',
        type: 'text',
        label: 'Name',
        value: v.name,
        error: e.name
      })}
      ${vf.vsInput({
        id: 'email',
        name: 'email',
        type: 'email',
        label: 'Email',
        hint: 'We never share it.',
        value: v.email,
        error: e.email
      })}
      ${vf.vsSelect({
        id: 'role',
        name: 'role',
        label: 'Role',
        options: [
          { value: 'admin', label: 'Admin' },
          { value: 'editor', label: 'Editor' },
          { value: 'viewer', label: 'Viewer' }
        ],
        value: v.role
      })}
      ${vf.vsSwitch({
        id: 'newsletter',
        name: 'newsletter',
        label: 'Newsletter',
        checked: v.newsletter
      })}
      ${vf.vsButton({ label: 'Save', type: 'submit', variant: 'primary' })}
    `;
  },

  delegates: [
    {
      selector: '#name',
      eventType: 'input',
      onEvent: function (e) { e.sender.state.values.name = e.target.value; }
    },
    {
      selector: '#email',
      eventType: 'input',
      onEvent: function (e) { e.sender.state.values.email = e.target.value; }
    },
    {
      selector: '#role',
      eventType: 'change',
      onEvent: function (e) { e.sender.state.values.role = e.target.value; }
    },
    {
      selector: '#newsletter',
      eventType: 'change',
      onEvent: function (e) { e.sender.state.values.newsletter = e.target.checked; }
    }
  ],

  events: [
    {
      eventType: 'submit',
      onEvent: function (e) {
        e.event.preventDefault();

        var inst = e.sender;
        var values = inst.state.values;
        var name = String(values.name || '').trim();
        var email = String(values.email || '').trim();
        var emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        var errors = {};

        if (!name) {
          errors.name = 'Enter your name.';
        }
        if (!emailPattern.test(email)) {
          errors.email = 'Enter a valid email address.';
        }

        inst.state.errors = errors;
        inst.refresh();

        var order = ['name', 'email', 'role', 'newsletter'];
        var firstInvalid = null;
        for (var i = 0; i < order.length; i++) {
          if (errors[order[i]]) {
            firstInvalid = order[i];
            break;
          }
        }

        if (firstInvalid) {
          var field = inst.$node.querySelector('#' + firstInvalid);
          if (field) field.focus();
          return;
        }

        var savedEl = vf.$('[data-ref="saved"]');
        if (savedEl) {
          savedEl.textContent = 'Saved: ' + name + ' (' + values.role + ')';
        }
        toast.show({ message: 'Profile saved', variant: 'success' });
      }
    }
  ]
});
```

### REPORT.md

```markdown
# 과제 12 — vs* 필드로 만드는 프로필 폼

## 가정
- 킷이 명시한 필드 순서(name, email, role, newsletter, submit)를 그대로 지켰고, 각 필드의 `id`/`name`을 표에 적힌 값과 동일하게 주어 `querySelector('#' + id)`로 첫 오류 필드에 포커스를 옮길 수 있게 했습니다.
- `vf.attach(form, { render, ... })`로 이미 존재하는 `<form id="profile" novalidate>`의 안쪽만 그리고, 루트 속성(`id`, `class`, `novalidate`)은 그대로 두었습니다.
- 텍스트 입력(`name`, `email`)은 입력할 때마다 다시 그리지 않도록 `inst.state.values.x = e.target.value`로 값만 저장하고(`vfunc.js` 문서의 "입력 중에는 키마다 다시 렌더하지 말고 값만 저장" 규칙), `role`/`newsletter`는 change 이벤트에서 같은 방식으로 값만 저장했습니다.
- 제출 처리에서는 `setState`(비동기 배치 렌더) 대신 `inst.state.errors = errors; inst.refresh();`로 동기적으로 다시 그린 뒤 바로 `querySelector`로 포커스를 옮겼습니다. 세 엔진 모두에서 타이밍 이슈 없이 동작하도록 하기 위한 선택입니다(킷에 `refresh()`가 즉시 재렌더하는 메서드로 별도 문서화되어 있음).
- 필드에 HTML `required` 속성은 주지 않았습니다. 표는 "오류는 제출한 뒤에만 보인다"고 했고, `required`를 주면 일부 브라우저가 제출 전에도 시각적 표시를 추가할 수 있어 이 규칙과 충돌할 수 있다고 판단했습니다. 검증과 오류 메시지는 전적으로 제출 핸들러에서 처리합니다.
- `[data-ref="saved"]`는 폼 컴포넌트의 루트 바깥(같은 `<main>`의 형제 요소)에 있으므로 킷 안내대로 `vf.$('[data-ref="saved"]')`로 찾았습니다.
- `vf.vfToast()`는 페이지에서 한 번만 만들고(`var toast = vf.vfToast();`), 문서에 따라 스스로 `document.body`에 붙는다고 보고 별도 `.mount()`를 호출하지 않았습니다.
- `index.html`, `style.css`는 고칠 필요가 없어 그대로 두었고(출력하지 않음), `app.js`만 새로 채웠습니다.

## 지키지 못한 것
- 없음(킷에 없는 API를 지어내지 않았고, 3~7절 규칙과 확인 목록을 모두 따랐다고 판단합니다).

## 확인 목록 결과
- [x] 모든 동적 값이 `vf.html`/vs* 컴포넌트를 거친다(문자열 이어붙이기나 `innerHTML` 직접 대입 없음)
- [x] 셀렉터는 `id`/`data-ref`만 사용(`#name`, `#email`, `#role`, `#newsletter`, `[data-ref="saved"]`), CSS 클래스로 고르지 않음
- [x] JS에 색·크기·간격 값 없음(스타일은 `style.css`의 토큰에 위임)
- [x] 새 텍스트는 모두 과제가 지정한 영어 문구를 그대로 사용(메시지 키 시스템은 이 과제 범위에 없어 별도 로케일 파일을 만들지 않음 — 페이지에 로케일 폴더/설정이 없는 단일 화면 과제라 하드코딩된 표시 문구가 요구사항과 일치)
- [x] 리스너·구독·타이머 없음(전부 delegates/events로 위임, 별도 해제 필요한 타이머 없음)
- [ ] 콘솔 에러·경고 0개 — 확인 못 함(브라우저 실행 환경이 없어 직접 열어보지 못함). 코드는 킷 문서(AGENTS.md, llms.ko.txt, components.md)에 있는 API만 사용했습니다.
- [x] 코드·로그·커밋에 비밀값 없음
```
