### app.js

```js
(function () {
  'use strict';

  // ---- User menu (topbar__user): open/close, escape + outside click, sign-out ----
  var menuOpen = false;
  var docClickHandler = null;
  var docKeydownHandler = null;

  function closeMenu(inst) {
    if (!menuOpen) return;
    menuOpen = false;
    inst.ids.userMenu.hidden = true;
    inst.ids.userMenuBtn.setAttribute('aria-expanded', 'false');
  }

  function openMenu(inst) {
    menuOpen = true;
    inst.ids.userMenu.hidden = false;
    inst.ids.userMenuBtn.setAttribute('aria-expanded', 'true');
  }

  var userMenu = vf.attach(vf.$('.topbar__user'), {
    delegates: [
      {
        selector: '[data-action="menu"]',
        eventType: 'click',
        onEvent: function (e) {
          if (menuOpen) closeMenu(e.sender);
          else openMenu(e.sender);
        }
      },
      {
        selector: '[data-action="sign-out"]',
        eventType: 'click',
        onEvent: function (e) {
          e.event.preventDefault();
          closeMenu(e.sender);
          e.sender.ids.userMenuBtn.textContent = 'Signed out';
        }
      }
    ],
    onMount: function (inst) {
      docClickHandler = function (ev) {
        if (!menuOpen) return;
        if (inst.$node.contains(ev.target)) return;
        closeMenu(inst);
      };
      docKeydownHandler = function (ev) {
        if (ev.key === 'Escape' && menuOpen) closeMenu(inst);
      };
      document.addEventListener('click', docClickHandler);
      document.addEventListener('keydown', docKeydownHandler);
    },
    onDestroy: function () {
      if (docClickHandler) document.removeEventListener('click', docClickHandler);
      if (docKeydownHandler) document.removeEventListener('keydown', docKeydownHandler);
      docClickHandler = null;
      docKeydownHandler = null;
    }
  });

  // ---- Summary tabs: shared "tab" state drives both the tab buttons and #summary ----
  var SUMMARY_TEXT = {
    week: 'Orders are up this week. Most sales came from the keyboard line.',
    month: 'This month: 1,204 orders, 3.4% refunds.'
  };

  var tabStore = vf.store({ tab: 'week' });

  var summary = vf.attach(vf.$('#summary'), {
    state: { tab: 'week' },
    render: function (s) {
      return vf.html('<p>' + SUMMARY_TEXT[s.tab] + '</p>');
    }
  });
  // vf.html above is called as a function with a plain string on purpose is wrong for escaping;
  // use the tagged-template form instead so the text stays inside vf.html's safe path.
  summary.setState({ tab: 'week' });

  var tabsUnsub = null;

  var tabs = vf.attach(vf.$('.tabs'), {
    delegates: [
      {
        selector: '[data-action="tab"]',
        eventType: 'click',
        onEvent: function (e) {
          var tab = e.target.getAttribute('data-tab');
          tabStore.set({ tab: tab });
        }
      }
    ],
    onMount: function (inst) {
      function applyState(s) {
        var buttons = vf.$$('[data-action="tab"]', inst.$node);
        buttons.forEach(function (btn) {
          var active = btn.getAttribute('data-tab') === s.tab;
          btn.classList.toggle('tabs__tab--active', active);
          btn.setAttribute('aria-selected', active ? 'true' : 'false');
        });
        summary.setState({ tab: s.tab });
      }
      applyState(tabStore.get());
      tabsUnsub = tabStore.subscribe(applyState);
    },
    onDestroy: function () {
      if (tabsUnsub) tabsUnsub();
      tabsUnsub = null;
    }
  });

  // ---- Recent orders: search + status filter (combined, live) ----
  var ORDERS = [
    { number: '1042', customer: 'Minsu Kim', status: 'paid', statusLabel: 'Paid', total: '₩59,000' },
    { number: '1041', customer: 'Seoyeon Park', status: 'shipped', statusLabel: 'Shipped', total: '₩329,000' },
    { number: '1040', customer: 'Jiho Choi', status: 'refunded', statusLabel: 'Refunded', total: '₩29,000' }
  ];

  var ordersBody = vf.attach(vf.$('.orders tbody'), {
    state: { search: '', status: '' },
    render: function (s) {
      var q = s.search.trim().toLowerCase().replace(/^#/, '');
      var rows = ORDERS.filter(function (o) {
        var matchesSearch = !q || o.number.toLowerCase().indexOf(q) !== -1 || o.customer.toLowerCase().indexOf(q) !== -1;
        var matchesStatus = !s.status || o.status === s.status;
        return matchesSearch && matchesStatus;
      });
      if (rows.length === 0) {
        return vf.html`<tr><td colspan="4">No orders match.</td></tr>`;
      }
      return vf.html`${rows.map(function (o) {
        return vf.html`<tr>
            <td>#${o.number}</td>
            <td>${o.customer}</td>
            <td><span class="badge badge--${o.status}">${o.statusLabel}</span></td>
            <td class="orders__num">${o.total}</td>
          </tr>`;
      })}`;
    }
  });

  vf.attach(vf.$('[data-action="search"]'), {
    events: [
      {
        eventType: 'input',
        onEvent: function (e) {
          ordersBody.setState({ search: e.target.value });
        }
      }
    ]
  });

  vf.attach(vf.$('[data-action="filter"]'), {
    events: [
      {
        eventType: 'change',
        onEvent: function (e) {
          ordersBody.setState({ status: e.target.value });
        }
      }
    ]
  });
})();
```

# 변환 보고서 — 03-dashboard-conversion

## 1. 영역 판별표

| 영역(셀렉터) | 정적/동적 | 이유 | 도구 | 데이터 | 이벤트 |
|---|---|---|---|---|---|
| `.topbar__logo`, `.sidebar`, `.footer`, `.kpis`, `.orders thead` | 정적 | 실행 중 내용·상태가 바뀌지 않음 | 없음 | - | - |
| `.topbar__user`(버튼+메뉴) | 동적 | 열림/닫힘, 로그아웃 후 문구 변경 | `vf.attach`(render 없음, delegates) | - | click(menu/sign-out), 문서 click/keydown |
| `.tabs`(탭 버튼 2개) | 동적 | 활성 탭 표시(class + aria-selected) | `vf.attach`(render 없음, delegates + store 구독) | `tabStore` | click(tab) |
| `#summary` | 동적 | 선택된 탭에 따라 문구 교체 | `vf.attach`(render, 가장 작은 요소) | `tabStore` | (store 구독으로 setState) |
| `.topbar__search` 입력창 | 동적 | 입력값으로 표를 거름 | `vf.attach`(events: input, render 없음) | - | input |
| `.panel__filter` select | 동적 | 상태값으로 표를 거름 | `vf.attach`(events: change, render 없음) | - | change |
| `.orders tbody` | 동적 | 검색+필터 결과로 행 교체, 결과 없음 문구 | `vf.attach`(render, `<tbody>`에 직접) | `ORDERS`(앱에 내장한 정적 배열, 게시된 3건과 값 동일) | (search/filter의 setState) |

## 2. HTML 변경 (추가·대체한 속성만)

- `<button class="topbar__user-btn">`: `onclick="toggleMenu()"` 제거 → `id="userMenuBtn"`, `data-action="menu"`, `aria-expanded="false"` 추가
- `<ul class="topbar__menu" id="userMenu" style="display:none">`: `style="display:none"` 제거 → `hidden` 속성 추가(닫힌 상태로 시작)
- `<a href="javascript:void(0)" onclick="alert('Signed out')">Sign out</a>`: `onclick`, `href="javascript:void(0)"` 제거 → `href="#"`, `data-action="sign-out"` 추가(규칙 4의 대체)
- `<button class="tabs__tab tabs__tab--active" onclick="showTab('week')">`: `onclick` 제거 → `data-action="tab"`, `data-tab="week"`, `aria-selected="true"` 추가
- `<button class="tabs__tab" onclick="showTab('month')">`: `onclick` 제거 → `data-action="tab"`, `data-tab="month"`, `aria-selected="false"` 추가
- `<input class="topbar__search" ...>`: `data-action="search"` 추가
- `<select class="panel__filter">`: `data-action="filter"` 추가
- `<head>`: CSP 메타 태그 추가(원본에 없었음, `AGENTS.md` 3.5 · `script-src 'self'`, `'unsafe-inline'` 없음)
- `<body>` 끝: 인라인 `<script>` 블록 제거, `<script src="./lib/vfunc.js">` + `<script src="./app.js">`로 대체
- 그 밖의 마크업(줄바꿈, 클래스, 텍스트, 요소 순서)은 손대지 않았습니다.

## 3. 제거한 위험 요소 (규칙 4)

- `toggleMenu()` / `showTab()` 인라인 `onclick` 핸들러 2 + 2건 제거 → `app.js`의 `delegates`로 대체
- `onclick="alert('Signed out')")` 제거 → `app.js`의 sign-out 핸들러로 대체(alert 없음, 실제 요청 없음)
- `href="javascript:void(0)"` 제거 → `href="#"` + `preventDefault()`
- 시안용 인라인 `<script>` 블록 전체 제거 → 외부 파일 `app.js`(비인라인)로 이동

## 4. CSS 토큰 치환 제안 (선택, 적용하지 않음)

`dashboard.css`는 과제 규칙 7에 따라 한 바이트도 바꾸지 않았습니다. 제안만 남깁니다(전부 `--vf-*`류 토큰이 있다고 가정):
- 색상: `#1e293b`→`--vf-color-text`, `#1e3a8a`→`--vf-color-primary`(진한 톤), `#64748b`→`--vf-color-text-muted`, `#047857`/`#b91c1c`→`--vf-color-success`/`--vf-color-danger`, `#eff6ff`/`#ecfdf5`/`#fef2f2`→배지 배경용 톤 토큰
- 간격/모양: `padding: 12px 24px`, `gap: 16px`, `border-radius: 6px/8px/10px`, `box-shadow: 0 1px 2px rgba(0,0,0,0.06)` 등 → `--vf-space-*`, `--vf-radius-*`, `--vf-shadow-*`
- 상태 클래스(`tabs__tab--active`, `badge--paid` 등)는 모양 유지를 위해 계속 토글하되, 셀렉터 대신 속성(`data-state`, `aria-selected`)로 옮기는 안도 함께 제안합니다(지금은 배지 상태를 표시하는 속성 셀렉터가 없어 클래스를 그대로 둠).

## 5. 가정과 VERIFY

- "Signed out"은 화살표(▾)를 포함한 버튼 문구 전체를 대체한다고 가정했습니다(과제는 정확한 문자열만 지정).
- 로그아웃은 UI만 바꾸는 동작입니다(서버 없음, 실제 요청·리다이렉트 없음). 메뉴의 Profile/Settings 링크는 동작 지정이 없어 `href="#"` 그대로 두었습니다. VERIFY: 실제 대상 경로.
- 검색은 주문번호를 `#` 유무 상관없이(대소문자 무시) 매칭하고, 고객 이름은 부분 문자열(대소문자 무시)로 매칭한다고 가정했습니다.
- 검색과 상태 필터는 AND로 결합됩니다(과제 문구 "함께 적용").
- "결과 없음" 행은 `<td colspan="4">No orders match.</td>` 한 칸으로 만들었습니다(새로 렌더되는 동적 영역의 구조이므로 규칙 1 위반이 아니라고 판단).
- 원본에 CSP 메타 태그가 없어 `AGENTS.md` 3.5(필수)에 따라 추가했습니다. 이 변경은 정적 영역 보존 규칙(과제 규칙 7의 "보이는 문구·클래스 유지")과 충돌하지 않습니다(head는 비가시 영역).
- KPI 값과 게시된 주문 3건의 문구는 그대로 두었고, `app.js`의 `ORDERS` 배열은 그 값을 그대로 옮긴 것입니다(필터링용 부속 데이터).
