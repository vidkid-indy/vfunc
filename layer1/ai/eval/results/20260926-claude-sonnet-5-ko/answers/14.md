### places-map.js

```js
// The app's map component. L1 wrapper: app-specific factory for Leaflet 1.9.4.
// Leaflet is loaded globally as `L` by index.html; this file does not touch it here.
(function () {
  if (!window.app) window.app = {};

  // app.vfPlacesMap(props) — props: { places: [{ id, name, lat, lng }], onSelect }
  // Returns a vfunc instance. The Leaflet map lives inside a data-vf-keep element,
  // is created in onMount and removed (map.remove()) in onDestroy.
  app.vfPlacesMap = function (props) {
    var places = (props && props.places) || [];
    var onSelect = props && props.onSelect;

    // Map and marker state kept in closure variables, not on the instance (reserved-name rules).
    var map = null;
    var markersById = {};

    function primaryColor() {
      return getComputedStyle(document.documentElement)
        .getPropertyValue('--vf-color-primary')
        .trim();
    }

    function applySelection(id) {
      Object.keys(markersById).forEach(function (key) {
        markersById[key].setStyle({ fillOpacity: key === id ? 1 : 0.4 });
      });
    }

    var inst = vf.vfunc({
      tag: 'div',
      // Static host markup only; the map itself is built imperatively in onMount.
      innerHTML: '<div data-vf-keep="map"></div>',
      onMount: function (i) {
        var host = vf.$('[data-vf-keep="map"]', i.$node);
        var color = primaryColor();
        var bounds = [];

        map = L.map(host, { attributionControl: false });

        places.forEach(function (place) {
          var marker = L.circleMarker([place.lat, place.lng], {
            radius: 8,
            weight: 2,
            color: color,
            fillColor: color,
            fillOpacity: 0.4
          });
          // Name may contain "<": escape before handing it to Leaflet, which sets
          // tooltip content via innerHTML, so it must show as plain text.
          marker.bindTooltip(vf.esc(place.name));
          marker.on('click', function (ev) {
            applySelection(place.id);
            if (onSelect) {
              onSelect({ sender: i, event: ev, data: { id: place.id } });
            }
          });
          marker.addTo(map);
          markersById[place.id] = marker;
          bounds.push([place.lat, place.lng]);
        });

        if (bounds.length) {
          map.fitBounds(bounds, { padding: [20, 20] });
        }
      },
      onDestroy: function () {
        if (map) {
          map.remove();
          map = null;
        }
        markersById = {};
      },
      methods: {
        select: function (id) {
          applySelection(id);
        }
      }
    });

    return inst;
  };
})();
```

### app.js

```js
// Places — the page script. Leaflet is loaded as the global `L`.
const PLACES = [
  { id: 'p1', name: 'Seoul Station', lat: 37.5547, lng: 126.9707 },
  { id: 'p2', name: 'Cafe <b>Moon</b>', lat: 37.5665, lng: 126.978 },
  { id: 'p3', name: 'Namsan Tower', lat: 37.5512, lng: 126.9882 }
];

(function () {
  var mapInstance = null;

  function findPlace(id) {
    for (var i = 0; i < PLACES.length; i++) {
      if (PLACES[i].id === id) return PLACES[i];
    }
    return null;
  }

  function selectPlace(id) {
    if (mapInstance) mapInstance.select(id);
    var place = findPlace(id);
    page.refs.current.textContent = place ? ('Selected: ' + place.name) : 'No place selected';
  }

  function createMap() {
    var inst = app.vfPlacesMap({
      places: PLACES,
      onSelect: function (e) { selectPlace(e.data.id); }
    });
    inst.mount('#map-slot');
    return inst;
  }

  // Publishing-HTML control: keep the existing markup, only wire up behavior.
  var page = vf.attach(document.querySelector('.page'), {
    delegates: [
      {
        selector: '[data-action="select"]',
        eventType: 'click',
        onEvent: function (e) {
          var id = e.target.getAttribute('data-id');
          if (id) selectPlace(id);
        }
      },
      {
        selector: '[data-action="toggle-map"]',
        eventType: 'click',
        onEvent: function (e) {
          if (mapInstance) {
            mapInstance.destroy();
            mapInstance = null;
            e.target.textContent = 'Show map';
          } else {
            mapInstance = createMap();
            e.target.textContent = 'Hide map';
          }
        }
      }
    ]
  });

  mapInstance = createMap();
})();
```

# 14-leaflet-places — 보고

## 만든/고친 파일
- `places-map.js` (새로 작성): `app.vfPlacesMap(props)` — L1 앱 전용 Leaflet 래퍼. `vf`에는 아무것도 더하지 않고 `window.app`에만 둠.
- `app.js` (기존 PLACES 배열 유지, 페이지 로직 추가): 지도 mount, 마커/목록 선택 동기화, `toggle-map` 처리.
- `index.html`, `style.css`: 고치지 않음(과제 조건대로 index.html은 그대로, style.css는 이미 토큰만 쓰고 있어 변경 불필요).

## 가정 (직접 판단, 승인 요청 없이 진행)
1. `L.map(host, { attributionControl: false })` — 타일 레이어가 없어 Leaflet 저작권 표시가 의미 없으므로 껐습니다. CSP나 동작에는 영향 없음.
2. `data-vf-keep="map"` 요소를 `innerHTML`로 고정 마크업만 두고 `vf.$('[data-vf-keep="map"]', i.$node)`로 찾았습니다. `render` 대신 정적 `innerHTML`을 쓴 이유: 지도는 vfunc의 재렌더 대상이 아니라 Leaflet API로만 갱신되므로.
3. `toggle-map`으로 지도를 다시 만들 때 마커 선택 상태만 초기화(새 인스턴스는 선택 없음)하고, `[data-ref="current"]`의 문구는 그대로 두었습니다. 과제 문구가 목록/마커 선택 결과에 대해서만 문구 갱신을 요구하고, 지도를 껐다 켜는 것 자체에는 문구 초기화를 명시하지 않았기 때문입니다.
4. `marker.bindTooltip(vf.esc(place.name))` — Leaflet 툴팁은 문자열을 `innerHTML`로 반영하므로, 이름에 `<`가 있어도 텍스트로 보이도록 `vf.esc`로 이스케이프한 값을 넘겼습니다(prompt-integrate-third-party.md의 XSS 규칙).
5. circleMarker의 `radius: 8`, `weight: 2`는 과제에 없는 값이라 임의로 정했습니다(디자인 토큰이 아니라 지도 좌표계 안의 기하 크기라 규칙 21의 "JS에 색·폰트·간격·그림자 값을 넣지 않는다"에는 해당하지 않는다고 판단했습니다. 색만 토큰에서 가져옴).
6. 마커 클릭·목록 버튼 클릭 모두 `selectPlace(id)`로 합쳐 처리했고, `mapInstance`가 없을 때(지도 숨김 상태)도 목록 버튼을 누르면 문구는 갱신되지만 지도 쪽 하이라이트는 건너뜁니다(가드 `if (mapInstance)`).
