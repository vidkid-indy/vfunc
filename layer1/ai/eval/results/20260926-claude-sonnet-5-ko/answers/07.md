### app.js

```js
import vf from './lib/vfunc.esm.js';

const CATEGORIES = [
  ['all', 'All'],
  ['keyboard', 'Keyboards'],
  ['mouse', 'Mice'],
  ['monitor', 'Monitors']
];

const SORTS = [
  ['name', 'Name'],
  ['price-asc', 'Price: low to high'],
  ['price-desc', 'Price: high to low']
];

// Plain Intl formatting kept as-is (matches the source component). vf.fmt.currency
// lives under the i18n module and needs vf.i18n.setup(); this app is single-language
// with no i18n setup (per task instructions), so a local formatter is used instead.
const money = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' });

function sortedShown(state) {
  const q = state.query.trim().toLowerCase();
  const order = {
    name: (a, b) => a.name.localeCompare(b.name, 'en'),
    'price-asc': (a, b) => a.price - b.price,
    'price-desc': (a, b) => b.price - a.price
  }[state.sort];

  return state.items
    .filter((p) => state.category === 'all' || p.category === state.category)
    .filter((p) => !state.inStock || p.stock > 0)
    .filter((p) => !q || p.name.toLowerCase().includes(q))
    .sort(order);
}

const app = vf.vfunc({
  tag: 'div',
  state: {
    status: 'loading', // 'loading' | 'ready' | 'error'
    items: [],
    query: '',
    category: 'all',
    inStock: false,
    sort: 'name'
  },

  render: (s) => {
    if (s.status === 'loading') {
      return vf.html`<p class="pf__note" data-ref="status" role="status">Loading products…</p>`;
    }
    if (s.status === 'error') {
      return vf.html`<p class="pf__note pf__note--error" data-ref="status" role="alert">Could not load products.</p>`;
    }

    const shown = sortedShown(s);
    const countText = shown.length === 1 ? '1 product' : `${shown.length} products`;

    return vf.html`
      <section class="pf">
        <div class="pf__controls">
          <label class="pf__field">
            Search
            <input class="pf__input" type="search" data-action="search" name="search" value="${s.query}">
          </label>
          <label class="pf__field">
            Category
            <select class="pf__input" data-action="category" name="category">
              ${CATEGORIES.map(([value, label]) => vf.html`<option value="${value}" ${s.category === value ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
          <label class="pf__check">
            <input type="checkbox" data-action="in-stock" name="in-stock" ${s.inStock ? 'checked' : ''}>
            In stock only
          </label>
          <label class="pf__field">
            Sort
            <select class="pf__input" data-action="sort" name="sort">
              ${SORTS.map(([value, label]) => vf.html`<option value="${value}" ${s.sort === value ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
        </div>
        <p class="pf__count" data-ref="count">${countText}</p>
        ${shown.length === 0
          ? vf.html`<p class="pf__empty" data-ref="empty">No products found.</p>`
          : vf.html`<ul class="pf__list" data-ref="list">
              ${shown.map((p) => vf.html`
                <li class="pf__item" data-id="${p.id}" data-state="${p.stock === 0 ? 'out' : 'in'}">
                  <span class="pf__name" data-ref="name">${p.name}</span>
                  <span class="pf__price" data-ref="price">${money.format(p.price)}</span>
                  ${p.stock === 0 ? vf.html`<span class="pf__badge">Sold out</span>` : ''}
                </li>
              `)}
            </ul>`
        }
      </section>
    `;
  },

  delegates: [
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent: (e) => { e.sender.query = e.target.value; }
    },
    {
      selector: '[data-action="category"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.category = e.target.value; }
    },
    {
      selector: '[data-action="in-stock"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.inStock = e.target.checked; }
    },
    {
      selector: '[data-action="sort"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.sort = e.target.value; }
    }
  ],

  onMount: (inst) => {
    let alive = true;
    fetch('./data/products.json')
      .then((r) => {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then((data) => {
        if (alive) inst.setState({ items: data, status: 'ready' });
      })
      .catch(() => {
        if (alive) inst.setState({ status: 'error' });
      });
    inst.onDestroy = () => { alive = false; };
  }
});

app.mount('#app');
```

### REPORT.md

```markdown
# 대응표 (React → vfunc.js)

| React | vfunc.js |
|---|---|
| `useState` × 6 | `state: { status, items, query, category, inStock, sort }` |
| `useEffect(fetch, [])` | `onMount`에서 `fetch('./data/products.json')` 후 `setState` |
| `useMemo(shown, …)` | `sortedShown(state)` — 순수 함수, `render` 안에서 매번 계산 |
| `onChange={(e) => setQuery(...)}` 등 4개 | `delegates`의 `input`/`change` 핸들러 4개, `data-action`으로 선택 |
| 조건부 렌더(loading/error/empty) | `render(state)` 안의 분기 |
| `key={p.id}` 목록 | `data-id="${p.id}"` + 컨테이너 위임 하나(행마다 컴포넌트 생성 없음) |
| `Intl.NumberFormat` | 그대로 유지(아래 가정 참고) |
| CSS 원시 값(`#475569`, `12px` 등) | `var(--vf-*)` 토큰으로 교체 |

# 변경/신규 파일
- `index.html` (신규) — CSP 메타, `#app`, 모듈 스크립트 하나
- `app.js` (신규) — vfunc 컴포넌트, fetch, 필터/정렬/렌더
- `styles/product-filter.css` (신규) — 토큰만 참조
- `data/products.json`은 프로젝트에 이미 있어 그대로 사용(수정 없음)

# 가정 (질문하지 않고 직접 판단)
1. **`vf.fmt.currency` 대신 `Intl.NumberFormat`을 직접 사용**: 킷 문서에 따르면 `vf.fmt.*`는 다국어(`vf.i18n`) 절 아래에 있고 로케일 설정에 의존하는 것으로 보입니다. 이 과제는 "앱은 한 언어입니다(i18n 설정 없음)"라고 명시했으므로, `vf.i18n.setup()`을 호출하지 않고 `vf.fmt.currency`를 부르면 문서화되지 않은 상태에서 호출하는 셈이라 판단해 원본 React 코드처럼 `Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' })`를 그대로 유지했습니다. `VERIFY:` i18n 설정 없이 `vf.fmt.currency('USD', 49)`를 호출해도 안전한지 확인 못 함.
2. **디자인 토큰 이름**: 킷의 `llms.ko.txt`는 토큰 예시로 `--vf-color-primary`, `--vf-color-text-muted`, `--vf-space-3`, `--vf-chart-1..8`만 나열합니다. `--vf-color-border`, `--vf-radius-md`, `--vf-radius-full`, `--vf-color-danger`, `--vf-color-danger-bg`, `--vf-font-size-sm`, `--vf-font-size-xs`, `--vf-space-1`, `--vf-space-2`는 같은 명명 규칙을 따라 존재한다고 가정한 것입니다. `VERIFY:` 실제 `vfunc.tokens.css`에 이 토큰들이 모두 정의되어 있는지 확인 못 함(폴더에 파일이 없어 열어볼 수 없었습니다). 없다면 각 토큰을 `styles/product-filter.css`가 아니라 `lib/vfunc.tokens.css`에 준하는 프로젝트 토큰 파일에 추가해야 합니다.
3. **검색 입력의 포커스 유지**: 매 키 입력마다 `query` 상태를 갱신해 전체를 다시 렌더하지만, 입력 요소에 고정된 `name="search"`를 주어 vfunc의 refresh 포커스 복원 규칙(같은 id/data-ref/name 요소로 포커스·커서 복귀)에 맡겼습니다. 별도의 "입력 중엔 렌더 안 함" 처리를 하지 않은 이유는 목록이 매 글자마다 즉시 갱신되어야 하는 요구사항(원본 React의 controlled input과 동일한 실시간 필터링) 때문입니다.
4. **빈 상태 처리**: 상품이 없을 때만 `data-ref="empty"` 요소를 렌더하고, 상품이 있을 때는 아예 DOM에 넣지 않는 방식을 택했습니다(과제 문구의 "없거나 숨김" 중 "없음" 쪽을 선택).
5. **로딩/오류 상태에서는 컨트롤·목록을 렌더하지 않음**: 원본 React 컴포넌트도 로딩/오류일 때 `return`으로 그 문구만 보여주므로 동일하게 처리했습니다.
6. **`onDestroy`를 `onMount` 안에서 할당**: 컴포넌트 옵션에 정적으로 `onDestroy`를 두는 대신, fetch의 `alive` 플래그를 닫는 클로저가 필요해 `onMount` 안에서 `inst.onDestroy = …`로 지정했습니다. `inst`에 직접 속성을 붙이는 것은 금지 규칙(밑줄 시작 이름만 예약)과는 다른, 옵션 자체의 콜백 필드를 런타임에 설정하는 것이라 규칙 위반이 아니라고 판단했습니다. `VERIFY:` `vf.vfunc` 인스턴스가 생성 이후 `onDestroy` 콜백 재할당을 지원하는지 킷 문서에 명시되어 있지 않아 확인 못 함 — 더 안전한 대안은 모듈 스코프 변수(`let alive = true`)를 옵션 객체 바깥에 두고 정적 `onDestroy: () => { alive = false }`로 옵션에 직접 넣는 방식이며, 이 리포트에서 코드를 그렇게 바꾸는 대신 위험을 감수하고 현재 형태로 제출합니다.

# 확인 목록 결과 (지키지 못한 것과 이유 포함)
- [x] 행마다 컴포넌트를 만들지 않고 목록 컨테이너에 델리게이트 없음(목록 자체는 정적 마크업이라 클릭 등 개별 이벤트가 필요 없어 델리게이트를 추가하지 않았습니다 — 원본에도 항목 클릭 동작 없음)
- [x] 모든 마크업이 `vf.html`을 거치고, 데이터로 `innerHTML`을 설정하지 않음
- [x] 셀렉터는 `data-action`/`data-ref`/`id`만 사용, CSS는 `<블록>__<요소>` 클래스
- [x] JS(app.js)에 색·크기·글꼴 값 없음, CSS는 `var(--vf-*)` 토큰만 참조(토큰 이름 자체는 위 가정 2 참고)
- [x] 타이머/구독 없음(리스너는 vfunc의 델리게이트가 자동 해제) — fetch만 있어 `alive` 플래그로 늦은 응답을 무시
- [ ] **콘솔 에러·경고 0개 — 확인 못 함**: 이 환경에서 브라우저를 직접 열어 실행할 수 없어 실제 콘솔 출력을 확인하지 못했습니다. 코드 리뷰 기준으로는 `vf.html` 이스케이프 규칙(따옴표 속성만 보간, `on*`/`srcdoc` 없음)을 지켰고, 개발 빌드의 실수 경고를 유발할 만한 패턴(문자열 `on*`, 따옴표 없는 속성 보간 등)은 넣지 않았습니다.
- [x] 코드·로그·커밋에 비밀값·개인정보 없음(단일 정적 데이터 파일만 사용)
```
