### app.js

```js
// Servers — the page script (an ES module).
import vf from './lib/vfunc.esm.js';

function isHttpUrl(u) {
  return typeof u === 'string' && /^https?:\/\//i.test(u);
}

function badgeFor(status) {
  if (status === 'up') return { state: 'up', label: 'Up' };
  if (status === 'down') return { state: 'down', label: 'Down' };
  if (status === 'maintenance') return { state: 'maintenance', label: 'Maintenance' };
  return { state: 'unknown', label: 'Unknown' };
}

// The feed comes from another team's service: trust nothing about its shape.
function normalizeServer(raw) {
  if (!raw || typeof raw !== 'object') return null;
  const id = raw.id != null ? String(raw.id) : '';
  if (!id) return null;
  return {
    id: id,
    name: raw.name != null ? String(raw.name) : '',
    region: raw.region != null ? String(raw.region) : '',
    status: typeof raw.status === 'string' ? raw.status : '',
    url: typeof raw.url === 'string' ? raw.url : ''
  };
}

function renderLink(server) {
  // Only http(s) URLs become a working link; anything else stays inert (no href).
  if (isHttpUrl(server.url)) {
    return vf.html`<a class="servers__link" data-ref="link" href="${server.url}">Open</a>`;
  }
  return vf.html`<a class="servers__link" data-ref="link">Open</a>`;
}

function renderItem(server) {
  const badge = badgeFor(server.status);
  return vf.html`
    <li class="servers__item" data-id="${server.id}">
      <div class="servers__name" data-ref="name">${server.name}</div>
      <div class="servers__region">${server.region}</div>
      <span class="servers__badge" data-ref="badge" data-state="${badge.state}">${badge.label}</span>
      ${renderLink(server)}
    </li>`;
}

function filterServers(servers, filter) {
  if (filter === 'all') return servers;
  return servers.filter((s) => badgeFor(s.status).state === filter);
}

async function load(inst) {
  inst.setState({ loading: true, error: false });
  try {
    const res = await fetch('./data/servers.json');
    if (!res.ok) throw new Error('http ' + res.status);
    const data = await res.json();
    if (!Array.isArray(data)) throw new Error('invalid payload');
    const servers = data.map(normalizeServer).filter((s) => s !== null);
    inst.setState({ loading: false, error: false, servers: servers });
  } catch (err) {
    inst.setState({ loading: false, error: true, servers: [] });
  }
}

const app = vf.attach(document.getElementById('servers'), {
  state: {
    loading: true,
    error: false,
    servers: [],
    filter: 'all'
  },
  render: (s) => {
    const visible = filterServers(s.servers, s.filter);
    let statusMarkup;
    if (s.loading) {
      statusMarkup = vf.html`Loading servers…`;
    } else if (s.error) {
      statusMarkup = vf.html`Could not load servers. <button type="button" data-action="retry">Retry</button>`;
    } else if (s.servers.length === 0) {
      statusMarkup = vf.html`No servers yet.`;
    } else {
      statusMarkup = vf.html`Showing ${visible.length} of ${s.servers.length}`;
    }

    return vf.html`
      <div class="servers__toolbar">
        <label class="servers__filter-label">
          <span>Status</span>
          <select class="servers__filter" data-action="status-filter">
            <option value="all" ${s.filter === 'all' ? 'selected' : ''}>All</option>
            <option value="up" ${s.filter === 'up' ? 'selected' : ''}>Up</option>
            <option value="down" ${s.filter === 'down' ? 'selected' : ''}>Down</option>
            <option value="maintenance" ${s.filter === 'maintenance' ? 'selected' : ''}>Maintenance</option>
          </select>
        </label>
      </div>
      <div class="servers__status" data-ref="status" role="status">${statusMarkup}</div>
      <ul class="servers__list" data-ref="list">${visible.map(renderItem)}</ul>
    `;
  },
  delegates: [
    { selector: '[data-action="retry"]', eventType: 'click', onEvent: (e) => { load(e.sender); } },
    { selector: '[data-action="status-filter"]', eventType: 'change', onEvent: (e) => {
        e.sender.setState({ filter: e.event.target.value });
      } }
  ],
  onMount: (inst) => {
    inst.$node.setAttribute('aria-busy', inst.state.loading ? 'true' : 'false');
    load(inst);
  },
  onUpdate: (inst) => {
    inst.$node.setAttribute('aria-busy', inst.state.loading ? 'true' : 'false');
  }
});
```

### REPORT.md

## 가정

1. **`Showing X of Y`의 Y**: 필터와 무관하게 이번 불러오기에서 받은 전체 서버 수(정규화 후, 즉 `id`가 있는 유효한 항목 수)로 잡았습니다. X는 현재 필터를 적용한 뒤 보이는 개수입니다.
2. **필드가 없거나 이상한 서버 항목 처리**: `servers.json`은 다른 팀의 서비스가 주는 신뢰할 수 없는 데이터이므로, 최상위가 배열이 아니거나 HTTP/네트워크 요청 자체가 실패하면 "요청 실패" 상태(`Could not load servers.` + Retry)로 처리했습니다. 배열 안의 개별 항목이 객체가 아니거나 `id`가 없으면 그 항목만 조용히 걸러내고 나머지는 정상 표시합니다(항목 하나 때문에 전체를 오류로 만들지 않음). 이 부분은 과제에 명시되지 않아 가장 단순하고 안전한 쪽으로 판단했습니다.
3. **유효하지 않은 `url`**: `http:`/`https:`로 시작하지 않으면 `<a data-ref="link">`에 `href` 속성 자체를 넣지 않았습니다. `vf.safeUrl`은 상대 경로도 허용해 "동작하는 링크"가 될 수 있어(킷 문서: `vf.safeUrl(u) → http/https/mailto/tel/상대 경로 또는 '#'`) 이번 요구사항(“다른 URL은 동작하는 링크가 되면 안 됩니다”)과 맞지 않는다고 판단해, `vf.html`의 자동 `href` 보호에 맡기지 않고 자바스크립트에서 직접 `http(s)` 여부를 검사한 뒤 두 가지 정적 마크업(`href` 있음/없음) 중 하나를 골랐습니다.
4. **Retry 버튼 위치**: 상태 요소(`data-ref="status"`, `role="status"`) 안쪽에 두었습니다(과제에서 "상태 요소 안이나 옆" 둘 다 허용).
5. **디자인 토큰**: 킷의 `llms.ko.txt`에는 토큰 이름 일부만 나열되어 있고("영어판과 같다"고만 함) 전체 목록은 이 번들에 없습니다. 확실히 등장한 토큰(`--vf-color-bg/text/surface/border/text-muted`, `--vf-space-3/4/6`, `--vf-font-body`, `--vf-font-size-2xl`, `--vf-font-weight-strong`, `--vf-line-height`, `--vf-radius-lg`, `--vf-chart-1..8`)만 우선 썼습니다. `--vf-space-1`과 `--vf-font-size-sm`은 기존 스케일(`space-3/4/6`)로 미루어 존재를 추정한 것이라 **VERIFY**: 실제 토큰 파일에 없으면 해당 값만 무시되고(커스텀 속성 특성상 콘솔 오류는 나지 않음) 배지 테두리 두께·작은 글자 크기가 적용되지 않을 뿐, 기능에는 영향이 없습니다. 배지 상태 색은 success/danger/warning류 시맨틱 토큰이 문서에 없어 대신 확인된 `--vf-chart-1/3/5`를 순서대로 up/down/maintenance에 매핑했습니다(정확한 의미 매핑은 **VERIFY**).

## 킷에서 지키지 못한 것

- **6절(텍스트와 포맷) 메시지 키**: 이 작은 프로젝트에는 `locales/{en,ko}.json`이나 `vf.i18n.setup()`이 없고, 과제 지시가 "화면 문구는 과제에 적힌 영어 문구를 그대로 씁니다"라고 못박고 있어 문자열을 `app.js`에 하드코딩했습니다. AGENTS.md의 일반 규칙과는 어긋나지만, 이 과제 범위(로케일 인프라 없음)에서는 과제 지시를 우선했습니다.
