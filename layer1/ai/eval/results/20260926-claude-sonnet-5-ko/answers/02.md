### app.js

```js
// My tasks — the page script. The to-do list will live in #todo.

// Module-scope flag (not an instance property) used to refocus the
// new-task input right after a successful add, regardless of whether
// the add was triggered by Enter (focus stays in the input) or by
// clicking the "Add" button (focus would otherwise move to the button).
var focusInputAfterUpdate = false;

var todoApp = vf.attach(document.getElementById('todo'), {
  state: {
    items: [],
    filter: 'all',
    nextId: 1
  },

  methods: {
    addItem: function (text) {
      var value = (text || '').trim();
      if (!value) {
        return;
      }
      var items = this.state.items.slice();
      items.push({ id: this.state.nextId, text: value, done: false });
      this.setState({ items: items, nextId: this.state.nextId + 1 });
    },
    toggleItem: function (id) {
      var items = this.state.items.map(function (item) {
        if (item.id === id) {
          return { id: item.id, text: item.text, done: !item.done };
        }
        return item;
      });
      this.setState({ items: items });
    },
    removeItem: function (id) {
      var items = this.state.items.filter(function (item) {
        return item.id !== id;
      });
      this.setState({ items: items });
    },
    setFilter: function (filter) {
      this.setState({ filter: filter });
    }
  },

  render: function (state) {
    var visible = state.items.filter(function (item) {
      if (state.filter === 'open') { return !item.done; }
      if (state.filter === 'done') { return item.done; }
      return true;
    });
    var openCount = state.items.filter(function (item) {
      return !item.done;
    }).length;

    return vf.html`
      <form data-action="add" novalidate>
        <label for="todo-new-task">New task</label>
        <input data-ref="input" id="todo-new-task" type="text" name="task" autocomplete="off">
        <button type="submit">Add</button>
      </form>
      <div class="todo__filters" role="group" aria-label="Filter tasks">
        <button type="button" data-action="filter" data-filter="all" aria-pressed="${state.filter === 'all'}">All</button>
        <button type="button" data-action="filter" data-filter="open" aria-pressed="${state.filter === 'open'}">Open</button>
        <button type="button" data-action="filter" data-filter="done" aria-pressed="${state.filter === 'done'}">Done</button>
      </div>
      <p class="todo__summary">Left: <span data-ref="left">${openCount}</span></p>
      <ul data-ref="list" class="todo__list">
        ${visible.map(function (item) {
          return vf.html`<li data-id="${item.id}" data-state="${item.done ? 'done' : 'open'}" class="todo__item">
            <input type="checkbox" data-action="toggle" id="todo-${item.id}" ${item.done ? 'checked' : ''}>
            <label for="todo-${item.id}" data-ref="text">${item.text}</label>
            <button type="button" data-action="remove">Remove</button>
          </li>`;
        })}
      </ul>
      ${visible.length === 0 ? vf.html`<p data-ref="empty">Nothing here.</p>` : ''}
    `;
  },

  delegates: [
    {
      selector: 'form[data-action="add"]',
      eventType: 'submit',
      onEvent: function (e) {
        e.event.preventDefault();
        var value = e.sender.refs.input.value.trim();
        if (!value) {
          return;
        }
        focusInputAfterUpdate = true;
        e.sender.addItem(value);
      }
    },
    {
      selector: '[data-action="toggle"]',
      eventType: 'change',
      onEvent: function (e) {
        var row = e.target.closest('[data-id]');
        if (!row) {
          return;
        }
        e.sender.toggleItem(Number(row.getAttribute('data-id')));
      }
    },
    {
      selector: '[data-action="remove"]',
      eventType: 'click',
      onEvent: function (e) {
        var row = e.target.closest('[data-id]');
        if (!row) {
          return;
        }
        e.sender.removeItem(Number(row.getAttribute('data-id')));
      }
    },
    {
      selector: '[data-action="filter"]',
      eventType: 'click',
      onEvent: function (e) {
        e.sender.setFilter(e.target.getAttribute('data-filter'));
      }
    }
  ],

  onUpdate: function (inst) {
    if (focusInputAfterUpdate) {
      focusInputAfterUpdate = false;
      if (inst.refs.input) {
        inst.refs.input.value = '';
        inst.refs.input.focus();
      }
    }
  }
});
```

### REPORT.md

```markdown
## 가정

- 고유한 `data-id`는 1부터 시작하는 증가 카운터(`state.nextId`)로 만들고, 항목을 지워도 재사용하지 않습니다.
- Enter로 제출하든 `Add` 버튼을 눌러 제출하든 항상 입력칸으로 포커스를 돌려야 한다고 보고, 모듈 스코프 플래그(`focusInputAfterUpdate`) + `onUpdate` 훅으로 명시적으로 입력칸을 비우고 포커스를 줬습니다. (킷 설명대로 refresh가 같은 `id`/`data-ref`를 가진 요소로 자동 포커스를 돌려주는 경우와 겹치더라도, 버튼 클릭으로 제출한 경우까지 확실히 보장하려고 명시적으로 처리했습니다.)
- 체크박스 토글(Space 키 포함) 후 포커스가 같은 체크박스에 남는 것은 킷 문서에 적힌 refresh의 기본 동작(같은 `id`를 가진 요소로 포커스 복원)에 맡기고 별도 코드를 추가하지 않았습니다.
- `data-ref="left"` 요소는 숫자만 담고 있고(`<span data-ref="left">2</span>`), 그 앞에 붙인 "Left:"라는 문구는 과제가 지정하지 않은 자유 문구입니다.
- `data-ref="empty"` 요소는 보일 조건이 아닐 때 아예 렌더하지 않는 쪽(마크업에서 제거)을 골랐습니다. "없거나 숨겨져 있음" 중 "없음" 쪽을 선택했습니다.
- 항목 텍스트 저장 전에 `trim()`으로 앞뒤 공백을 제거하고, 빈 문자열/공백만 있는 입력은 폼 submit 핸들러에서 걸러 추가하지 않습니다.
- 새 입력칸에 `type="text"`, `autocomplete="off"`, `novalidate`(브라우저 기본 검증 대신 직접 빈 값 검사)를 넣었습니다. 과제에 명시되지 않은 세부값이라 단순하고 무난한 쪽을 골랐습니다.
- `index.html`과 `style.css`는 이미 있는 마크업/토큰만으로 충분하다고 보고 고치지 않았습니다(과제에서 `#todo` 안쪽은 `app.js`가 전적으로 그리므로).

## 킷에서 지키지 못한 것

- 없음. `vf.html`로만 동적 마크업을 만들고(사용자 텍스트는 텍스트 위치에서 자동 이스케이프), 이벤트는 모두 `data-action` 셀렉터의 `delegates`로 걸었고, 요소 참조는 `data-ref`/`id`만 썼고, JS에는 색·간격 등 디자인 값을 넣지 않았습니다.

## 확인 목록 결과 (AGENTS.md 10절 기준)

- [x] 모든 동적 값이 `vf.html`을 거친다. 데이터로 `innerHTML`을 설정하지 않는다 — 코드상 확인.
- [x] 셀렉터는 `data-action`/`data-ref`/`id`만. JS에 색·크기가 없다 — 코드상 확인.
- [x] 새로 만든 CSS는 없음(기존 `style.css`를 그대로 씀, 토큰만 사용된 상태 유지).
- [x] 날짜·숫자 포맷이 필요한 값 없음(단순 정수 카운트라 `vf.fmt` 불필요로 판단).
- [x] 이 컴포넌트는 리스너·타이머·구독을 따로 만들지 않아 `onDestroy` 정리가 필요 없음(추가한 리스너는 모두 `delegates`로, 컴포넌트가 destroy될 때 엔진이 해제).
- [ ] 페이지를 열었을 때 콘솔 에러·경고가 0개다 — **확인 못 함**(브라우저 실행 환경이 없어 직접 열어보지 못했습니다. 코드 리뷰상 `vf.html` 규칙 위반, `on*` 속성, 따옴표 없는 보간은 없다고 판단했습니다).
- [x] 코드, 로그, 커밋에 비밀값·토큰·개인정보가 없다.
```
