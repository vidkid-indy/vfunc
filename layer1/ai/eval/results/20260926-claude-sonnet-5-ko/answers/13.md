### app.js

```js
// Files — the page script.
const FILES = [
  { id: 'f1', name: 'report.pdf' },
  { id: 'f2', name: 'photo.png' },
  { id: 'f3', name: 'notes.txt' }
];

const countEl = document.querySelector('[data-ref="count"]');

function countLabel(n) {
  return n === 1 ? '1 file' : n + ' files';
}

const toast = vf.vfToast();

const list = vf.attach(document.getElementById('files'), {
  state: { files: FILES.slice(), focusNext: null },
  render: (s) => vf.html`${s.files.map((file) => vf.html`
    <li class="files__item" data-id="${file.id}">
      <span>${file.name}</span>
      ${vf.vsButton({
        label: 'Delete',
        action: 'delete',
        id: 'delete-' + file.id,
        ariaLabel: 'Delete ' + file.name
      })}
    </li>
  `)}`,
  onMount: (inst) => {
    countEl.textContent = countLabel(inst.state.files.length);
  },
  onUpdate: (inst) => {
    countEl.textContent = countLabel(inst.state.files.length);
    if (inst.state.focusNext) {
      const target = document.getElementById(inst.state.focusNext)
        || document.getElementById('files-title');
      if (target) target.focus();
      inst.state.focusNext = null;
    }
  },
  delegates: [
    {
      selector: '[data-action="delete"]',
      eventType: 'click',
      onEvent: async (e) => {
        const li = e.target.closest('[data-id]');
        if (!li) return;
        const id = li.getAttribute('data-id');
        const files = list.state.files;
        const index = files.findIndex((f) => f.id === id);
        if (index === -1) return;
        const file = files[index];

        const confirmed = await vf.vfConfirm({
          title: 'Delete ' + file.name + '?',
          message: 'This cannot be undone.',
          variant: 'danger',
          confirmLabel: 'Delete'
        }).open();

        if (!confirmed) return;

        const current = list.state.files;
        const currentIndex = current.findIndex((f) => f.id === id);
        if (currentIndex === -1) return;

        const next = current[currentIndex + 1];
        const prev = currentIndex > 0 ? current[currentIndex - 1] : null;
        const focusNext = next ? 'delete-' + next.id : (prev ? 'delete-' + prev.id : 'files-title');

        list.setState({
          files: current.filter((f) => f.id !== id),
          focusNext: focusNext
        });
        toast.show({ message: 'Deleted ' + file.name });
      }
    }
  ]
});
```

### REPORT.md

```markdown
## 가정
- `vf.i18n.setup()`을 호출하지 않았습니다. 이 페이지에는 로케일 파일이 없고 과제도 다국어를 요구하지 않아, `vfConfirm`/`vfToast`의 기본 영어 문구(`confirm.cancel` 등 내장 기본값)가 setup 없이도 나온다고 가정했습니다. 실제로 콘솔 경고가 난다면 `vf.i18n.setup({ locales: ['en'] })`을 앱 최상단에 추가해야 합니다(확인 못 함).
- 취소/Escape/닫기 버튼을 눌렀을 때 포커스가 원래 Delete 버튼으로 돌아가는 것은 `vfModal`/`vfConfirm`(alertdialog)이 자체적으로 트리거 요소로 포커스를 되돌리는 문서상의 동작("포커스를 안에 가두고 되돌리며")에 의존했고, 별도 코드를 추가하지 않았습니다.
- 삭제 확인 후 다음 포커스 대상(다음 파일 → 이전 파일 → `#files-title`)은 `state.focusNext`에 저장해 두었다가 `onUpdate`에서 옮기는 방식으로 구현했습니다. `onUpdate`가 매 렌더 후 호출된다는 문서 설명에 따른 것입니다.
- `<li>`에 `data-id`를 두어 위임 핸들러가 클릭된 버튼에서 `closest('[data-id]')`로 대상 파일을 찾게 했습니다(과제가 명시하지 않은 추가 훅이지만 클래스가 아니라 `data-*`이므로 21조에 어긋나지 않는다고 판단).

## 킷에서 지키지 못한 것

## 확인 목록 결과
- [x] 동적 마크업은 `vf.html`과 `vf.vsButton`/`vf.vfConfirm`만 사용, `innerHTML`에 문자열을 직접 넣지 않음
- [x] 이벤트는 `data-action="delete"`의 delegate 하나로 모든 행 처리, CSS 클래스로 고르지 않음
- [x] 요소 훅은 `id`, `data-ref`, `data-id`만 사용
- [x] JS에 색·간격 값 없음, CSS는 `var(--vf-*)` 토큰만 사용(기존 `style.css` 그대로 유지)
- [x] `window.confirm` 미사용, `vf.vfConfirm`으로 확인
- [x] `vfToast`는 페이지에서 한 번만 생성
- [ ] 실제 브라우저(Chromium/Firefox/WebKit)에서 콘솔 오류·경고 0개, 포커스 이동, `aria-label` 렌더링 결과는 실행해 보지 못해 확인 못 함
```
