### app.js

```js
import vf from './lib/vfunc.esm.js';

// ---------------------------------------------------------------------
// User menu (behaviour 1 & 2): open/close, outside click, Escape, sign out
// ---------------------------------------------------------------------

let removeUserMenuGlobalListeners = null;

function applyUserMenuState(root, s) {
  const btn = root.querySelector('[data-action="menu"]');
  const menu = root.querySelector('#userMenu');
  if (btn) {
    btn.setAttribute('aria-expanded', s.open ? 'true' : 'false');
    if (s.signedOut) btn.textContent = 'Signed out';
  }
  if (menu) {
    if (s.open) menu.removeAttribute('hidden');
    else menu.setAttribute('hidden', '');
  }
}

vf.attach('#topbarUser', {
  state: { open: false, signedOut: false },
  delegates: [
    {
      selector: '[data-action="menu"]',
      eventType: 'click',
      onEvent: (e) => {
        e.event.stopPropagation();
        e.sender.open = !e.sender.state.open;
      }
    },
    {
      selector: '[data-action="sign-out"]',
      eventType: 'click',
      onEvent: (e) => {
        e.event.preventDefault();
        e.sender.setState({ open: false, signedOut: true });
      }
    }
  ],
  onMount() {
    applyUserMenuState(this.$node, this.state);
    const onDocClick = (ev) => {
      if (this.state.open && !this.$node.contains(ev.target)) this.open = false;
    };
    const onKeyDown = (ev) => {
      if (ev.key === 'Escape' && this.state.open) this.open = false;
    };
    document.addEventListener('click', onDocClick);
    document.addEventListener('keydown', onKeyDown);
    removeUserMenuGlobalListeners = () => {
      document.removeEventListener('click', onDocClick);
      document.removeEventListener('keydown', onKeyDown);
    };
  },
  onUpdate(s) {
    applyUserMenuState(this.$node, s);
  },
  onDestroy() {
    if (removeUserMenuGlobalListeners) {
      removeUserMenuGlobalListeners();
      removeUserMenuGlobalListeners = null;
    }
  }
});

// ---------------------------------------------------------------------
// Summary tabs (behaviour 3): switch the two texts from the original script
// ---------------------------------------------------------------------

const SUMMARY_TEXT = {
  week: 'Orders are up this week. Most sales came from the keyboard line.',
  month: 'This month: 1,204 orders, 3.4% refunds.'
};

function applyTabState(root, s) {
  const buttons = root.querySelectorAll('[data-action="tab"]');
  buttons.forEach((btn) => {
    const active = btn.getAttribute('data-tab') === s.tab;
    btn.setAttribute('aria-selected', active ? 'true' : 'false');
    btn.classList.toggle('tabs__tab--active', active);
  });
  const summary = root.querySelector('#summary');
  if (summary) {
    summary.innerHTML = '';
    summary.appendChild(vf.node(vf.html`<p>${SUMMARY_TEXT[s.tab] || ''}</p>`));
  }
}

vf.attach('#summaryPanel', {
  state: { tab: 'week' },
  delegates: [
    {
      selector: '[data-action="tab"]',
      eventType: 'click',
      onEvent: (e) => {
        const tab = e.target.getAttribute('data-tab');
        if (tab && tab !== e.sender.state.tab) e.sender.tab = tab;
      }
    }
  ],
  onMount() { applyTabState(this.$node, this.state); },
  onUpdate(s) { applyTabState(this.$node, s); }
});

// ---------------------------------------------------------------------
// Recent orders: search (4) + status filter (5) + no-match row (6)
// ---------------------------------------------------------------------

const ORDERS = [
  { id: '#1042', customer: 'Minsu Kim', status: 'paid', statusLabel: 'Paid', total: '₩59,000' },
  { id: '#1041', customer: 'Seoyeon Park', status: 'shipped', statusLabel: 'Shipped', total: '₩329,000' },
  { id: '#1040', customer: 'Jiho Choi', status: 'refunded', statusLabel: 'Refunded', total: '₩29,000' }
];

function normalize(value) {
  return String(value || '').trim().toLowerCase();
}

function renderOrdersBody(state) {
  const term = state.term;
  const status = state.status;
  const rows = ORDERS.filter((o) => {
    const matchesTerm = term === ''
      || o.id.toLowerCase().replace(/^#/, '').indexOf(term) !== -1
      || o.customer.toLowerCase().indexOf(term) !== -1;
    const matchesStatus = status === '' || o.status === status;
    return matchesTerm && matchesStatus;
  });

  if (rows.length === 0) {
    return vf.html`<tr><td colspan="4">No orders match.</td></tr>`;
  }

  return vf.html`${rows.map((o) => vf.html`<tr><td>${o.id}</td><td>${o.customer}</td><td><span class="badge badge--${o.status}">${o.statusLabel}</span></td><td class="orders__num">${o.total}</td></tr>`)}`;
}

const ordersTable = vf.attach('#ordersBody', {
  state: { term: '', status: '' },
  render: (s) => renderOrdersBody(s)
});

vf.attach('[data-action="search"]', {
  events: [
    {
      eventType: 'input',
      onEvent: (e) => {
        const term = normalize(e.sender.$node.value).replace(/^#/, '');
        ordersTable.setState({ term });
      }
    }
  ]
});

vf.attach('[data-action="filter"]', {
  events: [
    {
      eventType: 'change',
      onEvent: (e) => {
        ordersTable.setState({ status: e.sender.$node.value });
      }
    }
  ]
});
```

```markdown
# Conversion report — Dashboard

## 1. Area table

| Area (selector) | Static / dynamic | Why | Tool | Data | Events |
|---|---|---|---|---|---|
| `.topbar__logo`, `.sidebar`, `.content__title`, `.kpis`, `.footer` | Static | Text/markup never changes | none | — | — |
| `#topbarUser` (`.topbar__user-btn` + `#userMenu`) | Dynamic (open state, signed-out text) | Menu open/close, Escape, outside click, sign out | `vf.attach` without `render` (adopt; only attributes/text toggled in `onMount`/`onUpdate`) | — | click on `[data-action="menu"]`, `[data-action="sign-out"]`; document click/keydown for outside-click and Escape |
| `#summaryPanel` (tabs + `#summary`) | Dynamic (active tab, summary text) | Tab switch changes text and `aria-selected`/class | `vf.attach` without `render` on the panel; the `#summary` child text is written with `vf.html`/`vf.node` (div context) | Two fixed texts from the original inline script | click on `[data-action="tab"]` |
| `.panel__filter` (status select) | Dynamic (drives table) | Filters rows | `vf.attach` with `events` (root is the select) | — | `change` |
| `.topbar__search` (search input) | Dynamic (drives table) | Filters rows while typing | `vf.attach` with `events` (root is the input) | — | `input` |
| `#ordersBody` (`<tbody>`) | Dynamic (row list) | Content depends on search + status state, and can become the "No orders match." row | `vf.attach` with `render` directly on the `<tbody>` (per llms.txt: attach on a `<tbody>` needs no `tag` override) | The three orders, captured as a small in-file array (`ORDERS`) since there is no server | none directly (driven by search/filter components via `setState`) |

## 2. HTML changes (added attributes / replaced items only)

- `<head>`: added `<meta http-equiv="Content-Security-Policy" ...>` (was missing).
- `.topbar__search`: added `data-action="search"`.
- `.topbar__user`: added `id="topbarUser"`.
- `.topbar__user-btn`: removed `onclick="toggleMenu()"`; added `data-action="menu"`, `aria-expanded="false"`.
- `#userMenu`: removed `style="display:none"`; added `hidden` attribute (see §3).
- Sign-out `<a>`: removed `href="javascript:void(0)"` and `onclick="alert('Signed out')"`; replaced `href` with `"#"`; added `data-action="sign-out"`.
- Wrapping `<section class="panel">` for the summary: added `id="summaryPanel"`.
- Both `.tabs__tab` buttons: removed `onclick="showTab(...)"`; added `data-action="tab"`, `data-tab="week"`/`"month"`, `aria-selected="true"`/`"false"`.
- `.panel__filter` select: added `data-action="filter"`.
- Orders `<tbody>`: added `id="ordersBody"`.
- End of `<body>`: removed the publisher's `<script>...</script>` block; replaced with `<script type="module" src="./app.js"></script>`.

No other markup, classes, texts or ordering were touched. `dashboard.css` is unchanged (not reprinted, per the output rules: files not given stay as they are).

## 3. Removed unsafe items (rule 4)

- `onclick="toggleMenu()"` on the user button — removed, replaced by `data-action="menu"` + delegated `click` in `app.js`.
- `onclick="alert('Signed out')"` and `href="javascript:void(0)"` on "Sign out" — removed; the alert is also disallowed by the task (no real request, no alert). Replaced with `href="#"` (navigation prevented in JS) + `data-action="sign-out"`.
- `onclick="showTab('week'|'month')"` on both tab buttons — removed, replaced by `data-action="tab"` + delegated `click`.
- Inline `style="display:none"` on `#userMenu` — removed (CSP blocks inline `style`), replaced by the `hidden` attribute, toggled in JS.
- The publisher's inline `<script>` block — removed entirely; its two behaviours (`toggleMenu`, `showTab`) are reimplemented in `app.js` using `vf.attach`/delegates instead of globals and inline handlers.

## 4. Suggested token replacements (optional, not applied)

`dashboard.css` was kept byte-for-byte as required, but if tokens are adopted later:
- `#1e3a8a`, `#1d4ed8`, `#93c5fd`, `#eff6ff` → a `--vf-color-primary` family
- `#047857`/`#ecfdf5`, `#b91c1c`/`#fef2f2` → `--vf-color-success`/`-soft`, `--vf-color-danger`/`-soft`
- `#64748b`, `#1e293b`, `#475569` → `--vf-color-text-muted`, `--vf-color-text`
- `#e2e8f0`, `#cbd5e1` → `--vf-color-border`
- `#ffffff`/`#f4f6fb` → `--vf-color-surface`/`--vf-color-bg`
- `0 1px 2px rgba(0,0,0,.06)`, `0 10px 20px rgba(0,0,0,.15)` → `--vf-shadow-1`, `--vf-shadow-3`
- `border-radius: 6/8/10px` → `--vf-radius-sm|md|lg`; the `999px` pill on `.badge` → `--vf-radius-full`

## 5. Assumptions and VERIFY items

- No `store.js`/`api.js`/`pages/*.js` split was used. AGENTS.md's structure targets a multi-page SPA; this task is a single static page with no server and no second screen, so all behaviour lives in one `app.js`. Noted as a deliberate simplification, not a rule break.
- There is no backend, so "no real request" for sign-out is implemented as a pure UI state change (menu closes, button text becomes "Signed out"). `VERIFY:` if a real sign-out endpoint should be called later.
- Assumed `onUpdate` fires after any state change that goes through `setState`/the `inst.key =` shortcut, even when the component has no `render`/`innerHTML` (used for the user-menu and tabs components, which only need attribute/text updates, not a full re-render). This matches the documented refresh/batching model but was not runnable to confirm; if `onUpdate` only fires when `render` is present, the same DOM sync in `onMount` plus recomputing manually inside each delegate handler would be the fallback.
- The three orders were copied into a small `ORDERS` array in `app.js` (same values, order and text as the published table) so the `<tbody>` can be re-rendered through `vf.attach({ render })` — the documented way to build table rows safely, since `vf.node`/`vf.frag` cannot parse `<tr>` outside a table context. The visible values are unchanged, as required.
- The "No orders match." row and all order rows go through `vf.html`, so no `innerHTML` is ever set from data.
- Search matching strips a leading `#` from both the typed term and the order id, and matches by substring on the order number or customer name, case-insensitively, as specified.
```
