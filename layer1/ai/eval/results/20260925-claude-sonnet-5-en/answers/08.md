```js
// Sales — the page script. Chart.js is loaded as the global `Chart`.

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const SERIES = {
  revenue: { label: 'Revenue', values: [120, 135, 150, 142, 168, 181, 175, 190, 205, 198, 220, 236] },
  orders: { label: 'Orders', values: [30, 32, 35, 33, 40, 44, 41, 46, 50, 48, 53, 57] }
};

const CHART_TOKENS = { revenue: '--vf-chart-1', orders: '--vf-chart-2' };

function seriesColor(seriesKey) {
  return getComputedStyle(document.documentElement).getPropertyValue(CHART_TOKENS[seriesKey]).trim();
}

function createSales() {
  // Keep the Chart.js instance in a closure variable, not on the component
  // instance (instance properties are reserved for the engine).
  let chart = null;

  return vf.vfunc({
    tag: 'div',
    state: { series: 'revenue', count: 6 },
    render: (s) => {
      const def = SERIES[s.series];
      const last = def.values[s.count - 1];
      return vf.html`
        <div class="sales__controls">
          <button type="button" class="sales__button" data-action="series" data-series="revenue" aria-pressed="${s.series === 'revenue'}">Revenue</button>
          <button type="button" class="sales__button" data-action="series" data-series="orders" aria-pressed="${s.series === 'orders'}">Orders</button>
          <button type="button" class="sales__button" data-action="add-month" ${s.count >= MONTHS.length ? 'disabled' : ''}>Add month</button>
        </div>
        <p class="sales__summary" data-ref="summary">${def.label}: ${s.count} months, last ${last}</p>
        <div class="sales__chart">
          <canvas data-ref="chart" data-vf-keep="chart" width="480" height="240"></canvas>
        </div>
      `;
    },
    delegates: [
      {
        selector: '[data-action="series"]',
        eventType: 'click',
        onEvent: (e) => {
          const series = e.target.dataset.series;
          if (series && series !== e.sender.state.series) {
            e.sender.setState({ series });
          }
        }
      },
      {
        selector: '[data-action="add-month"]',
        eventType: 'click',
        onEvent: (e) => {
          e.sender.setState((state) => ({ count: Math.min(MONTHS.length, state.count + 1) }));
        }
      }
    ],
    onMount(inst) {
      const s = inst.state;
      const def = SERIES[s.series];
      chart = new Chart(inst.refs.chart.getContext('2d'), {
        type: 'line',
        data: {
          labels: MONTHS.slice(0, s.count),
          datasets: [{
            label: def.label,
            data: def.values.slice(0, s.count),
            borderColor: seriesColor(s.series),
            fill: false
          }]
        },
        options: { responsive: false, maintainAspectRatio: false }
      });
    },
    onUpdate(inst) {
      if (!chart) return;
      const s = inst.state;
      const def = SERIES[s.series];
      chart.data.labels = MONTHS.slice(0, s.count);
      chart.data.datasets[0].label = def.label;
      chart.data.datasets[0].data = def.values.slice(0, s.count);
      chart.data.datasets[0].borderColor = seriesColor(s.series);
      chart.update();
    },
    onDestroy() {
      if (chart) {
        chart.destroy();
        chart = null;
      }
    }
  });
}

let salesInstance = null;

function mountSales() {
  salesInstance = createSales();
  salesInstance.mount('#sales');
}

function hideSales() {
  if (salesInstance) {
    salesInstance.destroy();
    salesInstance = null;
  }
}

const toggle = vf.attach('[data-action="toggle-chart"]', {
  events: [
    {
      eventType: 'click',
      onEvent: (e) => {
        if (salesInstance) {
          hideSales();
          e.sender.$node.textContent = 'Show chart';
        } else {
          mountSales();
          e.sender.$node.textContent = 'Hide chart';
        }
      }
    }
  ]
});

mountSales();
```
