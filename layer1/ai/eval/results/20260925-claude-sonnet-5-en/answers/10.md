### app.js

```js
// Team board — the page script.
import vf from './lib/vfunc.esm.js';
import { store, setTheme, setLang, toggleTask, removeTask } from './store.js';

// Preferences: theme and language.
const prefs = vf.attach('#prefs', {
  render: () => {
    const p = store.get('prefs');
    return vf.html`
      <p class="prefs__row">Theme: <span data-ref="theme">${p.theme}</span>
        <button class="prefs__button" type="button" data-action="theme">Switch theme</button></p>
      <p class="prefs__row">Language: <span data-ref="lang">${p.lang}</span>
        <button class="prefs__button" type="button" data-action="lang" data-lang="en">English</button>
        <button class="prefs__button" type="button" data-action="lang" data-lang="ko">한국어</button></p>`;
  },
  delegates: [
    { selector: '[data-action="theme"]', eventType: 'click', onEvent: () => setTheme(store.get('prefs').theme === 'dark' ? 'light' : 'dark') },
    { selector: '[data-action="lang"]', eventType: 'click', onEvent: (e) => setLang(e.target.getAttribute('data-lang')) }
  ]
});

// Notice: how many tasks are open.
// Fix #1: render only the element's inside (attach parses it as the content
// of the existing #notice section) — it must not repeat the section itself,
// or attach ends up with the section nested inside its own root and a
// duplicate id="notice" once the tree is patched.
const notice = vf.attach('#notice', {
  render: () => {
    const open = store.get('tasks').filter((t) => !t.done).length;
    return vf.html`<p class="notice__text">Open tasks: ${open}</p>`;
  }
});

// Details: a disclosure.
// Fix #2: aria-expanded must be the string "false" when closed, not "".
vf.attach('#details-panel', {
  state: { open: false },
  delegates: [{
    selector: '[data-action="details"]',
    eventType: 'click',
    onEvent: (e) => {
      const open = !e.sender.state.open;
      e.sender.state.open = open;
      e.sender.ids.details.hidden = !open;
      e.target.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
  }],
  onMount: (inst) => inst.$node.querySelector('[data-action="details"]').setAttribute('aria-expanded', 'false')
});

// Tasks, open ones first.
const tasks = vf.attach('#tasks', {
  render: () => vf.html`
    <h2 class="panel__title">Tasks</h2>
    <ul class="tasks">${store.get('tasks').slice().sort((a, b) => a.done - b.done).map((t) => vf.html`
      <li class="task" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
        <input type="checkbox" id="task-toggle-${t.id}" data-action="toggle" aria-label="Done" ${t.done ? 'checked' : ''}>
        <span class="task__title">${t.title}</span>
        <button class="task__delete" type="button" data-action="remove">Remove</button>
      </li>`)}</ul>`,
  delegates: [
    // Fix #7: give the checkbox a per-task id so focus-restore on refresh
    // finds "the same id" instead of falling back to "same data-action at
    // the same position", which points at a different row once the list
    // is re-sorted (open tasks first).
    { selector: '[data-action="toggle"]', eventType: 'change', onEvent: (e) => toggleTask(e.target.closest('[data-id]').getAttribute('data-id')) },
    // Fix #5: the designer renamed the class task__remove to task__delete;
    // selectors must never target a class, so bind on data-action="remove"
    // (already on the button) instead.
    { selector: '[data-action="remove"]', eventType: 'click', onEvent: (e) => removeTask(e.target.closest('[data-id]').getAttribute('data-id')) }
  ]
});

// Comments.
// Fix #6: the previous code built <li> markup by string concatenation and
// wrote it with innerHTML, so a comment like "<b>hi</b>" ran as markup
// instead of showing as text. Comments are now kept in component state and
// rendered through vf.html, which escapes each comment by position.
const comments = vf.attach('#comments', {
  state: { items: [] },
  render: (s) => vf.html`
    <form class="comments__form" data-action="comment">
      <label class="comments__label" for="comment-text">Comment</label>
      <input class="comments__input" id="comment-text" data-ref="comment" autocomplete="off">
      <button class="comments__button" type="submit">Post</button>
    </form>
    <ul class="comments__list" data-ref="comment-list">${s.items.map((c) => vf.html`<li class="comments__item">${c}</li>`)}</ul>`,
  delegates: [{
    selector: '[data-action="comment"]',
    eventType: 'submit',
    onEvent: (e) => {
      e.event.preventDefault();
      const text = e.sender.refs.comment.value.trim();
      if (!text) return;
      e.sender.refs.comment.value = '';
      e.sender.setState({ items: e.sender.state.items.concat(text) });
    }
  }]
});

// Footer: how often the clock has updated.
const footer = vf.attach('#footer', {
  state: { updates: 0 },
  render: (s) => vf.html`Clock updates: <span data-ref="updates">${s.updates}</span>`
});

// Clock, with a close button.
// Fix #4: the timer id was stored as inst._timer (set) but read back as
// inst.timer (typo), so clearInterval always received undefined and the
// interval kept running after "Close clock". Names starting with "_" are
// also reserved for the engine on an instance, so the timer id now lives in
// a closure variable instead of on the instance.
let clockTimerId = null;
const clock = vf.vfunc({
  state: { now: new Date() },
  render: (s) => vf.html`
    <p class="clock">Time: <span data-ref="time">${vf.fmt.date(s.now, { timeStyle: 'medium' })}</span>
      <button class="clock__close" type="button" data-action="close-clock">Close clock</button></p>`,
  delegates: [{ selector: '[data-action="close-clock"]', eventType: 'click', onEvent: (e) => e.sender.destroy() }],
  onMount: (inst) => {
    clockTimerId = setInterval(() => {
      inst.setState({ now: new Date() });
      footer.setState({ updates: footer.state.updates + 1 });
    }, 250);
  },
  onDestroy: () => {
    clearInterval(clockTimerId);
    clockTimerId = null;
  }
});
clock.mount('#clock-slot');

store.subscribe(() => {
  prefs.refresh();
  notice.refresh();
  tasks.refresh();
});
```

### store.js

```js
// Shared state and the only functions that change it.
import vf from './lib/vfunc.esm.js';

export const store = vf.store({
  prefs: { theme: 'light', lang: 'en' },
  tasks: [
    { id: 't1', title: 'Write the release notes', done: false },
    { id: 't2', title: 'Review the login page', done: false },
    { id: 't3', title: 'Update the icons', done: true }
  ]
});

export function setTheme(theme) {
  store.set((s) => ({ prefs: { theme: theme, lang: s.prefs.lang } }));
}

// Fix #3: store.set shallow-merges only at the top level, so replacing
// "prefs" with just { lang } dropped the theme back to whatever the merge
// left (the prefs key itself was overwritten, not merged). Read the
// current theme and keep it, exactly like setTheme keeps the current lang.
export function setLang(lang) {
  store.set((s) => ({ prefs: { theme: s.prefs.theme, lang: lang } }));
}

export function toggleTask(id) {
  store.set((s) => ({ tasks: s.tasks.map((t) => (t.id === id ? { id: t.id, title: t.title, done: !t.done } : t)) }));
}

export function removeTask(id) {
  store.set((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }));
}
```

### REPORT.md

```markdown
# Report — Fix seven bugs

## Cause and fix per report

1. **Notice duplicated / nested (`[vfunc] attach: render returned the target element itself`).** `vf.attach('#notice', { render })` must return only the *inside* of the existing `#notice` element, not the element again. The old `render` returned a whole `<section id="notice">…</section>`, so attach parsed a second `#notice` as the content of the first. Fix: `render` now returns just `<p class="notice__text">Open tasks: …</p>`. `#notice` in `index.html` was already correct and needed no change.
2. **`aria-expanded=""` instead of `"false"`.** Both the click handler and `onMount` wrote `''` for the closed state instead of the string `'false'`. Fixed both call sites to set `'false'`/`'true'`.
3. **Switching language resets the theme.** `setLang` called `store.set({ prefs: { lang } })`. `store.set` shallow-merges only at the top level, so the whole `prefs` object was replaced, dropping `theme`. Fixed by reading `s.prefs.theme` and keeping it, the same pattern `setTheme` already used for `lang`.
4. **Clock keeps counting after "Close clock".** The interval id was stored as `inst._timer` but `onDestroy` read `inst.timer` (typo) — `clearInterval(undefined)` is a no-op, so the timer survived `destroy()`. Also, `_`-prefixed instance properties are reserved for the engine per the kit's rules, so custom instance properties are the wrong place for this regardless. Moved the timer id to a closure variable (`clockTimerId`), set in `onMount`, cleared in `onDestroy`.
5. **Remove buttons do nothing after the class rename.** The delegate selector was `.task__remove` (a CSS class, and the stale one at that — the designer renamed it to `task__delete`). Selectors must never target classes; the button already carries `data-action="remove"`, so the delegate now binds on `[data-action="remove"]` instead. No markup/class change needed.
6. **Comments run as HTML.** The submit handler built `<li>` strings by concatenation and wrote them with `innerHTML`, bypassing all escaping. Rewrote the comments panel as `vf.attach('#comments', { state: { items: [] }, render, delegates })`; the list is now produced through `vf.html`, which escapes each comment by position, so `<b>hi</b>` shows as literal text. Form/input/button markup and all hooks (`id="comment-text"`, `data-ref="comment"`, `data-ref="comment-list"`, `data-action="comment"`) are unchanged.
7. **Focus jumps to a different task's checkbox after Space.** Toggling a task moves it in the sorted (open-first) list. The checkboxes had no `id`/`data-ref`/`name`, so the engine's focus-restore fell back to "same `data-action` at the same position" — a different row after the reorder. Added `id="task-toggle-${t.id}"` to each checkbox so focus-restore matches the same element by id regardless of its new position. `data-action="toggle"` and the visible markup are unchanged.

## Assumptions

- "Open tasks are still listed first" and the console/warning list were read as the authoritative spec; where the task's prose ("follows the tasks") was ambiguous against the reported symptom (a duplicated/nested element with two `id="notice"`), I treated the console warning and the "one element" requirement as primary and did not reorder `#notice` in `index.html`, since the bug report and warning are specifically about duplication, not placement.
- Kept `index.html` and `style.css` unchanged — no reported bug required a markup/CSS change beyond attributes already produced by `app.js`/`store.js`.
- For the comments fix, cleared the input's value before calling `setState` (which batches the re-render to the next tick); since the render markup has no `value="…"` binding, the freshly rendered input has no leftover value either way.

## Checklist

- [x] Every dynamic value goes through `vf.html` / `vf.tpl`; no `innerHTML` from data (comments list now goes through `vf.html`)
- [x] Selectors use `data-action` / `data-ref` / `id` only; no colors or sizes in JS (removed the `.task__remove` class selector)
- [x] New text uses message keys in every locale; dates and numbers use `vf.fmt` — not applicable, no new user-visible text was added
- [x] Listeners, timers and subscriptions are released in `onDestroy` — clock timer now correctly cleared
- [ ] The page opens with no console errors or warnings — not verified (no browser/test runner available in this session); reasoned through each console message in the report table and addressed its stated cause
- [x] No secrets, tokens or personal data in code, logs or commits
```
