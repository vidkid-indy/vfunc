### app.js

```js
// My tasks — the page script: a to-do list rendered inside #todo.

(function () {
  'use strict';

  function makeTask(id, text) {
    return { id: id, text: text, done: false };
  }

  function visibleTasks(state) {
    return state.tasks.filter(function (t) {
      if (state.filter === 'open') return !t.done;
      if (state.filter === 'done') return t.done;
      return true;
    });
  }

  function render(state) {
    var visible = visibleTasks(state);
    var leftCount = state.tasks.filter(function (t) { return !t.done; }).length;

    return vf.html`
      <form data-action="add">
        <label for="todo-input">New task</label>
        <input type="text" id="todo-input" data-ref="input" name="text" autocomplete="off">
        <button type="submit">Add</button>
      </form>
      <div class="todo__filters" role="group" aria-label="Filter tasks">
        <button type="button" data-action="filter" data-filter="all" aria-pressed="${state.filter === 'all'}">All</button>
        <button type="button" data-action="filter" data-filter="open" aria-pressed="${state.filter === 'open'}">Open</button>
        <button type="button" data-action="filter" data-filter="done" aria-pressed="${state.filter === 'done'}">Done</button>
      </div>
      <ul class="todo__list" data-ref="list">
        ${visible.map(function (t) {
          return vf.html`
            <li class="todo__item" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
              <input type="checkbox" data-action="toggle" id="todo-${t.id}" ${t.done ? 'checked' : ''}>
              <label for="todo-${t.id}" data-ref="text">${t.text}</label>
              <button type="button" data-action="remove">Remove</button>
            </li>
          `;
        })}
      </ul>
      <p class="todo__count"><span data-ref="left">${leftCount}</span></p>
      ${visible.length === 0 ? vf.html`<p class="todo__empty" data-ref="empty">Nothing here.</p>` : ''}
    `;
  }

  vf.attach('#todo', {
    state: { tasks: [], filter: 'all', nextId: 1 },
    render: render,
    delegates: [
      {
        selector: 'form[data-action="add"]',
        eventType: 'submit',
        onEvent: function (e) {
          e.event.preventDefault();
          var input = e.sender.refs.input;
          var text = (input.value || '').trim();
          if (!text) return;
          input.value = '';
          input.focus();
          e.sender.setState(function (s) {
            return {
              tasks: s.tasks.concat(makeTask(s.nextId, text)),
              nextId: s.nextId + 1
            };
          });
        }
      },
      {
        selector: '[data-action="toggle"]',
        eventType: 'change',
        onEvent: function (e) {
          var row = e.target.closest('[data-id]');
          var id = Number(row.getAttribute('data-id'));
          var checked = e.target.checked;
          e.sender.setState(function (s) {
            return {
              tasks: s.tasks.map(function (t) {
                if (t.id !== id) return t;
                return { id: t.id, text: t.text, done: checked };
              })
            };
          });
        }
      },
      {
        selector: '[data-action="remove"]',
        eventType: 'click',
        onEvent: function (e) {
          var row = e.target.closest('[data-id]');
          var id = Number(row.getAttribute('data-id'));
          e.sender.setState(function (s) {
            return { tasks: s.tasks.filter(function (t) { return t.id !== id; }) };
          });
        }
      },
      {
        selector: '[data-action="filter"]',
        eventType: 'click',
        onEvent: function (e) {
          var filter = e.target.getAttribute('data-filter');
          e.sender.setState({ filter: filter });
        }
      }
    ]
  });
})();
```

### REPORT.md

## Assumptions

- **No build/approval wait**: per the evaluation conditions, decided everything myself rather than pausing for approval, and record choices here.
- **`index.html` unchanged**: the given `index.html` already has `<section id="todo">`, loads `lib/vfunc.js` (global build) and `app.js`, and its CSP (`script-src 'self'`, no `unsafe-inline`) already fits a file-only script. No edit needed.
- **`vf.attach('#todo', …)` on the existing section** rather than a fresh `vf.vfunc` mounted elsewhere, since the markup hook (`id="todo"`) is already published in the page and AGENTS.md §4.7 recommends `vf.attach` for that case.
- **List markup**: `<ul data-ref="list">` / `<li data-id="…" data-state="…">` — the task only requires "an element" for the list and for each task, so `ul`/`li` was the simplest reasonable choice.
- **Task text element**: used `<label for="todo-<id>" data-ref="text">` instead of a plain `<span>`, so clicking the text also toggles its checkbox (a small accessibility bonus). It still satisfies "the text in an element with `data-ref="text"`".
- **Empty message**: rendered `data-ref="empty"` only when the current filter has no visible tasks (element is entirely absent otherwise), which satisfies "absent or hidden" the simplest way.
- **IDs**: task ids are an in-memory incrementing counter (`state.nextId`, starting at 1); checkbox `id="todo-<id>"` as literally required.
- **Focus after add**: input is cleared and explicitly focused *before* calling `setState`, so that at refresh time the "currently focused" element (matched by `data-ref="input"`) is the input itself; vfunc's refresh restores focus to the new element sharing that `data-ref`. This guarantees focus lands on the input regardless of whether Enter (input already focused) or the button (button focused) triggered the submit.
- **Keyboard toggle focus**: no extra code — each checkbox has a stable, unique `id="todo-<id>"`, and vfunc's refresh restores focus to the element sharing the same `id`, which covers the Space-key toggle case per the kit's documented refresh behavior.
- **Text safety**: task text is only ever interpolated through `vf.html` in a text position, so it is escaped automatically (e.g. `<img src=x onerror=alert(1)>` is shown as literal characters, never executed).

## Deviations from the kit (and why)

- **AGENTS.md §6.1 (message keys / i18n)**: text ("New task", "Add", "All", "Open", "Done", "Remove", "Nothing here.") is hardcoded in `app.js` rather than routed through `vf.t()` / locale files. The task's project files have no `locales/` folder or `vf.i18n.setup()` call, and the request is a single-language page with no i18n mentioned; adding a full i18n setup would be out of scope for this task's simplest reasonable solution. Flagged here as the one rule from AGENTS.md I did not follow.
- Everything else in AGENTS.md/llms.txt (delegation only on `data-action`/`data-ref`/`id`, no CSS-class selectors, `vf.html` for all dynamic markup, no colors/sizes in JS, tokens-only CSS, no inline styles/handlers, releasing nothing since there are no timers/subscriptions to clean up) was followed.
