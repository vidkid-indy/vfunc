### app.js

```javascript
import vf from './lib/vfunc.esm.js';
import store from './store.js';
import homePage from './pages/home.js';
import serversPage from './pages/servers.js';
import serverPage from './pages/server.js';
import notFoundPage from './pages/not-found.js';

const LOCALE_KEY = 'status-board-locale';

function detectLocale() {
  const lang = (navigator.language || 'en').toLowerCase();
  return lang.indexOf('ko') === 0 ? 'ko' : 'en';
}

function readSavedLocale() {
  try {
    return localStorage.getItem(LOCALE_KEY);
  } catch (err) {
    return null;
  }
}

function saveLocale(locale) {
  try {
    localStorage.setItem(LOCALE_KEY, locale);
  } catch (err) {
    // ignore storage errors (private mode, disabled storage, ...)
  }
}

function localeUrl(locale) {
  return new URL('./locales/' + locale + '.json', import.meta.url);
}

let currentLocale = readSavedLocale() || detectLocale();
let currentRouteId = 'home';
let currentPage = null;
let header = null;
let r = null;

document.documentElement.lang = currentLocale;

function mountPage(factory, routeId) {
  return function route(ctx) {
    currentRouteId = routeId;
    if (currentPage) {
      currentPage.destroy();
      currentPage = null;
    }
    currentPage = factory(ctx, r);
    currentPage.mount('#view');
    updateNav();
  };
}

function updateNav() {
  if (!header) return;
  if (currentRouteId === 'home') {
    header.refs.navHome.setAttribute('aria-current', 'page');
    header.refs.navServers.removeAttribute('aria-current');
  } else if (currentRouteId === 'servers' || currentRouteId === 'server') {
    header.refs.navServers.setAttribute('aria-current', 'page');
    header.refs.navHome.removeAttribute('aria-current');
  } else {
    header.refs.navHome.removeAttribute('aria-current');
    header.refs.navServers.removeAttribute('aria-current');
  }
}

function updateFavCount() {
  if (!header) return;
  const el = header.$node.querySelector('[data-ref="fav-count"]');
  if (el) el.textContent = String(store.get('favorites').length);
}

function updateLocaleUI() {
  document.documentElement.lang = currentLocale;
  if (!header) return;
  vf.i18n.apply(header.$node);
  const buttons = vf.$('[data-action="locale"]', header.$node);
  buttons.forEach((btn) => {
    const pressed = btn.getAttribute('data-locale') === currentLocale;
    btn.setAttribute('aria-pressed', pressed ? 'true' : 'false');
  });
}

function switchLocale(locale) {
  if (!locale || locale === currentLocale) return;
  currentLocale = locale;
  saveLocale(locale);
  const result = vf.i18n.set(locale);
  const done = () => {
    updateLocaleUI();
    if (currentPage) currentPage.refresh();
  };
  if (result && typeof result.then === 'function') {
    result.then(done);
  } else {
    done();
  }
}

async function init() {
  await vf.i18n.setup({
    locales: ['en', 'ko'],
    fallback: 'en',
    load: (locale) => fetch(localeUrl(locale)).then((res) => res.json())
  });
  await vf.i18n.set(currentLocale);

  header = vf.attach('#app-header', {
    delegates: [
      {
        selector: '[data-action="locale"]',
        eventType: 'click',
        onEvent: (e) => { switchLocale(e.target.getAttribute('data-locale')); }
      }
    ]
  });

  updateLocaleUI();
  updateFavCount();
  store.subscribe(() => updateFavCount());

  r = vf.router({
    mode: 'hash',
    routes: {
      '/': mountPage(homePage, 'home'),
      '/servers': mountPage(serversPage, 'servers'),
      '/servers/:id': mountPage(serverPage, 'server')
    },
    notFound: mountPage(notFoundPage, 'notFound'),
    linkSelector: 'a[data-link]',
    focus: '#view'
  });

  r.start();
}

init();
```

### store.js

```javascript
import vf from './lib/vfunc.esm.js';

const store = vf.store({ favorites: [] });

export function isFavorite(id) {
  return store.get('favorites').indexOf(id) !== -1;
}

export function toggleFavorite(id) {
  store.set((state) => {
    const has = state.favorites.indexOf(id) !== -1;
    return {
      favorites: has
        ? state.favorites.filter((favId) => favId !== id)
        : state.favorites.concat(id)
    };
  });
}

export default store;
```

```javascript
import vf from '../lib/vfunc.esm.js';
import { fetchServers } from '../api.js';

export default function serversPage(ctx, router) {
  let inst;
  inst = vf.vfunc({
    state: { status: 'loading', servers: [] },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p>${vf.t('servers.loading')}</p>
        `;
      }
      if (s.status === 'error') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p role="alert">${vf.t('servers.error')}</p>
        `;
      }
      if (s.servers.length === 0) {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p>${vf.t('servers.empty')}</p>
        `;
      }
      return vf.html`
        <h1>${vf.t('servers.title')}</h1>
        <ul class="servers__list">
          ${s.servers.map((server) => vf.html`
            <li class="servers__item" data-id="${server.id}">
              <a data-link href="${router.href('/servers/' + server.id)}">${server.name}</a>
            </li>
          `)}
        </ul>
      `;
    },
    onMount: () => {
      fetchServers()
        .then((servers) => { inst.setState({ status: 'ready', servers: servers }); })
        .catch(() => { inst.setState({ status: 'error' }); });
    }
  });
  return inst;
}
```

### pages/server.js

```javascript
import vf from '../lib/vfunc.esm.js';
import { fetchServers } from '../api.js';
import { isFavorite, toggleFavorite } from '../store.js';

export default function serverPage(ctx) {
  const id = ctx.params.id;
  let inst;
  inst = vf.vfunc({
    state: { status: 'loading', server: null },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`<p>${vf.t('servers.loading')}</p>`;
      }
      if (!s.server) {
        return vf.html`<h1>${vf.t('server.missing')}</h1>`;
      }
      const fav = isFavorite(s.server.id);
      return vf.html`
        <h1>${s.server.name}</h1>
        <p class="server__region">${s.server.region}</p>
        <p class="server__status" data-state="${s.server.status}">${s.server.status}</p>
        <button type="button" data-action="favorite" aria-pressed="${fav}">${fav ? vf.t('server.favorite.remove') : vf.t('server.favorite.add')}</button>
      `;
    },
    delegates: [
      {
        selector: '[data-action="favorite"]',
        eventType: 'click',
        onEvent: (e) => {
          if (e.sender.state.server) {
            toggleFavorite(e.sender.state.server.id);
            e.sender.refresh();
          }
        }
      }
    ],
    onMount: () => {
      fetchServers()
        .then((servers) => {
          const found = servers.filter((sv) => sv.id === id)[0] || null;
          inst.setState({ status: 'ready', server: found });
        })
        .catch(() => {
          inst.setState({ status: 'ready', server: null });
        });
    }
  });
  return inst;
}
```

### REPORT.md

```markdown
# Report — SPA scaffold: router, store, two languages

## Assumptions

1. **Router-to-DOM wiring.** `llms.txt` documents `vf.router`'s `routes`/`onChange`/`focus` options but not a container/"mount target" option. I assumed the app (not the router) owns mounting: each route handler is wrapped by `mountPage()` in `app.js`, which destroys the previous page instance and calls `newInstance.mount('#view')`. `focus: '#view'` is passed to the router for its documented focus-after-navigation behaviour.
2. **`vf.attach` on the header without `render`.** I assumed `refs`/`ids` are still populated from the existing `data-ref`/`id` attributes of the attached markup (the kit says attach "keeps" the markup but doesn't explicitly say whether `refs` are scanned without `render`). Used for `navHome`/`navServers`.
3. **Hyphenated `data-ref` value.** The task requires the literal attribute `data-ref="fav-count"`. Since the kit never shows whether `refs` keys are camelCased from dash-case (`refs.favCount`) or kept literal (`refs['fav-count']`), I avoided the ambiguity for this one element and read it with `header.$node.querySelector('[data-ref="fav-count"]')` instead of `header.refs`. All other `data-ref` names I chose myself are hyphen-free (`navHome`, `navServers`) so `refs.*` is used safely there.
4. **Locale persistence.** The kit's `vf.i18n.setup({ persist: true })` doesn't document its storage key or exact fallback-to-browser-language behaviour, and the task's rule is specific ("saved choice, else browser language, survives reload but not a plain in-memory navigation"). I implemented persistence myself with `localStorage` under `status-board-locale`, read once at module load, and call `vf.i18n.set(locale)` explicitly on startup and on every switch, instead of relying on the engine's own `persist` option.
5. **`vf.i18n.set()` return value.** Not documented as sync or async, so `switchLocale()` handles both (awaits it if it looks like a thenable, otherwise proceeds synchronously).
6. **`onMount` binding.** Only `render`'s `this`/argument binding is documented ("this is the instance"); `onMount` isn't shown in any example. To avoid depending on undocumented binding behaviour, `pages/servers.js` and `pages/server.js` capture their own instance in a closure variable (`let inst; inst = vf.vfunc({ onMount: () => { ... inst.setState(...) } })`) instead of using `this`.
7. **Server detail page fields.** No message keys were given for "region"/"status" labels, so I show the raw data values only (`server__region`, `server__status[data-state]`) without inventing new hard-coded label text, keeping "no hard-coded visible text" limited to the keys the task actually specified.
8. **No `components/*.js` files.** The task didn't need a reusable widget beyond the header, which is small enough to wire directly in `app.js`/`index.html`; the folder is simply left unused (per `AGENTS.md`, populating every listed folder isn't mandatory).
9. **One raw CSS value.** `border: 1px solid var(--vf-color-border)` uses a literal `1px` because the documented token list has no border-width token. This is a deliberate, minimal deviation from "CSS reads only `var(--vf-*)` tokens", noted here rather than invented as a fake token.
10. Update plugin (`vf.ext.update`) and `version.json` intentionally left out, per the task's instruction that they are not available in this evaluation.

## Checklist (from prompt-spa-scaffold.md)

- [x] Every route has a page, including not found — `/`, `/servers`, `/servers/:id`, and `notFound` all map to page factories in `pages/`.
- [x] Data pages show loading, error and empty states — `pages/servers.js` has `loading` / `error` / `empty` / `ready`; `pages/server.js` has `loading` and a "not found" state (`server.missing`) for an unknown id (no separate network-error copy for the single-item view, since the table only requires the missing-id case).
- [x] Shared state changes only through `store.js` functions — favorites are only ever changed via `toggleFavorite()`/read via `isFavorite()` in `store.js`; pages import these, never touch the store object's `set` directly.
- [x] No hard-coded visible text; both locale files have every key — every table key is present in both `locales/en.json` and `locales/ko.json`; all page/header text goes through `vf.t()` or `data-i18n`.
- [x] No colors or sizes in JS; CSS uses tokens only — JS never sets colors/fonts/spacing; CSS reads `var(--vf-*)` tokens except the one noted `1px` border-width (assumption 9).
- [ ] No console errors or warnings when you click through every screen — **not verified**: this response was produced without a browser/runtime to execute the app in, so this item could not actually be run against Chromium/Firefox/WebKit. The code was written to match every documented API signature in the kit as closely as possible, but real execution should be checked before shipping.
```
