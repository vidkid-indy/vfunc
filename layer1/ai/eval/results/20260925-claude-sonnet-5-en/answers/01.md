### app.js

```js
import vf from './lib/vfunc.esm.js';

function render(s) {
  const trimmed = s.name.trim();
  const greetName = trimmed || 'stranger';
  return vf.html`
    <section class="widget widget__greeting">
      <label for="name">Your name</label>
      <input type="text" id="name" name="name" value="${s.name}">
      <p data-ref="greeting">Hello, ${greetName}!</p>
    </section>
    <section class="widget widget__counter">
      <p data-ref="count" data-state="${s.count === 0 ? 'zero' : 'positive'}">${s.count}</p>
      <button type="button" data-action="inc">+1</button>
      <button type="button" data-action="dec" ${s.count === 0 ? 'disabled' : ''}>&minus;1</button>
      <button type="button" data-action="reset">Reset</button>
    </section>
  `;
}

const app = vf.vfunc({
  tag: 'div',
  state: { name: '', count: 0 },
  render,
  events: [
    {
      id: 'name',
      eventType: 'input',
      onEvent: (e) => { e.sender.setState({ name: e.event.target.value }); }
    }
  ],
  delegates: [
    {
      selector: '[data-action="inc"]',
      eventType: 'click',
      onEvent: (e) => { e.sender.count = e.sender.count + 1; }
    },
    {
      selector: '[data-action="dec"]',
      eventType: 'click',
      onEvent: (e) => { if (e.sender.count > 0) e.sender.count = e.sender.count - 1; }
    },
    {
      selector: '[data-action="reset"]',
      eventType: 'click',
      onEvent: (e) => { e.sender.count = 0; }
    }
  ]
});

app.mount('#app');
```

- The component root uses the default `tag: 'div'` and is mounted into `#app` with `app.mount('#app')`, matching the exact pattern shown in `llms.txt`'s own example (`counter.mount('#app')`), rather than using `vf.attach` on `#app` directly — `vf.attach` was not required since `index.html` was written from scratch with no pre-existing markup to preserve inside `#app`.
- The minus sign in the `dec` button label is written as the HTML entity `&minus;` (−) inside the static template string (not interpolated data), which is unaffected by `vf.html`'s escaping rules.
