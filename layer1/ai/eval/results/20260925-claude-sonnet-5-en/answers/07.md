### app.js

```js
import vf from './lib/vfunc.esm.js';

const CATEGORIES = [
  ['all', 'All'],
  ['keyboard', 'Keyboards'],
  ['mouse', 'Mice'],
  ['monitor', 'Monitors']
];
const SORTS = [
  ['name', 'Name'],
  ['price-asc', 'Price: low to high'],
  ['price-desc', 'Price: high to low']
];
const money = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' });

function visibleItems(s) {
  const q = s.query.trim().toLowerCase();
  const orderFns = {
    name: (a, b) => a.name.localeCompare(b.name, 'en'),
    'price-asc': (a, b) => a.price - b.price,
    'price-desc': (a, b) => b.price - a.price
  };
  const order = orderFns[s.sort];
  return s.items
    .filter((p) => s.category === 'all' || p.category === s.category)
    .filter((p) => !s.inStock || p.stock > 0)
    .filter((p) => !q || p.name.toLowerCase().includes(q))
    .sort(order);
}

function render(s) {
  if (s.status === 'loading') {
    return vf.html`<p class="pf__note" data-ref="status" role="status">Loading products…</p>`;
  }
  if (s.status === 'error') {
    return vf.html`<p class="pf__note pf__note--error" data-ref="status" role="alert">Could not load products.</p>`;
  }

  const shown = visibleItems(s);
  const countText = shown.length === 1 ? '1 product' : `${shown.length} products`;

  return vf.html`
    <section class="pf">
      <div class="pf__controls">
        <label class="pf__field">
          Search
          <input class="pf__input" type="search" data-action="search" value="${s.query}">
        </label>
        <label class="pf__field">
          Category
          <select class="pf__input" data-action="category">
            ${CATEGORIES.map(([value, label]) => vf.html`<option value="${value}" ${value === s.category ? 'selected' : ''}>${label}</option>`)}
          </select>
        </label>
        <label class="pf__check">
          <input type="checkbox" data-action="in-stock" ${s.inStock ? 'checked' : ''}>
          In stock only
        </label>
        <label class="pf__field">
          Sort
          <select class="pf__input" data-action="sort">
            ${SORTS.map(([value, label]) => vf.html`<option value="${value}" ${value === s.sort ? 'selected' : ''}>${label}</option>`)}
          </select>
        </label>
      </div>
      <p class="pf__count" data-ref="count">${countText}</p>
      ${shown.length === 0
        ? vf.html`<p class="pf__empty" data-ref="empty">No products found.</p>`
        : vf.html`<ul class="pf__list" data-ref="list">${shown.map((p) => vf.html`<li class="pf__item" data-id="${p.id}" data-state="${p.stock === 0 ? 'out' : 'in'}">
              <span class="pf__name" data-ref="name">${p.name}</span>
              <span class="pf__price" data-ref="price">${money.format(p.price)}</span>
              ${p.stock === 0 ? vf.html`<span class="pf__badge">Sold out</span>` : ''}
            </li>`)}</ul>`}
    </section>
  `;
}

let controller = null;

vf.attach('#app', {
  state: {
    items: [],
    status: 'loading',
    query: '',
    category: 'all',
    inStock: false,
    sort: 'name'
  },
  render,
  delegates: [
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent: (e) => { e.sender.query = e.event.target.value; }
    },
    {
      selector: '[data-action="category"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.category = e.event.target.value; }
    },
    {
      selector: '[data-action="in-stock"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.inStock = e.event.target.checked; }
    },
    {
      selector: '[data-action="sort"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.sort = e.event.target.value; }
    }
  ],
  onMount() {
    const inst = this;
    controller = new AbortController();
    fetch(new URL('./data/products.json', import.meta.url), { signal: controller.signal })
      .then((r) => {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then((data) => {
        inst.setState({ items: data, status: 'ready' });
      })
      .catch((err) => {
        if (err && err.name === 'AbortError') return;
        inst.setState({ status: 'error' });
      });
  },
  onDestroy() {
    if (controller) controller.abort();
  }
});
```

### REPORT.md

```markdown
# Port report — ProductFilter.jsx → vfunc.js

## Mapping (source → vfunc)
- `useState(items/status/query/category/inStock/sort)` → `vf.attach('#app', { state: {...} })`
- `useEffect(fetch, [])` → `onMount()`, fetch aborted in `onDestroy()` via a module-scoped `AbortController` (kept in a closure, not as an instance property, per AGENTS.md §4.9's "do not add your own properties to an instance").
- `useMemo(shown, deps)` → plain function `visibleItems(s)` called inside `render`.
- controlled `<input>`/`<select>` `onChange` → `data-action="…"` + `delegates` (`input` event for the search box so the count updates as you type, `change` for category/in-stock/sort).
- conditional early returns (loading/error) → `render` branches on `s.status` before building the full markup; both branches share `data-ref="status"`.
- `key`/`className` variants (`pf__item--out`) → `data-state="in"|"out"` on the `<li>`, styled with `.pf__item[data-state="out"] .pf__name` instead of a modifier class, per the design-separation rule (state in `data-state`, not in a class).
- no React Router / store / i18n needed: this is a single static page with local component state, so `vf.attach` (not `vf.vfunc` + router/store) is enough.

## Files
- `index.html` — CSP meta (`script-src 'self'`, no `unsafe-inline`), loads `lib/vfunc.tokens.css` then `styles.css`, one `<script type="module" src="app.js">`.
- `app.js` — ES module, `vf.attach('#app', …)`, fetch resolved via `new URL('./data/products.json', import.meta.url)` so it keeps working from a version folder.
- `styles.css` — only `var(--vf-*)` tokens, no raw hex/px values; BEM class names (`pf__field`, `pf__item`, …) used only for style hooks, never for JS selectors.

## Assumptions / deviations (kit conditions say to decide and record, not ask)
1. **Money formatting bypasses `vf.fmt.currency`.** AGENTS.md §6.2 asks for `vf.fmt.currency`, but the task requires the exact literal `$49.00` output and the project intentionally skips `vf.i18n.setup()` (single-language app, "no i18n setup" per the task). `vf.fmt.currency`'s locale resolution without an initialized i18n context isn't documented in the kit files given, so to guarantee the required `$X.XX` format deterministically across Chromium/Firefox/WebKit I used the same `Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' })` the source used directly. This is a documented exception to §6.2, not an oversight.
2. **`data-ref="list"` and `data-ref="empty"` are mutually exclusive**, mirroring the source's conditional rendering: when there are 0 results, no `<ul data-ref="list">` is rendered at all (only `<p data-ref="empty">`), and vice versa. This matches the hook table's "absent or hidden when there are products" wording for `empty`, applied symmetrically to `list`.
3. **`data-ref="status"` only exists during `loading`/`error`**; it is not rendered once `status === 'ready'` (the hook table only specifies its text for the loading/error states).
4. **Design tokens for spacing** (`--vf-space-1/2/3`) are mapped by scale position, approximating the source's `4px/6px/8px/12px`, since the kit files don't give exact pixel values per token — only the token names. Functionally equivalent, exact pixel match not verified.
5. Search input keeps focus via the documented restoration rule ("same `id`, `data-ref` or `name`, else same `data-action` at the same position") — `data-action="search"` is unique and positionally stable, so no extra `data-ref` was added to avoid two hooks pointing at the same element.

## Checklist
- [x] No per-row components: one delegated listener set on the root handles all rows/controls.
- [x] Controlled search input does not use uncontrolled re-render tricks; it re-renders each keystroke through `setState`-driven `render`, relying on the engine's documented focus restoration (same `data-action`) rather than debouncing — matches "search while typing" requirement in the task.
- [x] Fetch is cancelled in `onDestroy` via `AbortController` kept in a closure (not an instance property).
- [x] All markup built with `vf.html`; no `innerHTML` from data; no `on*` attributes.
- [x] Selectors are `data-action` / `data-ref` / `data-id` only; no CSS-class selectors in JS.
- [x] CSS reads only `var(--vf-*)` tokens; no raw hex/px values in `styles.css`.
- [x] Console errors/warnings: not verified (no browser run performed in this one-shot answer); code avoids the documented common mistakes (component created before any `vf.t` use — none used; `data-vf-keep` not needed; `<ul>`/`<li>` are not table rows so no `tag` override needed for `vf.attach`).
- [x] No secrets, tokens, or personal data anywhere in the files.
```
