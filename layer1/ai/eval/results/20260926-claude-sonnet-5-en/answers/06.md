### app.js

```javascript
import vf from './lib/vfunc.esm.js';
import { store } from './store.js';
import { createHeader } from './components/header.js';
import homePage from './pages/home.js';
import serversPage from './pages/servers.js';
import serverDetailPage from './pages/server-detail.js';
import notFoundPage from './pages/not-found.js';

let currentLocale = 'en';
let currentPage = null;
let header = null;
let router = null;

function setLocale(next) {
  return vf.i18n.set(next).then((locale) => {
    currentLocale = locale;
    document.documentElement.lang = locale;
    return locale;
  });
}

function mountPage(factory) {
  return (ctx) => {
    if (currentPage) {
      currentPage.destroy();
      currentPage = null;
    }
    currentPage = factory(ctx, router);
    currentPage.mount('#view');
  };
}

async function start() {
  const saved = localStorage.getItem('vf.locale');

  await vf.i18n.setup({
    locales: ['en', 'ko'],
    fallback: 'en',
    load: (locale) => fetch(new URL(`./locales/${locale}.json`, import.meta.url)).then((res) => res.json()),
    persist: true
  });

  vf.i18n.subscribe(() => {
    if (header) header.refresh();
    if (currentPage) currentPage.refresh();
  });

  const initialLocale = saved || (String(navigator.language || '').toLowerCase().indexOf('ko') === 0 ? 'ko' : 'en');
  await setLocale(initialLocale);

  router = vf.router({
    mode: 'hash',
    routes: {
      '/': mountPage(homePage),
      '/servers': mountPage(serversPage),
      '/servers/:id': mountPage(serverDetailPage)
    },
    notFound: mountPage(notFoundPage),
    linkSelector: 'a[data-link]',
    focus: '#view',
    onChange: () => { if (header) header.refresh(); }
  });

  router.start();

  header = createHeader({
    store,
    router,
    getLocale: () => currentLocale,
    onLocaleChange: setLocale
  });
  header.mount('#app-header');

  store.subscribe(() => { if (header) header.refresh(); });
}

start();
```

### store.js

```javascript
import vf from './lib/vfunc.esm.js';

export const store = vf.store({ favorites: [] });

export function toggleFavorite(id) {
  store.set((state) => {
    const has = state.favorites.indexOf(id) !== -1;
    return {
      favorites: has
        ? state.favorites.filter((favoriteId) => favoriteId !== id)
        : state.favorites.concat(id)
    };
  });
}
```

### api.js

```javascript
const SERVERS_URL = new URL('./data/servers.json', import.meta.url);

export function fetchServers() {
  return fetch(SERVERS_URL).then((res) => {
    if (!res.ok) throw new Error('Failed to load servers');
    return res.json();
  });
}

export function fetchServer(id) {
  return fetchServers().then((list) => list.find((server) => server.id === id) || null);
}
```

```javascript
import vf from '../lib/vfunc.esm.js';

export function createHeader(opts) {
  const { store, router, getLocale, onLocaleChange } = opts;

  return vf.attach('#app-header', {
    render: () => {
      const favorites = store.get('favorites');
      const current = router.current();
      const path = (current && current.path) || '/';
      const locale = getLocale();
      const homeCurrent = path === '/';
      const serversCurrent = path === '/servers' || path.indexOf('/servers/') === 0;

      return vf.html`
        <nav class="header__nav">
          <a data-link href="#/" aria-current="${homeCurrent ? 'page' : 'false'}">${vf.t('nav.home')}</a>
          <a data-link href="#/servers" aria-current="${serversCurrent ? 'page' : 'false'}">${vf.t('nav.servers')}</a>
        </nav>
        <div class="header__favorites">
          <span class="header__favorites-label">${vf.t('favorites.label')}</span>
          <span data-ref="fav-count">${favorites.length}</span>
        </div>
        <div class="header__locale">
          <button type="button" data-action="locale" data-locale="en" aria-pressed="${locale === 'en' ? 'true' : 'false'}">English</button>
          <button type="button" data-action="locale" data-locale="ko" aria-pressed="${locale === 'ko' ? 'true' : 'false'}">한국어</button>
        </div>
      `;
    },
    delegates: [
      {
        selector: '[data-action="locale"]',
        eventType: 'click',
        onEvent: (e) => {
          const next = e.target.getAttribute('data-locale');
          onLocaleChange(next);
        }
      }
    ]
  });
}
```

### pages/servers.js

```javascript
import vf from '../lib/vfunc.esm.js';
import { fetchServers } from '../api.js';

export default function serversPage() {
  return vf.vfunc({
    tag: 'div',
    state: { status: 'loading', servers: [] },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p>${vf.t('servers.loading')}</p>
        `;
      }
      if (s.status === 'error') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p role="alert">${vf.t('servers.error')}</p>
        `;
      }
      if (s.servers.length === 0) {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p>${vf.t('servers.empty')}</p>
        `;
      }
      return vf.html`
        <h1>${vf.t('servers.title')}</h1>
        <ul class="servers__list">
          ${s.servers.map((server) => vf.html`
            <li class="servers__item" data-id="${server.id}">
              <a data-link href="#/servers/${server.id}">${server.name}</a>
            </li>
          `)}
        </ul>
      `;
    },
    onMount(inst) {
      fetchServers()
        .then((servers) => { inst.setState({ status: 'ready', servers }); })
        .catch(() => { inst.setState({ status: 'error' }); });
    }
  });
}
```

### pages/server-detail.js

```javascript
import vf from '../lib/vfunc.esm.js';
import { fetchServer } from '../api.js';
import { store, toggleFavorite } from '../store.js';

export default function serverDetailPage(ctx) {
  const id = ctx.params.id;
  let unsubscribe = null;

  return vf.vfunc({
    tag: 'div',
    state: { status: 'loading', server: null },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p>${vf.t('servers.loading')}</p>
        `;
      }
      if (s.status === 'missing') {
        return vf.html`<h1>${vf.t('server.missing')}</h1>`;
      }
      const favored = store.get('favorites').indexOf(id) !== -1;
      const label = favored ? vf.t('server.favorite.remove') : vf.t('server.favorite.add');
      return vf.html`
        <h1>${s.server.name}</h1>
        <p>${s.server.region} — ${s.server.status}</p>
        <button type="button" data-action="favorite" aria-pressed="${favored ? 'true' : 'false'}">${label}</button>
      `;
    },
    delegates: [
      {
        selector: '[data-action="favorite"]',
        eventType: 'click',
        onEvent: () => { toggleFavorite(id); }
      }
    ],
    onMount(inst) {
      fetchServer(id)
        .then((server) => { inst.setState({ status: server ? 'ready' : 'missing', server }); })
        .catch(() => { inst.setState({ status: 'missing', server: null }); });
      unsubscribe = store.subscribe(() => { inst.refresh(); });
    },
    onDestroy() {
      if (unsubscribe) {
        unsubscribe();
        unsubscribe = null;
      }
    }
  });
}
```

```markdown
## Screen list and state map

| Route | Page file | Purpose | Data | Actions |
|---|---|---|---|---|
| `#/` | `pages/home.js` | Landing intro | none | none |
| `#/servers` | `pages/servers.js` | List servers | `data/servers.json` via `api.js` | navigate to a server |
| `#/servers/:id` | `pages/server-detail.js` | One server | one server record via `api.js` | toggle favorite |
| anything else | `pages/not-found.js` | Not found | none | none |

Shared state (`store.js`, `vf.store`): `favorites: string[]`, changed only through `toggleFavorite(id)`. Survives navigation and locale switch (in-memory), not a reload (no persistence requested for it — only the locale choice persists, per the task).

Local state: `servers.js` (`status: 'loading'|'error'|'ready'`, `servers`), `server-detail.js` (`status: 'loading'|'ready'|'missing'`, `server`). The current locale and current route are read from `vf.i18n` / the router at render time, not duplicated into any component's own state.

## Message keys (both locales)

All keys from the task's table are in `locales/en.json` and `locales/ko.json`: `nav.home`, `nav.servers`, `favorites.label`, `home.title`, `home.intro`, `servers.title`, `servers.loading`, `servers.error`, `servers.empty`, `server.favorite.add`, `server.favorite.remove`, `server.missing`, `notFound.title`.

## Files

Given above in full: `index.html`, `app.js`, `store.js`, `api.js`, `components/header.js`, `pages/home.js`, `pages/servers.js`, `pages/server-detail.js`, `pages/not-found.js`, `locales/en.json`, `locales/ko.json`, `styles/base.css`, `styles/components/header.css`, `styles/pages/home.css`, `styles/pages/servers.css`. `data/servers.json` is unchanged from the one given in the task, so it is not repeated here.

## Assumptions

1. **Initial locale detection.** `llms.txt` documents `vf.i18n.setup({ persist: true })` as reading a previously saved choice, but not an option for browser-language detection on first visit. I read `localStorage.getItem('vf.locale')` myself (read-only, before `setup()`) only to decide whether a saved choice exists; if not, I derive `en`/`ko` from `navigator.language` and pass it to `vf.i18n.set(...)` right after `setup()` resolves, which is the documented way to change and persist the locale. This matches "follow the browser language" without writing to `localStorage` directly.
2. **`router.current()` before the router starts.** The header is created and mounted only after `router.start()`, so its first render already reads the real matched route instead of guessing `/`.
3. **No `vf.ext.update` / `version.json`.** Left out entirely, as the task says the update plugin is not available in this evaluation.
4. **No favorites persistence across reload.** The task explicitly says favorites survive navigation and a language switch, "not a reload", so `store.js` keeps them in memory only (no `localStorage`).
5. **Header has no component-local state.** It re-renders through explicit `refresh()` calls from the router's `onChange`, the store's `subscribe`, and `vf.i18n.subscribe`, since everything it shows (route, favorites count, locale) is external state rather than its own — this follows the `vf.attach` pattern in the kit rather than duplicating that data into a `state` object.
6. **CSP.** `script-src 'self'` / `style-src 'self'` with no `'unsafe-inline'`; no CDN scripts are used, so no SRI/`crossorigin` entries were needed.

Nothing from the kit's must rules was knowingly broken.

- [x] Shared state changes only through `store.js` functions — only `toggleFavorite` mutates `store`; everything else calls `store.get`/`store.subscribe`.
- [x] No hard-coded visible text; both locale files have every key — checked against the message table; both JSON files have matching key shapes.
```
