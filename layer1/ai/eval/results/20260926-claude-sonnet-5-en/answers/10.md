### app.js

```js
// Team board — the page script.
import vf from './lib/vfunc.esm.js';
import { store, setTheme, setLang, toggleTask, removeTask } from './store.js';

// Preferences: theme and language.
const prefs = vf.attach('#prefs', {
  render: () => {
    const p = store.get('prefs');
    return vf.html`
      <p class="prefs__row">Theme: <span data-ref="theme">${p.theme}</span>
        <button class="prefs__button" type="button" data-action="theme">Switch theme</button></p>
      <p class="prefs__row">Language: <span data-ref="lang">${p.lang}</span>
        <button class="prefs__button" type="button" data-action="lang" data-lang="en">English</button>
        <button class="prefs__button" type="button" data-action="lang" data-lang="ko">한국어</button></p>`;
  },
  delegates: [
    { selector: '[data-action="theme"]', eventType: 'click', onEvent: () => setTheme(store.get('prefs').theme === 'dark' ? 'light' : 'dark') },
    { selector: '[data-action="lang"]', eventType: 'click', onEvent: (e) => setLang(e.target.getAttribute('data-lang')) }
  ]
});

// Notice: how many tasks are open.
// Bug 1 fix: render must return only the inside of #notice, not the
// section/id again — returning the target's own id created a nested
// duplicate "#notice" on every refresh (and vf.attach warned about it).
const notice = vf.attach('#notice', {
  render: () => {
    const open = store.get('tasks').filter((t) => !t.done).length;
    return vf.html`<p class="notice__text">Open tasks: ${open}</p>`;
  }
});

// Details: a disclosure.
vf.attach('#details-panel', {
  state: { open: false },
  delegates: [{
    selector: '[data-action="details"]',
    eventType: 'click',
    onEvent: (e) => {
      const open = !e.sender.state.open;
      e.sender.state.open = open;
      e.sender.ids.details.hidden = !open;
      // Bug 2 fix: aria-expanded must be the string "false" when closed,
      // not an empty string.
      e.target.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
  }],
  onMount: (inst) => inst.$node.querySelector('[data-action="details"]').setAttribute('aria-expanded', 'false')
});

// Tasks, open ones first.
const tasks = vf.attach('#tasks', {
  render: () => vf.html`
    <h2 class="panel__title">Tasks</h2>
    <ul class="tasks">${store.get('tasks').slice().sort((a, b) => a.done - b.done).map((t) => vf.html`
      <li class="task" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
        <input type="checkbox" id="task-toggle-${t.id}" data-action="toggle" aria-label="Done" ${t.done ? 'checked' : ''}>
        <span class="task__title">${t.title}</span>
        <button class="task__delete" type="button" data-action="remove">Remove</button>
      </li>`)}</ul>`,
  delegates: [
    // Bug 7 fix: each checkbox now has a stable per-task `id`
    // ("task-toggle-<id>"), so vfunc's focus-restore after a re-render
    // finds the same element by id instead of falling back to "the
    // Nth data-action at the same position" — which pointed at a
    // different task once the list re-sorted after a toggle.
    { selector: '[data-action="toggle"]', eventType: 'change', onEvent: (e) => toggleTask(e.target.closest('[data-id]').getAttribute('data-id')) },
    // Bug 5 fix: select by data-action, not by a CSS class. The
    // designer renaming task__remove -> task__delete broke the old
    // `.task__remove` selector; data-action="remove" is the real hook
    // and was in the markup all along.
    { selector: '[data-action="remove"]', eventType: 'click', onEvent: (e) => removeTask(e.target.closest('[data-id]').getAttribute('data-id')) }
  ]
});

// Comments.
// Bug 6 fix: comments were inserted with hand-built HTML strings via
// innerHTML (`'<li class="comments__item">' + c + '</li>'`), so a
// comment like `<b>hi</b>` — or a `<script>` — ran as markup instead of
// showing as text. The list is now a small vfunc component whose
// `render` builds markup with `vf.html`, which escapes the comment
// text by position.
const commentList = vf.attach('[data-ref="comment-list"]', {
  state: { items: [] },
  render: (s) => vf.html`${s.items.map((c) => vf.html`<li class="comments__item">${c}</li>`)}`
});

vf.attach('#comments', {
  delegates: [{
    selector: '[data-action="comment"]',
    eventType: 'submit',
    onEvent: (e) => {
      e.event.preventDefault();
      const text = e.sender.refs.comment.value.trim();
      if (!text) return;
      commentList.setState((s) => ({ items: s.items.concat(text) }));
      e.sender.refs.comment.value = '';
    }
  }]
});

// Footer: how often the clock has updated.
const footer = vf.attach('#footer', {
  state: { updates: 0 },
  render: (s) => vf.html`Clock updates: <span data-ref="updates">${s.updates}</span>`
});

// Clock, with a close button.
// Bug 4 fix: the timer id was stored as `inst._timer` (a name starting
// with "_", which AGENTS.md reserves for the engine and which the
// engine may not even keep around) but released as `inst.timer`
// (no underscore) — clearInterval(undefined) does nothing, so the
// timer, and the footer's "Clock updates" counter, kept running after
// close. Keeping the timer id in a closure variable (as AGENTS.md
// recommends) instead of on the instance fixes the mismatch.
let clockTimer = null;
const clock = vf.vfunc({
  state: { now: new Date() },
  render: (s) => vf.html`
    <p class="clock">Time: <span data-ref="time">${vf.fmt.date(s.now, { timeStyle: 'medium' })}</span>
      <button class="clock__close" type="button" data-action="close-clock">Close clock</button></p>`,
  delegates: [{ selector: '[data-action="close-clock"]', eventType: 'click', onEvent: (e) => e.sender.destroy() }],
  onMount: (inst) => {
    clockTimer = setInterval(() => {
      inst.setState({ now: new Date() });
      footer.setState({ updates: footer.state.updates + 1 });
    }, 250);
  },
  onDestroy: () => {
    clearInterval(clockTimer);
    clockTimer = null;
  }
});
clock.mount('#clock-slot');

store.subscribe(() => {
  prefs.refresh();
  notice.refresh();
  tasks.refresh();
});
```

### store.js

```js
// Shared state and the only functions that change it.
import vf from './lib/vfunc.esm.js';

export const store = vf.store({
  prefs: { theme: 'light', lang: 'en' },
  tasks: [
    { id: 't1', title: 'Write the release notes', done: false },
    { id: 't2', title: 'Review the login page', done: false },
    { id: 't3', title: 'Update the icons', done: true }
  ]
});

export function setTheme(theme) {
  store.set((s) => ({ prefs: { theme: theme, lang: s.prefs.lang } }));
}

// Bug 3 fix: `store.set({ prefs: { lang: lang } })` shallow-merges only
// the store's *top-level* keys, so it replaced the whole `prefs`
// object with `{ lang }` and silently dropped `theme` back to nothing
// (rendered as the initial "light"). Read the current theme and carry
// it forward, the same way setTheme() already carries the language
// forward.
export function setLang(lang) {
  store.set((s) => ({ prefs: { theme: s.prefs.theme, lang: lang } }));
}

export function toggleTask(id) {
  store.set((s) => ({ tasks: s.tasks.map((t) => (t.id === id ? { id: t.id, title: t.title, done: !t.done } : t)) }));
}

export function removeTask(id) {
  store.set((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }));
}
```

### REPORT.md

```markdown
# Report — Fix seven bugs

## Assumptions
- No files besides `app.js` and `store.js` needed changes. `index.html` and `style.css` were already correct for the fixed behaviour (the `#notice` section, the `details` panel and its `hidden`/`aria-controls` wiring, and the token-only CSS did not need edits) — output rules say files not given stay as they are, so they are omitted above.
- For bug 7 (focus jumping on Space), the fix adds a stable per-task `id` (`task-toggle-<taskId>`) to each checkbox, since the task list re-sorts on every toggle. This is the "new attribute" the task description allows when a fix needs one. No existing hook, id, `data-action`, `data-ref` or visible text was renamed or removed.
- For bug 6 (comment XSS), the comment list markup is now produced by a small `vf.attach` component with `render` returning `vf.html` output, instead of hand-built strings joined into `innerHTML`. The list's own hook (`data-ref="comment-list"`), the form's hook (`data-action="comment"`) and the visible texts are unchanged.
- For bug 4 (clock timer leak), the timer id moved from `inst._timer` (an instance property, which AGENTS.md §4.9 reserves for the engine and warns against) to a module-level closure variable, per AGENTS.md's own guidance ("keep listener functions and timers in a closure or module variable"). This also happens to be the actual bug: `onDestroy` was clearing `inst.timer`, a different (always-undefined) property from the one `onMount` set (`inst._timer`), so the interval was never cleared.

## Bugs and fixes
| # | Cause | Fix |
|---|---|---|
| 1 | `#notice`'s `render` returned a whole `<section id="notice">…</section>`, i.e. the target element's own id again; `vf.attach` parses `render`'s return value as the *inside* of the target, so this nested a second `#notice` inside the first on every refresh. | `render` now returns only `<p class="notice__text">Open tasks: …</p>`. |
| 2 | The click handler set `aria-expanded` to `''` when closing (`open ? 'true' : ''`), and `onMount` initialised it to `''` too. | Both now use the literal string `'false'` when closed. |
| 3 | `setLang` did `store.set({ prefs: { lang } })`, replacing the whole `prefs` object and dropping `theme`. | `setLang` now reads `s.prefs.theme` and carries it forward, mirroring `setTheme`. |
| 4 | `onMount` stored the interval id on `inst._timer`; `onDestroy` cleared `inst.timer` (no underscore) — a different, always-`undefined` property, so `clearInterval` did nothing and the footer counter kept incrementing after "Close clock". | Timer id moved to a closure variable (`clockTimer`) set in `onMount` and cleared in `onDestroy`. |
| 5 | The delegate for Remove selected `.task__remove`, a CSS class. The designer's rename to `task__delete` (a pure style change) silently broke the click handler, because behaviour was wired to a class instead of a `data-*` hook. | Delegate selector changed to `[data-action="remove"]`, the hook that was already present in the markup and untouched by the rename. |
| 6 | Comments were rendered with `'<li class="comments__item">' + c + '</li>'` assigned to `innerHTML` — plain string concatenation, no escaping, so `<b>hi</b>` and `<script>` both ran as markup. | Comment list is now a `vf.attach` component whose `render` builds each `<li>` with `vf.html`, which escapes the comment text by position. |
| 7 | Checkboxes had no `id`/`data-ref`/`name`, only `data-action="toggle"` — vfunc's focus-restore falls back to "the same `data-action` at the same position" when there's no id, and toggling a task reorders the list (open tasks first), so position no longer points at the same task. | Each checkbox now gets a stable `id="task-toggle-<taskId>"`, so focus restore finds the *same* checkbox regardless of where the task moved. |

## Checklist
- [x] Every dynamic value goes through `vf.html` / `vf.tpl`; no `innerHTML` from data — fixed for the comment list (bug 6); nothing else used `innerHTML` from data.
- [x] Selectors use `data-action` / `data-ref` / `id` only; no colors or sizes in JS — fixed the Remove delegate (bug 5); `[data-ref="comment-list"]` is an attribute selector on a behaviour hook, not a class.
- [x] New CSS files too use only `var(--vf-*)` tokens; hex, `rgb()` and `px` values live only in `styles/tokens.css` — no CSS was added or changed.
- [x] New text uses message keys in every locale; dates and numbers use `vf.fmt` — no new user-visible text was introduced; the clock already used `vf.fmt.date`.
- [x] Listeners, timers and subscriptions are released in `onDestroy` — clock's `onDestroy` now clears the correct timer id.
- [ ] The page opens with no console errors or warnings — not verified (no browser available in this environment); reasoned through each reported console message and symptom instead. Bug 1's specific warning ("attach: render returned the target element itself") is addressed by no longer returning the target's own id/section from `render`.
- [x] No secrets, tokens or personal data in code, logs or commits — none introduced.
```
