### app.js

```js
import vf from './lib/vfunc.esm.js';

const CATEGORIES = [
  ['all', 'All'],
  ['keyboard', 'Keyboards'],
  ['mouse', 'Mice'],
  ['monitor', 'Monitors']
];

const SORTS = [
  ['name', 'Name'],
  ['price-asc', 'Price: low to high'],
  ['price-desc', 'Price: high to low']
];

function sortItems(items, sort) {
  const copy = items.slice();
  if (sort === 'price-asc') {
    copy.sort((a, b) => a.price - b.price);
  } else if (sort === 'price-desc') {
    copy.sort((a, b) => b.price - a.price);
  } else {
    copy.sort((a, b) => a.name.localeCompare(b.name, 'en'));
  }
  return copy;
}

function shownItems(s) {
  const q = s.query.trim().toLowerCase();
  const filtered = s.items
    .filter((p) => s.category === 'all' || p.category === s.category)
    .filter((p) => !s.inStock || p.stock > 0)
    .filter((p) => !q || p.name.toLowerCase().includes(q));
  return sortItems(filtered, s.sort);
}

const productFilter = vf.vfunc({
  tag: 'div',
  state: {
    items: [],
    status: 'loading',
    query: '',
    category: 'all',
    inStock: false,
    sort: 'name'
  },
  render(s) {
    if (s.status === 'loading') {
      return vf.html`<p class="pf__note" role="status" data-ref="status">Loading products…</p>`;
    }
    if (s.status === 'error') {
      return vf.html`<p class="pf__note pf__note--error" role="alert" data-ref="status">Could not load products.</p>`;
    }

    const shown = shownItems(s);

    return vf.html`
      <section class="pf">
        <div class="pf__controls">
          <label class="pf__field">
            Search
            <input class="pf__input" type="search" data-action="search" data-ref="searchInput" value="${s.query}">
          </label>
          <label class="pf__field">
            Category
            <select class="pf__input" data-action="category">
              ${CATEGORIES.map(([value, label]) => vf.html`<option value="${value}" ${value === s.category ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
          <label class="pf__check">
            <input type="checkbox" data-action="in-stock" ${s.inStock ? 'checked' : ''}>
            In stock only
          </label>
          <label class="pf__field">
            Sort
            <select class="pf__input" data-action="sort">
              ${SORTS.map(([value, label]) => vf.html`<option value="${value}" ${value === s.sort ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
        </div>
        <p class="pf__count" data-ref="count">${shown.length === 1 ? '1 product' : `${shown.length} products`}</p>
        ${shown.length === 0
          ? vf.html`<p class="pf__empty" data-ref="empty">No products found.</p>`
          : vf.html`<ul class="pf__list" data-ref="list">
              ${shown.map((p) => vf.html`
                <li class="pf__item${p.stock === 0 ? ' pf__item--out' : ''}" data-id="${p.id}" data-state="${p.stock === 0 ? 'out' : 'in'}">
                  <span class="pf__name" data-ref="name">${p.name}</span>
                  <span class="pf__price" data-ref="price">${vf.fmt.currency(p.price, 'USD')}</span>
                  ${p.stock === 0 ? vf.html`<span class="pf__badge">Sold out</span>` : ''}
                </li>
              `)}
            </ul>`}
      </section>
    `;
  },
  delegates: [
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent: (e) => { e.sender.query = e.target.value; }
    },
    {
      selector: '[data-action="category"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.category = e.target.value; }
    },
    {
      selector: '[data-action="in-stock"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.inStock = e.target.checked; }
    },
    {
      selector: '[data-action="sort"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.sort = e.target.value; }
    }
  ],
  onMount(inst) {
    fetch(new URL('./data/products.json', import.meta.url))
      .then((r) => {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then((data) => {
        inst.setState({ items: data, status: 'ready' });
      })
      .catch(() => {
        inst.setState({ status: 'error' });
      });
  }
});

productFilter.mount('#app');
```

# Report — 07-react-port

## Assumptions

- **No i18n setup**: the task says the app has one language and to keep the English texts, so no `vf.i18n` calls are used; texts are written directly in `render`, matching the rule "a component renders when it is created" trivially since there is no locale wait.
- **Money formatting**: used `vf.fmt.currency(p.price, 'USD')`, assuming (from `llms.txt`'s `vf.fmt.currency(n, 'KRW')` example) that it produces the same `Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' })` output as the source (`$49.00`). I could not run the page to confirm the exact string vfunc's formatter produces for USD.
- **CSP policy**: chose a conservative `default-src 'self'; script-src 'self'; style-src 'self'; img-src 'self'; connect-src 'self'; base-uri 'none'; form-action 'self';` since the kit only requires no `'unsafe-inline'` in `script-src` and did not specify exact directives.
- **Design tokens**: mapped the source's raw colors/spacing to the closest documented tokens (`--vf-color-text-muted`, `--vf-color-danger-text`, `--vf-color-danger-soft`, `--vf-color-border`, `--vf-color-text-disabled`, `--vf-space-1..3`, `--vf-radius-sm|full`, `--vf-font-size-xs|sm`). No token list gives exact pixel values, so exact visual spacing may differ slightly from the original CSS's hardcoded px values; this is expected/intended since the task asks to replace raw values with tokens.
- **Data loading**: resolved `data/products.json` via `new URL('./data/products.json', import.meta.url)` per the kit's "Common mistakes" note, so the path still works if the app were later served from a version folder.
- **Focus while typing**: relied on vfunc's documented refresh behavior (focus/caret return to the element at the same `data-action` position) rather than adding extra logic, since the search input is the only `data-action="search"` element and keeps a stable position across renders. I also added `data-ref="searchInput"` (not required by the grader, does not conflict with hooks) to give the engine an additional, more specific match target for restoring focus/caret.
- **Single-file, single-page**: since the task says "as a page of its own", I did not add `app.js` structure conventions from `AGENTS.md` §2 (`store.js`, `api.js`, `pages/*.js`, router) — this is a single self-contained component/page, not a multi-screen SPA, so that structure did not apply.

## Kit rules I could not follow, and why

- None. All security rules (no `innerHTML` from data, `vf.html` only, no inline scripts/styles, `data-action`/`data-ref`/`id` selectors only, no CSS classes as selectors, tokens-only CSS) were followed.

- [x] No per-row components: one delegated listener handles every row/control (four delegates total, all on the root, none per list item)
- [x] Controlled inputs do not re-render on every key in a way that loses focus: the search input re-renders each keystroke (required, since filtering must update as you type), but relies on vfunc's position/hook-based focus restoration; not literally "store the value and render on blur" since the task explicitly requires live search-while-typing

- [ ] No console errors or warnings — **not verified** (no browser was run in this environment; the code follows all documented APIs and hook names from the kit, but I could not open the page in Chromium/Firefox/WebKit to confirm)

- [x] Count text matches `1 product` / `N products` exactly
- [x] List items carry `data-id` and `data-state="in"|"out"`, with `name`/`price` refs and the literal `Sold out` badge text
