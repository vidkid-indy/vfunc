### app.js

```js
// Sales — the page script. Chart.js is loaded as the global `Chart`.

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const SERIES = {
  revenue: { label: 'Revenue', values: [120, 135, 150, 142, 168, 181, 175, 190, 205, 198, 220, 236] },
  orders: { label: 'Orders', values: [30, 32, 35, 33, 40, 44, 41, 46, 50, 48, 53, 57] }
};

function chartToken(seriesKey) {
  const name = seriesKey === 'orders' ? '--vf-chart-2' : '--vf-chart-1';
  return getComputedStyle(document.documentElement).getPropertyValue(name).trim();
}

function createSales() {
  // Chart.js instance lives in a closure variable, not as a property on the
  // component instance (the instance is reserved for vfunc's own bookkeeping).
  let chart = null;

  return vf.attach('#sales', {
    state: { series: 'revenue', months: 6 },

    render: (s) => {
      const data = SERIES[s.series];
      const shown = data.values.slice(0, s.months);
      const isLast = s.months >= MONTHS.length;
      return vf.html`
        <div class="sales__toolbar">
          <button type="button" class="sales__button" data-action="series" data-series="revenue"
                  aria-pressed="${s.series === 'revenue'}">Revenue</button>
          <button type="button" class="sales__button" data-action="series" data-series="orders"
                  aria-pressed="${s.series === 'orders'}">Orders</button>
          <button type="button" class="sales__button" data-action="add-month" ${isLast ? 'disabled' : ''}>Add month</button>
        </div>
        <p class="sales__summary" data-ref="summary">${data.label}: ${shown.length} months, last ${shown[shown.length - 1]}</p>
        <div class="sales__chartWrap" data-vf-keep="chart">
          <canvas data-ref="chart"></canvas>
        </div>
      `;
    },

    delegates: [
      {
        selector: '[data-action="series"]',
        eventType: 'click',
        onEvent: (e) => { e.sender.series = e.target.getAttribute('data-series'); }
      },
      {
        selector: '[data-action="add-month"]',
        eventType: 'click',
        onEvent: (e) => {
          if (e.sender.months < MONTHS.length) e.sender.months = e.sender.months + 1;
        }
      }
    ],

    onMount: (inst) => {
      const data = SERIES[inst.state.series];
      const shown = data.values.slice(0, inst.state.months);
      chart = new Chart(inst.refs.chart, {
        type: 'line',
        data: {
          labels: MONTHS.slice(0, inst.state.months),
          datasets: [{ label: data.label, data: shown, borderColor: chartToken(inst.state.series) }]
        },
        options: { responsive: true, maintainAspectRatio: false }
      });
    },

    onUpdate: (inst) => {
      if (!chart) return;
      const data = SERIES[inst.state.series];
      const shown = data.values.slice(0, inst.state.months);
      chart.data.labels = MONTHS.slice(0, inst.state.months);
      chart.data.datasets[0].label = data.label;
      chart.data.datasets[0].data = shown;
      chart.data.datasets[0].borderColor = chartToken(inst.state.series);
      chart.update();
    },

    onDestroy: () => {
      if (chart) {
        chart.destroy();
        chart = null;
      }
    }
  });
}

let sales = createSales();

// The toggle button lives in the page header, outside the sales section, so
// it is wired up on its own container instead of by selecting a CSS class.
vf.attach('.page__head', {
  delegates: [
    {
      selector: '[data-action="toggle-chart"]',
      eventType: 'click',
      onEvent: (e) => {
        if (sales) {
          sales.destroy();
          sales = null;
          e.target.textContent = 'Show chart';
        } else {
          sales = createSales();
          e.target.textContent = 'Hide chart';
        }
      }
    }
  ]
});
```

### REPORT.md

## Assumptions

- The `<section id="sales">` element itself is the sales component's root: it is bound with `vf.attach('#sales', { … })` (options match `vf.vfunc`, and the element keeps its existing `id`/`aria-label`), rather than mounting a separate wrapper `<div>` inside the section. This matches the request ("show … inside `<section id="sales">`") and keeps the section itself as the thing that is created/destroyed.
- The `toggle-chart` button lives outside `#sales` (in the page header), so it cannot be a `delegate` of the sales component. It is wired with its own `vf.attach('.page__head', { delegates: [...] })`; selecting by class here targets the fixed layout container, not app state, and the click handler itself is bound to `data-action="toggle-chart"`, not the class.
- Toggling the button's own text (`Hide chart` / `Show chart`) is done with a direct `textContent` assignment on the clicked button. Both strings are fixed literals (not user/server data), so this does not violate the "dynamic markup only through `vf.html`" rule, which is about untrusted/dynamic content, not static label swaps on a plain button outside any vfunc render.
- The Chart.js instance is kept in a closure variable inside `createSales()`, not as a property on the component instance, per the kit's explicit rule: "Do not add your own properties to an instance … keep listener functions and timers in a closure or module variable." A fresh closure is created each time `createSales()` runs (i.e., each time the section is shown again), so the chart is always paired with the current attach instance.
- The canvas sits inside a `data-vf-keep="chart"` wrapper so re-renders (series switch, add-month) move the existing `<canvas>` and keep the same Chart.js instance alive for the component's whole life, updating it in `onUpdate` instead of recreating it. `onDestroy` calls `chart.destroy()` so no Chart.js instance survives when the section is hidden.
- Chart options add `maintainAspectRatio: false` plus a fixed-height CSS wrapper (`.sales__chartWrap { height: 16rem }`) since Chart.js needs a sized, non-auto-aspect container to render sensibly; the task did not specify chart sizing.
- No new locale keys were needed: all visible strings (`Revenue`, `Orders`, `Add month`, `Hide chart`, `Show chart`) are given verbatim by the task, and the project has no `locales/` folder to extend.

## Not followed

- Nothing from the kit was knowingly skipped. The project has no i18n setup, store, or router in this task, so those sections of AGENTS.md do not apply here.
