```js
// The app's Leaflet map wrapper (L1: an app factory, not a vf.* member).
// Leaflet 1.9.4 (BSD-2-Clause) is loaded by index.html as the global `L`.
(function () {
  window.app = window.app || {};

  function readPrimaryColor() {
    var value = getComputedStyle(document.documentElement).getPropertyValue('--vf-color-primary');
    return value ? value.trim() : '#1d4ed8';
  }

  // app.vfPlacesMap(props) -> a vfunc instance.
  // props.places: [{ id, name, lat, lng }]
  // props.onSelect({ sender, event, data: { id } }) — fired only for a real marker click
  //   (like setValue never firing onChange, select(id) below never fires onSelect).
  app.vfPlacesMap = function (props) {
    props = props || {};
    var places = props.places || [];
    var onSelect = props.onSelect;

    var map = null;
    var markers = {};
    var selectedId = null;

    function refreshStyles() {
      Object.keys(markers).forEach(function (id) {
        markers[id].setStyle({ fillOpacity: id === selectedId ? 1 : 0.4 });
      });
    }

    function setSelected(id) {
      selectedId = id;
      refreshStyles();
    }

    function buildMap(container) {
      map = L.map(container);
      var color = readPrimaryColor();
      var bounds = [];

      places.forEach(function (place) {
        var marker = L.circleMarker([place.lat, place.lng], {
          radius: 8,
          weight: 2,
          color: color,
          fillColor: color,
          fillOpacity: 0.4
        });
        // The name may contain "<": escape it, Leaflet renders tooltip content as markup.
        marker.bindTooltip(vf.esc(place.name));
        marker.on('click', function (leafletEvent) {
          setSelected(place.id);
          if (typeof onSelect === 'function') {
            onSelect({ sender: inst, event: leafletEvent.originalEvent || null, data: { id: place.id } });
          }
        });
        marker.addTo(map);
        markers[place.id] = marker;
        bounds.push([place.lat, place.lng]);
      });

      if (bounds.length > 1) {
        map.fitBounds(bounds, { padding: [24, 24] });
      } else if (bounds.length === 1) {
        map.setView(bounds[0], 15);
      } else {
        map.setView([0, 0], 2);
      }
    }

    var inst = vf.vfunc({
      tag: 'div',
      innerHTML: '<div data-ref="host" data-vf-keep="map"></div>',
      onMount: function (instance) {
        buildMap(instance.refs.host);
      },
      onDestroy: function () {
        if (map) {
          map.remove();
          map = null;
        }
        markers = {};
      },
      methods: {
        // Programmatic selection: highlights the marker only, no onSelect callback.
        select: function (id) {
          setSelected(id);
        }
      }
    });

    return inst;
  };
})();
```

### app.js

```js
// Places — the page script. Leaflet is loaded as the global `L`.
const PLACES = [
  { id: 'p1', name: 'Seoul Station', lat: 37.5547, lng: 126.9707 },
  { id: 'p2', name: 'Cafe <b>Moon</b>', lat: 37.5665, lng: 126.978 },
  { id: 'p3', name: 'Namsan Tower', lat: 37.5512, lng: 126.9882 }
];

let page = null;
let mapInstance = null;
let selectedId = null;
let mapOpen = true;

function findPlace(id) {
  for (let i = 0; i < PLACES.length; i += 1) {
    if (PLACES[i].id === id) return PLACES[i];
  }
  return null;
}

function updateCurrent() {
  const place = selectedId ? findPlace(selectedId) : null;
  page.refs.current.textContent = place ? `Selected: ${place.name}` : 'No place selected';
}

function selectPlace(id) {
  selectedId = id;
  if (mapInstance) mapInstance.select(id);
  updateCurrent();
}

function openMap() {
  mapInstance = app.vfPlacesMap({
    places: PLACES,
    onSelect: function (e) { selectPlace(e.data.id); }
  });
  mapInstance.mount(page.ids['map-slot']);
}

function closeMap() {
  if (mapInstance) {
    mapInstance.destroy();
    mapInstance = null;
  }
}

page = vf.attach('body', {
  delegates: [
    {
      selector: '[data-action="select"]',
      eventType: 'click',
      onEvent: function (e) {
        selectPlace(e.target.getAttribute('data-id'));
      }
    },
    {
      selector: '[data-action="toggle-map"]',
      eventType: 'click',
      onEvent: function (e) {
        if (mapOpen) {
          closeMap();
          mapOpen = false;
          e.target.textContent = 'Show map';
        } else {
          selectedId = null;
          openMap();
          updateCurrent();
          mapOpen = true;
          e.target.textContent = 'Hide map';
        }
      }
    }
  ],
  onMount: function (inst) {
    page = inst;
    openMap();
  }
});
```

```markdown
- `onSelect` on `app.vfPlacesMap` fires only for a genuine marker click. The public `select(id)`
  method only updates the highlight (mirrors the documented rule that `setValue` never triggers
  `onChange`). This avoids a feedback loop where a list-button click → `select(id)` → `onSelect` →
  the page's own handler → `select(id)` again.
- Kept Leaflet's default `zoomControl` / attribution behavior since the task does not ask to change
  them; no tile layer is added (the page has no network access for tiles), only `L.circleMarker`.
- `--vf-color-primary` is read once (via `getComputedStyle`) when the map is built; no live
  theme-switch handling was requested.
- Hiding the map (`toggle-map` → "Show map") does not reset `[data-ref="current"]`; only reopening
  the map ("nothing selected") resets the selection and the text, per the wording of the task.
```
