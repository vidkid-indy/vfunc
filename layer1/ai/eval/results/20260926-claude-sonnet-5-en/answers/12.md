```js
// Profile — the page script.
(function () {
  'use strict';

  var EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

  var toast = vf.vfToast();
  var savedEl = vf.$('[data-ref="saved"]');
  var pendingFocusId = null;

  vf.attach('#profile', {
    state: {
      values: { name: '', email: '', role: 'viewer', newsletter: false },
      errors: {}
    },
    render: function (s) {
      return vf.html`
        ${vf.vsInput({
          id: 'name',
          name: 'name',
          label: 'Name',
          value: s.values.name,
          required: true,
          error: s.errors.name
        })}
        ${vf.vsInput({
          id: 'email',
          name: 'email',
          type: 'email',
          label: 'Email',
          hint: 'We never share it.',
          value: s.values.email,
          required: true,
          error: s.errors.email
        })}
        ${vf.vsSelect({
          id: 'role',
          name: 'role',
          label: 'Role',
          value: s.values.role,
          options: [
            { value: 'admin', label: 'Admin' },
            { value: 'editor', label: 'Editor' },
            { value: 'viewer', label: 'Viewer' }
          ]
        })}
        ${vf.vsSwitch({
          id: 'newsletter',
          name: 'newsletter',
          label: 'Newsletter',
          checked: s.values.newsletter
        })}
        ${vf.vsButton({ label: 'Save', type: 'submit', variant: 'primary' })}
      `;
    },
    events: [
      {
        eventType: 'submit',
        onEvent: function (e) {
          e.event.preventDefault();

          var inst = e.sender;
          var raw = vf.form.values(inst.$node);
          var name = String(raw.name || '');
          var email = String(raw.email || '');
          var role = raw.role || 'viewer';
          var newsletter = !!raw.newsletter;
          var errors = {};

          if (!name.trim()) {
            errors.name = 'Enter your name.';
          }
          if (!EMAIL_RE.test(email.trim())) {
            errors.email = 'Enter a valid email address.';
          }

          if (errors.name) {
            pendingFocusId = 'name';
          } else if (errors.email) {
            pendingFocusId = 'email';
          } else {
            pendingFocusId = null;
          }

          inst.setState({
            values: { name: name, email: email, role: role, newsletter: newsletter },
            errors: errors
          });

          if (!errors.name && !errors.email) {
            if (savedEl) {
              savedEl.textContent = 'Saved: ' + name.trim() + ' (' + role + ')';
            }
            toast.show({ message: 'Profile saved', variant: 'success' });
          }
        }
      }
    ],
    onUpdate: function (inst) {
      if (pendingFocusId) {
        var el = inst.$node.querySelector('#' + pendingFocusId);
        if (el) {
          el.focus();
        }
        pendingFocusId = null;
      }
    }
  });
})();
```
