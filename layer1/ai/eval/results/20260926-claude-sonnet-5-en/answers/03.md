### app.js

```javascript
import vf from './lib/vfunc.esm.js';

// Recent orders kept in the app as static data; there is no server for this sample.
const ORDERS = [
  { id: '1042', customer: 'Minsu Kim', status: 'paid', statusLabel: 'Paid', total: '₩59,000' },
  { id: '1041', customer: 'Seoyeon Park', status: 'shipped', statusLabel: 'Shipped', total: '₩329,000' },
  { id: '1040', customer: 'Jiho Choi', status: 'refunded', statusLabel: 'Refunded', total: '₩29,000' }
];

// The two texts the publisher's inline script swapped into #summary.
const SUMMARY_TEXT = {
  week: 'Orders are up this week. Most sales came from the keyboard line.',
  month: 'This month: 1,204 orders, 3.4% refunds.'
};

function rowMarkup(order) {
  return vf.html`<tr><td>#${order.id}</td><td>${order.customer}</td><td><span class="badge badge--${order.status}">${order.statusLabel}</span></td><td class="orders__num">${order.total}</td></tr>`;
}

// Recent orders: render is attached to the <tbody> only, so rows are parsed
// as content of the real table (no wrapping <table>/<tag> needed for a <tbody> target).
const ordersBody = vf.attach('#ordersBody', {
  state: { search: '', status: '' },
  render: (s) => {
    const term = s.search.trim().toLowerCase();
    const query = term.charAt(0) === '#' ? term.slice(1) : term;
    const matches = ORDERS.filter((o) => {
      const matchesTerm = !term ||
        o.id.toLowerCase().indexOf(query) !== -1 ||
        o.customer.toLowerCase().indexOf(term) !== -1;
      const matchesStatus = !s.status || o.status === s.status;
      return matchesTerm && matchesStatus;
    });
    if (!matches.length) {
      return vf.html`<tr><td colspan="4">No orders match.</td></tr>`;
    }
    return vf.html`${matches.map(rowMarkup)}`;
  }
});

// Search box: filters as you type. The box itself is never re-rendered
// (this attach has no `render`), so focus and caret stay in place.
vf.attach('[data-action="search"]', {
  events: [{ eventType: 'input', onEvent: (e) => { ordersBody.search = e.event.target.value; } }]
});

// Status filter: combines with the search text above (both live on ordersBody's state).
vf.attach('[data-action="filter"]', {
  events: [{ eventType: 'change', onEvent: (e) => { ordersBody.status = e.event.target.value; } }]
});

// Summary tabs: only the active tab's look (class + aria-selected) and the
// summary paragraph's text change, so the panel is adopted without `render`
// and the two spots are written directly in onMount/onUpdate.
function applyTabState(inst) {
  const tabs = inst.$node.querySelectorAll('[data-action="tab"]');
  for (let i = 0; i < tabs.length; i++) {
    const isActive = tabs[i].getAttribute('data-tab') === inst.state.active;
    tabs[i].classList.toggle('tabs__tab--active', isActive);
    tabs[i].setAttribute('aria-selected', String(isActive));
  }
  const summary = inst.$node.querySelector('#summary p');
  if (summary) {
    summary.textContent = SUMMARY_TEXT[inst.state.active];
  }
}

vf.attach('#summaryPanel', {
  state: { active: 'week' },
  delegates: [
    { selector: '[data-action="tab"]', eventType: 'click', onEvent: (e) => {
        e.sender.active = e.target.getAttribute('data-tab');
      } }
  ],
  onMount: applyTabState,
  onUpdate: applyTabState
});

// User menu: opens/closes on the name button, closes on Escape or an
// outside click, and "signs out" without any request, alert or javascript: URL.
// The document listeners are kept in module-scope closures (never as
// properties on the instance) and released in onDestroy.
let onDocClick = null;
let onDocKeydown = null;

function applyUserNavState(inst) {
  inst.refs.userBtn.setAttribute('aria-expanded', String(inst.state.open));
  inst.ids.userMenu.hidden = !inst.state.open;
  if (inst.state.signedOut) {
    inst.refs.userBtn.textContent = 'Signed out';
  }
}

vf.attach('#userNav', {
  state: { open: false, signedOut: false },
  delegates: [
    { selector: '[data-action="menu"]', eventType: 'click', onEvent: (e) => {
        e.sender.open = !e.sender.state.open;
      } },
    { selector: '[data-action="sign-out"]', eventType: 'click', onEvent: (e) => {
        e.event.preventDefault();
        e.sender.setState({ open: false, signedOut: true });
      } }
  ],
  onMount: (inst) => {
    onDocClick = (ev) => {
      if (inst.state.open && !inst.$node.contains(ev.target)) {
        inst.open = false;
      }
    };
    onDocKeydown = (ev) => {
      if (ev.key === 'Escape' && inst.state.open) {
        inst.open = false;
      }
    };
    document.addEventListener('click', onDocClick);
    document.addEventListener('keydown', onDocKeydown);
    applyUserNavState(inst);
  },
  onUpdate: applyUserNavState,
  onDestroy: () => {
    document.removeEventListener('click', onDocClick);
    document.removeEventListener('keydown', onDocKeydown);
  }
});
```

# Conversion report — dashboard.html → vfunc.js

`dashboard.css` is not given above: it is kept byte-for-byte identical, as required, so it "stays as it is".

## 1. Area table

| Area (selector) | Static / dynamic | Why | Tool | Data | Events |
|---|---|---|---|---|---|
| `.topbar__logo`, `.sidebar`, `.footer`, `.kpis` | Static | Never changes at run time | none | — | — |
| `#userNav` (`.topbar__user` button + `#userMenu`) | Dynamic (state) | Open/closed + signed-out text | `vf.attach` without `render` (attribute/text writes in `onUpdate`) | none | click on `[data-action="menu"]`, `[data-action="sign-out"]`; document click/Escape |
| `#summaryPanel` (tabs + `#summary`) | Dynamic (state) | Active tab look + summary text | `vf.attach` without `render` (writes in `onUpdate`) | inline text table (`SUMMARY_TEXT`) | click on `[data-action="tab"]` |
| `[data-action="search"]` (search input) | Behaviour only | Reads typed text, never re-rendered itself | `vf.attach` with `events` (root), no `render` | — | `input` |
| `[data-action="filter"]` (status select) | Behaviour only | Reads selection | `vf.attach` with `events` (root), no `render` | — | `change` |
| `#ordersBody` (`<tbody>`) | Dynamic (data + state) | Rows filtered by search text and status | `vf.attach` with `render` (smallest changing element: the `<tbody>`, not the whole table/panel) | `ORDERS` array in `app.js` | none (reacts to `ordersBody.search` / `ordersBody.status`) |

## 2. HTML changes (added attributes and script tags only)

- `<div class="topbar__user">` → added `id="userNav"`.
- `<button class="topbar__user-btn" ... onclick="toggleMenu()">` → removed `onclick`; added `data-action="menu"`, `data-ref="userBtn"`, `aria-expanded="false"`.

- `<a href="javascript:void(0)" onclick="alert('Signed out')">Sign out</a>` → removed `onclick` and the `javascript:` URL; `href` changed to `#`; added `data-action="sign-out"`.

- `<select class="panel__filter">` → added `data-action="filter"`.
- `<tbody>` → added `id="ordersBody"`.
- Added `<meta http-equiv="Content-Security-Policy" ...>` in `<head>` (none was present).
- Removed the publisher's inline `<script>` block entirely; added `<script type="module" src="./app.js"></script>` before `</body>`.

No other attribute, class, tag, or visible text was touched. `dashboard.css` is unchanged.

## 3. Removed unsafe items (rule 4)

- Inline `onclick="toggleMenu()"` on the user button — replaced by a `data-action="menu"` delegate in `app.js`.
- Inline `onclick="alert('Signed out')"` and the `javascript:void(0)` href on "Sign out" — replaced by a `data-action="sign-out"` delegate; no alert, no request, as the task asks.
- Inline `onclick="showTab('week'|'month')"` on both tab buttons — replaced by `data-action="tab"` + `data-tab` delegates.
- Inline `style="display:none"` on `#userMenu` — replaced by the `hidden` attribute (a CSP with no `'unsafe-inline'` in `style-src` blocks inline `style`, per the kit's common mistakes).
- The publisher's inline `<script>` block (both functions) — removed entirely; its behaviour is reimplemented in `app.js`.

## 4. Suggested token replacements (optional list — not applied to `dashboard.css`)

`dashboard.css` was kept exactly as delivered, per the task. If it is ever opened for a design pass, these raw values map to the closest vfunc tokens:

- `#1e293b`, `#475569`, `#64748b` (text colors) → `var(--vf-color-text)`, `var(--vf-color-text-muted)`.

- No `BEHAVIOUR` was given for the "Profile" and "Settings" links inside the user menu; they were left as plain `href="#"` anchors with no added behaviour (out of scope for the listed hooks).
- "Filters rows while you type... ignoring case" was read as substring matching on both order number and customer name (not only exact match), since a live search box conventionally narrows as you type.
- The order id compare strips a leading `#` from the typed query so both `1041` and `#1041` match; this is an assumption since the task names both forms without specifying the exact matching rule.
- No API/data source was named (`DATA` was not given) — the three orders are kept as a literal array in `app.js`, per the task's instruction to keep them as they are.
- The CSP meta tag content (`default-src 'self'; script-src 'self'; style-src 'self'; img-src 'self' data:; base-uri 'none'; form-action 'self'`) is a reasonable default for this static page; no existing CSP was given to preserve, so this is a new addition rather than a preserved value — flagged as `VERIFY:` if the real deployment needs a different policy (e.g. `connect-src` for a real API).
- No `VERIFY:` items for packages, external APIs or URLs: this task uses only local files (`lib/`, `dashboard.css`, `app.js`), no third-party or CDN scripts.
