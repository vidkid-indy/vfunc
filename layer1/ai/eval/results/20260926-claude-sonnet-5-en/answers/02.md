// My tasks — the page script. The to-do list lives in #todo, in memory only.

let nextId = 1;

function visibleTasks(state) {
  if (state.filter === 'open') return state.tasks.filter((t) => !t.done);
  if (state.filter === 'done') return state.tasks.filter((t) => t.done);
  return state.tasks;
}

vf.attach('#todo', {
  state: { tasks: [], filter: 'all' },

  render(state) {
    const shown = visibleTasks(state);
    const openCount = state.tasks.filter((t) => !t.done).length;
    const isEmpty = shown.length === 0;

    return vf.html`
      <form data-action="add" novalidate class="todo__form">
        <label class="todo__label" for="todo-input">New task</label>
        <div class="todo__row">
          <input type="text" id="todo-input" data-ref="input" class="todo__input" autocomplete="off">
          <button type="submit" class="todo__add-btn">Add</button>
        </div>
      </form>
      <div class="todo__filters" role="group" aria-label="Filter tasks">
        <button type="button" data-action="filter" data-filter="all" aria-pressed="${state.filter === 'all'}" class="todo__filter-btn">All</button>
        <button type="button" data-action="filter" data-filter="open" aria-pressed="${state.filter === 'open'}" class="todo__filter-btn">Open</button>
        <button type="button" data-action="filter" data-filter="done" aria-pressed="${state.filter === 'done'}" class="todo__filter-btn">Done</button>
      </div>
      <ul class="todo__list" data-ref="list">
        ${shown.map((t) => vf.html`
          <li class="todo__item" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
            <input type="checkbox" data-action="toggle" id="todo-${t.id}" class="todo__checkbox" ${t.done ? 'checked' : ''}>
            <label class="todo__text" data-ref="text" for="todo-${t.id}">${t.text}</label>
            <button type="button" data-action="remove" class="todo__remove-btn">Remove</button>
          </li>
        `)}
      </ul>
      ${isEmpty ? vf.html`<p data-ref="empty" class="todo__empty">Nothing here.</p>` : ''}
      <p class="todo__count"><span data-ref="left">${openCount}</span> open</p>
    `;
  },

  delegates: [
    {
      selector: 'form[data-action="add"]',
      eventType: 'submit',
      onEvent(e) {
        e.event.preventDefault();
        const input = e.sender.refs.input;
        const text = (input.value || '').trim();
        if (!text) return;
        e.sender.setState((state) => ({
          tasks: state.tasks.concat({ id: nextId++, text: text, done: false })
        }));
      }
    },
    {
      selector: '[data-action="filter"]',
      eventType: 'click',
      onEvent(e) {
        const filter = e.target.getAttribute('data-filter');
        e.sender.setState({ filter: filter });
      }
    },
    {
      selector: '[data-action="toggle"]',
      eventType: 'change',
      onEvent(e) {
        const row = e.target.closest('[data-id]');
        if (!row) return;
        const id = row.getAttribute('data-id');
        e.sender.setState((state) => ({
          tasks: state.tasks.map((t) => (String(t.id) === id ? { ...t, done: !t.done } : t))
        }));
      }
    },
    {
      selector: '[data-action="remove"]',
      eventType: 'click',
      onEvent(e) {
        const row = e.target.closest('[data-id]');
        if (!row) return;
        const id = row.getAttribute('data-id');
        e.sender.setState((state) => ({
          tasks: state.tasks.filter((t) => String(t.id) !== id)
        }));
      }
    }
  ]
});

# Report — 02-todo

## Assumptions
- The to-do list is built with a single `vf.attach('#todo', { render, delegates })` on the existing `<section id="todo">`, since `attach` with `render` draws only the inside of the target element and the section, its class and `aria-label` were already given in `index.html` (left unchanged).
- The text input is **uncontrolled** (its value is not mirrored into `state` on every keystroke). This avoids a re-render per keystroke (llms.txt "Common mistakes" / AGENTS.md §4.4) and means the input naturally comes back empty after `render` runs again post-add — matching "after adding, the input is empty" without extra reset code.
- Focus return to the input, and to a toggled checkbox, is not done by hand: it relies on the engine's documented refresh behaviour ("focus and caret return to the element with the same id, data-ref or name"). The input keeps `data-ref="input"` (and `id="todo-input"`), and each checkbox keeps a stable `id="todo-<data-id>"` across renders, so this should hold. I could not run a browser to confirm it, so I marked that checklist line "not verified".
- The task id counter (`nextId`) is a plain module-level closure variable, not a property on the component instance, per AGENTS.md §4.9 ("do not add your own properties to an instance").
- `data-ref="text"` is repeated once per row (one per task), which the task explicitly asks for. This is a row-level marker for the grader (`[data-id="…"] [data-ref="text"]`), not a component-level unique ref used through `inst.refs`, so the usual "one ref per name" expectation does not apply here.
- The open-task counter (`data-ref="left"`) always counts open tasks across the whole list, not just the current filter view, since the requirement calls it "the number of open tasks" without saying "of the current filter."
