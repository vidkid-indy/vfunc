### app.js

```js
// Users screen — the page script. USERS, MONTHS and MONTH_LABELS come from data.js.
(function () {
  'use strict';

  var selectedEl = vf.$('[data-ref="selected"]');

  function setSelectedCount(n) {
    selectedEl.textContent = n + ' selected';
  }

  function filterUsers(query) {
    var q = (query || '').trim().toLowerCase();
    if (!q) return USERS.slice();
    return USERS.filter(function (u) {
      return u.name.toLowerCase().indexOf(q) !== -1;
    });
  }

  function applySearch(query) {
    grid.setData(filterUsers(query));
    grid.setPage(1);
  }

  var columns = [
    { key: 'name', label: 'Name', sortable: true },
    { key: 'email', label: 'Email' },
    {
      key: 'role',
      label: 'Role',
      render: function (row) {
        return vf.vsBadge({ label: row.role });
      }
    },
    { key: 'joined', label: 'Joined', sortable: true }
  ];

  var grid = vf.vfGrid({
    id: 'users-grid',
    caption: 'Users',
    columns: columns,
    data: USERS,
    pageSize: 10,
    selectable: 'multiple',
    rowKey: 'id',
    onSelect: function (e) {
      setSelectedCount(e.data.keys.length);
    }
  });
  grid.mount('#users');

  var search = vf.vfSearchInput({
    id: 'q',
    label: 'Search users',
    onSearch: function (e) {
      applySearch(e.data.value);
    },
    onClear: function () {
      applySearch('');
    }
  });
  search.mount('#search');

  var counts = MONTHS.map(function (month) {
    return USERS.filter(function (u) {
      return u.joined.slice(0, 7) === month;
    }).length;
  });

  var chart = vf.vfChart({
    id: 'signups-chart',
    type: 'bar',
    label: 'Sign-ups per month',
    dataTable: true,
    data: {
      labels: MONTH_LABELS,
      series: [{ name: 'Sign-ups', data: counts }]
    }
  });
  chart.mount('#signups');
})();
```

### REPORT.md

## Assumptions

- `vf.vfGrid` in its default (`client`) mode sorts and paginates its `data` internally when a sort button (`data-action="sort"`) or pagination control is used, so `onSort` / `onPage` handlers are not required for the sortable `name` and `joined` columns to work; I only handle `onSelect` to keep the selected-count text in sync.
- Search filtering is app-level, not delegated to the grid's own `setQuery`: on every `onSearch` (debounced) and on `onClear`, the full `USERS` list is filtered by case-insensitive substring match on `name`, then pushed into the grid with `setData(...)` and the grid is returned to page 1 with `setPage(1)`, matching "searching ... shows page 1; clearing the search shows every user again."
- The `id: 'q'` prop passed to `vf.vfSearchInput` sets the id of the actual `<input>` element (as `vsInput`'s doc says the id lands on the control itself), which is what the grader checks for.
- The `role` column has no built-in `sortable` requirement (the task table only marks `name` and `joined` as sortable), so it is left non-sortable, rendered as `vf.vsBadge({ label: row.role })`.
- The `joined` column is shown with its raw `YYYY-MM-DD` string (no custom `render`), since sorting the ISO-format string alphabetically is also chronological and the task does not ask for reformatting.
- No `vf.i18n.setup()` call was added: the layer 2 built-in default messages (e.g. `searchInput.clear`) are assumed to be registered by `vfunc-ui.js` itself without requiring an explicit i18n setup call, since the project has no locale files and the task does not mention adding one.
- `data.js` and `index.html` are used exactly as given; only `app.js` was written (the empty stub was replaced in full).
