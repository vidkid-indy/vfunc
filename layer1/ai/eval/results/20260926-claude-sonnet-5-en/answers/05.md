### app.js

```js
// Servers — the page script (an ES module).
import vf from './lib/vfunc.esm.js';

// Resolve relative to this module so it still works from a version folder.
const DATA_URL = new URL('./data/servers.json', import.meta.url);

const FILTERS = [
  { value: 'all', label: 'All' },
  { value: 'up', label: 'Up' },
  { value: 'down', label: 'Down' },
  { value: 'maintenance', label: 'Maintenance' }
];

// The server list is untrusted data (another team's service in production):
// every field is coerced to a plain string / known enum before it is shown.
function badgeFor(status) {
  const key = typeof status === 'string' ? status : '';
  if (key === 'up') return { state: 'up', text: 'Up' };
  if (key === 'down') return { state: 'down', text: 'Down' };
  if (key === 'maintenance') return { state: 'maintenance', text: 'Maintenance' };
  return { state: 'unknown', text: 'Unknown' };
}

// Only an http(s) URL becomes a working link; anything else (javascript:,
// data:, an unknown scheme, garbage) falls back to '#'.
function linkHref(url) {
  try {
    const parsed = new URL(String(url), window.location.href);
    if (parsed.protocol === 'http:' || parsed.protocol === 'https:') return parsed.href;
  } catch (err) {
    // not a parseable URL: fall through to '#'
  }
  return '#';
}

function textOf(value) {
  return value == null ? '' : String(value);
}

function filterServers(servers, filter) {
  if (filter === 'all') return servers;
  return servers.filter((server) => server && server.status === filter);
}

function renderStatus(state) {
  if (state.phase === 'loading') return vf.html`Loading servers…`;
  if (state.phase === 'error') {
    return vf.html`Could not load servers.
      <button type="button" class="servers__retry" data-action="retry">Retry</button>`;
  }
  if (state.servers.length === 0) return vf.html`No servers yet.`;
  const shown = filterServers(state.servers, state.filter).length;
  return vf.html`Showing ${shown} of ${state.servers.length}`;
}

function renderItem(server) {
  const id = textOf(server && server.id);
  const name = textOf(server && server.name);
  const region = textOf(server && server.region);
  const badge = badgeFor(server && server.status);
  const href = linkHref(server && server.url);
  return vf.html`
    <li class="servers__item" data-id="${id}">
      <span class="servers__name" data-ref="name">${name}</span>
      <span class="servers__region">${region}</span>
      <span class="servers__badge" data-ref="badge" data-state="${badge.state}">${badge.text}</span>
      <a class="servers__link" data-ref="link" href="${href}">Open</a>
    </li>`;
}

function renderList(state) {
  return vf.html`${filterServers(state.servers, state.filter).map(renderItem)}`;
}

function renderFilterOptions(state) {
  return vf.html`${FILTERS.map((f) => vf.html`
    <option value="${f.value}" ${f.value === state.filter ? 'selected' : ''}>${f.label}</option>`)}`;
}

// Keeps the in-flight request cancellable; there is only one instance of
// this component (attached to #servers), so a module-level variable is
// enough — instances themselves must not carry extra properties.
let abortController = null;

async function load(inst) {
  if (abortController) abortController.abort();
  const controller = new AbortController();
  abortController = controller;
  inst.setState({ phase: 'loading' });
  try {
    const res = await fetch(DATA_URL, { signal: controller.signal });
    if (!res.ok) throw new Error('http-error');
    const data = await res.json();
    const servers = Array.isArray(data) ? data : [];
    inst.setState({ phase: 'ready', servers });
  } catch (err) {
    if (err && err.name === 'AbortError') return;
    inst.setState({ phase: 'error', servers: [] });
  }
}

vf.attach('#servers', {
  state: { phase: 'loading', servers: [], filter: 'all' },
  render: (state) => vf.html`
    <div class="servers__controls">
      <label class="servers__filter-label">
        Status
        <select class="servers__filter" data-action="status-filter">${renderFilterOptions(state)}</select>
      </label>
    </div>
    <p class="servers__status" data-ref="status" role="status">${renderStatus(state)}</p>
    <ul class="servers__list" data-ref="list">${renderList(state)}</ul>`,
  delegates: [
    {
      selector: '[data-action="status-filter"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.filter = e.target.value; }
    },
    {
      selector: '[data-action="retry"]',
      eventType: 'click',
      onEvent: (e) => { load(e.sender); }
    }
  ],
  onMount: (inst) => {
    inst.$node.setAttribute('aria-busy', inst.phase === 'loading' ? 'true' : 'false');
    load(inst);
  },
  onUpdate: (inst) => {
    inst.$node.setAttribute('aria-busy', inst.phase === 'loading' ? 'true' : 'false');
  },
  onDestroy: () => {
    if (abortController) abortController.abort();
    abortController = null;
  }
});
```
