### places-map.js

```js
// The app's map component: an L1 wrapper factory kept on `app`, not on `vf`.
// LIBRARY: leaflet@1.9.4 (global `L`, loaded by index.html). LEVEL: L1 (app-only wrapper).
(function () {
  if (!window.app) {
    window.app = {};
  }

  app.vfPlacesMap = function (props) {
    const places = (props && props.places) || [];
    const onSelect = props && props.onSelect;
    let selectedId = null;
    let map = null;
    const markers = {};

    function styleFor(id) {
      return { fillOpacity: id === selectedId ? 1 : 0.4 };
    }

    function applySelection() {
      Object.keys(markers).forEach((id) => {
        markers[id].setStyle(styleFor(id));
      });
    }

    function selectPlace(id) {
      selectedId = id;
      applySelection();
    }

    const inst = vf.vfunc({
      tag: 'div',
      // Trusted static markup written by this file (no interpolated data):
      // the Leaflet map is only ever created inside this data-vf-keep host,
      // so a refresh() of this instance would never rebuild the vendor DOM.
      innerHTML: '<div data-ref="host" data-vf-keep="map"></div>',
      methods: {
        select(id) {
          selectPlace(id);
        }
      },
      onMount(sender) {
        const color = getComputedStyle(document.documentElement)
          .getPropertyValue('--vf-color-primary')
          .trim();
        map = L.map(sender.refs.host, { attributionControl: false });
        const bounds = [];
        places.forEach((place) => {
          const marker = L.circleMarker([place.lat, place.lng], {
            radius: 8,
            weight: 2,
            color: color,
            fillColor: color,
            fillOpacity: styleFor(place.id).fillOpacity
          });
          // Place names may contain "<": Leaflet's tooltip sets its content
          // via innerHTML, so the text is escaped before it reaches Leaflet.
          marker.bindTooltip(vf.esc(place.name));
          marker.on('click', (event) => {
            selectPlace(place.id);
            if (onSelect) {
              onSelect({ sender, event: event.originalEvent || event, data: { id: place.id } });
            }
          });
          marker.addTo(map);
          markers[place.id] = marker;
          bounds.push([place.lat, place.lng]);
        });
        if (bounds.length) {
          map.fitBounds(bounds, { padding: [24, 24] });
        }
      },
      onDestroy() {
        if (map) {
          map.remove();
          map = null;
        }
      }
    });

    return inst;
  };
})();
```

### app.js

```js
// Places — the page script. Leaflet is loaded as the global `L`.
const PLACES = [
  { id: 'p1', name: 'Seoul Station', lat: 37.5547, lng: 126.9707 },
  { id: 'p2', name: 'Cafe <b>Moon</b>', lat: 37.5665, lng: 126.978 },
  { id: 'p3', name: 'Namsan Tower', lat: 37.5512, lng: 126.9882 }
];

const currentEl = vf.$('[data-ref="current"]');
const mapSlot = vf.$('#map-slot');
const toggleButton = vf.$('[data-action="toggle-map"]');

let placesMap = null;
let mapVisible = true;

function placeById(id) {
  return PLACES.find((place) => place.id === id) || null;
}

function showSelection(id) {
  const place = placeById(id);
  currentEl.textContent = place ? `Selected: ${place.name}` : 'No place selected';
}

function createMap() {
  placesMap = app.vfPlacesMap({
    places: PLACES,
    onSelect: (e) => {
      showSelection(e.data.id);
    }
  });
  placesMap.mount(mapSlot);
}

createMap();

// Publishing-style delegation on the existing markup: no render/innerHTML here,
// so the page's own markup (header, list, map slot) is left untouched.
vf.attach(document.body, {
  delegates: [
    {
      selector: '[data-action="select"]',
      eventType: 'click',
      onEvent(e) {
        const id = e.target.getAttribute('data-id');
        if (placesMap) {
          placesMap.select(id);
        }
        showSelection(id);
      }
    },
    {
      selector: '[data-action="toggle-map"]',
      eventType: 'click',
      onEvent() {
        if (mapVisible) {
          if (placesMap) {
            placesMap.destroy();
            placesMap = null;
          }
          toggleButton.textContent = 'Show map';
          mapVisible = false;
        } else {
          createMap();
          toggleButton.textContent = 'Hide map';
          mapVisible = true;
        }
      }
    }
  ]
});
```

### REPORT.md

**수준과 결과물**: `LEVEL: L1`(앱 전용 래퍼) — `app.vfPlacesMap(props)`를 `window.app`에 두고 `vf`는 건드리지 않았습니다. Leaflet은 `index.html`이 이미 SRI 붙여 전역 `L`로 불러오므로 로드 코드는 추가하지 않았습니다.

**설계 요약**
- `places-map.js`: `vf.vfunc({ tag: 'div', innerHTML: '<div data-ref="host" data-vf-keep="map"></div>', methods: { select }, onMount, onDestroy })`. Leaflet 지도는 `data-vf-keep="map"` 요소 안에서만 만들고, `onDestroy`에서 `map.remove()`로 정리합니다.
- 마커는 `L.circleMarker`만 쓰고 타일 레이어는 만들지 않았습니다(네트워크 불가 조건). 색은 `getComputedStyle(document.documentElement).getPropertyValue('--vf-color-primary')`로 읽어 stroke/fill에 씁니다.
- 툴팁 텍스트는 `vf.esc(place.name)`을 거쳐 `bindTooltip`에 넘깁니다. Leaflet의 툴팁 콘텐츠는 문자열이면 내부적으로 `innerHTML`로 설정되므로, 이름에 `<`가 있어도(`Cafe <b>Moon</b>`) 이스케이프해 텍스트로만 보이게 했습니다(prompt-integrate-third-party.md의 11번 "vf.esc(text)를 넘기고 원본을 그대로 넘기지 않는다" 규칙).
- 선택 상태는 컴포넌트 밖(앱)이 아니라 컴포넌트 내부 클로저(`selectedId`, `markers`)에서 관리하고, `methods.select(id)`로 앱이 호출합니다. `fillOpacity`는 선택 1 / 비선택 0.4, 초기값은 전부 0.4(선택 없음).
- 지도 경계는 `map.fitBounds(bounds, { padding: [24, 24] })`로 모든 장소가 보이게 맞췄습니다.
- `app.js`는 지도를 `#map-slot`에 `mount`하고, `vf.attach(document.body, { delegates })`로 목록 버튼(`data-action="select"`, `data-id`)과 헤더 버튼(`data-action="toggle-map"`)을 위임 처리합니다. `render`/`innerHTML`을 주지 않아 기존 퍼블리싱 마크업(헤더, 목록, `.map-slot`)은 그대로 둡니다.
- 마커 클릭과 목록 버튼 클릭 모두 `placesMap.select(id)` 경로로 마커를 강조하고, `[data-ref="current"]`의 텍스트를 `textContent`로 설정해(`Selected: ${place.name}`) 이름이 자동으로 텍스트로만 표시되게 했습니다(별도 이스케이프 불필요).
- 헤더 버튼: 지도가 보이는 상태에서 누르면 `placesMap.destroy()` 후 글자를 `Show map`으로, 다시 누르면 `app.vfPlacesMap(...)`을 새로 만들어 `mount`하고 글자를 `Hide map`으로 되돌립니다. 새 인스턴스는 항상 `selectedId = null`로 시작하므로 선택 없는 상태로 시작합니다.

**가정(직접 판단하고 진행한 것)**
- `vf.attach`의 대상을 클래스 선택자 대신 `document.body`(직접 참조, 셀렉터 질의 아님)로 잡았습니다. `id`/`data-ref`가 없는 `<main class="page">` 전체에 위임을 걸 필요가 있었고, 규칙은 "훅으로 찾는다"이지 "id가 반드시 있어야 한다"는 아니므로 body 직접 참조를 선택했습니다.
- `circleMarker`의 `radius: 8`, `weight: 2`, `fitBounds`의 `padding: [24, 24]`는 과제에 수치가 없어 임의로 정했습니다(디자인 값이 아니라 지도 레이아웃 수치라 인라인으로 둬도 되는 종류로 판단).
- `L.map`에 `attributionControl: false`를 줬습니다(타일이 없어 저작권 표시가 의미 없고, 화면 문구 요구사항에도 없음). `zoomControl` 등은 Leaflet 기본값을 그대로 뒀습니다.
- 지도 재생성(`toggle-map` 다시 누름) 시 `[data-ref="current"]`의 기존 "Selected: …" 문구는 과제에 명시된 것이 없어 그대로 두었습니다(새 지도 인스턴스에서만 선택 없음을 보장).

**확인 못 함**
- Chromium/Firefox/WebKit에서 실제로 열어 콘솔 오류·경고 0개인지, 지도가 `#map-slot` 크기에 맞게 그려지는지, 마커 클릭·목록 버튼·토글 버튼 동작이 실제 브라우저에서 기대대로 되는지는 실행 환경이 없어 확인하지 못했습니다.
- Leaflet 1.9.4의 `bindTooltip` 문자열 처리(`innerHTML` 경유)에 대한 이해는 API 문서 기억에 근거했고, 브라우저에서 `Cafe <b>Moon</b>`이 실제로 굵게 렌더되지 않고 그대로 텍스트로 보이는지는 직접 실행해 검증하지 못했습니다.
- CSS `.map-slot [data-vf-keep] { block-size: … }`가 컴포넌트 루트(`tag: 'div'`, `data-vf-keep` 없음) 안의 호스트 div에 실제로 적용되어 Leaflet이 0 높이로 그려지지 않는지는 확인하지 못했습니다.
