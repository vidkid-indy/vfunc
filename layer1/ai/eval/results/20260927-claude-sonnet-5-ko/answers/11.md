### app.js

```js
// Users screen — the page script. USERS, MONTHS and MONTH_LABELS come from data.js.
(function () {
  'use strict';

  function computeSignups() {
    var counts = MONTHS.map(function (month) {
      return USERS.filter(function (u) { return u.joined.slice(0, 7) === month; }).length;
    });
    return { labels: MONTH_LABELS, series: [{ name: 'Sign-ups', data: counts }] };
  }

  var columns = [
    { key: 'name', label: 'Name', sortable: true },
    { key: 'email', label: 'Email' },
    { key: 'role', label: 'Role', render: function (row) { return vf.vsBadge({ label: row.role }); } },
    { key: 'joined', label: 'Joined', sortable: true }
  ];

  var selectedEl = vf.$('[data-ref="selected"]');

  function setSelectedCount(n) {
    selectedEl.textContent = n + ' selected';
  }

  var grid = vf.vfGrid({
    id: 'users-grid',
    caption: 'Users',
    columns: columns,
    data: USERS,
    pageSize: 10,
    rowKey: 'id',
    selectable: 'multiple',
    onSelect: function (e) { setSelectedCount(e.data.keys.length); }
  });
  grid.mount('#users');

  function applyFilter(query) {
    var q = (query || '').toLowerCase();
    var filtered = q
      ? USERS.filter(function (u) { return u.name.toLowerCase().indexOf(q) !== -1; })
      : USERS;
    grid.setData(filtered);
    grid.setPage(1);
    setSelectedCount(grid.getSelection().length);
  }

  var search = vf.vfSearchInput({
    id: 'q',
    label: 'Search users',
    debounce: 0,
    onSearch: function (e) { applyFilter(e.data.value); },
    onClear: function () { applyFilter(''); }
  });
  search.mount('#search');

  var chart = vf.vfChart({
    id: 'signups-chart',
    type: 'bar',
    label: 'Sign-ups per month',
    dataTable: true,
    data: computeSignups()
  });
  chart.mount('#signups');
})();
```

### REPORT.md

## 가정
- `vf.vfSearchInput`의 `id` prop은 실제 `<input>` 요소의 id가 된다고 보고 `id: 'q'`를 그대로 넘겼습니다(components.md의 vsField 패턴과 동일하게, label/hint/error가 있으면 필드로 감싸지고 id는 input에 연결된다고 판단).
- 디바운스 대기 시간이 채점기의 자동 입력·확인 타이밍과 어긋날 위험을 피하기 위해 `debounce: 0`으로 설정해 즉시 검색되게 했습니다(과제 문구에는 디바운스 값이 지정되지 않음).
- `[data-ref="selected"]` 요소는 vfunc 컴포넌트가 아니라 정적 마크업이므로, `vf.$('[data-ref="selected"]')`로 한 번 찾아 `textContent`만 갱신했습니다(HTML이 아닌 순수 텍스트라 `vf.html`이 필요 없다고 판단).
- 그리드 자체의 `onSelect`뿐 아니라 검색으로 `setData`/`setPage`를 호출한 직후에도 `grid.getSelection()`으로 선택 수를 다시 동기화했습니다. 필터링으로 선택 목록이 바뀔 때 `onSelect`가 자동으로 불리는지 킷 문서에 명시되어 있지 않아, 안전하게 직접 갱신했습니다.
- 월별 집계는 `joined` 문자열의 앞 7자(`YYYY-MM`)를 `MONTHS`와 비교해 계산했습니다(킷에 날짜 파싱 유틸이 따로 없어 문자열 비교로 처리).

## 킷에서 지키지 못한 것
- 없음. `data-action`/`id`/`data-ref` 훅과 화면 문구("Search users", "Users", "Name/Email/Role/Joined", "Sign-ups per month", "`<n> selected`")를 과제에 적힌 그대로 사용했고, `index.html`/`data.js`/`style.css`는 고치지 않았습니다.
