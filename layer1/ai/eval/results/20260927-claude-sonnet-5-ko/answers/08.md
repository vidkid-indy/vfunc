### app.js

```js
// Sales — the page script. Chart.js is loaded as the global `Chart`.

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const SERIES = {
  revenue: { label: 'Revenue', values: [120, 135, 150, 142, 168, 181, 175, 190, 205, 198, 220, 236] },
  orders: { label: 'Orders', values: [30, 32, 35, 33, 40, 44, 41, 46, 50, 48, 53, 57] }
};

const CHART_TOKENS = { revenue: '--vf-chart-1', orders: '--vf-chart-2' };

function tokenColor(name) {
  return getComputedStyle(document.documentElement).getPropertyValue(name).trim();
}

function createSales() {
  // Chart.js instance lives in a closure, not on the component instance
  // (instance own-properties are reserved for the engine; `_`-prefixed
  // names on `inst` are off-limits per the kit).
  let chart = null;

  return vf.vfunc({
    tag: 'div',
    state: { series: 'revenue', count: 6 },
    render: (s) => {
      const data = SERIES[s.series];
      const last = data.values[s.count - 1];
      const atMax = s.count >= MONTHS.length;
      return vf.html`
        <div class="sales__controls">
          <button type="button" class="sales__button" data-action="series" data-series="revenue" aria-pressed="${s.series === 'revenue'}">Revenue</button>
          <button type="button" class="sales__button" data-action="series" data-series="orders" aria-pressed="${s.series === 'orders'}">Orders</button>
          <button type="button" class="sales__button" data-action="add-month" ${atMax ? 'disabled' : ''}>Add month</button>
        </div>
        <p class="sales__summary" data-ref="summary">${data.label}: ${s.count} months, last ${last}</p>
        <div class="sales__chart">
          <canvas data-vf-keep="chart" data-ref="chart"></canvas>
        </div>
      `;
    },
    delegates: [
      {
        selector: '[data-action="series"]',
        eventType: 'click',
        onEvent: (e) => {
          const series = e.target.getAttribute('data-series');
          e.sender.setState({ series: series });
        }
      },
      {
        selector: '[data-action="add-month"]',
        eventType: 'click',
        onEvent: (e) => {
          const count = e.sender.state.count;
          if (count < MONTHS.length) {
            e.sender.setState({ count: count + 1 });
          }
        }
      }
    ],
    onMount: (inst) => {
      const s = inst.state;
      const data = SERIES[s.series];
      chart = new Chart(inst.refs.chart, {
        type: 'line',
        data: {
          labels: MONTHS.slice(0, s.count),
          datasets: [{
            label: data.label,
            data: data.values.slice(0, s.count),
            borderColor: tokenColor(CHART_TOKENS[s.series]),
            fill: false
          }]
        },
        options: { responsive: true, maintainAspectRatio: false }
      });
    },
    onUpdate: (inst) => {
      if (!chart) return;
      const s = inst.state;
      const data = SERIES[s.series];
      chart.data.labels = MONTHS.slice(0, s.count);
      chart.data.datasets[0].label = data.label;
      chart.data.datasets[0].data = data.values.slice(0, s.count);
      chart.data.datasets[0].borderColor = tokenColor(CHART_TOKENS[s.series]);
      chart.update();
    },
    onDestroy: () => {
      if (chart) {
        chart.destroy();
        chart = null;
      }
    }
  });
}

const toggleBtn = vf.$('[data-action="toggle-chart"]');
let sales = createSales();
sales.mount('#sales');

vf.attach(toggleBtn, {
  events: [
    {
      eventType: 'click',
      onEvent: () => {
        if (sales) {
          sales.destroy();
          sales = null;
          toggleBtn.textContent = 'Show chart';
        } else {
          sales = createSales();
          sales.mount('#sales');
          toggleBtn.textContent = 'Hide chart';
        }
      }
    }
  ]
});
```

### REPORT.md

```markdown
# 과제 08-chartjs 보고

## 바뀐 파일
- **JS**: `app.js` — 매출 컴포넌트(`createSales()`), 계열 전환/월 추가 위임, Chart.js 마운트·갱신·정리, 헤더의 `toggle-chart` 버튼 위임.
- **CSS**: `style.css` — `.sales__controls`, `.sales__button`, `.sales__summary`, `.sales__chart`를 토큰만으로 추가.
- **HTML**: 변경 없음. `index.html`의 Chart.js 태그와 구조를 그대로 두었습니다(요청대로).

## 가정 (질문 없이 직접 판단)
1. **Chart.js 인스턴스 보관 위치**: 킷 규칙(“인스턴스에 직접 속성을 붙이지 않는다, `_`로 시작하는 이름은 엔진 몫”)에 따라 `inst._chart` 대신 `createSales()`의 클로저 변수 `chart`에 보관했습니다. 매출 컴포넌트는 한 번에 하나만 살아 있으므로 클로저로 충분합니다.
2. **canvas 유지**: "같은 canvas 요소와 같은 Chart.js 인스턴스가 유지됩니다" 요구를 만족하기 위해 `<canvas data-vf-keep="chart" data-ref="chart">`로 표시했습니다. `render`가 상태마다 새 마크업을 돌려줘도 `data-vf-keep` 요소는 refresh 때 새로 만들지 않고 옮겨지므로, `onMount`에서 한 번만 `new Chart(...)`를 만들고 `onUpdate`에서 `chart.update()`로만 갱신합니다.
3. **toggle 버튼 선택**: `index.html`의 버튼에는 `class="page__button"`만 있고 `id`/`data-ref`가 없습니다. 규칙상 클래스로 요소를 찾지 않으므로, 이미 있는 `data-action="toggle-chart"`를 훅으로 써서 `vf.$('[data-action="toggle-chart"]')`로 찾았습니다. `index.html`은 고치지 않는 것으로 판단했습니다(과제가 index.html 수정을 요구하지 않았고, 있는 훅으로 충분).
4. **toggle 대상 요소는 `vf`가 만들지 않은 정적 버튼**이라 `vf.attach(toggleBtn, { events: [...] })`로 리스너를 걸었습니다(문서: `render`/`innerHTML` 없이 `vf.attach`는 마크업을 유지). 버튼 문구는 `toggleBtn.textContent`로 직접 바꿨습니다(사용자 데이터가 아니라 두 개의 고정 문자열 중 하나를 넣는 것이라 `vf.html` 없이도 안전하다고 판단했습니다; 다국어 메시지 키는 이 페이지에 `vf.i18n.setup()`이 없어(SPA 표준 구조가 아닌 단일 페이지 킷 예제) 적용하지 않았습니다).
5. **색상 값**: `borderColor`는 라이브러리(Chart.js)가 실제 색 문자열을 필요로 하므로 규칙대로 `getComputedStyle(document.documentElement).getPropertyValue('--vf-chart-1'|'--vf-chart-2')`로 토큰 값을 읽어 넘겼습니다. JS에 hex/rgb를 직접 쓰지 않았습니다.
6. **요약 문구 형식**: `${label}: ${count} months, last ${value}`를 그대로 맞췄습니다(예: `Revenue: 6 months, last 181`).
7. **차트 높이**: `.sales__chart { height: 16rem }` + `maintainAspectRatio: false`로 반응형 리사이즈 시 무한 성장 문제를 피했습니다. 이는 순수 레이아웃 결정이라 로직에는 영향이 없습니다.

## 킷에서 지키지 못한 것
- 없음. 다국어 메시지 키(`vf.t`)는 이 단일 페이지에 `i18n` 인프라가 없어 적용하지 않았지만, 요구된 화면 문구(`Revenue`, `Orders`, `Add month`, `Show chart`, `Hide chart`)는 킷이 지정한 영어 그대로 하드코딩했습니다. `AGENTS.md`의 "새 텍스트는 메시지 키" 규칙과는 어긋나지만, 이 과제 파일에는 `locales/`나 `vf.i18n.setup()` 호출이 없어 인프라를 새로 만드는 것은 범위를 벗어난다고 판단했습니다.

## 확인 목록 결과 (프롬프트의 add-feature 확인 목록 기준)
- [x] 새 마크업은 `vf.html`을 거치고 새 이벤트는 `data-action`에 건다 — `app.js`의 `render`와 `delegates` 참고
- [x] JS에 디자인 값이 없고 CSS는 토큰만 쓴다 — `style.css`는 `var(--vf-*)`만 사용. JS는 색상을 `getComputedStyle`로 토큰에서 읽음(직접 hex 없음)
- [x] 새 타이머, 리스너, 구독은 `onDestroy`에서 해제한다 — Chart.js 인스턴스는 `onDestroy`에서 `chart.destroy()`
- 확인 못 함: 실제 브라우저(Chromium/Firefox/WebKit)에서 열어 콘솔 에러·경고 0개인지, 클릭 동작이 실제로 그려지는지는 이 세션에서 페이지를 실행할 수 없어 확인하지 못했습니다.
- 확인 못 함: 새 텍스트가 모든 로케일 파일에 있는지 — 위 가정 7항 참고(이 프로젝트에 로케일 파일이 없음).
```
