### app.js

```js
// Sign up — the page script.
(function () {
  'use strict';

  var FIELDS = [
    { id: 'email', message: 'Enter a valid email address.' },
    { id: 'password', message: 'Use at least 8 characters, including a number.' },
    { id: 'confirm', message: 'Passwords do not match.' },
    { id: 'terms', message: 'Accept the terms to continue.' }
  ];

  // Tracks fields that have shown an error at least once, so we know which
  // ones should re-validate live as the user types (per spec: only fields
  // that have shown an error re-validate on every input).
  var everInvalid = {};

  function fieldValid(id, formEl) {
    if (id === 'email') {
      var email = formEl.email.value.trim();
      return (/^[^\s@]+@[^\s@]+\.[^\s@]+$/).test(email);
    }
    if (id === 'password') {
      var password = formEl.password.value;
      return password.length >= 8 && (/\d/).test(password);
    }
    if (id === 'confirm') {
      var confirmVal = formEl.confirm.value;
      return confirmVal.length > 0 && confirmVal === formEl.password.value;
    }
    if (id === 'terms') {
      return formEl.terms.checked;
    }
    return true;
  }

  function fieldMessage(id) {
    var i;
    for (i = 0; i < FIELDS.length; i++) {
      if (FIELDS[i].id === id) return FIELDS[i].message;
    }
    return '';
  }

  function validateField(inst, id) {
    var formEl = inst.$node;
    var ok = fieldValid(id, formEl);
    var errorEl = inst.ids[id + '-error'];
    var inputEl = inst.ids[id];
    if (errorEl) errorEl.textContent = ok ? '' : fieldMessage(id);
    if (inputEl) inputEl.setAttribute('aria-invalid', ok ? 'false' : 'true');
    if (!ok) everInvalid[id] = true;
    return ok;
  }

  function onBlurLike(e) {
    validateField(e.sender, e.target.id);
  }

  function onInput(e) {
    var id = e.target.id;
    if (everInvalid[id]) validateField(e.sender, id);
  }

  function onTermsChange(e) {
    validateField(e.sender, 'terms');
  }

  function onSubmit(e) {
    e.event.preventDefault();
    var inst = e.sender;
    var firstInvalidId = null;
    FIELDS.forEach(function (field) {
      var ok = validateField(inst, field.id);
      if (!ok && !firstInvalidId) firstInvalidId = field.id;
    });
    if (firstInvalidId) {
      var el = inst.ids[firstInvalidId];
      if (el) el.focus();
      return;
    }
    var email = inst.$node.email.value.trim();
    inst.$node.hidden = true;
    var doneEl = vf.$('[data-ref="done"]');
    if (doneEl) {
      doneEl.textContent = 'Welcome, ' + email + '!';
      doneEl.hidden = false;
    }
  }

  vf.attach(vf.$('#signup'), {
    delegates: [
      { selector: '#email, #password, #confirm', eventType: 'focusout', onEvent: onBlurLike },
      { selector: '#email, #password, #confirm', eventType: 'input', onEvent: onInput },
      { selector: '#terms', eventType: 'change', onEvent: onTermsChange }
    ],
    events: [
      { eventType: 'submit', onEvent: onSubmit }
    ]
  });
})();
```

### REPORT.md

```markdown
# 가입 폼 검증 — 보고

## 가정
- 킷의 `AGENTS.md` 구조(pages/, store.js, api.js 등)는 이 과제의 단일 페이지 정적 폼에는 해당하지 않아 적용하지 않았습니다(과제 자체가 `index.html`/`app.js`/`style.css` 3파일 구조를 지정).
- 필드마다 새 `<p class="field__error" id="...">` 요소와, `email`/`password`/`confirm` input에 `aria-describedby`를 추가했습니다. 기존 폼 마크업(라벨, input, 클래스, id, name)은 그대로 두고 요구된 오류 요소만 더했습니다.
- 완료 메시지용 `<p class="signup__done" data-ref="done" hidden></p>`를 `<form>` 형제로 `<main>` 안에 추가했습니다. 폼을 `hidden`으로 감추고 이 요소의 `hidden`을 지워 보이게 합니다.
- "한 번 오류를 보인 필드는 입력할 때마다 다시 검증"을 "그 필드가 한 번이라도 오류를 보인 적이 있으면 이후 모든 input에서 재검증"으로 해석했습니다(플래그가 한 번 켜지면 계속 유지). 값을 고쳐 통과하면 즉시 메시지가 사라지고, 이후 다시 틀리게 고치면 다시 표시됩니다.
- `--vf-color-danger`, `--vf-font-size-sm`, `--vf-color-border-strong`, `--vf-radius-*` 등은 `lib/vfunc.tokens.css`에 있다고 가정했습니다(킷이 전체 토큰 목록을 주지 않아 이름은 `VERIFY:` 대상). 존재하지 않으면 해당 값은 조용히 무시되어 콘솔 오류는 나지 않지만 스타일이 기대와 다를 수 있습니다.
- `[hidden]` 규칙을 이기는 `.signup { display: grid }`, `.signup__done`(향후 스타일 대비) 때문에 `.signup[hidden]`, `.signup__done[hidden]`에 `display: none`을 명시적으로 다시 선언했습니다(동일 특이도라 소스 순서로 UA 스타일을 덮어쓰는 문제 방지).

## 지키지 못한 것
- 없음. `vf.attach`로 퍼블리싱 마크업을 그대로 채택하고(`render` 없음), `data-action` 대신 이 과제가 이미 준 `id` 셀렉터로 위임을 걸었습니다(킷 규칙상 CSS 클래스만 피하면 됨. `id`/`data-ref`는 허용된 훅). 비밀번호 값은 어디에도 출력·로그하지 않았습니다(`console` 미사용, `textContent`에는 이메일만).

## 확인 목록 결과 (직접 실행은 못 함 — 코드 리뷰로 확인)
- [x] 동적 값은 `textContent`만 사용(이메일 표시), `innerHTML`/`vf.html` 불필요(마크업 삽입 없음)
- [x] 셀렉터는 `id`만 사용, JS에 색·크기 없음
- [x] 새 CSS는 `var(--vf-*)` 토큰만 참조(`:empty`, `[aria-invalid]`, `[hidden]`은 값이 아니라 셀렉터), hex/px 값 없음
- [x] 새 텍스트는 과제가 지정한 영어 문구 그대로(로케일 파일 없는 단일 정적 페이지라 다국어 키 미적용 — 과제 범위 밖)
```
