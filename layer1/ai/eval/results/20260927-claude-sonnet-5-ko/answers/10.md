### app.js

```js
// Team board — the page script.
import vf from './lib/vfunc.esm.js';
import { store, setTheme, setLang, toggleTask, removeTask } from './store.js';

// Preferences: theme and language.
const prefs = vf.attach('#prefs', {
  render: () => {
    const p = store.get('prefs');
    return vf.html`
      <p class="prefs__row">Theme: <span data-ref="theme">${p.theme}</span>
        <button class="prefs__button" type="button" data-action="theme">Switch theme</button></p>
      <p class="prefs__row">Language: <span data-ref="lang">${p.lang}</span>
        <button class="prefs__button" type="button" data-action="lang" data-lang="en">English</button>
        <button class="prefs__button" type="button" data-action="lang" data-lang="ko">한국어</button></p>`;
  },
  delegates: [
    { selector: '[data-action="theme"]', eventType: 'click', onEvent: () => setTheme(store.get('prefs').theme === 'dark' ? 'light' : 'dark') },
    { selector: '[data-action="lang"]', eventType: 'click', onEvent: (e) => setLang(e.target.getAttribute('data-lang')) }
  ]
});

// Notice: how many tasks are open.
// Bug 1 fix: render must return only the INSIDE of #notice (vf.attach parses
// into the target element's own tag). Returning another <section id="notice">
// made attach adopt it as a duplicate child, so the id existed twice.
const notice = vf.attach('#notice', {
  render: () => {
    const open = store.get('tasks').filter((t) => !t.done).length;
    return vf.html`<p class="notice__text">Open tasks: ${open}</p>`;
  }
});

// Details: a disclosure.
vf.attach('#details-panel', {
  state: { open: false },
  delegates: [{
    selector: '[data-action="details"]',
    eventType: 'click',
    onEvent: (e) => {
      const open = !e.sender.state.open;
      e.sender.state.open = open;
      e.sender.ids.details.hidden = !open;
      // Bug 2 fix: aria-expanded must be the string "false", not "".
      e.target.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
  }],
  // Bug 2 fix: initial state must also be "false", not "".
  onMount: (inst) => inst.$node.querySelector('[data-action="details"]').setAttribute('aria-expanded', 'false')
});

// Tasks, open ones first.
const tasks = vf.attach('#tasks', {
  render: () => vf.html`
    <h2 class="panel__title">Tasks</h2>
    <ul class="tasks">${store.get('tasks').slice().sort((a, b) => a.done - b.done).map((t) => vf.html`
      <li class="task" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
        <input type="checkbox" id="task-toggle-${t.id}" data-action="toggle" aria-label="Done" ${t.done ? 'checked' : ''}>
        <span class="task__title">${t.title}</span>
        <button class="task__delete" type="button" data-action="remove">Remove</button>
      </li>`)}</ul>`,
  delegates: [
    { selector: '[data-action="toggle"]', eventType: 'change', onEvent: (e) => toggleTask(e.target.closest('[data-id]').getAttribute('data-id')) },
    // Bug 5 fix: the designer renamed the CSS class from task__remove to
    // task__delete, which silently broke the class-based selector. Hooks
    // must use data-action, not classes, so this now matches the button
    // regardless of its class name.
    { selector: '[data-action="remove"]', eventType: 'click', onEvent: (e) => removeTask(e.target.closest('[data-id]').getAttribute('data-id')) }
  ]
});
// Bug 7 fix: each checkbox now has a stable per-task id
// (id="task-toggle-<id>"). vf's refresh restores focus by matching id first;
// without it, focus fell back to "same data-action, same position", which
// pointed at the wrong checkbox once toggling re-sorted the list.

// Comments.
const comments = [];
vf.attach('#comments', {
  delegates: [{
    selector: '[data-action="comment"]',
    eventType: 'submit',
    onEvent: (e) => {
      e.event.preventDefault();
      const text = e.sender.refs.comment.value.trim();
      if (!text) return;
      comments.push(text);
      // Bug 6 fix: escape user-supplied text with vf.esc before building the
      // markup string. The previous code concatenated raw comment text into
      // innerHTML, so a comment like "<b>hi</b>" (or a <script> tag) was
      // parsed as markup instead of shown as text.
      e.sender.refs['comment-list'].innerHTML = comments.map((c) => '<li class="comments__item">' + vf.esc(c) + '</li>').join('');
      e.sender.refs.comment.value = '';
    }
  }]
});

// Footer: how often the clock has updated.
const footer = vf.attach('#footer', {
  state: { updates: 0 },
  render: (s) => vf.html`Clock updates: <span data-ref="updates">${s.updates}</span>`
});

// Clock, with a close button.
const clock = vf.vfunc({
  state: { now: new Date() },
  render: (s) => vf.html`
    <p class="clock">Time: <span data-ref="time">${vf.fmt.date(s.now, { timeStyle: 'medium' })}</span>
      <button class="clock__close" type="button" data-action="close-clock">Close clock</button></p>`,
  delegates: [{ selector: '[data-action="close-clock"]', eventType: 'click', onEvent: (e) => e.sender.destroy() }],
  onMount: (inst) => {
    inst._timer = setInterval(() => {
      inst.setState({ now: new Date() });
      footer.setState({ updates: footer.state.updates + 1 });
    }, 250);
  },
  // Bug 4 fix: the timer was stored as inst._timer in onMount but read back
  // as inst.timer (no underscore) in onDestroy, so clearInterval was called
  // with undefined and the interval kept running after destroy.
  onDestroy: (inst) => clearInterval(inst._timer)
});
clock.mount('#clock-slot');

store.subscribe(() => {
  prefs.refresh();
  notice.refresh();
  tasks.refresh();
});
```

### store.js

```js
// Shared state and the only functions that change it.
import vf from './lib/vfunc.esm.js';

export const store = vf.store({
  prefs: { theme: 'light', lang: 'en' },
  tasks: [
    { id: 't1', title: 'Write the release notes', done: false },
    { id: 't2', title: 'Review the login page', done: false },
    { id: 't3', title: 'Update the icons', done: true }
  ]
});

export function setTheme(theme) {
  store.set((s) => ({ prefs: { theme: theme, lang: s.prefs.lang } }));
}

export function setLang(lang) {
  // Bug 3 fix: store.set does a shallow merge at the top level only, so
  // { prefs: { lang: lang } } replaced the whole prefs object and dropped
  // theme (it became undefined, which showed as the page falling back to
  // the default/light look). Rebuild prefs from the current state instead,
  // the same way setTheme already does.
  store.set((s) => ({ prefs: { theme: s.prefs.theme, lang: lang } }));
}

export function toggleTask(id) {
  store.set((s) => ({ tasks: s.tasks.map((t) => (t.id === id ? { id: t.id, title: t.title, done: !t.done } : t)) }));
}

export function removeTask(id) {
  store.set((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }));
}
```

```markdown
# 버그 일곱 개 — 원인과 수정

모든 훅(`id`, `data-action`, `data-ref`, `data-id`)과 화면 문구는 그대로 두었습니다. `index.html`, `style.css`는 고칠 필요가 없어 내지 않았습니다(그대로 유지).

## 1. `#notice`가 두 개가 됨
- 원인: `vf.attach('#notice', { render })`의 `render`는 대상 요소의 **안쪽**만 돌려줘야 하는데, 원래 코드는 `<section id="notice">…</section>`를 통째로 다시 반환했습니다. 이미 페이지에 `id="notice"` 요소가 있는데 render 결과 안에 또 하나가 생겨 attach가 그것을 자식으로 붙였습니다. 콘솔 경고(`render returned the target element itself`)가 정확히 이 상황을 가리킵니다.
- 수정: `render`가 `<p class="notice__text">Open tasks: ${open}</p>`만 반환하도록 줄였습니다. `#notice` 요소 자체(및 `aria-live`)는 `index.html`에 이미 있으므로 그대로 둡니다.

## 2. `aria-expanded=""`
- 원인: 초기 `onMount`와 클릭 핸들러 모두 닫힘 상태를 빈 문자열 `''`로 설정했습니다. `aria-expanded`는 boolean 속성이 아니라 문자열 값(`"true"`/`"false"`)이어야 합니다.
- 수정: 두 곳 모두 `'false'`로 바꿨습니다(`onMount`의 초기값, 클릭 핸들러의 삼항식).

## 3. 언어를 바꾸면 테마가 light로 돌아감
- 원인: `vf.store`의 `set`은 **최상위 키만** 얕은 병합합니다. `setTheme`은 `{ prefs: { theme, lang: s.prefs.lang } }`처럼 `prefs` 전체를 다시 만들어 넘기지만, `setLang`은 `{ prefs: { lang: lang } }`만 넘겨 `prefs.theme`이 통째로 사라졌습니다(`undefined`가 되어 화면상 기본/light처럼 보임).
- 수정: `setLang`도 `setTheme`과 같은 패턴으로 현재 `theme`을 유지한 채 `prefs`를 다시 만들도록 고쳤습니다.

## 4. Close clock을 눌러도 타이머가 안 멈춤
- 원인: `onMount`에서 `inst._timer = setInterval(...)`로 저장했는데 `onDestroy`에서는 `inst.timer`(밑줄 없음)를 읽어 `clearInterval(undefined)`가 되어 아무 효과가 없었습니다.
- 수정: `onDestroy`를 `clearInterval(inst._timer)`로 고쳤습니다.

## 5. Remove 버튼이 동작하지 않음(클래스 이름 변경 후)
- 원인: delegate 셀렉터가 `'.task__remove'`라는 CSS 클래스를 가리켰는데, 디자이너가 마크업의 클래스를 `task__delete`로 바꾸면서 셀렉터가 더 이상 아무 것도 찾지 못했습니다. 킷 규칙(4-1, 4절 "흔한 실수")대로 동작 훅은 `data-action`이어야 합니다.
- 수정: 셀렉터를 `'[data-action="remove"]'`로 바꿨습니다. 버튼에는 이미 `data-action="remove"`가 있어 마크업은 그대로 두었습니다.

## 6. 댓글에 HTML/스크립트가 그대로 실행됨(보안)
- 원인: 댓글 목록을 `'<li class="comments__item">' + c + '</li>'`처럼 사용자 입력 `c`를 이스케이프 없이 그대로 이어 붙여 `innerHTML`에 설정했습니다. `<b>hi</b>`가 마크업으로 파싱되고, `<script>`도 실행될 수 있는 구조였습니다.
- 수정: 이어 붙이기 직전에 `vf.esc(c)`로 이스케이프했습니다. 최소 변경을 위해 문자열 조합 구조는 유지하되 값만 이스케이프 함수를 거치도록 했습니다. (더 원칙적인 방법은 `render`가 있는 `vf.vfunc`/`vf.attach`로 바꿔 `vf.html`의 배열 보간을 쓰는 것이지만, 이 부분은 `render` 없는 `vf.attach`라 구조를 바꾸지 않고 이스케이프만 추가했습니다.)

## 7. Space로 체크하면 포커스가 다른 할 일로 넘어감
- 원인: 목록은 `done` 여부로 정렬되어, 체크할 때마다 순서가 바뀝니다. 체크박스에 `id`/`data-ref`/`name`이 없어서 refresh 후 포커스 복원이 "같은 `data-action`의 같은 순서 요소"로 대체 규칙을 썼는데, 정렬로 순서 자체가 바뀌어 버려 엉뚱한 체크박스로 포커스가 옮겨갔습니다.
- 수정: 체크박스에 `id="task-toggle-${t.id}"`를 추가했습니다(요구사항에 있는 `data-action="toggle"`은 그대로 유지). refresh는 먼저 같은 `id`로 요소를 매칭하므로, 정렬로 위치가 바뀌어도 같은 할 일의 체크박스에 포커스가 남습니다.

## 가정
- "가능한 한 적은 변경"을 위해 댓글 이스케이프는 구조를 바꾸지 않고 `vf.esc`만 추가하는 쪽을 택했습니다(더 큰 구조 변경인 `render` 도입은 하지 않음).
- 체크박스에 새 `id` 속성을 추가하는 것은 과제 본문의 "고치는 데 새 속성이 필요하면 추가합니다"에 해당한다고 보고 추가했습니다.

## 확인 목록
- [확인 못 함] 실제로 Chromium/Firefox/WebKit에서 페이지를 열어 콘솔 에러·경고 0개인지, 7개 증상이 모두 해소되는지 — 이 환경에서는 브라우저를 실행하지 않았습니다(과제 조건상 프로젝트/킷 파일만 읽고 답하도록 되어 있음).
- 코드 리뷰 상으로는: `#notice`가 단일 요소로 유지되는지, `aria-expanded` 문자열 값, `setLang` 후 테마 유지, 타이머 정지, Remove 동작, 댓글 이스케이프, 체크박스 포커스 유지 모두 위 수정으로 논리상 해결됩니다.
- 어긴 규칙: 없음(AGENTS.md의 셀렉터/이스케이프/디자인 분리 규칙을 모두 지켰습니다). 댓글 렌더링에 `vf.html` 대신 `vf.esc` + 문자열 조합을 쓴 것은 원칙(6절 "동적 마크업은 vf.html로")과 다소 어긋나지만, 최소 변경 조건과 문자열 자체가 안전하게 이스케이프된다는 점에서 의도적으로 선택했습니다.
```
