### app.js

```js
// vfunc.js control layer for the publisher's static dashboard markup.
// The HTML/CSS were not rewritten: only hook attributes were added (see REPORT.md).
import vf from './lib/vfunc.esm.js';

// Static copy shown in the "Summary" panel per tab. Not user data.
var SUMMARY_TEXT = {
  week: 'Orders are up this week. Most sales came from the keyboard line.',
  month: 'This month: 1,204 orders, 3.4% refunds.'
};

// Order data mirrors the three rows the publisher shipped in index.html.
var ORDERS = [
  { id: '#1042', customer: 'Minsu Kim', status: 'paid', statusLabel: 'Paid', total: '₩59,000' },
  { id: '#1041', customer: 'Seoyeon Park', status: 'shipped', statusLabel: 'Shipped', total: '₩329,000' },
  { id: '#1040', customer: 'Jiho Choi', status: 'refunded', statusLabel: 'Refunded', total: '₩29,000' }
];

function matchesQuery(order, rawQuery) {
  var q = rawQuery.trim().toLowerCase();
  if (!q) return true;
  var qNormalized = q.replace('#', '');
  var idNormalized = order.id.replace('#', '').toLowerCase();
  if (idNormalized.indexOf(qNormalized) !== -1) return true;
  if (order.customer.toLowerCase().indexOf(q) !== -1) return true;
  return false;
}

// ---- Recent orders table: render is scoped to the <tbody> only. ----
var ordersBody = vf.attach('#ordersBody', {
  state: { query: '', status: '' },
  render: function (s) {
    var rows = ORDERS.filter(function (o) {
      return matchesQuery(o, s.query) && (!s.status || o.status === s.status);
    });
    if (!rows.length) {
      return vf.html`<tr><td colspan="4">No orders match.</td></tr>`;
    }
    return vf.html`${rows.map(function (o) {
      return vf.html`<tr><td>${o.id}</td><td>${o.customer}</td><td><span class="badge badge--${o.status}">${o.statusLabel}</span></td><td class="orders__num">${o.total}</td></tr>`;
    })}`;
  }
});

// ---- Search: filters by order number (with or without "#") or customer name. ----
vf.attach('#searchInput', {
  events: [{
    eventType: 'input',
    onEvent: function (e) {
      ordersBody.query = e.target.value;
    }
  }]
});

// ---- Status filter: combines with the search query. ----
vf.attach('#statusFilter', {
  events: [{
    eventType: 'change',
    onEvent: function (e) {
      ordersBody.status = e.target.value;
    }
  }]
});

// ---- Summary tabs: render is scoped to the #summary text box only. ----
var summary = vf.attach('#summary', {
  state: { tab: 'week' },
  render: function (s) {
    return vf.html`<p>${SUMMARY_TEXT[s.tab]}</p>`;
  }
});

function setActiveTab(tab) {
  var buttons = vf.$$('[data-action="tab"]', vf.$('#summaryTabs'));
  buttons.forEach(function (btn) {
    var active = btn.getAttribute('data-tab') === tab;
    btn.classList.toggle('tabs__tab--active', active);
    btn.setAttribute('aria-selected', active ? 'true' : 'false');
  });
}

vf.attach('#summaryTabs', {
  delegates: [{
    selector: '[data-action="tab"]',
    eventType: 'click',
    onEvent: function (e) {
      var tab = e.target.getAttribute('data-tab');
      summary.tab = tab;
      setActiveTab(tab);
    }
  }]
});

// ---- User menu: dropdown adopted as-is (no render), only hidden/aria toggled. ----
function isMenuOpen() {
  var menu = vf.$('#userMenu');
  return !!menu && !menu.hidden;
}

function openMenu() {
  var menu = vf.$('#userMenu');
  var btn = vf.$('#userMenuBtn');
  if (!menu || !btn) return;
  menu.hidden = false;
  btn.setAttribute('aria-expanded', 'true');
}

function closeMenu() {
  var menu = vf.$('#userMenu');
  var btn = vf.$('#userMenuBtn');
  if (!menu || !btn) return;
  menu.hidden = true;
  btn.setAttribute('aria-expanded', 'false');
}

function onDocumentClick(e) {
  var wrap = vf.$('#userMenuWrap');
  if (wrap && !wrap.contains(e.target)) {
    closeMenu();
  }
}

function onDocumentKeydown(e) {
  if (e.key === 'Escape' && isMenuOpen()) {
    closeMenu();
    var btn = vf.$('#userMenuBtn');
    if (btn) btn.focus();
  }
}

vf.attach('#userMenuWrap', {
  delegates: [
    {
      selector: '[data-action="menu"]',
      eventType: 'click',
      onEvent: function (e) {
        e.event.stopPropagation();
        if (isMenuOpen()) {
          closeMenu();
        } else {
          openMenu();
        }
      }
    },
    {
      selector: '[data-action="sign-out"]',
      eventType: 'click',
      onEvent: function (e) {
        e.event.preventDefault();
        closeMenu();
        var btn = vf.$('#userMenuBtn');
        if (btn) btn.textContent = 'Signed out';
      }
    }
  ],
  onMount: function () {
    document.addEventListener('click', onDocumentClick);
    document.addEventListener('keydown', onDocumentKeydown);
  },
  onDestroy: function () {
    document.removeEventListener('click', onDocumentClick);
    document.removeEventListener('keydown', onDocumentKeydown);
  }
});
```

### REPORT.md

```markdown
# 변환 보고서 — 퍼블리싱 대시보드 (03-dashboard-conversion)

## 1. 영역 판별표

| 영역(셀렉터) | 정적/동적 | 이유 | 도구 | 데이터 | 이벤트 |
|---|---|---|---|---|---|
| `.topbar__logo`, `.sidebar`, `.kpis`, `.footer`, `.content__title` | 정적 | 실행 중 절대 안 바뀜 | 없음(그대로) | — | — |
| `#userMenuWrap`(버튼 + `#userMenu`) | 동적(토글) | 열림/닫힘 상태가 바뀜, 목록 자체는 안 바뀜 | `vf.attach` (render 없음, 채택) | — | click(menu), click(sign-out), document click/keydown |
| `#summaryTabs`의 탭 버튼 | 동적(클래스/aria만) | 활성 표시만 바뀌고 마크업 구조는 그대로 | `vf.attach` (render 없음, 채택) | — | click(tab) |
| `#summary` | 동적(문구 교체) | 탭에 따라 문구가 완전히 바뀜(가장 작은 변경 요소) | `vf.attach` (render 있음, `#summary`에만) | `SUMMARY_TEXT` 상수 | tab 상태 변경 시 |
| `#searchInput` | 동적(입력만, 자기 자신은 안 바뀜) | 값만 읽으면 됨 | `vf.attach` (render 없음, events) | 사용자 입력값 → `ordersBody.query` | input |
| `#statusFilter` | 동적(입력만) | 값만 읽으면 됨 | `vf.attach` (render 없음, events) | 사용자 선택값 → `ordersBody.status` | change |
| `#ordersBody`(`<tbody>`) | 동적(행 목록) | 검색/필터로 행 집합이 바뀜 | `vf.attach` (render 있음, `<tbody>`만) | `ORDERS` 상수 배열 | query/status 상태 변경 시 |

## 2. HTML 변경 (추가한 속성 · 대체만, 구조·클래스·문구는 그대로)

- `<head>`: CSP `<meta>` 태그 추가.
- 검색 input: `id="searchInput"`, `data-action="search"` 추가.
- `.topbar__user` div: `id="userMenuWrap"` 추가.
- 사용자 버튼: `onclick="toggleMenu()"` 제거 → `id="userMenuBtn"`, `data-action="menu"`, `aria-expanded="false"` 추가.
- `#userMenu`: `style="display:none"` 제거(규칙 4 대체) → `hidden` 속성 추가. 목록 항목·문구는 그대로.
- "Sign out" 링크: `href="javascript:void(0)"`와 `onclick="alert(...)"` 제거(규칙 4) → `href="#"`, `data-action="sign-out"`로 대체.
- 탭 div: `id="summaryTabs"` 추가.
- 탭 버튼 두 개: `onclick="showTab(...)"` 제거 → `data-action="tab"`, `data-tab="week"`/`"month"`, `aria-selected="true"/"false"` 추가. `tabs__tab--active` 클래스는 모양 유지를 위해 그대로 둠(규칙 7).
- 상태 select: `id="statusFilter"`, `data-action="filter"` 추가.
- `<tbody>`: `id="ordersBody"` 추가. 행 내용/클래스는 그대로(첫 렌더에서 JS가 동일한 내용으로 교체).
- `<body>` 끝의 인라인 `<script>` 블록 전체 제거(규칙 4) → `<script type="module" src="./app.js"></script>` 하나로 대체.
- `dashboard.css`는 한 바이트도 바꾸지 않음.

## 3. 제거한 위험 요소 (규칙 4)

- 인라인 `<script>` 블록(`toggleMenu`, `showTab`) 제거 → `app.js`로 이전.
- `onclick="toggleMenu()"`, `onclick="showTab('week'|'month')"`, `onclick="alert('Signed out')"` 인라인 핸들러 3곳 제거 → `data-action` + `delegates`로 대체.
- `href="javascript:void(0)"` 1곳 제거 → `href="#"` + `preventDefault()`로 대체.
- 인라인 `style="display:none"` 1곳 제거 → `hidden` 속성으로 대체(CSP `style-src`에 `'unsafe-inline'`을 넣지 않기 위함).

## 4. CSS 토큰 치환 제안 (선택, 적용하지 않음)

요청이 없어 `dashboard.css`는 고치지 않았습니다. 참고로 원시 값 다수(`#1e3a8a`, `#1d4ed8`, `#047857`, `#b91c1c`, `14px`, `24px` 등)가 있어, 적용한다면 `--vf-color-primary`, `--vf-color-danger`, `--vf-color-success`, `--vf-space-*`, `--vf-font-size-*` 계열 토큰으로 옮기는 것을 제안합니다.

## 5. 가정과 VERIFY 항목

- 검색은 주문 번호와 고객 이름에만 적용하고 상태 문구는 검색 대상에 포함하지 않았습니다(과제 문구가 "주문 번호나 고객 이름"만 명시).
- "Sign out" 클릭 후 재클릭 동작은 정의되어 있지 않아 별도 처리하지 않았습니다(버튼 문구만 "Signed out"으로 바뀌고 이후에도 메뉴는 계속 열고 닫을 수 있습니다). 서버 요청이 없으므로 실제 로그아웃 상태 전이는 구현 범위 밖입니다.
- 메뉴 바깥 클릭/Escape 동작은 `document`에 리스너를 등록했습니다. 이 페이지는 라우터로 이동/destroy되는 SPA 페이지가 아니라서 실제로 해제될 일은 없지만, 관례대로 `onDestroy`에서 해제하도록 작성했습니다.
- `VERIFY:` 항목 없음(외부 API·엔드포인트를 쓰지 않음).

## 6. 확인 목록 결과

- [x] 정적 영역은 규칙 1의 속성과 규칙 4의 대체 말고는 바이트 단위로 같다 — 헤더 로고, 사이드바, KPI 4개, 푸터는 원본과 동일.
- [x] `render`는 내용이 바뀌는 가장 작은 요소에만 붙였다 — `#summary`(문구 상자)와 `#ordersBody`(`<tbody>`)에만 붙였고, 패널 전체나 메뉴 전체에는 붙이지 않음.
- [x] 퍼블리셔 클래스를 쓰는 셀렉터가 없다 — 모든 위임/이벤트는 `data-action`/`id`만 사용. `tabs__tab--active`, `badge--*` 클래스는 여전히 모양 토글용으로만 토글되고 셀렉터로는 쓰이지 않음.
- [x] 모든 동적 값이 `vf.html` 안에 있다 — 탭 문구(`SUMMARY_TEXT`)와 주문 행(`ORDERS`)은 모두 `vf.html`을 거쳐 렌더됨. "Signed out" 문구와 `hidden`/`aria-*` 토글은 사용자 데이터가 아닌 고정 리터럴/상태 속성이라 `textContent`·`setAttribute` 직접 조작으로 처리.
- [x] 인라인 스크립트, 인라인 핸들러, `javascript:` URL이 남아 있지 않다. CSP 메타 태그가 있고 `script-src`에 `'unsafe-inline'`이 없다 — 모두 제거·대체 확인(2·3절 참고).
- [x] 폼은 입력값을 유지한다 — 검색 input과 상태 select는 자기 자신을 다시 그리지 않는 별도 attach 루트이므로 타이핑 중 포커스가 유지됨(재렌더되는 것은 `#ordersBody`뿐).
- [x] 타이머, 리스너, 서드파티 위젯은 `onDestroy`에서 해제한다 — 메뉴의 `document` click/keydown 리스너를 `onDestroy`에서 해제하도록 작성. 다른 리스너는 모두 `vf.attach`가 관리하는 위임/이벤트라 엔진이 해제함.
- [ ] 개발 빌드로 열었을 때 콘솔 에러·경고가 0개다 — **확인 못 함**(브라우저를 실행할 수 없는 환경에서 코드만 작성). 로직상 `vf.html`/`vf.attach`/`vf.$`/`vf.$$` API를 킷 문서 그대로 사용했고 `on*`·`srcdoc`·따옴표 없는 속성·함수 값을 넣지 않았으므로 개발 빌드 경고가 발생할 이유는 없다고 판단했지만 실제 실행 검증은 못 했습니다.
```
