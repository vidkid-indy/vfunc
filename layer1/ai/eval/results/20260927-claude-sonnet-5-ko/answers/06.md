### app.js

```js
import vf from './lib/vfunc.esm.js';
import { store } from './store.js';
import { createHeader } from './components/header.js';
import homePage from './pages/home.js';
import serversPage from './pages/servers.js';
import serverDetailPage from './pages/server-detail.js';
import notFoundPage from './pages/not-found.js';

const LOCALE_KEY = 'vf.locale';

function detectInitialLocale() {
  try {
    const saved = window.localStorage.getItem(LOCALE_KEY);
    if (saved === 'en' || saved === 'ko') {
      return saved;
    }
  } catch (err) {
    // localStorage may be unavailable (private mode, disabled storage); fall back below
  }
  const nav = (navigator.language || '').toLowerCase();
  return nav.indexOf('ko') === 0 ? 'ko' : 'en';
}

let currentLocale = detectInitialLocale();
let currentPage = null;
let header = null;

const viewEl = document.getElementById('view');

const router = vf.router({
  mode: 'hash',
  routes: {
    '/': (ctx) => homePage(ctx, router),
    '/servers': (ctx) => serversPage(ctx, router),
    '/servers/:id': (ctx) => serverDetailPage(ctx, router)
  },
  notFound: (ctx) => notFoundPage(ctx, router),
  linkSelector: 'a[data-link]',
  focus: '#view',
  onChange(ctx) {
    if (currentPage) {
      currentPage.destroy();
      currentPage = null;
    }
    const page = ctx && ctx.route ? ctx.route(ctx) : notFoundPage(ctx, router);
    currentPage = page;
    page.mount(viewEl);
    if (header) {
      header.setState({ path: ctx ? ctx.path : '/' });
    }
  }
});

vf.i18n.setup({
  locales: ['en', 'ko'],
  fallback: 'en',
  load: (locale) => fetch(new URL('./locales/' + locale + '.json', import.meta.url)).then((res) => res.json()),
  persist: true
});

function applyLocale(locale) {
  return vf.i18n.set(locale).then(() => {
    currentLocale = locale;
    document.documentElement.lang = currentLocale;
    if (header) {
      header.setState({ locale: currentLocale });
    }
    if (currentPage) {
      currentPage.refresh();
    }
  });
}

applyLocale(currentLocale).then(() => {
  header = createHeader({
    router,
    store,
    getLocale: () => currentLocale,
    onLocaleChange: (locale) => {
      applyLocale(locale);
    }
  });
  router.start();
});
```

### store.js

```js
import vf from './lib/vfunc.esm.js';

export const store = vf.store({ favorites: [] });

export function toggleFavorite(id) {
  store.set((state) => {
    const has = state.favorites.indexOf(id) !== -1;
    const favorites = has
      ? state.favorites.filter((favId) => favId !== id)
      : state.favorites.concat([id]);
    return { favorites };
  });
}

export function isFavorite(id) {
  return store.get('favorites').indexOf(id) !== -1;
}
```

### api.js

```js
export function fetchServers() {
  return fetch(new URL('./data/servers.json', import.meta.url)).then((res) => {
    if (!res.ok) {
      throw new Error('Failed to load servers: ' + res.status);
    }
    return res.json();
  });
}

export function fetchServer(id) {
  return fetchServers().then((servers) => {
    const found = servers.filter((server) => server.id === id);
    return found.length ? found[0] : null;
  });
}
```

```js
import vf from '../lib/vfunc.esm.js';

export function createHeader({ router, store, getLocale, onLocaleChange }) {
  const target = document.getElementById('site-header');

  const header = vf.attach(target, {
    state: {
      path: '/',
      favorites: store.get('favorites'),
      locale: getLocale()
    },
    render: (s) => {
      const homeCurrent = s.path === '/';
      const serversCurrent = !homeCurrent && s.path.indexOf('/servers') === 0;
      return vf.html`
        <nav class="header__nav">
          <a data-link href="${router.href('/')}" aria-current="${homeCurrent ? 'page' : 'false'}">${vf.t('nav.home')}</a>
          <a data-link href="${router.href('/servers')}" aria-current="${serversCurrent ? 'page' : 'false'}">${vf.t('nav.servers')}</a>
        </nav>
        <div class="header__favorites">
          <span class="header__favorites-label">${vf.t('favorites.label')}</span>
          <span data-ref="fav-count">${s.favorites.length}</span>
        </div>
        <div class="header__locale">
          <button type="button" class="header__locale-btn" data-action="locale" data-locale="en" aria-pressed="${s.locale === 'en'}">English</button>
          <button type="button" class="header__locale-btn" data-action="locale" data-locale="ko" aria-pressed="${s.locale === 'ko'}">한국어</button>
        </div>
      `;
    },
    delegates: [
      {
        selector: '[data-action="locale"]',
        eventType: 'click',
        onEvent: (e) => {
          const locale = e.target.getAttribute('data-locale');
          onLocaleChange(locale);
        }
      }
    ]
  });

  store.subscribe((state) => {
    header.setState({ favorites: state.favorites });
  });

  return header;
}
```

```js
import vf from '../lib/vfunc.esm.js';
import { fetchServers } from '../api.js';

export default function serversPage(ctx, router) {
  return vf.vfunc({
    tag: 'section',
    opts: { class: 'servers-page' },
    state: { status: 'loading', servers: [] },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p data-state="loading">${vf.t('servers.loading')}</p>
        `;
      }
      if (s.status === 'error') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p data-state="error">${vf.t('servers.error')}</p>
        `;
      }
      if (s.servers.length === 0) {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p data-state="empty">${vf.t('servers.empty')}</p>
        `;
      }
      return vf.html`
        <h1>${vf.t('servers.title')}</h1>
        <ul class="servers__list">
          ${s.servers.map((server) => vf.html`
            <li class="servers__item" data-id="${server.id}" data-status="${server.status}">
              <a data-link href="${router.href('/servers/' + server.id)}">${server.name}</a>
            </li>
          `)}
        </ul>
      `;
    },
    onMount(inst) {
      fetchServers()
        .then((servers) => {
          inst.setState({ status: 'ready', servers });
        })
        .catch(() => {
          inst.setState({ status: 'error', servers: [] });
        });
    }
  });
}
```

### pages/server-detail.js

```js
import vf from '../lib/vfunc.esm.js';
import { fetchServer } from '../api.js';
import { store, toggleFavorite, isFavorite } from '../store.js';

export default function serverDetailPage(ctx) {
  const id = ctx && ctx.params ? ctx.params.id : null;
  let unsubscribe = null;

  return vf.vfunc({
    tag: 'section',
    opts: { class: 'server-detail-page' },
    state: { status: 'loading', server: null },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`<h1>${vf.t('servers.loading')}</h1>`;
      }
      if (!s.server) {
        return vf.html`<h1>${vf.t('server.missing')}</h1>`;
      }
      const fav = isFavorite(s.server.id);
      return vf.html`
        <h1>${s.server.name}</h1>
        <dl class="server-detail__meta">
          <dt>${vf.t('nav.servers')}</dt>
          <dd>${s.server.region}</dd>
        </dl>
        <p class="server-detail__status" data-state="${s.server.status}">${s.server.status}</p>
        <button type="button" data-action="favorite" aria-pressed="${fav}">
          ${fav ? vf.t('server.favorite.remove') : vf.t('server.favorite.add')}
        </button>
      `;
    },
    delegates: [
      {
        selector: '[data-action="favorite"]',
        eventType: 'click',
        onEvent: (e) => {
          const current = e.sender.state.server;
          if (current) {
            toggleFavorite(current.id);
          }
        }
      }
    ],
    onMount(inst) {
      fetchServer(id)
        .then((server) => {
          inst.setState({ status: 'ready', server });
        })
        .catch(() => {
          inst.setState({ status: 'ready', server: null });
        });
      unsubscribe = store.subscribe(() => {
        inst.refresh();
      });
    },
    onDestroy() {
      if (unsubscribe) {
        unsubscribe();
        unsubscribe = null;
      }
    }
  });
}
```

### REPORT.md

```markdown
# 06-spa-scaffold 보고서

## 화면 목록과 상태 지도

| 라우트 | 페이지 파일 | 목적 | 데이터 | 동작 |
|---|---|---|---|---|
| `#/` | `pages/home.js` | 소개 문구 | 없음 | 없음 |
| `#/servers` | `pages/servers.js` | 서버 목록 | `data/servers.json`(`api.js`의 `fetchServers`) | 목록 항목 클릭 시 상세로 이동 |
| `#/servers/:id` | `pages/server-detail.js` | 서버 상세 | `fetchServer(id)` | 즐겨찾기 토글(`store.js`) |
| 그 밖 | `pages/not-found.js` | 404 | 없음 | 없음 |

상태 지도:
- 공유 상태(`store.js`, `vf.store`): `favorites`(즐겨찾기 id 배열). `toggleFavorite`/`isFavorite` 두 함수로만 바뀝니다. 새로고침 시 초기화(요구사항이 "새로고침은 아님"이라고 명시).
- 페이지 안에서만 쓰는 상태(컴포넌트 `state`): `servers.js`의 `status/servers`, `server-detail.js`의 `status/server`.
- 로케일(`currentLocale`)과 라우터 경로(`path`)는 `app.js`의 모듈 변수로 관리하고, 헤더 컴포넌트의 `state`(`path`, `locale`, `favorites`)로 전달합니다. 비밀값은 store/앱 어디에도 없습니다.

## 메시지 키(두 로케일)

`locales/en.json`, `locales/ko.json`에 과제 표의 12개 키(`nav.home`, `nav.servers`, `favorites.label`, `home.title`, `home.intro`, `servers.title`, `servers.loading`, `servers.error`, `servers.empty`, `server.favorite.add`, `server.favorite.remove`, `server.missing`, `notFound.title`)를 모두 넣었습니다. 언어 버튼 문구(`English`, `한국어`)는 언어 이름 자체이므로 키 없이 고정 문자열로 두었습니다(요구사항의 화면 문구와 정확히 일치).

## 새로 만들거나 바꾼 파일

`index.html`, `app.js`, `store.js`, `api.js`, `components/header.js`, `pages/home.js`, `pages/servers.js`, `pages/server-detail.js`, `pages/not-found.js`, `locales/en.json`, `locales/ko.json`, `data/servers.json`, `styles/tokens.css`, `styles/base.css`, `styles/components/header.css`, `styles/pages/home.css`, `styles/pages/servers.css`, `styles/pages/server-detail.css`. `lib/`는 킷에 이미 있다고 해서 손대지 않았습니다. `AGENTS.md`/`llms.txt`는 이 프로젝트에 복사하지 않았습니다(과제가 요구한 산출물 목록에 없어서 생략했습니다 — 필요하면 알려주세요).

## 가정과 VERIFY

- **`vf.router`의 `onChange` payload 모양은 킷에 정확히 문서화되어 있지 않습니다.** `routes`가 `{ path: (ctx) => component }` 형태이고 라우터 설정에 마운트 대상 옵션이 없는 점으로 미루어, `onChange(ctx)`가 `ctx.route`(매치된 라우트 함수, 없으면 `notFound` 처리)와 `ctx.params`, `ctx.path`를 담아 온다고 가정하고 `app.js`에서 직접 `ctx.route(ctx)`를 호출해 페이지를 만들고 `page.mount(viewEl)`로 붙였습니다. `ctx.route`가 없을 때는 직접 `notFoundPage(ctx, router)`를 호출하는 방어 코드를 넣어, 라우터가 내부적으로 이미 `notFound`를 `ctx.route`에 채워 주는 경우와 아닌 경우 둘 다 동작하도록 했습니다. **VERIFY: 실제 `ctx` 필드 이름(`route`/`params`/`path`)과 `notFound` 처리 방식.**
- **`focus: '#view'`가 라우터가 이동마다 그 요소에 포커스를 준다는 뜻으로 해석했습니다**(요구사항의 "이동할 때마다 이 요소가 포커스를 받습니다"와 정확히 맞아떨어져서 별도 포커스 코드를 추가하지 않았습니다). **VERIFY.**
- **초기 로케일 결정**: `vf.i18n.setup({ persist: true })`가 내부적으로 저장된 로케일/브라우저 언어를 어떻게 고르는지 문서에 상세히 없어서, `localStorage['vf.locale']` → 없으면 `navigator.language`(`ko`로 시작하면 `ko`, 아니면 `en`)를 앱이 직접 계산하고, `setup()` 직후 `vf.i18n.set(currentLocale)`을 명시적으로 호출해 확정했습니다. `header`/`router.start()`는 이 Promise가 끝난 뒤에만 만듭니다(컴포넌트는 `vf.i18n.setup()` 이후에 만들라는 킷의 "흔한 실수" 항목을 지키기 위함).
- **`vf.i18n` 서브스크라이브 콜백의 인자 형태가 문서에 없어서**, `subscribe`에 의존하지 않고 `vf.i18n.set(locale).then(...)`으로 로케일 변경 후 처리(헤더/현재 페이지 `refresh`, `<html lang>` 갱신)를 직접 수행했습니다. 동작은 동일하되 근거 API가 다르다는 점을 밝힙니다.
- **디자인 토큰 이름**: 킷에 명시된 것은 `--vf-color-primary`, `--vf-color-text-muted`, `--vf-space-3`, `--vf-chart-1..8`뿐입니다. 그 밖에 CSS에서 쓴 `--vf-color-bg`, `--vf-color-text`, `--vf-color-border`, `--vf-color-success`, `--vf-color-danger`, `--vf-color-warning`, `--vf-color-primary-contrast`, `--vf-font-sans`, `--vf-radius-md`, `--vf-space-2/4/6` 등은 같은 명명 규칙을 따른다고 가정한 것이며 실제 `lib/vfunc.tokens.css` 내용을 읽지 않았으므로 **VERIFY**입니다. 값이 없더라도 `var()` 미해결은 콘솔 에러를 내지 않고 해당 스타일만 무시되므로 기능과 "콘솔 에러 0개" 조건에는 영향이 없다고 판단했습니다.
- **`ctx.params.id`** 로 동적 세그먼트를 읽는다고 가정했습니다(`routes`의 `'/users/:id'` 예시로 미루어). **VERIFY.**
- **`opts: { class: '...' }`** 로 컴포넌트 루트에 블록 클래스를 준다고 가정했습니다(`opts`가 "루트 속성"이라고만 되어 있어 속성 이름 자체는 확인하지 못함). **VERIFY.**
- `AGENTS.md`, `llms.ko.txt`, `design/DESIGN.md` 자체는 과제가 요구한 파일 목록(`index.html`, `app.js`, `store.js`, `api.js`, `pages/*`, `components/*`, `locales/*`, `styles/`)에 없어 프로젝트에 복사하지 않았습니다. `DESIGN.md`가 없으므로 중립 토큰만 쓰고 `styles/tokens.css`는 거의 비워 두었습니다(요구사항 6절과 일치).
```
