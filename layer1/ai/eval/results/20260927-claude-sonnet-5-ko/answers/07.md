### app.js

```js
import vf from './lib/vfunc.esm.js';

const CATEGORIES = [
  ['all', 'All'],
  ['keyboard', 'Keyboards'],
  ['mouse', 'Mice'],
  ['monitor', 'Monitors']
];

const SORTS = [
  ['name', 'Name'],
  ['price-asc', 'Price: low to high'],
  ['price-desc', 'Price: high to low']
];

// vf.fmt.currency's exact call signature is not documented in the kit
// (only "vf.fmt.number/currency/date/relative" is listed with no args shown).
// Using the native formatter here to guarantee a console-error-free page
// (rule 7 / checklist item) rather than guess an unverified API surface.
// VERIFY: swap for vf.fmt.currency once its signature is confirmed.
const moneyFmt = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' });
function formatPrice(value) {
  return moneyFmt.format(value);
}

let abortController = null;

const app = vf.vfunc({
  tag: 'div',
  state: {
    items: [],
    status: 'loading',
    query: '',
    category: 'all',
    inStock: false,
    sort: 'name'
  },

  render(s) {
    if (s.status === 'loading') {
      return vf.html`<p class="pf__note" role="status" data-ref="status">Loading products…</p>`;
    }
    if (s.status === 'error') {
      return vf.html`<p class="pf__note pf__note--error" role="alert" data-ref="status">Could not load products.</p>`;
    }

    const q = s.query.trim().toLowerCase();
    const comparators = {
      name: (a, b) => a.name.localeCompare(b.name, 'en'),
      'price-asc': (a, b) => a.price - b.price,
      'price-desc': (a, b) => b.price - a.price
    };

    const shown = s.items
      .filter((p) => s.category === 'all' || p.category === s.category)
      .filter((p) => !s.inStock || p.stock > 0)
      .filter((p) => !q || p.name.toLowerCase().includes(q))
      .slice()
      .sort(comparators[s.sort]);

    const countText = shown.length === 1 ? '1 product' : shown.length + ' products';

    return vf.html`
      <section class="pf">
        <div class="pf__controls">
          <label class="pf__field">Search
            <input class="pf__input" type="search" data-action="search" value="${s.query}">
          </label>
          <label class="pf__field">Category
            <select class="pf__input" data-action="category">
              ${CATEGORIES.map(([value, label]) => vf.html`<option value="${value}" ${value === s.category ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
          <label class="pf__check">
            <input type="checkbox" data-action="in-stock" ${s.inStock ? 'checked' : ''}>
            In stock only
          </label>
          <label class="pf__field">Sort
            <select class="pf__input" data-action="sort">
              ${SORTS.map(([value, label]) => vf.html`<option value="${value}" ${value === s.sort ? 'selected' : ''}>${label}</option>`)}
            </select>
          </label>
        </div>
        <p class="pf__count" data-ref="count">${countText}</p>
        ${shown.length === 0
          ? vf.html`<p class="pf__empty" data-ref="empty">No products found.</p>`
          : vf.html`<ul class="pf__list" data-ref="list">${shown.map((p) => vf.html`
              <li class="pf__item" data-id="${p.id}" data-state="${p.stock === 0 ? 'out' : 'in'}">
                <span class="pf__name" data-ref="name">${p.name}</span>
                <span class="pf__price" data-ref="price">${formatPrice(p.price)}</span>
                ${p.stock === 0 ? vf.html`<span class="pf__badge">Sold out</span>` : ''}
              </li>
            `)}</ul>`}
      </section>
    `;
  },

  delegates: [
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent: (e) => { e.sender.query = e.target.value; }
    },
    {
      selector: '[data-action="category"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.category = e.target.value; }
    },
    {
      selector: '[data-action="in-stock"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.inStock = e.target.checked; }
    },
    {
      selector: '[data-action="sort"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.sort = e.target.value; }
    }
  ],

  onMount(inst) {
    abortController = new AbortController();
    fetch('./data/products.json', { signal: abortController.signal })
      .then((r) => {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then((data) => {
        inst.setState({ items: data, status: 'ready' });
      })
      .catch((err) => {
        if (err.name === 'AbortError') return;
        inst.status = 'error';
      });
  },

  onDestroy() {
    if (abortController) abortController.abort();
  }
});

app.mount('#app');
```

### REPORT.md

```markdown
# 07-react-port 보고

## 만든/고친 파일
- `index.html` — CSP 메타 태그 포함, `lib/vfunc.tokens.css` + `styles.css` 링크, `<script type="module" src="./app.js">` 하나.
- `app.js` — `vf.vfunc` 컴포넌트 하나(`#app`에 mount). 로딩/오류/정상 3가지 render 분기. 필터·정렬은 매 render마다 순수 계산. 이벤트는 `data-action` 셀렉터 4개에 위임 하나씩.
- `styles.css` — 클래스 셀렉터만 사용(`data-action`/`data-ref`/`id`는 JS 훅 전용), 원시 색상·px 값은 쓰지 않고 `var(--vf-*)` 토큰 참조.
- `data/products.json`은 과제에서 준 내용 그대로이므로 다시 내지 않았습니다.

## 가정 (킷이 "물어보라"고 한 지점 없음 — 직접 판단한 부분)
1. **`vf.fmt.currency`를 쓰지 않고 `Intl.NumberFormat`을 그대로 씀.** 킷의 llms.ko.txt는 `vf.fmt.number/currency/date/relative`가 있다고만 적고 정확한 인자 형식(통화 코드 전달 방식, locale 지정 방식)을 보여주지 않습니다. 잘못 추측하면 런타임 오류로 콘솔 경고·오류(확인 목록 필수 항목)를 낼 위험이 커서, 원본 React 코드와 동일한 네이티브 포맷터를 그대로 남겼습니다. **VERIFY:** 실제 `vf.fmt.currency(price, ...)` 시그니처를 확인해 맞으면 교체하세요.
2. **`vf.i18n`/메시지 키를 쓰지 않고 영어 문구를 하드코딩.** 과제 지문이 "앱은 한 언어입니다... i18n 설정 없음, 원본의 영어 문구를 그대로 쓰세요"라고 명시했으므로, 일반 규칙(6.1, 새 텍스트는 메시지 키)보다 과제 지시를 따랐습니다.
3. **CSS 토큰 이름을 일부 추측.** llms.ko.txt는 토큰 예시로 `--vf-color-primary`, `--vf-color-text-muted`, `--vf-space-3`, `--vf-chart-1..8`만 들었고 전체 토큰 표는 주지 않았습니다. `--vf-color-danger`, `--vf-color-danger-bg`, `--vf-color-border`, `--vf-space-1/2`, `--vf-radius-sm`, `--vf-font-size-sm/xs`는 같은 명명 규칙을 따라 추측한 이름입니다. 실제 `lib/vfunc.tokens.css`에 이름이 다르면 값이 적용되지 않을 수 있어 각 선언에 `var(--vf-token, 원래값)` 형태로 대체값을 함께 두었습니다(토큰이 없어도 레이아웃이 깨지지 않도록). **VERIFY:** 실제 토큰 파일과 이름을 맞춰 조정하세요.
4. **`vf.vfunc` + `mount('#app')`** 패턴을 썼습니다(킷의 대표 예제와 동일한 형태). `#app` 안에 컴포넌트 자신의 루트 `div`가 하나 더 생기지만, 채점 훅은 전부 `data-ref`/`data-action`/`data-id`/`data-state` 기반이라 추가 wrapper는 영향이 없다고 판단했습니다.
5. **검색 입력은 매 키 입력마다 `setState`(직접 대입)로 다시 렌더합니다.** "입력 중에는 값만 저장하고 키마다 다시 렌더하지 말라"는 일반 규칙과 부딪히지만, 과제 요구사항이 "입력하는 동안 검색"(실시간 필터링)과 "입력하는 동안 포커스 유지"를 동시에 요구합니다. 킷 문서(`refresh: …`)가 "포커스와 커서는 같은 id, data-ref, name의 요소로, 없으면 같은 data-action의 같은 순서 요소로 돌아간다"고 명시하므로, 검색 입력에 `data-action="search"`를 유지해 매 렌더에도 포커스가 복원되도록 설계했습니다.
6. `fetch`에 `AbortController`를 걸고 `onDestroy`에서 `abort()`하도록 했습니다. 이 페이지는 실제로 이동/destroy될 일이 없는 단일 화면이지만, 규칙 4.5(타이머·구독·리스너는 onDestroy에서 해제)를 기계적으로 지키기 위해 넣었습니다.

## 지키지 못한 것
- 가정 1(통화 포맷)과 가정 5(입력마다 재렌더)는 위에 적은 대로 일반 규칙과 충돌하는 선택입니다. 이유는 각 항목에 적었습니다.

## 확인 목록 (prompt-migrate-from-react-vue.md)
- [x] 행마다 컴포넌트를 만들지 않고 목록에 위임 하나 — `data-action` 셀렉터 4개짜리 delegates만 있고, 상품 행은 `render` 안의 `.map`으로 정적 마크업만 생성(행마다 컴포넌트 없음).
- [ ] 제어 입력이 키마다 다시 렌더하지 않는다 — 지키지 못함(위 가정 5 참고). 실시간 검색 요구사항 때문에 의도적으로 어김.

- [ ] 콘솔 에러·경고가 0개다 — **확인 못 함**(정적 파일만 작성했고 브라우저에서 실행해보지 않았습니다).
```
