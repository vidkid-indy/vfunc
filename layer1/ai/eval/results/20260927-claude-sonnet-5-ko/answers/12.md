### app.js

```js
// Profile — the page script.

(function () {
  'use strict';

  function optionsFor() {
    return [
      { value: 'admin', label: 'Admin' },
      { value: 'editor', label: 'Editor' },
      { value: 'viewer', label: 'Viewer' }
    ];
  }

  function renderFields(state) {
    var v = state.values;
    var err = state.errors;
    return vf.html`
      ${vf.vsInput({
        id: 'name',
        name: 'name',
        type: 'text',
        label: 'Name',
        value: v.name,
        required: true,
        error: err.name
      })}
      ${vf.vsInput({
        id: 'email',
        name: 'email',
        type: 'email',
        label: 'Email',
        hint: 'We never share it.',
        value: v.email,
        required: true,
        error: err.email
      })}
      ${vf.vsSelect({
        id: 'role',
        name: 'role',
        label: 'Role',
        options: optionsFor(),
        value: v.role
      })}
      ${vf.vsSwitch({
        id: 'newsletter',
        name: 'newsletter',
        label: 'Newsletter',
        checked: v.newsletter
      })}
      ${vf.vsButton({ label: 'Save', type: 'submit', variant: 'primary' })}
    `;
  }

  var toast = vf.vfToast();

  function validate(values) {
    var errors = {};
    var name = values.name.trim();
    if (!name) {
      errors.name = 'Enter your name.';
    }
    var email = values.email.trim();
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      errors.email = 'Enter a valid email address.';
    }
    return errors;
  }

  function onSubmit(e) {
    e.event.preventDefault();
    var inst = e.sender;
    var values = inst.state.values;
    var errors = validate(values);
    inst.state.errors = errors;
    inst.refresh();

    var order = ['name', 'email'];
    var i;
    for (i = 0; i < order.length; i++) {
      var key = order[i];
      if (errors[key]) {
        var el = inst.$node.querySelector('#' + key);
        if (el) { el.focus(); }
        return;
      }
    }

    var name = values.name.trim();
    var saved = vf.$('[data-ref="saved"]');
    if (saved) {
      saved.textContent = 'Saved: ' + name + ' (' + values.role + ')';
    }
    toast.show({ message: 'Profile saved' });
  }

  vf.attach(vf.$('#profile'), {
    state: {
      values: { name: '', email: '', role: 'viewer', newsletter: false },
      errors: {}
    },
    render: renderFields,
    events: [
      { eventType: 'submit', onEvent: onSubmit }
    ],
    delegates: [
      {
        selector: '#name',
        eventType: 'input',
        onEvent: function (e) { e.sender.state.values.name = e.event.target.value; }
      },
      {
        selector: '#email',
        eventType: 'input',
        onEvent: function (e) { e.sender.state.values.email = e.event.target.value; }
      },
      {
        selector: '#role',
        eventType: 'change',
        onEvent: function (e) { e.sender.state.values.role = e.event.target.value; }
      },
      {
        selector: '#newsletter',
        eventType: 'change',
        onEvent: function (e) { e.sender.state.values.newsletter = e.event.target.checked; }
      }
    ]
  });
}());
```

### REPORT.md

```markdown
## 가정
- `vf.vfToast()`는 킷 예제(`const toast = vf.vfToast(); toast.show({...})`)를 그대로 따라, 생성 즉시 스스로 `document.body`에 붙는다고 가정하고 별도 `mount()`를 호출하지 않았습니다.
- `id`와 `name`을 표의 값과 동일하게(`name`, `email`, `role`, `newsletter`) 그대로 사용했습니다. `<form id="profile">` 안에 `id="name"`인 입력을 둬도 HTML 스펙상 `name`은 `HTMLFormElement`의 기존 속성이라 named-property로 가려지지 않으므로 `form.name`과 충돌하지 않는다고 판단했습니다.
- `role` 옵션은 값(`admin`/`editor`/`viewer`)과 라벨(`Admin`/`Editor`/`Viewer`)이 다르므로 `vsSelect`의 `{ value, label }` 객체 형태로 넘겼습니다.
- `vsInput`/`vsSelect`/`vsSwitch`는 `label`/`hint`/`error`를 직접 받아 스스로 필드로 감싸 준다는 `components.md`의 공통 props 설명에 따라 `vsField`를 따로 쓰지 않았습니다.
- 키 입력 중에는 AGENTS.md/llms.ko.txt의 안내대로(`inst.state.value = …`, 다시 렌더하지 않음) `state.values`만 직접 바꾸고 `setState`/`refresh`를 호출하지 않아, 타이핑 중 입력칸이 다시 그려지며 커서가 튀는 문제를 피했습니다.
- 제출 시에는 `errors`를 갱신한 뒤 `inst.refresh()`로 동기적으로 다시 그리고, 실패한 첫 필드(표 순서: name → email)에 `el.focus()`로 직접 포커스를 옮겼습니다. `refresh()`의 자동 포커스 복원(같은 id로 되돌리는 동작)은 "현재 포커스 유지"용이라 "첫 번째 잘못된 필드로 이동"이라는 요구와는 다르므로 별도 처리를 그대로 두었습니다.
- 저장 문구(`Saved: <name> (<role>)`)는 순수 텍스트라 `vf.html`을 거치지 않고 `textContent`로 넣었습니다(`innerHTML`이 아니므로 이스케이프가 필요 없고 규칙 20의 "동적 HTML은 vf.html/vf.tpl로만" 대상은 아니라고 판단했습니다).
- `role`, `newsletter`에는 검증 규칙이 없으므로 `error` prop을 넘기지 않았습니다(항상 `undefined`).

## 킷에서 지키지 못한 것
- 없음. 페이지 하나짜리 컴포넌트라 `data-i18n`/로케일 키 대신 과제가 명시한 영어 문구를 그대로 하드코딩했습니다(과제 6번 조건이 "화면 문구는 과제에 적힌 영어 문구를 그대로 씁니다"를 명시했으므로 다국어 키를 새로 만들지 않았습니다).
```
