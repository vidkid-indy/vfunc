```js
// Servers — the page script (an ES module).
// Renders the server list into <section id="servers">, handling loading,
// error, empty and normal states, plus a status filter. All data from
// ./data/servers.json is treated as untrusted (another team's service).
import vf from './lib/vfunc.esm.js';

const DATA_URL = './data/servers.json';
const KNOWN_STATUSES = ['up', 'down', 'maintenance'];
const STATUS_LABELS = { up: 'Up', down: 'Down', maintenance: 'Maintenance', unknown: 'Unknown' };

// Untrusted response -> a safe, well-shaped array. Anything malformed is
// dropped rather than guessed at.
function sanitizeServers(raw) {
  if (!Array.isArray(raw)) return [];
  const out = [];
  for (const item of raw) {
    if (!item || typeof item !== 'object') continue;
    const id = item.id;
    if (typeof id !== 'string' || id === '') continue;
    const name = typeof item.name === 'string' && item.name !== '' ? item.name : id;
    const region = typeof item.region === 'string' ? item.region : '';
    const status = typeof item.status === 'string' ? item.status : '';
    const url = typeof item.url === 'string' ? item.url : '';
    out.push({ id, name, region, status, url });
  }
  return out;
}

function normalizeStatus(status) {
  return KNOWN_STATUSES.indexOf(status) !== -1 ? status : 'unknown';
}

// Only http(s) URLs become a clickable link; anything else (javascript:,
// data:, ftp:, relative junk, empty) renders as inert text-only "Open".
function isHttpUrl(url) {
  return /^https?:\/\//i.test(String(url || ''));
}

function renderServerItem(server) {
  const badgeState = normalizeStatus(server.status);
  const badgeLabel = STATUS_LABELS[badgeState];
  const link = isHttpUrl(server.url)
    ? vf.html`<a data-ref="link" class="servers__link" href="${server.url}" target="_blank" rel="noopener noreferrer">Open</a>`
    : vf.html`<a data-ref="link" class="servers__link">Open</a>`;

  return vf.html`
    <li class="servers__item" data-id="${server.id}">
      <span data-ref="name" class="servers__name">${server.name}</span>
      <span class="servers__region">${server.region}</span>
      <span data-ref="badge" class="servers__badge" data-state="${badgeState}">${badgeLabel}</span>
      ${link}
    </li>
  `;
}

function renderStatusText(state, visibleCount) {
  if (state.status === 'loading') {
    return 'Loading servers…';
  }
  if (state.status === 'error') {
    return vf.html`Could not load servers.
      <button type="button" data-action="retry" class="servers__retry">Retry</button>`;
  }
  if (state.servers.length === 0) {
    return 'No servers yet.';
  }
  return `Showing ${visibleCount} of ${state.servers.length}`;
}

const app = vf.attach('#servers', {
  state: { status: 'loading', servers: [], filter: 'all' },

  render(state) {
    const filter = state.filter;
    const visible = filter === 'all'
      ? state.servers
      : state.servers.filter((s) => normalizeStatus(s.status) === filter);

    return vf.html`
      <div class="servers__toolbar">
        <label class="servers__filter-label" for="status-filter">Status</label>
        <select id="status-filter" data-action="status-filter" class="servers__filter">
          <option value="all" ${filter === 'all' ? 'selected' : ''}>All</option>
          <option value="up" ${filter === 'up' ? 'selected' : ''}>Up</option>
          <option value="down" ${filter === 'down' ? 'selected' : ''}>Down</option>
          <option value="maintenance" ${filter === 'maintenance' ? 'selected' : ''}>Maintenance</option>
        </select>
      </div>
      <p data-ref="status" role="status" class="servers__status">${renderStatusText(state, visible.length)}</p>
      <ul data-ref="list" class="servers__list">
        ${visible.map(renderServerItem)}
      </ul>
    `;
  },

  delegates: [
    {
      selector: '[data-action="status-filter"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ filter: e.target.value }); }
    },
    {
      selector: '[data-action="retry"]',
      eventType: 'click',
      onEvent: (e) => { load(e.sender); }
    }
  ],

  onMount(inst) { load(inst); },

  onUpdate(inst) {
    inst.$node.setAttribute('aria-busy', inst.state.status === 'loading' ? 'true' : 'false');
  }
});

function load(inst) {
  inst.setState({ status: 'loading' });
  fetch(DATA_URL)
    .then((res) => {
      if (!res.ok) throw new Error('http-error');
      return res.json();
    })
    .then((data) => {
      inst.setState({ status: 'loaded', servers: sanitizeServers(data) });
    })
    .catch(() => {
      inst.setState({ status: 'error' });
    });
}
```

```markdown
# 서버 목록 과제 — 보고서

## 요약
- `app.js`에서 `vf.attach('#servers', …)`로 이미 있는 마크업(#servers 섹션)에 동작을 붙였습니다. 새 화면/라우트가 아니라 퍼블리싱 페이지 하나이므로 AGENTS.md의 SPA 구조(pages/, store.js, api.js)는 만들지 않았습니다 — 이 과제 범위(정적 페이지 + 데이터 파일 하나)에는 맞지 않는다고 판단했습니다.
- 상태: `loading` / `error` / `loaded`(+ 클라이언트 쪽 `filter`)를 컴포넌트 state로 관리하고, `render`가 상태에 따라 `data-ref="status"` 문구와 목록을 다시 그립니다. `aria-busy`는 루트 속성이라 `onUpdate`에서 `inst.$node.setAttribute`로 설정했습니다(킷의 "루트 속성은 onMount/onUpdate에서" 규칙).
- 데이터는 `sanitizeServers()`로 정제합니다: 배열이 아니면 빈 배열, 각 항목은 객체이고 `id`가 비어있지 않은 문자열일 때만 채택하며 나머지 필드는 문자열이 아니면 빈 문자열로 대체합니다. 신뢰할 수 없는 서버 응답에 이상한 타입·필드가 섞여도 죽지 않게 하려는 의도입니다.
- 상태 배지는 `up`/`down`/`maintenance`만 인정하고 그 밖은 전부 `unknown`으로 접습니다(과제 문구대로).
- 링크: `isHttpUrl()`로 `http:`/`https:`만 골라 그때만 `href`를 넣은 `<a>`를 만들고, 그 밖에는 `href` 속성 자체가 없는 `<a data-ref="link">Open</a>`을 만듭니다. `vf.safeUrl`의 `'#'` 대체값(같은 페이지로의 앵커 이동)도 "동작하는 링크"로 볼 여지가 있어, 더 엄격하게 href를 아예 안 붙이는 쪽을 택했습니다. 이 가정은 제가 직접 판단해 고른 것입니다.
- 모든 사용자 데이터(name, region, url 등)는 `vf.html` 보간을 통해서만 마크업에 들어가므로 텍스트로만 표시되고 이스케이프됩니다. `vf.unsafeHtml`은 쓰지 않았습니다.
- 필터 select(`data-action="status-filter"`)는 `change` 이벤트로 `state.filter`만 바꾸고 다시 fetch하지 않습니다(이미 받은 목록을 클라이언트에서 거름). `Showing X of Y`의 X는 필터링된 개수, Y는 전체(필터 무관) 개수입니다.
- CSS는 킷(`llms.ko.txt`)에 실제로 이름이 언급된 토큰만 썼습니다: `--vf-color-bg/text/surface/border/text-muted/primary`, `--vf-font-body`, `--vf-font-size-2xl`, `--vf-font-weight-strong`, `--vf-line-height`, `--vf-space-3/4/6`, `--vf-radius-lg`, `--vf-chart-1/2/3`. 다른 간격·모서리 크기 토큰(`--vf-space-1`, `--vf-radius-sm` 등)이 실제로 있는지 이 과제에서는 `lib/vfunc.tokens.css`를 읽지 않았으므로 확인하지 못했고, 확인된 토큰만 재사용해 새 값을 지어내지 않았습니다.
- 텍스트는 과제가 못박은 정확한 영어 문구(`Loading servers…`, `Could not load servers.`, `Retry`, `No servers yet.`, `Showing … of …`, `Up`/`Down`/`Maintenance`/`Unknown`, `Open`, `All`)를 그대로 하드코딩했습니다. AGENTS.md의 다국어 규칙(모든 메시지는 키로)과 어긋나지만, 채점기가 이 정확한 문구를 찾는다는 조건 6을 우선했습니다. 로케일 파일은 이 프로젝트에 없어서 만들지 않았습니다.

- [ ] 콘솔 에러·경고 0개 — **확인 못 함**(브라우저·서버 실행이 금지된 평가 조건이라 실제로 열어보지 못했습니다). 코드 리뷰상 `vf.html`의 거부 규칙(온 이벤트 속성, 따옴표 없는 속성 등)에 걸리는 부분은 없다고 판단했습니다.
- [x] 비밀값 없음.

## 가정
1. `href` 미지정 시 앵커에 `href` 속성 자체를 생략(=`'#'` 대체보다 더 엄격하게 "동작하는 링크 아님" 요구를 만족).
2. `id`가 없거나 빈 문자열인 서버 항목은 목록에서 제외(신뢰할 수 없는 데이터 방어).

4. 이 과제 범위에서는 SPA 구조(app.js/store.js/api.js/pages)를 새로 만들지 않고 주어진 4개 파일(`index.html`, `app.js`, `style.css`, `data/servers.json`) 안에서만 구현했습니다. `index.html`과 `data/servers.json`은 이미 요구사항과 맞아 바꾸지 않았습니다.
```
