### app.js

```js
// Files — the page script.
const FILES = [
  { id: 'f1', name: 'report.pdf' },
  { id: 'f2', name: 'photo.png' },
  { id: 'f3', name: 'notes.txt' }
];

function countText(n) {
  return n === 1 ? '1 file' : n + ' files';
}

function updateCount(n) {
  vf.$('[data-ref="count"]').textContent = countText(n);
}

// One notification region for the whole page.
const toast = vf.vfToast();

// The list is the only dynamic region; the count paragraph next to it is
// plain text updated directly (not part of the vf.attach root).
const list = vf.attach(vf.$('#files'), {
  state: { files: FILES.slice() },
  render: (s) => vf.html`${s.files.map((f) => vf.html`
    <li class="files__item" data-id="${f.id}">
      <span class="files__name">${f.name}</span>
      ${vf.vsButton({ id: 'delete-' + f.id, action: 'delete', label: 'Delete', ariaLabel: 'Delete ' + f.name })}
    </li>
  `))}`,
  delegates: [
    {
      selector: '[data-action="delete"]',
      eventType: 'click',
      onEvent: async (e) => {
        const inst = e.sender;
        const id = e.target.id.slice('delete-'.length);
        const index = inst.state.files.findIndex((f) => f.id === id);
        if (index === -1) return;
        const file = inst.state.files[index];

        const confirmed = await vf.vfConfirm({
          title: 'Delete ' + file.name + '?',
          message: 'This cannot be undone.',
          variant: 'danger',
          confirmLabel: 'Delete'
        }).open();
        if (!confirmed) return;

        const remaining = inst.state.files.filter((f) => f.id !== id);
        // Next file at the same index, else the previous one, else none.
        const next = remaining[index] || remaining[index - 1] || null;

        inst.state.files = remaining;
        inst.refresh();
        updateCount(remaining.length);
        toast.show({ message: 'Deleted ' + file.name, variant: 'success' });

        const focusTarget = next ? vf.$('#delete-' + next.id) : vf.$('#files-title');
        if (focusTarget) focusTarget.focus();
      }
    }
  ]
});

updateCount(FILES.length);
```

### REPORT.md

## 가정
- Delete 행 버튼은 `vf.vsButton({ id, action: 'delete', label: 'Delete', ariaLabel })`로 만들었습니다. `id="delete-<id>"` 형식이 과제에 명시되어 있어, 삭제 클릭 핸들러에서 `e.target.id`를 `'delete-'` 접두어로 잘라 파일 id를 구했습니다(별도 `data-id`를 버튼에 붙일 방법이 vsButton props에 없어서 택함).
- 취소/Escape/닫기 버튼에서 포커스가 원래 Delete 버튼으로 돌아가는 동작은 `vfModal`/`vfConfirm`의 기본 포커스 복원(트리거로 되돌림) 동작에 맡겼고, 별도 코드를 추가하지 않았습니다. 킷 문서에 "포커스를 안에 가두고 되돌리며"라고 되어 있어 확인 버튼을 누르지 않고 닫는 모든 경로(취소/Escape/닫기 버튼/배경)에서 자동으로 트리거(클릭했던 Delete 버튼)로 돌아간다고 가정했습니다.
- toast의 `variant: 'success'`는 과제에 명시되지 않아 삭제 완료를 알리는 일반적인 선택으로 정했습니다.
- `<li>` / `<span>`에 `files__item`, `files__name` 클래스를 두었는데, `files__item`은 이미 `style.css`에 있던 이름을 재사용했고 `files__name`은 새 클래스지만 CSS를 추가하지 않았습니다(디자인 값 없이 구조만 표시하는 훅이라 기존 `style.css`로 충분하다고 판단, 필요하면 별도로 `.files__name`에 `flex: 1` 등을 토큰만으로 추가할 수 있음).
- 카운트 문구(`<n> files` / `1 file`)는 메시지 키(`vf.t`)를 쓰지 않고 JS 리터럴로 만들었습니다. 이 프로젝트에는 `vf.i18n.setup()`이나 로케일 파일이 없어(스타터 SPA 구조가 아닌 단일 페이지 조각) 다국어 인프라가 갖춰지지 않았다고 보고, 과제가 요구한 정확한 영어 문구만 만들었습니다. AGENTS.md의 메시지 키 규칙(6절)은 이 축소된 킷 범위에서는 적용할 로케일 파일이 없어 지키지 못했습니다.

## 지키지 못한 것
- 다국어 메시지 키(`vf.t`, `data-i18n`) 사용 규칙(AGENTS.md 6절)을 지키지 못했습니다. 이 과제 범위(`index.html`, `app.js`, `style.css`)에 로케일 파일이 없고 과제가 정확한 영어 문구를 요구했기 때문입니다.

## 확인 목록 (요청된 "완료" 전 확인 목록 기준)
- [확인 못 함] 모든 동적 값이 `vf.html`을 거친다 — 코드상으로는 파일명·삭제 확인 문구 모두 `vf.html` 템플릿이나 `vf.vsButton`/`vf.vfConfirm` props를 통해서만 들어가고 `innerHTML` 문자열 이어붙이기가 없음을 검토했으나, 브라우저에서 직접 실행해 보지는 못했습니다.
- [확인 못 함] 셀렉터는 `data-action`/`id`만 — `[data-action="delete"]` 위임과 `id="delete-<id>"` / `#files-title` 참조만 사용했고 CSS 클래스로 선택하는 코드는 없음을 코드 검토로 확인했지만, 실행 확인은 못 했습니다.
- [확인 못 함] 리스너/타이머 해제 — 이 페이지는 destroy되지 않는 단일 페이지라 `onDestroy` 훅을 두지 않았습니다(등록한 리스너가 없어 해제할 것도 없음).
- [확인 못 함] 콘솔 에러·경고 0개 — 실제 브라우저(Chromium/Firefox/WebKit)로 열어 확인하지 못했습니다.
- [확인 못 함] 포커스 이동(다음 → 이전 → 제목)과 취소/Escape 시 포커스 복원 — 로직은 위 설명대로 짰지만 실제 동작은 실행해서 검증하지 못했습니다.
