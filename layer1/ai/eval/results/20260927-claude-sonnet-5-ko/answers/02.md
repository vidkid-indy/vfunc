### app.js

```js
// My tasks — the page script. The to-do list lives in #todo.
(function () {
  'use strict';

  var nextId = 1;

  var todo = vf.attach(document.getElementById('todo'), {
    state: {
      items: [],
      filter: 'all'
    },

    render: function (s) {
      var filtered = s.items.filter(function (item) {
        if (s.filter === 'open') return !item.done;
        if (s.filter === 'done') return item.done;
        return true;
      });
      var leftCount = s.items.filter(function (item) { return !item.done; }).length;

      return vf.html`
        <form class="todo__form" data-action="add">
          <label class="todo__label" for="new-task-input">New task</label>
          <input class="todo__input" type="text" id="new-task-input" data-ref="input" name="task" autocomplete="off">
          <button class="todo__add" type="submit">Add</button>
        </form>
        <div class="todo__list" data-ref="list">
          ${filtered.map(function (item) {
            return vf.html`
              <div class="todo__item" data-id="${item.id}" data-state="${item.done ? 'done' : 'open'}">
                <input class="todo__checkbox" type="checkbox" data-action="toggle" id="todo-${item.id}" ${item.done ? 'checked' : ''}>
                <span class="todo__text" data-ref="text">${item.text}</span>
                <button class="todo__remove" type="button" data-action="remove">Remove</button>
              </div>
            `;
          })}
        </div>
        <div class="todo__filters">
          <button class="todo__filter" type="button" data-action="filter" data-filter="all" aria-pressed="${s.filter === 'all'}">All</button>
          <button class="todo__filter" type="button" data-action="filter" data-filter="open" aria-pressed="${s.filter === 'open'}">Open</button>
          <button class="todo__filter" type="button" data-action="filter" data-filter="done" aria-pressed="${s.filter === 'done'}">Done</button>
        </div>
        <p class="todo__count">Left: <span data-ref="left">${leftCount}</span></p>
        ${filtered.length === 0 ? vf.html`<p class="todo__empty" data-ref="empty">Nothing here.</p>` : ''}
      `;
    },

    delegates: [
      {
        selector: '[data-action="add"]',
        eventType: 'submit',
        onEvent: function (e) {
          e.event.preventDefault();
          var inst = e.sender;
          var value = inst.refs.input.value.trim();
          if (!value) return;
          var id = 't' + (nextId++);
          inst.setState({
            items: inst.state.items.concat([{ id: id, text: value, done: false }])
          });
        }
      },
      {
        selector: '[data-action="toggle"]',
        eventType: 'change',
        onEvent: function (e) {
          var row = e.target.closest('[data-id]');
          if (!row) return;
          var id = row.getAttribute('data-id');
          var inst = e.sender;
          var items = inst.state.items.map(function (item) {
            if (item.id !== id) return item;
            return { id: item.id, text: item.text, done: !item.done };
          });
          inst.setState({ items: items });
        }
      },
      {
        selector: '[data-action="remove"]',
        eventType: 'click',
        onEvent: function (e) {
          var row = e.target.closest('[data-id]');
          if (!row) return;
          var id = row.getAttribute('data-id');
          var inst = e.sender;
          inst.setState({
            items: inst.state.items.filter(function (item) { return item.id !== id; })
          });
        }
      },
      {
        selector: '[data-action="filter"]',
        eventType: 'click',
        onEvent: function (e) {
          var filter = e.target.getAttribute('data-filter');
          e.sender.setState({ filter: filter });
        }
      }
    ]
  });
}());
```

### REPORT.md

## 바뀐 파일

- **JS**: `app.js` — `vf.attach(#todo, { render, delegates })`로 할 일 목록 전체를 구현.
- **CSS**: `style.css` — `todo__*` 블록에 기존 `--vf-*` 토큰만 사용해 규칙 추가(원시 색상값·px 없음).
- **HTML**: 변경 없음(`index.html`은 그대로 사용, `#todo` 섹션의 기존 속성은 `vf.attach`가 유지).

## 가정

1. 기존 `#todo`(section)에 `vf.attach`를 쓰고 `render`로 안쪽 전체를 그렸습니다. `render`가 요소의 안쪽만 반환하므로 섹션 자체의 `class="todo"`, `aria-label`은 그대로 유지됩니다.
2. 할 일 id는 `t1, t2, …`처럼 모듈 안 카운터로 생성했습니다(메모리 전용, 새로고침 시 초기화되는 요구사항과 일치).
3. 입력칸 값은 state에 넣지 않고 제출 시 `inst.refs.input.value`만 읽습니다(입력 중 재렌더 방지, 킷의 "입력 중에는 값만 저장" 원칙). 추가 후에는 `render`가 `value` 속성 없는 새 input을 그리므로 자동으로 비워지고, `data-ref="input"` 매칭으로 포커스가 자동 복원됩니다(엔진의 refresh 포커스 규칙).
4. 체크박스는 `id="todo-<data-id>"`를 가지므로 refresh 후에도 같은 id로 포커스가 복원됩니다(토글 후 포커스 유지 요구사항).
5. 스타일 훅과 동작 훅을 분리하기 위해 `todo__*` 형태의 클래스를 추가로 붙였습니다(과제에서 요구하지 않았지만 디자인 분리 규칙 준수 목적이며, 이벤트/참조는 전부 `data-action`/`data-ref`/`id`만 사용).
6. 텍스트는 모두 `vf.html`을 통해서만 삽입되므로 자동 이스케이프됩니다(`<img onerror>` 같은 입력은 글자 그대로 표시).
7. 필터 값이 없는 할 일 목록일 때 `data-ref="empty"` 요소는 조건부로 아예 렌더링하지 않는 방식(엘리먼트 없음)을 택했습니다. 과제에서 "없거나 숨겨져 있습니다" 둘 다 허용하므로 이 방식을 선택했습니다.

## 킷에서 지키지 못한 것

- 없음. 보안(동적 마크업은 `vf.html`만, `on*` 없음, CSP 유지), 이벤트 위임(`data-action` + 컨테이너 단일 위임), 디자인 분리(JS에 색상 없음, CSS는 토큰만) 모두 지켰습니다.
- 다국어(`vf.t`/로케일 파일)는 이 프로젝트에 로케일 인프라가 없고 과제가 영어 문구를 그대로 쓰라고 요구했으므로 하드코딩했습니다(단일 언어 소규모 페이지라 킷의 "새 텍스트는 메시지 키" 원칙에서 의도적으로 벗어난 유일한 지점).
