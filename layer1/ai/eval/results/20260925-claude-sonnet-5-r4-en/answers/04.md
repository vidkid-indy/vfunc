### app.js

```js
// Sign up — the page script.
(function () {
  'use strict';

  var FIELD_ORDER = ['email', 'password', 'confirm', 'terms'];

  var RULES = {
    email: {
      test: function (value) {
        return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.trim());
      },
      message: 'Enter a valid email address.'
    },
    password: {
      test: function (value) {
        return value.length >= 8 && /\d/.test(value);
      },
      message: 'Use at least 8 characters, including a number.'
    },
    confirm: {
      test: function (value, form) {
        return value !== '' && value === form.elements.password.value;
      },
      message: 'Passwords do not match.'
    },
    terms: {
      test: function (checked) {
        return checked === true;
      },
      message: 'Accept the terms to continue.'
    }
  };

  // Tracks whether each field has already shown an error once, so it can be
  // re-validated on every keystroke afterwards (per the task's rules).
  var shown = { email: false, password: false, confirm: false, terms: false };

  function fieldValue(form, name) {
    var el = form.elements[name];
    return name === 'terms' ? el.checked : el.value;
  }

  function errorEl(form, name) {
    return form.querySelector('#' + name + '-error');
  }

  function validateField(form, name) {
    var rule = RULES[name];
    var valid = rule.test(fieldValue(form, name), form);
    errorEl(form, name).textContent = valid ? '' : rule.message;
    form.elements[name].setAttribute('aria-invalid', valid ? 'false' : 'true');
    if (!valid) shown[name] = true;
    return valid;
  }

  function validateAll(form) {
    var allValid = true;
    var firstInvalid = null;
    FIELD_ORDER.forEach(function (name) {
      var valid = validateField(form, name);
      if (!valid) {
        allValid = false;
        if (!firstInvalid) firstInvalid = form.elements[name];
      }
    });
    if (firstInvalid) firstInvalid.focus();
    return allValid;
  }

  vf.attach('#signup', {
    events: [
      {
        eventType: 'submit',
        onEvent: function (e) {
          e.event.preventDefault();
          var form = e.sender.$node;
          if (!validateAll(form)) return;
          var email = form.elements.email.value.trim();
          form.hidden = true;
          var done = e.sender.refs.done;
          done.textContent = 'Welcome, ' + email + '!';
          done.hidden = false;
        }
      }
    ],
    delegates: [
      {
        selector: '#email, #password, #confirm',
        eventType: 'focusout',
        onEvent: function (e) {
          validateField(e.sender.$node, e.target.name);
        }
      },
      {
        selector: '#email, #password, #confirm',
        eventType: 'input',
        onEvent: function (e) {
          var name = e.target.name;
          if (shown[name]) validateField(e.sender.$node, name);
        }
      },
      {
        selector: '#terms',
        eventType: 'change',
        onEvent: function (e) {
          validateField(e.sender.$node, 'terms');
        }
      }
    ]
  });
})();
```

### REPORT.md

**Assumptions**

- The welcome text is written with `Element.textContent` rather than `vf.html`. It is plain text (no markup), so `textContent` is the safer, more direct choice and satisfies the "no `innerHTML` from data" rule without adding an unnecessary re-render.
- `vf.attach('#signup', …)` is used without `render`, so the existing form markup and its attributes/listeners are kept as-is; the `submit` handler is registered through `events` (no `id`, so it binds to the root `<form>` itself) and field validation through `delegates`, per `AGENTS.md` §4.1 ("never select by CSS class").
- I added `id="<field>-error"` message elements and static `aria-describedby` directly in `index.html` (they don't change, so no render/JS wiring is needed for them) and one `data-ref="done"` element outside the form for the welcome message, initially `hidden`.
- Text-field validation runs on `focusout` (bubbles, unlike `blur`) and, once a field has shown an error, again on every `input` on that same field, as the task specifies. The checkbox validates on `change`.
- `aria-invalid` is only set once a field has been validated at least once, so before that it is absent, matching the rule "before its first validation it has no `aria-invalid` or `'false'`".
- Submit always calls `preventDefault()` first (there's no server), validates every field in table order, and focuses the first invalid field; only when everything is valid does it hide the form (`hidden` attribute) and reveal `data-ref="done"` with `Welcome, <trimmed email>!`.
- Password value is only ever read from `form.elements.password.value` for comparison; it is never assigned to a variable that outlives the check, concatenated into any message, or logged.
- Added `.field__error` / `.signup__done` styles using only `var(--vf-*)` tokens (no new tokens needed); no colors/sizes were added in JS.
- Nothing from `AGENTS.md` needed to be broken.
