### app.js

```js
// My tasks — the page script. The to-do list lives in #todo (in-memory only).
(function () {
  'use strict';

  var nextId = 1;

  function matchesFilter(item, filter) {
    if (filter === 'open') return !item.done;
    if (filter === 'done') return item.done;
    return true;
  }

  vf.attach('#todo', {
    state: { items: [], filter: 'all' },

    render: function (s) {
      var filtered = s.items.filter(function (it) { return matchesFilter(it, s.filter); });
      var openCount = s.items.filter(function (it) { return !it.done; }).length;

      return vf.html`
        <form data-action="add" class="todo__form">
          <label class="todo__label" for="todo-input">New task</label>
          <input type="text" id="todo-input" data-ref="input" name="task" class="todo__input" autocomplete="off">
          <button type="submit" class="todo__add">Add</button>
        </form>
        <div class="todo__filters" role="group" aria-label="Filter">
          <button type="button" data-action="filter" data-filter="all" aria-pressed="${s.filter === 'all'}" class="todo__filter">All</button>
          <button type="button" data-action="filter" data-filter="open" aria-pressed="${s.filter === 'open'}" class="todo__filter">Open</button>
          <button type="button" data-action="filter" data-filter="done" aria-pressed="${s.filter === 'done'}" class="todo__filter">Done</button>
        </div>
        <ul data-ref="list" class="todo__list">
          ${filtered.map(function (it) {
            return vf.html`
              <li data-id="${it.id}" data-state="${it.done ? 'done' : 'open'}" class="todo__item">
                <input type="checkbox" data-action="toggle" id="todo-${it.id}" ${it.done ? 'checked' : ''} class="todo__checkbox">
                <span data-ref="text" class="todo__text">${it.text}</span>
                <button type="button" data-action="remove" class="todo__remove">Remove</button>
              </li>
            `;
          })}
        </ul>
        ${filtered.length === 0 ? vf.html`<p data-ref="empty" class="todo__empty">Nothing here.</p>` : ''}
        <p class="todo__count">Left: <span data-ref="left">${openCount}</span></p>
      `;
    },

    delegates: [
      {
        selector: '[data-action="add"]',
        eventType: 'submit',
        onEvent: function (e) {
          e.event.preventDefault();
          var input = e.sender.refs.input;
          var value = (input && input.value ? input.value : '').trim();
          if (!value) return;
          var id = String(nextId++);
          e.sender.setState(function (s) {
            return { items: s.items.concat([{ id: id, text: value, done: false }]) };
          });
        }
      },
      {
        selector: '[data-action="filter"]',
        eventType: 'click',
        onEvent: function (e) {
          var filter = e.target.getAttribute('data-filter');
          e.sender.setState({ filter: filter });
        }
      },
      {
        selector: '[data-action="toggle"]',
        eventType: 'change',
        onEvent: function (e) {
          var row = e.target.closest('[data-id]');
          if (!row) return;
          var id = row.getAttribute('data-id');
          var checked = e.target.checked;
          e.sender.setState(function (s) {
            return {
              items: s.items.map(function (it) {
                return it.id === id ? { id: it.id, text: it.text, done: checked } : it;
              })
            };
          });
        }
      },
      {
        selector: '[data-action="remove"]',
        eventType: 'click',
        onEvent: function (e) {
          var row = e.target.closest('[data-id]');
          if (!row) return;
          var id = row.getAttribute('data-id');
          e.sender.setState(function (s) {
            return { items: s.items.filter(function (it) { return it.id !== id; }) };
          });
        }
      }
    ]
  });
})();
```

### REPORT.md

## Assumptions

- `vf.attach('#todo', { render, ... })` renders only the inside of `<section id="todo">`, so `index.html` and its CSP/tokens/link setup did not need to change.
- Task ids are generated from a module-scoped counter (`nextId`, closure variable in the IIFE), not stored as a custom instance property, per the kit's rule against adding properties to a `vfunc` instance.
- "The current filter shows no task" is evaluated against the *filtered* list (`filtered.length === 0`), not the whole `items` array, so `data-ref="empty"` appears whenever the visible list under the active filter is empty (including "no tasks yet" and "no open/done tasks").
- The `data-ref="left"` element's own text content is the bare number (e.g. `2`); the surrounding `"Left: "` text lives on a sibling, outside that element, so the element itself still satisfies "shows the number ... as a bare number".
- Empty/space-only input is rejected by trimming and checking for a non-empty result before adding.
- Focus return to the input after adding, and to the checkbox after a keyboard (Space) toggle, relies on the engine's documented refresh behavior ("focus and caret return to the element with the same id, data-ref or name"), since both the input (`id="todo-input"`/`data-ref="input"`) and each checkbox (`id="todo-<data-id>"`) keep the same id across re-renders. This was not run in a browser, so it is marked "not verified" below rather than assumed confirmed.
- Wrote `app.js` in ES5-style function syntax (no arrow functions/`const`/`let`) even though section 0 of `AGENTS.md` was left as a template placeholder; this is a defensive, harmless choice since the task didn't specify IE11 support, and it costs nothing given `vf.html` tagged templates are used regardless.
- No new locale files were added: the task gave no `locales/` folder and no i18n requirement, and all visible strings ("New task", "Add", "All", "Open", "Done", "Remove", "Nothing here.") are treated as literal English per the fixed task text rather than routed through `vf.t`, since introducing i18n scaffolding wasn't requested and would risk mismatching the exact required visible texts.

- Did not run the page in a browser (no tool access in this one-shot task), so "no console errors or warnings" and the exact runtime focus-retention behavior are **not verified** — they follow directly from the documented engine behavior but were not observed live.
- Did not add corresponding `locales/*.json` files since the project has no `locales/` folder in the given file set and the task's checklist (from `prompt-add-feature.md`) is being treated as advisory context, not a literal requirement for this fixed-spec task.
