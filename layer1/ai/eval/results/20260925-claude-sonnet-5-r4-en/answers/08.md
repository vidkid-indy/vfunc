### app.js

```js
// Sales — the page script. Chart.js is loaded as the global `Chart`.

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const SERIES = {
  revenue: { label: 'Revenue', values: [120, 135, 150, 142, 168, 181, 175, 190, 205, 198, 220, 236] },
  orders: { label: 'Orders', values: [30, 32, 35, 33, 40, 44, 41, 46, 50, 48, 53, 57] }
};

const CHART_TOKEN = { revenue: '--vf-chart-1', orders: '--vf-chart-2' };

// Reads a chart color token from the CSS variables. Only place a library
// (Chart.js) needs a raw color value, per AGENTS.md 5.3.
function chartColor(series) {
  return getComputedStyle(document.documentElement).getPropertyValue(CHART_TOKEN[series]).trim();
}

// Creates the sales component. The Chart.js instance is kept in a closure
// variable (not an instance property: AGENTS.md / llms.txt forbid adding
// custom properties to a vfunc instance) so it survives re-renders and is
// released on destroy.
function createSales() {
  let chart = null;

  return vf.attach('#sales', {
    state: { series: 'revenue', monthCount: 6 },
    render: (s) => {
      const data = SERIES[s.series];
      return vf.html`
        <div class="sales__controls">
          <button type="button" class="sales__button" data-action="series" data-series="revenue" aria-pressed="${s.series === 'revenue'}">Revenue</button>
          <button type="button" class="sales__button" data-action="series" data-series="orders" aria-pressed="${s.series === 'orders'}">Orders</button>
          <button type="button" class="sales__button" data-action="add-month" ${s.monthCount >= MONTHS.length ? 'disabled' : ''}>Add month</button>
        </div>
        <p class="sales__summary" data-ref="summary">${data.label}: ${s.monthCount} months, last ${data.values[s.monthCount - 1]}</p>
        <div class="sales__chart-wrap" data-vf-keep="chart-wrap"><canvas data-ref="chart" width="600" height="300"></canvas></div>
      `;
    },
    delegates: [
      {
        selector: '[data-action="series"]', eventType: 'click',
        onEvent: (e) => { e.sender.setState({ series: e.target.getAttribute('data-series') }); }
      },
      {
        selector: '[data-action="add-month"]', eventType: 'click',
        onEvent: (e) => { e.sender.setState((s) => ({ monthCount: Math.min(MONTHS.length, s.monthCount + 1) })); }
      }
    ],
    onMount: (inst) => {
      const s = inst.state;
      const data = SERIES[s.series];
      chart = new Chart(inst.refs.chart, {
        type: 'line',
        data: {
          labels: MONTHS.slice(0, s.monthCount),
          datasets: [{ label: data.label, data: data.values.slice(0, s.monthCount), borderColor: chartColor(s.series) }]
        },
        options: { responsive: false }
      });
    },
    onUpdate: (inst) => {
      if (!chart) { return; }
      const s = inst.state;
      const data = SERIES[s.series];
      chart.data.labels = MONTHS.slice(0, s.monthCount);
      chart.data.datasets[0].label = data.label;
      chart.data.datasets[0].data = data.values.slice(0, s.monthCount);
      chart.data.datasets[0].borderColor = chartColor(s.series);
      chart.update();
    },
    onDestroy: () => {
      if (chart) {
        chart.destroy();
        chart = null;
      }
    }
  });
}

let sales = null;

function showSales() {
  sales = createSales();
}

function hideSales() {
  if (sales) {
    sales.destroy();
    sales = null;
  }
}

// The toggle button lives in the page header, outside #sales, so it is
// wired with its own vf.attach (behaviour only, markup kept as published).
vf.attach('body', {
  delegates: [
    {
      selector: '[data-action="toggle-chart"]', eventType: 'click',
      onEvent: (e) => {
        if (sales) {
          hideSales();
          e.target.textContent = 'Show chart';
        } else {
          showSales();
          e.target.textContent = 'Hide chart';
        }
      }
    }
  ]
});

showSales();
```

- `index.html` needed no changes: the Chart.js `<script>` tag, the CSP meta tag and the `#sales` section were already correct for this task, so it is not repeated in the output (output rule 1: files not given stay as they are).
- The toggle button (`data-action="toggle-chart"`) lives in the page header, outside `#sales`, so it cannot be wired from inside the sales component. It is wired with a separate `vf.attach('body', { delegates: [...] })` call — behaviour only, no `render`, so the published markup (including the button's starting text `Hide chart`) is kept as-is. `body` was chosen as the attach target only because the header has no `id`; the delegate selector itself is still `[data-action="toggle-chart"]`, not a class, per AGENTS.md 4.1.
- The sales component is built with `vf.attach('#sales', { state, render, delegates, onMount, onUpdate, onDestroy })` rather than `vf.vfunc` + `.mount()`, since `#sales` already exists in the page (AGENTS.md 4.6 / llms.txt "control published HTML"). `render` returns only the inside of `#sales` (no repeated tag), as the kit describes.
- The Chart.js instance is **not** stored as `sales.chart` — the kit explicitly forbids adding custom properties to a vfunc instance ("Do not add your own properties to an instance... keep listener functions and timers in a closure or module variable"). Instead, `createSales()` closes over a local `chart` variable, created fresh each time `showSales()` runs, and it is created in `onMount`, updated in `onUpdate`, and destroyed in `onDestroy`, per AGENTS.md 4.6.
- The canvas lives inside `data-vf-keep="chart-wrap"` so the same `<canvas>` node (and thus the same Chart.js instance) survives every re-render caused by `setState` (series switch, add month); only `onUpdate` mutates the existing chart's data/color and calls `chart.update()` — the chart is never recreated while the component is alive, matching "the same canvas element and the same Chart.js instance stay for the component's whole life."
- Chart.js `responsive: false` is used with fixed `width="600" height="300"` attributes on the `<canvas>` element (functional HTML attributes, not CSS) so the chart has a usable size without introducing raw pixel values into CSS outside `tokens.css` (AGENTS.md 5.3/5.4). `style.css` only sizes the wrapper with `max-width: 100%` so the canvas scales down on narrow viewports without vfunc/Chart.js re-layout logic in JS.
- `aria-pressed` is passed as a JS boolean (`s.series === 'revenue'`) directly into the `vf.html` template; per llms.txt this renders as the literal string `"true"`/`"false"` for `aria-*` attributes, so no manual `String(...)` conversion was needed.
- Line color: `borderColor` is read once per mount/update via `getComputedStyle(document.documentElement).getPropertyValue('--vf-chart-1' | '--vf-chart-2')`, the one place a token is read in JS because the Chart.js library itself needs a raw color, per AGENTS.md 5.3 and the "Common mistakes" note in llms.txt.

This is a deliberate deviation from AGENTS.md 6.1, noted here as requested by condition 1.
- Numbers (month counts, sales values) are small fixed integers already formatted correctly in the source data; `vf.fmt.number` was not applied since the task's example (`last 181`) shows plain integers with no locale-specific formatting.
