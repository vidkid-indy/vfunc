### app.js

```js
import vf from './lib/vfunc.esm.js';

const CATEGORIES = [
  ['all', 'All'],
  ['keyboard', 'Keyboards'],
  ['mouse', 'Mice'],
  ['monitor', 'Monitors']
];
const SORTS = [
  ['name', 'Name'],
  ['price-asc', 'Price: low to high'],
  ['price-desc', 'Price: high to low']
];
const money = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' });

function shownProducts(s) {
  const q = s.query.trim().toLowerCase();
  const order = {
    name: (a, b) => a.name.localeCompare(b.name, 'en'),
    'price-asc': (a, b) => a.price - b.price,
    'price-desc': (a, b) => b.price - a.price
  }[s.sort];
  return s.items
    .filter((p) => s.category === 'all' || p.category === s.category)
    .filter((p) => !s.inStock || p.stock > 0)
    .filter((p) => !q || p.name.toLowerCase().includes(q))
    .slice()
    .sort(order);
}

const productFilter = vf.vfunc({
  tag: 'section',
  opts: { className: 'pf' },
  state: {
    items: [],
    status: 'loading',
    query: '',
    category: 'all',
    inStock: false,
    sort: 'name'
  },
  render(s) {
    if (s.status === 'loading') {
      return vf.html`<p class="pf__note" role="status" data-ref="status">Loading products…</p>`;
    }
    if (s.status === 'error') {
      return vf.html`<p class="pf__note pf__note--error" role="alert" data-ref="status">Could not load products.</p>`;
    }

    const shown = shownProducts(s);

    return vf.html`
      <div class="pf__controls">
        <label class="pf__field">
          Search
          <input class="pf__input" type="search" data-action="search" value="${s.query}">
        </label>
        <label class="pf__field">
          Category
          <select class="pf__input" data-action="category">
            ${CATEGORIES.map(([value, label]) => vf.html`<option value="${value}" ${value === s.category ? 'selected' : ''}>${label}</option>`)}
          </select>
        </label>
        <label class="pf__check">
          <input type="checkbox" data-action="in-stock" ${s.inStock ? 'checked' : ''}>
          In stock only
        </label>
        <label class="pf__field">
          Sort
          <select class="pf__input" data-action="sort">
            ${SORTS.map(([value, label]) => vf.html`<option value="${value}" ${value === s.sort ? 'selected' : ''}>${label}</option>`)}
          </select>
        </label>
      </div>
      <p class="pf__count" data-ref="count">${shown.length === 1 ? '1 product' : `${shown.length} products`}</p>
      ${shown.length === 0
        ? vf.html`<p class="pf__empty" data-ref="empty">No products found.</p>`
        : vf.html`<ul class="pf__list" data-ref="list">
            ${shown.map((p) => vf.html`<li class="pf__item" data-id="${p.id}" data-state="${p.stock === 0 ? 'out' : 'in'}">
              <span class="pf__name" data-ref="name">${p.name}</span>
              <span class="pf__price" data-ref="price">${money.format(p.price)}</span>
              ${p.stock === 0 ? vf.html`<span class="pf__badge">Sold out</span>` : ''}
            </li>`)}
          </ul>`}
    `;
  },
  delegates: [
    { selector: '[data-action="search"]', eventType: 'input',
      onEvent: (e) => { e.sender.query = e.target.value; } },
    { selector: '[data-action="category"]', eventType: 'change',
      onEvent: (e) => { e.sender.category = e.target.value; } },
    { selector: '[data-action="in-stock"]', eventType: 'change',
      onEvent: (e) => { e.sender.inStock = e.target.checked; } },
    { selector: '[data-action="sort"]', eventType: 'change',
      onEvent: (e) => { e.sender.sort = e.target.value; } }
  ],
  onMount(inst) {
    const url = new URL('./data/products.json', import.meta.url);
    fetch(url)
      .then((r) => {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then((data) => {
        inst.setState({ items: data, status: 'ready' });
      })
      .catch(() => {
        inst.setState({ status: 'error' });
      });
  }
});

productFilter.mount('#app');
```

### REPORT.md

```markdown
# Port report — ProductFilter.jsx → vfunc.js

## Mapping table (this component)

| React | vfunc.js |
|---|---|
| `ProductFilter()` function component | `vf.vfunc({ tag: 'section', opts: { className: 'pf' }, state, render, delegates, onMount })` mounted directly to `#app` |
| `useState(items/status/query/category/inStock/sort)` | `state: { items, status, query, category, inStock, sort }`; changed with `e.sender.<key> = value` (shortcut for `setState`) |
| `useEffect(fetch, [])` | `onMount(inst)`: `fetch(new URL('./data/products.json', import.meta.url))`, `inst.setState({ items, status })` on success/failure. No `alive` flag needed: the component is mounted once for the page's lifetime and is never destroyed before the fetch settles, so no cleanup guard was ported. |
| `useMemo(shown, [...])` | plain function `shownProducts(s)` called at the top of `render` (recomputed each render, same as `useMemo` recomputing when deps change) |
| `onChange={(e) => setQuery(...)}` on search input | `delegates` entry on `[data-action="search"]`, `input` event |
| `onChange` on category/sort `<select>` | `delegates` entries on `[data-action="category"]` / `[data-action="sort"]`, `change` event |
| `onChange` on the checkbox | `delegates` entry on `[data-action="in-stock"]`, `change` event |
| conditional `if (status === 'loading'/'error') return <p>...</p>` | same early-return branches inside `render`, output element carries `data-ref="status"` |
| `{shown.length === 0 ? <p className="pf__empty"> : <ul>}` | same ternary in `vf.html`, `data-ref="empty"` / `data-ref="list"` |
| `.map((p) => <li key={p.id} ...>)` | one delegated component; `shown.map(...)` inside `vf.html`, no per-row components; `data-id="${p.id}"` replaces React `key` |
| `className={'pf__item' + (stock===0 ? ' pf__item--out' : '')}` | `data-state="${p.stock === 0 ? 'out' : 'in'}"` (behaviour hook is data-state per project rules, not a class) plus CSS selector `.pf__item[data-state="out"]` |
| `Intl.NumberFormat` money formatter | kept as-is (plain JS, not a vfunc concern) |
| `./ProductFilter.css` | `styles/product-filter.css`, same class names, values replaced with `var(--vf-*)` tokens |

## Files changed
- `index.html` — new page shell (CSP meta, `#app`, `styles/product-filter.css`, `lib/vfunc.tokens.css`, `<script type="module" src="app.js">`)
- `app.js` — new: the ported component
- `styles/product-filter.css` — new: tokenized CSS
- `data/products.json` — unchanged, used as-is

## Differences from the source / VERIFY

- **No `alive`/cleanup guard**: the React effect guards against setting state after unmount when the request outlives the component. This page mounts the component once and never destroys it, so the guard was dropped rather than ported as dead code (rule: no dead code). If this page is later put behind a router that can navigate away mid-fetch, add an `onDestroy` that sets a local `destroyed` flag and skip the `setState` calls when it is set.
- **Select re-render vs. DOM diffing**: React only touches the `value` prop on the mounted `<select>`. vfunc's `render` replaces the whole inner markup of the root each state change (options rebuilt with `selected` on the matching one). Functionally equivalent (the shown value is always correct); the search `<input>` keeps focus and caret because vfunc restores focus to the element with the same `data-action` at the same position, but this is **not verified in a real browser** — no server was available in this environment to open the page in Chromium/Firefox/WebKit.
- **`VERIFY:`** the exact token names (`--vf-space-1..7`, `--vf-radius-sm|full`, `--vf-font-size-xs|sm`, `--vf-color-danger-soft|-text`) were taken from the llms.txt token list; their concrete pixel/color values were not inspected (`vfunc.tokens.css` was not opened in this run), so the visual spacing/radius may not pixel-match the original `6px`/`8px`/`12px` values — only the intent (small padding, small gap, pill radius, danger tone) was preserved.
- Kept the page as a single-purpose page (`index.html` + `app.js` + one CSS file) rather than the full `app.js`/`store.js`/`api.js`/`pages/*.js` SPA structure from the starter, since the task asks for "a page of its own," not a multi-screen app with routing.
- No i18n: task explicitly says one language, English texts kept verbatim, so `vf.i18n.setup()` was intentionally not added.

## Checklist (prompt-migrate-from-react-vue.md)

- [x] No per-row components: one delegated component renders the whole list; row buttons are not needed for this task (no per-row actions)
- [x] Controlled inputs — search re-renders on every keystroke because the task requires **live** filtering while typing ("search while typing"); this is a deliberate deviation from the general "store the value, render on submit/blur" rule, justified by the explicit requirement. Focus is preserved by the engine via `data-action="search"` at a stable position.
- [x] The only effect (`onMount` fetch) has no timer/subscription to release, so no `onDestroy` was needed; noted above as a place to revisit if the page is ever mounted inside a router.
- [x] All markup goes through `vf.html`; no `innerHTML` from data
- [ ] No console errors or warnings — **not verified**: no static server/browser was available in this sandboxed, single-file evaluation run.
```
