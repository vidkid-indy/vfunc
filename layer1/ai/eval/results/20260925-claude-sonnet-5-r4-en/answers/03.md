### app.js

```javascript
import vf from './lib/vfunc.esm.js';

// The three published orders, kept as-is (no server for this sample).
const ORDERS = [
  { id: '1042', customer: 'Minsu Kim', status: 'paid', statusLabel: 'Paid', total: '₩59,000' },
  { id: '1041', customer: 'Seoyeon Park', status: 'shipped', statusLabel: 'Shipped', total: '₩329,000' },
  { id: '1040', customer: 'Jiho Choi', status: 'refunded', statusLabel: 'Refunded', total: '₩29,000' }
];

// Same texts the publisher's inline script used to swap into #summary.
const SUMMARY_TEXT = {
  week: 'Orders are up this week. Most sales came from the keyboard line.',
  month: 'This month: 1,204 orders, 3.4% refunds.'
};

// User menu: open/close on the name button; outside click and Escape close it; sign out swaps the button text.
function setupUserMenu() {
  let onDocClick;
  let onKeydown;

  function closeMenu(inst) {
    inst.ids.userMenu.hidden = true;
    inst.refs.menuBtn.setAttribute('aria-expanded', 'false');
  }

  return vf.attach('#userWrap', {
    delegates: [
      {
        selector: '[data-action="menu"]',
        eventType: 'click',
        onEvent: (e) => {
          e.event.stopPropagation();
          const inst = e.sender;
          const menu = inst.ids.userMenu;
          const opening = menu.hidden;
          menu.hidden = !opening;
          inst.refs.menuBtn.setAttribute('aria-expanded', String(opening));
        }
      },
      {
        selector: '[data-action="sign-out"]',
        eventType: 'click',
        onEvent: (e) => {
          e.event.preventDefault();
          const inst = e.sender;
          closeMenu(inst);
          inst.refs.menuBtn.textContent = 'Signed out';
        }
      }
    ],
    onMount(inst) {
      onDocClick = (ev) => {
        if (!inst.ids.userMenu.hidden && !inst.$node.contains(ev.target)) {
          closeMenu(inst);
        }
      };
      onKeydown = (ev) => {
        if (ev.key === 'Escape' && !inst.ids.userMenu.hidden) {
          closeMenu(inst);
          inst.refs.menuBtn.focus();
        }
      };
      document.addEventListener('click', onDocClick);
      document.addEventListener('keydown', onKeydown);
    },
    onDestroy() {
      document.removeEventListener('click', onDocClick);
      document.removeEventListener('keydown', onKeydown);
    }
  });
}

// Summary tabs: #summary is the smallest element whose content changes (render);
// #tabs is adopted (no render) and only gets class/aria-selected updates, for looks.
function setupTabs() {
  const summary = vf.attach('#summary', {
    state: { tab: 'week' },
    render: (s) => vf.html`<p>${SUMMARY_TEXT[s.tab] || SUMMARY_TEXT.week}</p>`
  });

  vf.attach('#tabs', {
    delegates: [
      {
        selector: '[data-action="tab"]',
        eventType: 'click',
        onEvent: (e) => {
          const tab = e.target.getAttribute('data-tab');
          vf.$$('[data-action="tab"]', e.sender.$node).forEach((btn) => {
            const active = btn.getAttribute('data-tab') === tab;
            btn.classList.toggle('tabs__tab--active', active);
            btn.setAttribute('aria-selected', String(active));
          });
          summary.setState({ tab });
        }
      }
    ]
  });
}

// Orders table: search and status filter combine; #ordersBody (<tbody>) is the
// smallest element whose content changes. The search input and select are adopted
// (no render), so typed values and focus are never disturbed.
function setupOrders() {
  const ordersBody = vf.attach('#ordersBody', {
    state: { query: '', status: '' },
    render: (s) => {
      const q = s.query.trim().toLowerCase();
      const qNum = q.replace(/^#/, '');
      const rows = ORDERS.filter((o) => {
        const matchesQuery = !q || o.id.toLowerCase().includes(qNum) || o.customer.toLowerCase().includes(q);
        const matchesStatus = !s.status || o.status === s.status;
        return matchesQuery && matchesStatus;
      });
      const rowsHtml = rows.length
        ? rows.map((o) => vf.html`<tr>
            <td>#${o.id}</td>
            <td>${o.customer}</td>
            <td><span class="badge badge--${o.status}">${o.statusLabel}</span></td>
            <td class="orders__num">${o.total}</td>
          </tr>`)
        : [vf.html`<tr><td colspan="4">No orders match.</td></tr>`];
      return vf.html`${rowsHtml}`;
    }
  });

  vf.attach('#searchInput', {
    events: [
      {
        eventType: 'input',
        onEvent: (e) => {
          ordersBody.setState({ query: e.target.value });
        }
      }
    ]
  });

  vf.attach('#statusFilter', {
    events: [
      {
        eventType: 'change',
        onEvent: (e) => {
          ordersBody.setState({ status: e.target.value });
        }
      }
    ]
  });
}

setupUserMenu();
setupTabs();
setupOrders();
```

# Conversion report — dashboard

`dashboard.css` is not included above: it is unchanged (byte-for-byte), as required.

## Area table

| Area (selector) | Static / dynamic | Why | Tool | Data | Events |
|---|---|---|---|---|---|
| `.topbar__logo`, `.sidebar`, `.kpis`, `.panel__title`, `thead` | Static | Text/markup never changes at run time | none | — | — |
| `#userWrap` (`.topbar__user`) | Dynamic (state only) | Menu open/closed, button text after sign-out | `vf.attach` without `render` (adopt) + direct attribute/text updates | — | click on `[data-action="menu"]`, `[data-action="sign-out"]`; document `click`/`keydown` for outside-close and Escape |
| `#summary` (`.panel__body`) | Dynamic (content) | Text swaps with the active tab | `vf.attach` with `render` | `SUMMARY_TEXT` (from the original inline script) | — |
| `#tabs` (`.tabs`) | Dynamic (state only) | Active tab's class/`aria-selected` toggle; adopted, not rebuilt, so it does not also rebuild `#summary` | `vf.attach` without `render` | — | click on `[data-action="tab"]` |
| `#ordersBody` (`<tbody>`) | Dynamic (content) | Rows filtered by search text + status | `vf.attach` with `render` (attached to the `<tbody>` itself, per the table-rows rule) | `ORDERS` (the three published rows, kept as-is) | — |
| `#searchInput` | Dynamic (state only) | Drives the table filter; itself never re-rendered so focus/typed value survive | `vf.attach` with `events` | — | `input` |
| `#statusFilter` | Dynamic (state only) | Drives the table filter | `vf.attach` with `events` | — | `change` |

## HTML changes (added attributes / replaced items only)

- `<head>`: added a CSP `<meta>` tag (`script-src 'self'`, no `'unsafe-inline'`).
- `<body>`: removed the inline `<script>` block; added `<script type="module" src="app.js"></script>`.
- `.topbar__search` input: `+ id="searchInput"`, `+ data-action="search"`.
- `.topbar__user` div: `+ id="userWrap"`.
- `.topbar__user-btn` button: `onclick="toggleMenu()"` → removed; `+ data-action="menu"`, `+ data-ref="menuBtn"`, `+ aria-expanded="false"`.
- `#userMenu` `<ul>`: `style="display:none"` → removed, `+ hidden` (rule 4 replacement: inline style to `hidden` attribute).
- "Sign out" `<a>`: `href="javascript:void(0)"` → `href="#"` (rule 4: `javascript:` URL removed); `onclick="alert('Signed out')"` → removed; `+ data-action="sign-out"`.
- `.tabs` div: `+ id="tabs"`.
- Both `.tabs__tab` buttons: `onclick="showTab(...)"` → removed; `+ data-action="tab"`, `+ data-tab="week"|"month"`, `+ aria-selected="true"|"false"` (matching the existing `tabs__tab--active` class already on the first button).
- `.panel__filter` select: `+ id="statusFilter"`, `+ data-action="filter"`.
- `<tbody>`: `+ id="ordersBody"`.

No text, class name, or static markup was reworded, reformatted, or moved.

## Removed unsafe items (rule 4)

- Inline `<script>` block (`toggleMenu`, `showTab`) — moved to `app.js` as vfunc delegates/attach.
- `onclick="toggleMenu()"`, `onclick="showTab('week'|'month')"`, `onclick="alert('Signed out')"` — inline handlers, replaced by `delegates`/`events`.
- `href="javascript:void(0)"` on "Sign out" — replaced with `href="#"` plus `e.event.preventDefault()` in the click handler.
- `style="display:none"` on `#userMenu` — replaced with the `hidden` attribute (CSP blocks inline `style`).

## Suggested token replacements (optional — CSS not changed)

`dashboard.css` keeps its raw values as instructed. If tokens are ever introduced: `#1e3a8a`/`#1d4ed8` → `--vf-color-primary`, `#ffffff` (surfaces) → `--vf-color-surface`, `#f4f6fb` → `--vf-color-bg`, `#1e293b`/`#64748b` → `--vf-color-text`/`--vf-color-text-muted`, `#047857`/`#b91c1c` (deltas, badges) → `--vf-color-success`/`--vf-color-danger` (+ `-soft`/`-text`), `#e2e8f0`/`#cbd5e1` → `--vf-color-border`, `8px`/`10px`/`16px`/`24px` paddings/gaps → `--vf-space-*`, `6px`/`8px`/`10px` radii → `--vf-radius-sm|md`, the two `box-shadow` values → `--vf-shadow-1`.

## Assumptions

1. "Keep the three published orders and the KPI values as they are" is read as: the KPI section (`.kpis`) stays fully static and untouched; the three order rows become the in-memory dataset (`ORDERS` in `app.js`) that the table filters over — no fetch, no server.
2. No `store.js`/`api.js`/router/i18n were added: the page has one screen and no navigation or server calls, so the minimal `index.html` + `app.js` pair was judged sufficient rather than forcing the full SPA structure from `AGENTS.md` §2.
3. `href="#"` was chosen over dropping the `href` entirely for "Sign out", since the original was a real `<a>`; `preventDefault()` stops it from touching the URL/hash.
4. Added `id="userWrap"`, `id="tabs"`, `id="searchInput"`, `id="statusFilter"`, `id="ordersBody"` as attach/event hooks (ids, not classes), per rule 2's "only add hook attributes" allowance.
5. The tab buttons keep `tabs__tab--active` toggled by class **and** mirror it in `aria-selected`, per BEHAVIOUR #3 and the kit's rule 7 (state in both class-for-looks and `aria-*`).
6. `colspan="4"` on the "No orders match." row matches the table's 4 columns (Order, Customer, Status, Total).
