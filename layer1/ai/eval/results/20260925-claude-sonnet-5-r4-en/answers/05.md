### app.js

```js
// Servers — the page script (an ES module).
import vf from './lib/vfunc.esm.js';

const DATA_URL = new URL('./data/servers.json', import.meta.url);

const STATUS_LABELS = { up: 'Up', down: 'Down', maintenance: 'Maintenance', unknown: 'Unknown' };

function badgeState(status) {
  return status === 'up' || status === 'down' || status === 'maintenance' ? status : 'unknown';
}

// Stricter than vf.safeUrl: only an absolute http:/https: URL is allowed to
// become a working link. Anything else (relative, javascript:, mailto:,
// malformed, non-string) is treated as untrusted and gets no href.
function isHttpUrl(url) {
  if (typeof url !== 'string') return false;
  try {
    const parsed = new URL(url);
    return parsed.protocol === 'http:' || parsed.protocol === 'https:';
  } catch (e) {
    return false;
  }
}

function serverLink(url) {
  return isHttpUrl(url)
    ? vf.html`<a data-ref="link" href="${url}">Open</a>`
    : vf.html`<a data-ref="link">Open</a>`;
}

// The server list comes from another team's service: treat every field as
// untrusted and coerce it to the expected shape/type instead of trusting it.
function sanitizeServer(raw) {
  if (!raw || typeof raw !== 'object') return null;
  const id = raw.id;
  if (typeof id !== 'string' && typeof id !== 'number') return null;
  return {
    id: String(id),
    name: typeof raw.name === 'string' ? raw.name : '',
    region: typeof raw.region === 'string' ? raw.region : '',
    status: typeof raw.status === 'string' ? raw.status : '',
    url: typeof raw.url === 'string' ? raw.url : ''
  };
}

function filteredServers(state) {
  if (state.filter === 'all') return state.servers;
  return state.servers.filter((s) => badgeState(s.status) === state.filter);
}

function renderServerItem(srv) {
  const state = badgeState(srv.status);
  return vf.html`
    <li data-id="${srv.id}">
      <span data-ref="name">${srv.name}</span>
      <span>${srv.region}</span>
      <span data-ref="badge" data-state="${state}">${STATUS_LABELS[state]}</span>
      ${serverLink(srv.url)}
    </li>
  `;
}

function renderStatus(state, shown) {
  if (state.loading) return vf.html`Loading servers…`;
  if (state.error) {
    return vf.html`Could not load servers. <button type="button" data-action="retry">Retry</button>`;
  }
  if (state.servers.length === 0) return vf.html`No servers yet.`;
  return vf.html`Showing ${shown} of ${state.servers.length}`;
}

async function loadServers(inst) {
  inst.setState({ loading: true, error: false });
  try {
    const res = await fetch(DATA_URL);
    if (!res.ok) throw new Error('http error');
    const data = await res.json();
    const servers = Array.isArray(data)
      ? data.map(sanitizeServer).filter((s) => s !== null)
      : [];
    inst.setState({ loading: false, error: false, servers });
  } catch (e) {
    inst.setState({ loading: false, error: true, servers: [] });
  }
}

vf.attach('#servers', {
  state: { loading: true, error: false, servers: [], filter: 'all' },
  render: (s) => {
    const list = filteredServers(s);
    return vf.html`
      <div data-ref="status" role="status">${renderStatus(s, list.length)}</div>
      <select data-action="status-filter">
        <option value="all" ${s.filter === 'all' ? 'selected' : ''}>All</option>
        <option value="up" ${s.filter === 'up' ? 'selected' : ''}>Up</option>
        <option value="down" ${s.filter === 'down' ? 'selected' : ''}>Down</option>
        <option value="maintenance" ${s.filter === 'maintenance' ? 'selected' : ''}>Maintenance</option>
      </select>
      <ul data-ref="list">${list.map(renderServerItem)}</ul>
    `;
  },
  delegates: [
    {
      selector: '[data-action="status-filter"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ filter: e.target.value }); }
    },
    {
      selector: '[data-action="retry"]',
      eventType: 'click',
      onEvent: (e) => { loadServers(e.sender); }
    }
  ],
  onMount: (inst) => {
    inst.$node.setAttribute('aria-busy', inst.state.loading ? 'true' : 'false');
    loadServers(inst);
  },
  onUpdate: (inst) => {
    inst.$node.setAttribute('aria-busy', inst.state.loading ? 'true' : 'false');
  }
});
```

# Report — 05-server-list

## Assumptions

- **Link safety is stricter than `vf.safeUrl`.** The task requires that only `http:`/`https:` URLs become a working link and "any other URL must not become a working link." `vf.safeUrl` (used automatically by `vf.html` for `href`) also allows relative/`mailto:`/`tel:` URLs through, which is looser than the spec. So the link `href` attribute is only emitted at all when `new URL(srv.url)` parses with an `http:`/`https:` protocol; otherwise the `<a data-ref="link">Open</a>` is rendered with no `href` attribute, so it is not a hyperlink/navigable at all (not even `href="#"`).
- **"Showing `<shown>` of `<total>`":** interpreted as `shown` = number of servers after applying the status filter, `total` = number of servers in the full (unfiltered) fetched list — matching the example "Showing 1 of 4" against the 4-item sample data.
- **"No servers yet." vs. an empty filtered result:** "No servers yet." is shown only when the full unfiltered list is empty (nothing returned by the server). If the filter narrows the visible list to zero but the server has data, the status shows "Showing 0 of `<total>`" instead, since the list itself is not empty, only the current filter's result is.
- **Untrusted data handling:** every field from `./data/servers.json` is treated as untrusted per the request. `sanitizeServer` coerces each field to the expected type (string/number id → string; other fields default to `''` if not a string) and drops entries without a usable `id`. Any `status` value other than `up`/`down`/`maintenance` becomes `data-state="unknown"` / text `Unknown`, per the task.
