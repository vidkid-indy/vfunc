### app.js

```js
import vf from './lib/vfunc.esm.js';
import { store } from './store.js';
import homePage from './pages/home.js';
import serversPage from './pages/servers.js';
import serverDetailPage from './pages/serverDetail.js';
import notFoundPage from './pages/notFound.js';

const LOCALE_STORAGE_KEY = 'statusboard.locale';

function readSavedLocale() {
  try {
    const saved = localStorage.getItem(LOCALE_STORAGE_KEY);
    return saved === 'en' || saved === 'ko' ? saved : null;
  } catch (err) {
    return null;
  }
}

function detectLocale() {
  const saved = readSavedLocale();
  if (saved) return saved;
  const lang = (navigator.language || navigator.userLanguage || 'en').toLowerCase();
  return lang.indexOf('ko') === 0 ? 'ko' : 'en';
}

function currentHashPath() {
  const hash = location.hash || '#/';
  const path = hash.slice(1);
  return path === '' ? '/' : path;
}

async function main() {
  await vf.i18n.setup({
    locales: ['en', 'ko'],
    fallback: 'en',
    load: (locale) => fetch(new URL(`./locales/${locale}.json`, import.meta.url)).then((res) => res.json())
  });

  let locale = detectLocale();
  vf.i18n.set(locale);
  document.documentElement.lang = locale;

  let currentPage = null;
  let router;

  function mountPage(factory, ctx) {
    const view = document.getElementById('view');
    if (currentPage) {
      currentPage.destroy();
      currentPage = null;
    }
    if (view) view.innerHTML = '';
    currentPage = factory(ctx, router);
    currentPage.mount(view);
    shell.path = currentHashPath();
  }

  router = vf.router({
    mode: 'hash',
    routes: {
      '/': (ctx) => mountPage(homePage, ctx),
      '/servers': (ctx) => mountPage(serversPage, ctx),
      '/servers/:id': (ctx) => mountPage(serverDetailPage, ctx)
    },
    notFound: (ctx) => mountPage(notFoundPage, ctx),
    linkSelector: 'a[data-link]',
    focus: 'main'
  });

  function setLocale(next) {
    if (next !== 'en' && next !== 'ko') return;
    locale = next;
    vf.i18n.set(locale);
    document.documentElement.lang = locale;
    try {
      localStorage.setItem(LOCALE_STORAGE_KEY, locale);
    } catch (err) {
      /* storage unavailable: locale still works for this session */
    }
    shell.locale = locale;
    if (currentPage && typeof currentPage.refresh === 'function') currentPage.refresh();
  }

  const shell = vf.vfunc({
    tag: 'div',
    state: { locale, path: currentHashPath() },
    render: (s) => vf.html`
      <header class="app-header">
        <nav class="app-header__nav">
          <a data-link href="${router.href('/')}" class="app-header__link" aria-current="${s.path === '/' ? 'page' : 'false'}">${vf.t('nav.home')}</a>
          <a data-link href="${router.href('/servers')}" class="app-header__link" aria-current="${s.path === '/servers' || s.path.indexOf('/servers/') === 0 ? 'page' : 'false'}">${vf.t('nav.servers')}</a>
        </nav>
        <div class="app-header__favorites">
          <span class="app-header__favorites-label">${vf.t('favorites.label')}</span>
          <span data-ref="fav-count" class="app-header__favorites-count">${store.get('favorites').length}</span>
        </div>
        <div class="app-header__locale" role="group">
          <button type="button" data-action="locale" data-locale="en" class="app-header__locale-btn" aria-pressed="${s.locale === 'en'}">English</button>
          <button type="button" data-action="locale" data-locale="ko" class="app-header__locale-btn" aria-pressed="${s.locale === 'ko'}">한국어</button>
        </div>
      </header>
      <main id="view" data-vf-keep="view" tabindex="-1"></main>
    `,
    delegates: [
      { selector: '[data-action="locale"]', eventType: 'click', onEvent: (e) => setLocale(e.target.getAttribute('data-locale')) }
    ]
  });

  await shell.mount('#app');

  store.subscribe(() => shell.refresh());

  router.start();
}

main();
```

### store.js

```js
import vf from './lib/vfunc.esm.js';

export const store = vf.store({
  favorites: []
});

export function toggleFavorite(id) {
  store.set((state) => {
    const exists = state.favorites.indexOf(id) !== -1;
    return {
      favorites: exists
        ? state.favorites.filter((favId) => favId !== id)
        : state.favorites.concat(id)
    };
  });
}

export function isFavorite(id) {
  return store.get('favorites').indexOf(id) !== -1;
}
```

### api.js

```js
const DATA_URL = new URL('./data/servers.json', import.meta.url);

export function fetchServers() {
  return fetch(DATA_URL).then((res) => {
    if (!res.ok) throw new Error('Failed to load servers: ' + res.status);
    return res.json();
  });
}

export function fetchServer(id) {
  return fetchServers().then((servers) => servers.find((server) => server.id === id) || null);
}
```

```js
import vf from '../lib/vfunc.esm.js';
import { fetchServers } from '../api.js';

export default function serversPage(ctx, router) {
  return vf.vfunc({
    tag: 'div',
    state: { status: 'loading', servers: [] },
    render: (s) => vf.html`
      <section class="page page--servers">
        <h1>${vf.t('servers.title')}</h1>
        ${s.status === 'loading' ? vf.html`<p class="servers__status" role="status">${vf.t('servers.loading')}</p>` : ''}
        ${s.status === 'error' ? vf.html`<p class="servers__status servers__status--error" role="alert">${vf.t('servers.error')}</p>` : ''}
        ${s.status === 'ready' && s.servers.length === 0 ? vf.html`<p class="servers__status">${vf.t('servers.empty')}</p>` : ''}
        ${s.status === 'ready' && s.servers.length > 0 ? vf.html`
          <ul class="servers__list">
            ${s.servers.map((server) => vf.html`
              <li class="servers__item" data-id="${server.id}">
                <a data-link href="${router.href('/servers/' + server.id)}">${server.name}</a>
              </li>
            `)}
          </ul>
        ` : ''}
      </section>
    `,
    onMount(inst) {
      fetchServers()
        .then((servers) => { inst.setState({ status: 'ready', servers }); })
        .catch(() => { inst.setState({ status: 'error', servers: [] }); });
    }
  });
}
```

### pages/serverDetail.js

```js
import vf from '../lib/vfunc.esm.js';
import { fetchServer } from '../api.js';
import { store, toggleFavorite, isFavorite } from '../store.js';

export default function serverDetailPage(ctx) {
  const id = ctx.params.id;
  let unsubscribe = null;

  return vf.vfunc({
    tag: 'div',
    state: { status: 'loading', server: null },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`<section class="page page--server"><h1>${vf.t('servers.loading')}</h1></section>`;
      }
      if (s.status !== 'ready' || !s.server) {
        return vf.html`<section class="page page--server"><h1>${vf.t('server.missing')}</h1></section>`;
      }
      const fav = isFavorite(s.server.id);
      return vf.html`
        <section class="page page--server">
          <h1>${s.server.name}</h1>
          <p class="server-detail__region">${s.server.region}</p>
          <p class="server-detail__status" data-state="${s.server.status}">${s.server.status}</p>
          <button type="button" data-action="favorite" data-ref="favorite" data-id="${s.server.id}" aria-pressed="${fav}">${fav ? vf.t('server.favorite.remove') : vf.t('server.favorite.add')}</button>
        </section>
      `;
    },
    delegates: [
      { selector: '[data-action="favorite"]', eventType: 'click', onEvent: (e) => {
          toggleFavorite(e.target.getAttribute('data-id'));
        } }
    ],
    onMount(inst) {
      fetchServer(id)
        .then((server) => {
          inst.setState(server ? { status: 'ready', server } : { status: 'missing', server: null });
        })
        .catch(() => { inst.setState({ status: 'missing', server: null }); });
      unsubscribe = store.subscribe(() => inst.refresh());
    },
    onDestroy() {
      if (unsubscribe) {
        unsubscribe();
        unsubscribe = null;
      }
    }
  });
}
```

```markdown
## Screen list and state map

| Route | Page file | Purpose | Data | Actions |
|---|---|---|---|---|
| `#/` | `pages/home.js` | Landing intro | none | none |
| `#/servers` | `pages/servers.js` | List servers (loading/error/empty/ready) | `data/servers.json` via `api.js` | navigate to a server |
| `#/servers/:id` | `pages/serverDetail.js` | One server, favorite toggle | `data/servers.json` via `api.js` | toggle favorite |
| anything else | `pages/notFound.js` | Not found | none | none |

Shared state: `favorites` (array of ids) in `store.js`, changed only through `toggleFavorite`; survives navigation and locale switch (in-memory), not a reload, as specified. Locale itself is not in the store — it is owned by `vf.i18n` plus a small `localStorage` key so it survives a reload, per the task (favorites explicitly must not).

## Message keys

All keys are listed in `locales/en.json` / `locales/ko.json`, matching the table in the task exactly (`nav.*`, `favorites.label`, `home.*`, `servers.*`, `server.favorite.*`, `server.missing`, `notFound.title`). The two locale-switch button texts ("English" / "한국어") are hard-coded on purpose: the task specifies those exact literal texts regardless of current language, so they are not treated as translatable UI copy.

## Assumptions / things I could not verify against the kit

1. **How a matched route is displayed.** `llms.txt`'s SPA section shows `routes` mapping to functions but never states how the returned value reaches the DOM, and no starter/example file was included in this bundle to check. I assumed the route handlers are responsible for mounting/destroying pages themselves (matching `AGENTS.md`: "app.js — layout + router + page switching, the only place that knows every page"), so each route function calls a local `mountPage(factory, ctx)` that destroys the previous page instance and mounts the new one into `#view`. Marked `VERIFY:`.
2. **Focus management** relies on the router's documented `focus: 'main'` option matching the `<main id="view">` element by tag name, as shown in the kit's SPA example. Not independently verified.
3. **Locale persistence** is done with my own `localStorage` key (`statusboard.locale`) and explicit `vf.i18n.set()` calls rather than the `persist: true` option, because the kit does not document what `persist` stores or reads, or whether it also defaults from the browser language. This satisfies "survives a reload" and "falls back to browser language" exactly as specified.
4. `components/*.js` was left empty — the only reusable piece (the header) is small enough to live inline in `app.js`'s `shell` component; splitting it out did not seem to add value for this scope. Not a structural omission of any required behavior.
5. `styles/tokens.css` was not created: no new raw design values were needed beyond `lib/vfunc.tokens.css`'s tokens, so there was nothing to derive.
6. A network/fetch failure on the single-server page reuses the `server.missing` message (no separate wording exists for that case in the given message table).
7. Not verified: actual browser behavior (Chromium/Firefox/WebKit run, console warnings/errors) — no server or browser was available in this environment to execute the app.
```
