### app.js

```js
// Team board — the page script.
import vf from './lib/vfunc.esm.js';
import { store, setTheme, setLang, toggleTask, removeTask } from './store.js';

// Preferences: theme and language.
const prefs = vf.attach('#prefs', {
  render: () => {
    const p = store.get('prefs');
    return vf.html`
      <p class="prefs__row">Theme: <span data-ref="theme">${p.theme}</span>
        <button class="prefs__button" type="button" data-action="theme">Switch theme</button></p>
      <p class="prefs__row">Language: <span data-ref="lang">${p.lang}</span>
        <button class="prefs__button" type="button" data-action="lang" data-lang="en">English</button>
        <button class="prefs__button" type="button" data-action="lang" data-lang="ko">한국어</button></p>`;
  },
  delegates: [
    { selector: '[data-action="theme"]', eventType: 'click', onEvent: () => setTheme(store.get('prefs').theme === 'dark' ? 'light' : 'dark') },
    { selector: '[data-action="lang"]', eventType: 'click', onEvent: (e) => setLang(e.target.getAttribute('data-lang')) }
  ]
});

// Notice: how many tasks are open. render() must return only the inside of #notice,
// not the section itself (attach adopts the existing element as the root).
const notice = vf.attach('#notice', {
  render: () => {
    const open = store.get('tasks').filter((t) => !t.done).length;
    return vf.html`<p class="notice__text">Open tasks: ${open}</p>`;
  }
});

// Details: a disclosure.
vf.attach('#details-panel', {
  state: { open: false },
  delegates: [{
    selector: '[data-action="details"]',
    eventType: 'click',
    onEvent: (e) => {
      const open = !e.sender.state.open;
      e.sender.state.open = open;
      e.sender.ids.details.hidden = !open;
      e.target.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
  }],
  onMount: (inst) => inst.$node.querySelector('[data-action="details"]').setAttribute('aria-expanded', 'false')
});

// Tasks, open ones first. Each checkbox gets its own id so focus survives
// a re-render even when the task moves position (open tasks sort first).
const tasks = vf.attach('#tasks', {
  render: () => vf.html`
    <h2 class="panel__title">Tasks</h2>
    <ul class="tasks">${store.get('tasks').slice().sort((a, b) => a.done - b.done).map((t) => vf.html`
      <li class="task" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
        <input type="checkbox" id="toggle-${t.id}" data-action="toggle" aria-label="Done" ${t.done ? 'checked' : ''}>
        <span class="task__title">${t.title}</span>
        <button class="task__delete" type="button" data-action="remove">Remove</button>
      </li>`)}</ul>`,
  delegates: [
    { selector: '[data-action="toggle"]', eventType: 'change', onEvent: (e) => toggleTask(e.target.closest('[data-id]').getAttribute('data-id')) },
    { selector: '[data-action="remove"]', eventType: 'click', onEvent: (e) => removeTask(e.target.closest('[data-id]').getAttribute('data-id')) }
  ]
});

// Comments. Comment text is escaped with vf.esc before it goes into innerHTML,
// so a comment like "<b>hi</b>" is shown as text instead of being parsed as markup.
const comments = [];
vf.attach('#comments', {
  delegates: [{
    selector: '[data-action="comment"]',
    eventType: 'submit',
    onEvent: (e) => {
      e.event.preventDefault();
      const text = e.sender.refs.comment.value.trim();
      if (!text) return;
      comments.push(text);
      e.sender.refs['comment-list'].innerHTML = comments.map((c) => '<li class="comments__item">' + vf.esc(c) + '</li>').join('');
      e.sender.refs.comment.value = '';
    }
  }]
});

// Footer: how often the clock has updated.
const footer = vf.attach('#footer', {
  state: { updates: 0 },
  render: (s) => vf.html`Clock updates: <span data-ref="updates">${s.updates}</span>`
});

// Clock, with a close button.
const clock = vf.vfunc({
  state: { now: new Date() },
  render: (s) => vf.html`
    <p class="clock">Time: <span data-ref="time">${vf.fmt.date(s.now, { timeStyle: 'medium' })}</span>
      <button class="clock__close" type="button" data-action="close-clock">Close clock</button></p>`,
  delegates: [{ selector: '[data-action="close-clock"]', eventType: 'click', onEvent: (e) => e.sender.destroy() }],
  onMount: (inst) => {
    inst._timer = setInterval(() => {
      inst.setState({ now: new Date() });
      footer.setState({ updates: footer.state.updates + 1 });
    }, 250);
  },
  onDestroy: (inst) => clearInterval(inst._timer)
});
clock.mount('#clock-slot');

store.subscribe(() => {
  prefs.refresh();
  notice.refresh();
  tasks.refresh();
});
```

### store.js

```js
// Shared state and the only functions that change it.
import vf from './lib/vfunc.esm.js';

export const store = vf.store({
  prefs: { theme: 'light', lang: 'en' },
  tasks: [
    { id: 't1', title: 'Write the release notes', done: false },
    { id: 't2', title: 'Review the login page', done: false },
    { id: 't3', title: 'Update the icons', done: true }
  ]
});

export function setTheme(theme) {
  store.set((s) => ({ prefs: { theme: theme, lang: s.prefs.lang } }));
}

export function setLang(lang) {
  store.set((s) => ({ prefs: { theme: s.prefs.theme, lang: lang } }));
}

export function toggleTask(id) {
  store.set((s) => ({ tasks: s.tasks.map((t) => (t.id === id ? { id: t.id, title: t.title, done: !t.done } : t)) }));
}

export function removeTask(id) {
  store.set((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }));
}
```

### REPORT.md

## Bugs and fixes

1. **Double `#notice` / attach warning** — `render()` on `vf.attach('#notice', …)` returned the whole `<section id="notice">…</section>` instead of only its inside, so vfunc parsed it as content and created a second `#notice`. Fixed `render` to return only `<p class="notice__text">…</p>`; the existing `#notice` section keeps its `id`/`class`/`aria-live` from `index.html`. Also moved the `#notice` section in `index.html` to after `#tasks`, per "follows the tasks" in the expected behaviour.
2. **`aria-expanded=""` instead of `"false"`** — both the initial `onMount` handler and the click handler wrote `''` for the closed state instead of the string `'false'`. Fixed both to write `'false'`/`'true'`, and added `aria-expanded="false"` to the button's initial markup in `index.html` so it is correct even before JS runs.
3. **Language reset the theme** — `setLang` called `store.set({ prefs: { lang } })`. `store.set` shallow-merges only top-level keys, so replacing `prefs` entirely dropped `theme`. Fixed `setLang` to read the current `theme` from state and keep it, mirroring how `setTheme` already preserves `lang`.
4. **Clock kept running after "Close clock"** — the timer id was stored as `inst._timer` in `onMount` but `onDestroy` read `inst.timer` (no underscore), so `clearInterval` was called with `undefined` and the interval was never cleared. Fixed `onDestroy` to use `inst._timer`.
5. **Remove button did nothing** — the delegate selector was `'.task__remove'`, a CSS class that no longer exists after the designer renamed it to `task__delete`; selecting by class is also against the rules. The button already carries `data-action="remove"`, so the selector was changed to `'[data-action="remove"]'`. No CSS change was needed.
6. **Comment XSS** — the comment list was rebuilt with raw string concatenation (`'<li>' + c + '</li>'`) and assigned to `innerHTML`, so an entered `<b>hi</b>` (or a script) was parsed as markup. Fixed by escaping each comment with `vf.esc(c)` before concatenation, so comments always render as text.
   - Assumption: this still builds the list with a string join instead of `vf.html`, because the list was already being written by hand through `refs['comment-list'].innerHTML` and no `vf.vfunc`/`vf.attach` `render` owns that node. `vf.esc` is the documented helper for exactly this case (escaping a value before it goes into hand-built HTML) and it fully closes the XSS bug with the smallest change. A larger refactor (turning the comment list into its own `vf.attach` with `render` and `state.comments`) would be more in the spirit of section 3.1 of `AGENTS.md`, but changes more code than necessary for this task; flagging it here as recommended follow-up rather than doing it now, since the task asks for "as few changes as possible."
7. **Focus jumped to another checkbox after Space** — the checkboxes had no `id`/`data-ref`/`name`, so after ticking a task the list re-sorts (open tasks first) and the engine's fallback ("same `data-action` at the same position") restored focus to whatever checkbox ended up in the old position instead of following the task. Fixed by adding `id="toggle-${t.id}"` to each checkbox (a new attribute, not one of the required hooks, so this doesn't violate "keep every hook as it is"); focus now follows the same `id` wherever the row moves.
