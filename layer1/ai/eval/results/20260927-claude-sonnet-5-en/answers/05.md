### app.js

```js
// Servers — the page script (an ES module).
import vf from './lib/vfunc.esm.js';

// The server list comes from another team's service in production: every
// field is untrusted data. Coerce each field to the expected primitive type
// before it is ever put into state or markup (vf.html still escapes text and
// runs href through vf.safeUrl on top of this).
function sanitizeServer(raw) {
  const r = raw && typeof raw === 'object' ? raw : {};
  return {
    id: r.id != null ? String(r.id) : '',
    name: r.name != null ? String(r.name) : '',
    region: r.region != null ? String(r.region) : '',
    status: typeof r.status === 'string' ? r.status : '',
    url: typeof r.url === 'string' ? r.url : ''
  };
}

function badgeFor(status) {
  if (status === 'up') return { state: 'up', text: 'Up' };
  if (status === 'down') return { state: 'down', text: 'Down' };
  if (status === 'maintenance') return { state: 'maintenance', text: 'Maintenance' };
  return { state: 'unknown', text: 'Unknown' };
}

function renderServerItem(server) {
  const badge = badgeFor(server.status);
  return vf.html`
    <li class="servers__item" data-id="${server.id}">
      <div class="servers__info">
        <span class="servers__name" data-ref="name">${server.name}</span>
        <span class="servers__region">${server.region}</span>
      </div>
      <span class="servers__badge" data-ref="badge" data-state="${badge.state}">${badge.text}</span>
      <a class="servers__link" data-ref="link" href="${server.url}">Open</a>
    </li>
  `;
}

function renderServers(s) {
  const total = s.servers.length;
  const filtered = s.filter === 'all'
    ? s.servers
    : s.servers.filter((item) => item.status === s.filter);
  const shown = filtered.length;

  let status;
  if (s.loading) {
    status = vf.html`Loading servers…`;
  } else if (s.error) {
    status = vf.html`Could not load servers. <button type="button" class="servers__retry" data-action="retry">Retry</button>`;
  } else if (total === 0) {
    status = vf.html`No servers yet.`;
  } else {
    status = vf.html`Showing ${shown} of ${total}`;
  }

  return vf.html`
    <div class="servers__status" data-ref="status" role="status">${status}</div>
    <div class="servers__toolbar">
      <label class="servers__filter-label" for="status-filter">Status</label>
      <select id="status-filter" class="servers__filter" data-action="status-filter" aria-label="Filter servers by status">
        <option value="all" ${s.filter === 'all' ? 'selected' : ''}>All</option>
        <option value="up" ${s.filter === 'up' ? 'selected' : ''}>Up</option>
        <option value="down" ${s.filter === 'down' ? 'selected' : ''}>Down</option>
        <option value="maintenance" ${s.filter === 'maintenance' ? 'selected' : ''}>Maintenance</option>
      </select>
    </div>
    <ul class="servers__list" data-ref="list">
      ${filtered.map(renderServerItem)}
    </ul>
  `;
}

function load(inst) {
  inst.setState({ loading: true, error: false });
  // Resolve relative to this module so it still works from a version folder.
  fetch(new URL('./data/servers.json', import.meta.url))
    .then((res) => {
      if (!res.ok) throw new Error('HTTP ' + res.status);
      return res.json();
    })
    .then((data) => {
      const servers = Array.isArray(data) ? data.map(sanitizeServer) : [];
      inst.setState({ loading: false, error: false, servers });
    })
    .catch(() => {
      inst.setState({ loading: false, error: true, servers: [] });
    });
}

vf.attach('#servers', {
  state: { loading: true, error: false, servers: [], filter: 'all' },
  render: renderServers,
  delegates: [
    { selector: '[data-action="retry"]', eventType: 'click',
      onEvent: (e) => load(e.sender) },
    { selector: '[data-action="status-filter"]', eventType: 'change',
      onEvent: (e) => { e.sender.filter = e.target.value; } }
  ],
  onMount: (inst) => {
    inst.$node.setAttribute('aria-busy', inst.state.loading ? 'true' : 'false');
    load(inst);
  },
  onUpdate: (inst) => {
    inst.$node.setAttribute('aria-busy', inst.state.loading ? 'true' : 'false');
  }
});
```

## Changed files

- JS: `app.js` — full rewrite. Builds the server list inside `#servers` with `vf.attach` (`render` + `state`), loads `./data/servers.json` with `fetch`, and renders the loading/error/empty/list states.
- CSS: `style.css` — kept the original rules and appended rules for the status line, retry button, filter toolbar, list, item, status badge (via `data-state`) and link. All values are `var(--vf-*)` tokens; no hex/px/rgb literals were added, and only logical properties (`margin-inline-start`, `margin-block-end`) are used.
- HTML: `index.html` — not changed (the given `<section id="servers">` and CSP meta tag already fit the task; no inline scripts/styles were needed).
- Other: `data/servers.json` — not changed (used as given, and still treated as untrusted at runtime).

## Assumptions (no locale files were given, so no i18n system is wired up)

- Task conditions state "approval is given" and to decide simply where the kit would otherwise ask. No locale files (`locales/*.json`) are part of this task's project files, so I used the exact hard-coded English texts the task specifies (`Loading servers…`, `Could not load servers.`, `Retry`, `No servers yet.`, `Showing <shown> of <total>`, `Up`/`Down`/`Maintenance`/`Unknown`, option labels `All`/`Up`/`Down`/`Maintenance`) directly in `app.js` rather than inventing a locale layer AGENTS.md/llms.txt do not require for this task.
- "No servers yet." is shown only when the **unfiltered** list is empty (`servers.length === 0`), not when a filter happens to match zero items. When there are servers but the current filter matches none, the status shows `Showing 0 of <total>` instead — this matches the given example (`Showing 1 of 4`) where `<total>` clearly means the full list, not the filtered count, and keeps "no servers yet" meaning "the source truly has none" rather than "this filter is empty."
- The `Retry` button is rendered **inside** the `data-ref="status"` element (the task allows inside or next to it).
- `data-ref="name"`, `data-ref="badge"`, `data-ref="link"` are repeated once per row (one set per `<li data-id="…">`). These are used purely as query hooks (`[data-id="s1"] [data-ref="name"]`), not through `inst.refs` (which is only meaningful for a single unique element), consistent with how the task asks for them "inside each server."
- Retry re-runs the exact same load path (`state.loading = true`, fetch, then success/error), so a retry after a failure re-shows `aria-busy="true"` and the loading text while it runs.
- Untrusted JSON handling: every field is coerced to a plain string (`id`, `name`, `region`, `url`) or checked to be a string (`status`) in `sanitizeServer` before it reaches state/markup, in addition to `vf.html`'s own escaping and automatic `vf.safeUrl` on the `href` attribute. A non-array response is treated as an empty list rather than throwing.
- Region is shown as plain escaped text (no wrapping element beyond a `<span>` with a style-only class), since the task only requires "the region as text."
