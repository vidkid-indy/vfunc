### app.js

```js
import vf from './lib/vfunc.esm.js';
import { createHeader } from './components/header.js';
import homePage from './pages/home.js';
import serversPage from './pages/servers.js';
import serverDetailPage from './pages/serverDetail.js';
import notFoundPage from './pages/notFound.js';

function readSavedLocale() {
  try {
    return localStorage.getItem('vf.locale');
  } catch (e) {
    return null;
  }
}

function detectLocale() {
  const lang = (navigator.language || 'en').toLowerCase();
  return lang.indexOf('ko') === 0 ? 'ko' : 'en';
}

function currentPath() {
  const hash = window.location.hash.replace(/^#/, '');
  return hash === '' ? '/' : hash;
}

let router = null;
let header = null;
let currentPage = null;

function showPage(factory) {
  return (ctx) => {
    if (currentPage) {
      currentPage.destroy();
      currentPage = null;
    }
    currentPage = factory(ctx, router);
    currentPage.mount('#view');
  };
}

async function start() {
  const saved = readSavedLocale();
  const initialLocale = saved || detectLocale();

  await vf.i18n.setup({
    locales: ['en', 'ko'],
    fallback: 'en',
    load: (l) => fetch(new URL(`./locales/${l}.json`, import.meta.url)).then((r) => r.json()),
    persist: true
  });

  if (!saved) {
    await vf.i18n.set(initialLocale);
  }
  document.documentElement.lang = initialLocale;

  router = vf.router({
    mode: 'hash',
    routes: {
      '/': showPage(homePage),
      '/servers': showPage(serversPage),
      '/servers/:id': showPage(serverDetailPage)
    },
    notFound: showPage(notFoundPage),
    linkSelector: 'a[data-link]',
    focus: '#view',
    onChange: () => {
      if (header) {
        header.setState({ path: currentPath() });
      }
    }
  });

  vf.i18n.subscribe(() => {
    document.documentElement.lang = readSavedLocale() || initialLocale;
  });

  header = createHeader(router, initialLocale, currentPath());

  router.start();
}

start();
```

import vf from './lib/vfunc.esm.js';

// Shared state: favorites. This is the only module that changes it.
export const store = vf.store({ favorites: [] });

export function toggleFavorite(id) {
  store.set((state) => {
    const has = state.favorites.indexOf(id) !== -1;
    return {
      favorites: has
        ? state.favorites.filter((fid) => fid !== id)
        : state.favorites.concat(id)
    };
  });
}

export function isFavorite(id) {
  return store.get('favorites').indexOf(id) !== -1;
}

// Every server call lives here.
const dataUrl = new URL('./data/servers.json', import.meta.url);

let pending = null;

function loadServers() {
  if (pending) {
    return pending;
  }
  pending = fetch(dataUrl).then((res) => {
    if (!res.ok) {
      throw new Error('Failed to load servers');
    }
    return res.json();
  }).catch((err) => {
    pending = null;
    throw err;
  });
  return pending;
}

export function fetchServers() {
  return loadServers();
}

export function fetchServerById(id) {
  return loadServers().then((servers) => {
    const found = servers.filter((s) => s.id === id)[0];
    return found || null;
  });
}

### components/header.js

```js
import vf from '../lib/vfunc.esm.js';
import { store } from '../store.js';

function currentPath() {
  const hash = window.location.hash.replace(/^#/, '');
  return hash === '' ? '/' : hash;
}

export function createHeader(router, initialLocale, initialPath) {
  let instance = null;

  instance = vf.attach('#app-header', {
    state: {
      path: initialPath || currentPath(),
      locale: initialLocale || 'en',
      favCount: store.get('favorites').length
    },
    render: (s) => {
      const homeActive = s.path === '/';
      const serversActive = s.path === '/servers' || s.path.indexOf('/servers/') === 0;
      return vf.html`
        <nav class="app-header__nav">
          <a data-link href="${router.href('/')}" aria-current="${homeActive ? 'page' : 'false'}">${vf.t('nav.home')}</a>
          <a data-link href="${router.href('/servers')}" aria-current="${serversActive ? 'page' : 'false'}">${vf.t('nav.servers')}</a>
        </nav>
        <div class="app-header__favorites">
          <span class="app-header__favorites-label">${vf.t('favorites.label')}</span>
          <span data-ref="fav-count">${vf.fmt.number(s.favCount)}</span>
        </div>
        <div class="app-header__locale" role="group">
          <button type="button" data-action="locale" data-locale="en" aria-pressed="${s.locale === 'en' ? 'true' : 'false'}">English</button>
          <button type="button" data-action="locale" data-locale="ko" aria-pressed="${s.locale === 'ko' ? 'true' : 'false'}">한국어</button>
        </div>
      `;
    },
    delegates: [{
      selector: '[data-action="locale"]',
      eventType: 'click',
      onEvent: (e) => {
        const locale = e.target.getAttribute('data-locale');
        if (!locale || locale === instance.state.locale) {
          return;
        }
        vf.i18n.set(locale).then((resolved) => {
          document.documentElement.lang = resolved;
          instance.setState({ locale: resolved });
        });
      }
    }]
  });

  store.subscribe(() => {
    instance.setState({ favCount: store.get('favorites').length });
  });

  vf.i18n.subscribe(() => {
    instance.refresh();
  });

  return instance;
}
```

import vf from '../lib/vfunc.esm.js';
import { fetchServers } from '../api.js';

export default function serversPage(ctx, router) {
  return vf.vfunc({
    tag: 'section',
    state: { status: 'loading', servers: [] },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p role="status">${vf.t('servers.loading')}</p>
        `;
      }
      if (s.status === 'error') {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p role="alert">${vf.t('servers.error')}</p>
        `;
      }
      if (s.servers.length === 0) {
        return vf.html`
          <h1>${vf.t('servers.title')}</h1>
          <p>${vf.t('servers.empty')}</p>
        `;
      }
      return vf.html`
        <h1>${vf.t('servers.title')}</h1>
        <ul class="servers-list">
          ${s.servers.map((server) => vf.html`
            <li class="servers-list__item" data-id="${server.id}">
              <a data-link href="${router.href('/servers/' + server.id)}">${server.name}</a>
            </li>
          `)}
        </ul>
      `;
    },
    onMount(inst) {
      fetchServers()
        .then((servers) => {
          inst.setState({ status: 'ready', servers: servers });
        })
        .catch(() => {
          inst.setState({ status: 'error' });
        });
    }
  });
}

### pages/serverDetail.js

```js
import vf from '../lib/vfunc.esm.js';
import { fetchServerById } from '../api.js';
import { store, toggleFavorite } from '../store.js';

export default function serverDetailPage(ctx) {
  const id = ctx.params.id;
  let unsubscribe = null;

  return vf.vfunc({
    tag: 'section',
    state: { status: 'loading', server: null },
    render: (s) => {
      if (s.status === 'loading') {
        return vf.html`<p role="status">${vf.t('servers.loading')}</p>`;
      }
      if (s.status === 'missing') {
        return vf.html`<h1>${vf.t('server.missing')}</h1>`;
      }
      const isFav = store.get('favorites').indexOf(s.server.id) !== -1;
      return vf.html`
        <h1>${s.server.name}</h1>
        <button type="button" data-action="favorite" aria-pressed="${isFav ? 'true' : 'false'}">
          ${isFav ? vf.t('server.favorite.remove') : vf.t('server.favorite.add')}
        </button>
      `;
    },
    delegates: [{
      selector: '[data-action="favorite"]',
      eventType: 'click',
      onEvent: (e) => {
        toggleFavorite(id);
        e.sender.refresh();
      }
    }],
    onMount(inst) {
      fetchServerById(id)
        .then((server) => {
          if (!server) {
            inst.setState({ status: 'missing' });
            return;
          }
          inst.setState({ status: 'ready', server: server });
        })
        .catch(() => {
          inst.setState({ status: 'missing' });
        });
      unsubscribe = store.subscribe(() => {
        inst.refresh();
      });
    },
    onDestroy() {
      if (unsubscribe) {
        unsubscribe();
        unsubscribe = null;
      }
    }
  });
}
```

# Report — 06 SPA scaffold

## Assumptions

1. **No starter template given**, so the project was created from scratch, following exactly the structure in `AGENTS.md` §2 (`index.html`, `app.js`, `store.js`, `api.js`, `pages/*.js`, `components/*.js`, `locales/{en,ko}.json`, `styles/`). `data/servers.json` was left unchanged (given as-is).
2. **Update plugin / `version.json` intentionally left out**, exactly as the task says ("The update plugin is not available: leave `vf.ext.update` and `version.json` out.").
3. **Locale detection**: `llms.txt` documents that `vf.i18n.setup({ persist: true })` reads a previously saved choice from `localStorage['vf.locale']` automatically, but does not document a getter for "what is the active locale right now" or whether `setup()` auto-detects the browser language on a first run. To satisfy "the choice survives a reload… without a saved choice, follow the browser language" without inventing an undocumented API, `app.js` reads `localStorage.getItem('vf.locale')` itself (a read of the exact key the docs name, not a write) before calling `setup()`; if nothing was saved yet it explicitly calls `vf.i18n.set(detectedLocale)` once, which is documented and also persists through the library's own mechanism. This is the one place I made a judgment call the docs don't fully spell out.
4. **`components/header.js`** uses `vf.attach('#app-header', { render, state, delegates })` on a header element already present in `index.html`, since the header is tiny; a full inner re-render on every navigation / locale change / favorites change is simple and avoids partial-update bugs across three engines. This matches "attach `render` to the smallest element whose content changes" in spirit (the header is the smallest meaningful unit here — its whole content changes together).
5. **`#/servers/<id>` detail page** shows only the server name (or `server.missing`) and the favorite button. The task's message table does not include keys for labelling region/status text, and "no hard-coded visible text" (AGENTS.md §6.1 / checklist) rules out adding new literal labels, so extra server fields were left out rather than inventing new locale keys not requested by the task.
6. **Router `focus` option** was set to `focus: '#view'`, following the same pattern as the kit's own example `focus: 'main'` (a CSS selector); `<main id="view" tabindex="-1">` makes it programmatically focusable.
7. **Favorites store** (`store.js`) is a plain `vf.store({ favorites: [] })` in module scope — it survives navigation and the language switch (same page, same JS module instance) but is lost on a hard reload, exactly as required. No persistence layer was added since the task explicitly says "not a reload".
8. `data/servers.json` is served/fetched as `new URL('./data/servers.json', import.meta.url)` from `api.js` (module lives at the project root, same folder as `data/`), per the kit's "resolve data and locale URLs from the module" rule, so it keeps working if ever copied to a version folder.
9. CSP meta tag has no `'unsafe-inline'` anywhere; no inline `style="…"` or inline scripts were used.
