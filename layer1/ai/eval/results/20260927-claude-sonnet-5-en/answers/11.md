### app.js

```js
// Users screen — the page script. USERS, MONTHS and MONTH_LABELS come from data.js.

function monthKey(dateStr) {
  return dateStr.slice(0, 7);
}

function signupsData() {
  var counts = MONTHS.map(function (m) {
    return USERS.filter(function (u) {
      return monthKey(u.joined) === m;
    }).length;
  });
  return { labels: MONTH_LABELS, series: [{ name: 'Sign-ups', data: counts }] };
}

var columns = [
  { key: 'name', label: 'Name', sortable: true },
  { key: 'email', label: 'Email' },
  { key: 'role', label: 'Role', render: function (row) { return vf.vsBadge({ label: row.role }); } },
  { key: 'joined', label: 'Joined', sortable: true }
];

var selectedEl = document.querySelector('[data-ref="selected"]');

function setSelectedCount(n) {
  selectedEl.textContent = n + ' selected';
}

var grid = vf.vfGrid({
  id: 'users-grid',
  caption: 'Users',
  columns: columns,
  data: USERS,
  pageSize: 10,
  selectable: 'multiple',
  rowKey: 'id',
  onSelect: function (e) {
    setSelectedCount(e.data.keys.length);
  }
});
grid.mount('#users');

var chart = vf.vfChart({
  id: 'signups-chart',
  type: 'bar',
  label: 'Sign-ups per month',
  dataTable: true,
  data: signupsData()
});
chart.mount('#signups');

function applyFilter(text) {
  var q = (text || '').trim().toLowerCase();
  var filtered = q
    ? USERS.filter(function (u) { return u.name.toLowerCase().indexOf(q) !== -1; })
    : USERS;
  grid.setData(filtered);
  grid.setPage(1);
}

var search = vf.vfSearchInput({
  id: 'q',
  label: 'Search users',
  onSearch: function (e) {
    applyFilter(e.data.value);
  },
  onClear: function () {
    applyFilter('');
  }
});
search.mount('#search');
```
