### app.js

```js
// Sign up — the page script.
// Client-side validation for the published #signup form. No server yet:
// a valid submit only hides the form and shows a welcome message.

var FIELD_IDS = ['email', 'password', 'confirm', 'terms'];

function validateEmail(value) {
  var v = value.trim();
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v) ? '' : 'Enter a valid email address.';
}

function validatePassword(value) {
  return (value.length >= 8 && /\d/.test(value)) ? '' : 'Use at least 8 characters, including a number.';
}

function validateConfirm(value, passwordValue) {
  return (value !== '' && value === passwordValue) ? '' : 'Passwords do not match.';
}

function validateTerms(checked) {
  return checked ? '' : 'Accept the terms to continue.';
}

function fieldValue(root, id) {
  var el = root.querySelector('#' + id);
  return el.type === 'checkbox' ? el.checked : el.value;
}

// Runs the rule for one field, writes its message and aria-invalid, and
// returns whether the field is valid. Never touches the password's value
// except to read it for the "confirm" comparison; it is never displayed
// or logged.
function validateOne(root, id) {
  var input = root.querySelector('#' + id);
  var errorEl = root.querySelector('#' + id + '-error');
  var message = '';
  if (id === 'email') {
    message = validateEmail(fieldValue(root, 'email'));
  } else if (id === 'password') {
    message = validatePassword(fieldValue(root, 'password'));
  } else if (id === 'confirm') {
    message = validateConfirm(fieldValue(root, 'confirm'), fieldValue(root, 'password'));
  } else if (id === 'terms') {
    message = validateTerms(fieldValue(root, 'terms'));
  }
  errorEl.textContent = message;
  input.setAttribute('aria-invalid', message ? 'true' : 'false');
  return message === '';
}

// A field has already shown an error once its aria-invalid is "true";
// the attribute itself is the flag, so no extra state is kept.
function hasShownError(root, id) {
  return root.querySelector('#' + id).getAttribute('aria-invalid') === 'true';
}

vf.attach('#signup', {
  delegates: [
    {
      selector: '#email, #password, #confirm',
      eventType: 'focusout',
      onEvent: function (e) { validateOne(e.sender.$node, e.target.id); }
    },
    {
      selector: '#email, #password, #confirm',
      eventType: 'input',
      onEvent: function (e) {
        if (hasShownError(e.sender.$node, e.target.id)) {
          validateOne(e.sender.$node, e.target.id);
        }
      }
    },
    {
      selector: '#terms',
      eventType: 'change',
      onEvent: function (e) { validateOne(e.sender.$node, 'terms'); }
    }
  ],
  events: [
    {
      eventType: 'submit',
      onEvent: function (e) {
        e.event.preventDefault();
        var root = e.sender.$node;
        var firstInvalidId = '';
        FIELD_IDS.forEach(function (id) {
          var ok = validateOne(root, id);
          if (!ok && !firstInvalidId) { firstInvalidId = id; }
        });
        if (firstInvalidId) {
          root.querySelector('#' + firstInvalidId).focus();
          return;
        }
        var email = fieldValue(root, 'email').trim();
        root.setAttribute('hidden', '');
        var done = vf.$('[data-ref="done"]');
        if (done) {
          done.textContent = 'Welcome, ' + email + '!';
          done.removeAttribute('hidden');
        }
      }
    }
  ]
});
```

# Report — Sign-up form validation

## Assumptions (decided myself, per condition 1)

- **Checkbox row markup**: to give `#terms` its own per-field error message
  element without breaking the checkbox+label flex row, I wrapped the
  checkbox and its label in a new `<div class="field__check-row">` inside
  the existing `.field` container, and put `<p id="terms-error">` after
  that row (replacing the old flat `.field.field--check` structure). This
  is a small, necessary markup addition to satisfy "each field has a
  message element with id=`<field id>-error`" — the visible texts, `id`s,
  `name`s and `data-*` hooks the task lists are all unchanged.
- Added `<p class="field__error" id="...">` after each of the other three
  fields for the same reason, and wired `aria-describedby` on each input
  to point at its message element (static in HTML since it never changes).
- Added `<p class="signup__done" data-ref="done" hidden></p>` as a sibling
  of the form inside `<main>`, initially hidden, shown with its `hidden`
  attribute removed on a valid submit (per the "show/hide with `hidden`"
  guidance in the kit's common mistakes, no inline `style`).
- Used `vf.attach('#signup', { delegates, events })` with no `render`, so
  the published form markup and its live input values are kept as-is
  (AGENTS.md 4.7 / llms.txt "attach ... without render — keep the
  markup").
- "Validated again on every input once it has shown an error" is
  implemented by reading the field's own `aria-invalid="true"` attribute
  as the flag, instead of keeping separate JS state — this avoids adding
  custom properties to the instance (forbidden by the engine) and keeps
  a single source of truth for "has this field shown an error".
- Submit order for "focus the first invalid field" follows the table
  order: email, password, confirm, terms.
- `#password`'s value is only read (via `fieldValue`) to compare against
  `#confirm`; it is never written into markup, `textContent`, an
  attribute, or `console.*`.
- No client library beyond `lib/vfunc.js` (dev build) was used; no
  network calls were added since "there is no server yet".

## Behavior notes
