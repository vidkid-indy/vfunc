### app.js

```js
// Sales — the page script. Chart.js is loaded as the global `Chart`.

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const SERIES = {
  revenue: { label: 'Revenue', values: [120, 135, 150, 142, 168, 181, 175, 190, 205, 198, 220, 236] },
  orders: { label: 'Orders', values: [30, 32, 35, 33, 40, 44, 41, 46, 50, 48, 53, 57] }
};

function chartColor(token) {
  return getComputedStyle(document.documentElement).getPropertyValue(token).trim();
}

function seriesColorToken(seriesKey) {
  return seriesKey === 'revenue' ? '--vf-chart-1' : '--vf-chart-2';
}

function createSales() {
  // Chart.js instance lives in this closure, never as an instance property (see AGENTS.md §4.9).
  let chart = null;

  function chartConfig(state) {
    const data = SERIES[state.series];
    const labels = MONTHS.slice(0, state.count);
    const values = data.values.slice(0, state.count);
    return { label: data.label, labels: labels, values: values, color: chartColor(seriesColorToken(state.series)) };
  }

  return vf.vfunc({
    tag: 'section',
    state: { series: 'revenue', count: 6 },
    render: (s) => {
      const cfg = chartConfig(s);
      const last = cfg.values[cfg.values.length - 1];
      return vf.html`
        <div class="sales__controls">
          <button type="button" data-action="series" data-series="revenue" aria-pressed="${s.series === 'revenue'}">Revenue</button>
          <button type="button" data-action="series" data-series="orders" aria-pressed="${s.series === 'orders'}">Orders</button>
          <button type="button" data-action="add-month" ${s.count >= MONTHS.length ? 'disabled' : ''}>Add month</button>
        </div>
        <p class="sales__summary" data-ref="summary">${cfg.label}: ${cfg.values.length} months, last ${last}</p>
        <div class="sales__chart-wrap" data-vf-keep="chart-wrap">
          <canvas data-ref="chart"></canvas>
        </div>
      `;
    },
    delegates: [
      {
        selector: '[data-action="series"]', eventType: 'click',
        onEvent: (e) => {
          const key = e.target.dataset.series;
          e.sender.setState({ series: key });
        }
      },
      {
        selector: '[data-action="add-month"]', eventType: 'click',
        onEvent: (e) => {
          e.sender.setState((s) => ({ count: Math.min(MONTHS.length, s.count + 1) }));
        }
      }
    ],
    onMount: (inst) => {
      const cfg = chartConfig(inst.state);
      chart = new Chart(inst.refs.chart, {
        type: 'line',
        data: {
          labels: cfg.labels,
          datasets: [{ label: cfg.label, data: cfg.values, borderColor: cfg.color }]
        },
        options: { responsive: true, maintainAspectRatio: false }
      });
    },
    onUpdate: (inst) => {
      if (!chart) return;
      const cfg = chartConfig(inst.state);
      chart.data.labels = cfg.labels;
      chart.data.datasets[0].label = cfg.label;
      chart.data.datasets[0].data = cfg.values;
      chart.data.datasets[0].borderColor = cfg.color;
      chart.update();
    },
    onDestroy: () => {
      if (chart) {
        chart.destroy();
        chart = null;
      }
    }
  });
}

let salesInst = null;

function showSales() {
  salesInst = createSales();
  salesInst.mount('#sales');
}

showSales();

vf.attach('[data-action="toggle-chart"]', {
  events: [
    {
      eventType: 'click',
      onEvent: (e) => {
        if (salesInst) {
          salesInst.destroy();
          salesInst = null;
          e.sender.$node.textContent = 'Show chart';
        } else {
          showSales();
          e.sender.$node.textContent = 'Hide chart';
        }
      }
    }
  ]
});
```

## Assumptions

- `index.html` needed no changes: the Chart.js `<script>` tag was left exactly as given, and the existing `<section id="sales">` and `<button data-action="toggle-chart">` are the mount target and the toggle control, so I only wrote `app.js` and `style.css`. `index.html` is therefore not repeated above (output rule: "Files you do not give stay as they are").
- Styling (`.sales__controls`, `.sales__summary`, `.sales__chart-wrap`, the pressed/disabled looks, `.page__button`) was not specified by the task, so I added a small reasonable set of rules, all through `var(--vf-*)` tokens, none of it changing markup hooks or behaviour.
- Selectors used for the two attach/vfunc targets are hooks already in `index.html`/allowed by AGENTS.md (`'#sales'`, `'[data-action="toggle-chart"]'`), matching the rule "found like any hook: `'#id'` or `'[data-ref=\"x\"]'`, never a class".
- Chart options add `maintainAspectRatio: false` (not requested, not forbidden) purely so the `.sales__chart-wrap` fixed height renders sensibly; it does not affect any tested behaviour (labels, dataset, color, instance identity).
- The chart color is read once per render from `getComputedStyle` on `--vf-chart-1` / `--vf-chart-2` (no hardcoded fallback color), per "no color literal in JS" and "a library that needs a color reads a token".
- The Chart.js instance and canvas are kept alive across series/month changes by wrapping the `<canvas>` in `data-vf-keep="chart-wrap">`; `onUpdate` mutates the existing `chart` object instead of recreating it, satisfying "the same canvas element and the same Chart.js instance stay for the component's whole life".
- Toggle: `salesInst.destroy()` removes the component's DOM from `#sales` (root stays, since it's a `vf.attach`-mounted `vf.vfunc` root created via `mount()`, so the canvas/Chart.js instance are fully released) and clears the closure's `chart` reference via `onDestroy`; showing again calls `createSales()` fresh, so the new component starts at `revenue` / 6 months as required.
