### app.js

```js
import vf from './lib/vfunc.esm.js';

// ---- Recent orders (static sample data; mirrors the three published rows) ----
var orders = [
  { id: '1042', customer: 'Minsu Kim', status: 'paid', statusLabel: 'Paid', total: '₩59,000' },
  { id: '1041', customer: 'Seoyeon Park', status: 'shipped', statusLabel: 'Shipped', total: '₩329,000' },
  { id: '1040', customer: 'Jiho Choi', status: 'refunded', statusLabel: 'Refunded', total: '₩29,000' }
];

function orderRow(o) {
  return vf.html`<tr><td>#${o.id}</td><td>${o.customer}</td><td><span class="badge badge--${o.status}">${o.statusLabel}</span></td><td class="orders__num">${o.total}</td></tr>`;
}

// #ordersBody: render is attached to the <tbody> itself (the smallest element whose
// content changes) so the panel head, filter select and table head stay untouched.
var ordersBody = vf.attach('#ordersBody', {
  state: { search: '', status: '' },
  render: function (s) {
    var term = s.search.trim().toLowerCase().replace(/^#/, '');
    var rows = orders.filter(function (o) {
      var matchesStatus = !s.status || o.status === s.status;
      var matchesSearch = !term || o.id.toLowerCase().indexOf(term) !== -1 || o.customer.toLowerCase().indexOf(term) !== -1;
      return matchesStatus && matchesSearch;
    });
    if (rows.length === 0) {
      return vf.html`${[vf.html`<tr><td colspan="4">No orders match.</td></tr>`]}`;
    }
    return vf.html`${rows.map(orderRow)}`;
  }
});

// Search box: adopted, not re-rendered, so focus and caret stay in the input.
vf.attach('[data-action="search"]', {
  events: [{ eventType: 'input', onEvent: function (e) {
    ordersBody.setState({ search: e.sender.$node.value });
  } }]
});

// Status filter select: adopted, only listens for change.
vf.attach('[data-action="filter"]', {
  events: [{ eventType: 'change', onEvent: function (e) {
    ordersBody.setState({ status: e.sender.$node.value });
  } }]
});

// ---- Summary tabs ----
// #summary: render is attached to the smallest element whose text changes.
var summary = vf.attach('#summary', {
  state: { tab: 'week' },
  render: function (s) {
    return s.tab === 'week'
      ? vf.html`<p>Orders are up this week. Most sales came from the keyboard line.</p>`
      : vf.html`<p>This month: 1,204 orders, 3.4% refunds.</p>`;
  }
});

// #summaryTabs: adopted (no render); the buttons keep their markup, only the
// active-look class and the aria-selected state are toggled directly (rule 7).
vf.attach('#summaryTabs', {
  delegates: [{ selector: '[data-action="tab"]', eventType: 'click', onEvent: function (e) {
    var btn = e.target.closest('[data-action="tab"]');
    if (!btn) { return; }
    var tab = btn.getAttribute('data-tab');
    vf.$$('[data-action="tab"]', e.sender.$node).forEach(function (b) {
      var active = b === btn;
      b.classList.toggle('tabs__tab--active', active);
      b.setAttribute('aria-selected', active ? 'true' : 'false');
    });
    summary.setState({ tab: tab });
  } }]
});

// ---- User menu ----
// #userMenuWrap: adopted (no render); Escape and an outside click close the menu.
(function () {
  var removeDocListeners = null;

  function setMenuOpen(inst, open) {
    inst.refs.menuButton.setAttribute('aria-expanded', open ? 'true' : 'false');
    inst.refs.menuList.hidden = !open;
  }

  vf.attach('#userMenuWrap', {
    delegates: [
      { selector: '[data-action="menu"]', eventType: 'click', onEvent: function (e) {
        var open = e.sender.refs.menuButton.getAttribute('aria-expanded') !== 'true';
        setMenuOpen(e.sender, open);
      } },
      { selector: '[data-action="sign-out"]', eventType: 'click', onEvent: function (e) {
        e.event.preventDefault();
        setMenuOpen(e.sender, false);
        e.sender.refs.menuButton.textContent = 'Signed out';
      } }
    ],
    onMount: function (inst) {
      var onDocClick = function (ev) {
        if (!inst.$node.contains(ev.target)) { setMenuOpen(inst, false); }
      };
      var onKeyDown = function (ev) {
        if (ev.key === 'Escape') { setMenuOpen(inst, false); }
      };
      document.addEventListener('click', onDocClick);
      document.addEventListener('keydown', onKeyDown);
      removeDocListeners = function () {
        document.removeEventListener('click', onDocClick);
        document.removeEventListener('keydown', onKeyDown);
      };
    },
    onDestroy: function () {
      if (removeDocListeners) { removeDocListeners(); }
    }
  });
})();
```

### REPORT.md

## Area table

| Area (selector) | Static / dynamic | Why | Tool | Data | Events |
|---|---|---|---|---|---|
| `.topbar__logo`, `.sidebar`, KPI section, table `<thead>`, `.footer` | Static | Content never changes at runtime | none | — | — |
| `#userMenuWrap` (button + `#userMenu`) | Dynamic | Open/closed state, sign-out text change | `vf.attach` adopt (no render) | none | click on `[data-action="menu"]`, `[data-action="sign-out"]`; document `click`/`keydown` for outside-click and Escape |
| `#summaryTabs` | Dynamic | Active tab look (class) + `aria-selected` | `vf.attach` adopt (no render) | none | click on `[data-action="tab"]` |
| `#summary` | Dynamic | Text depends on selected tab | `vf.attach` with `render` (smallest element whose content changes) | inline literal texts (from the original script) | driven by the tabs handler via `setState` |
| `[data-action="search"]` (search input) | Dynamic (listener only) | Triggers filtering; the input itself is adopted, not re-rendered, so focus/caret stay | `vf.attach` adopt, `events` | none | `input` |
| `[data-action="filter"]` (select) | Dynamic (listener only) | Triggers filtering | `vf.attach` adopt, `events` | none | `change` |
| `#ordersBody` (`<tbody>`) | Dynamic | Rows filtered by search + status | `vf.attach` with `render` (rows re-rendered inside the real `<tbody>`) | in-module `orders` array (mirrors the three published rows) | none directly; reacts to `setState` from search/filter |

- Added: `data-action="search"` on the search input.
- Added: `id="userMenuWrap"` on `.topbar__user`; `data-action="menu"`, `data-ref="menuButton"`, `aria-expanded="false"` on the user button; `data-ref="menuList"` and `hidden` (replacing `style="display:none"`) on `#userMenu`; `data-action="sign-out"` and `href="#"` (replacing `href="javascript:void(0)"`) on the "Sign out" link.
- Added: `id="summaryTabs"` on `.tabs`; `data-action="tab"`, `data-tab="week"|"month"`, `aria-selected="true"|"false"` on the two tab buttons (as named in the task).
- Added: `data-action="filter"` on the status select.
- Added: `id="ordersBody"` on `<tbody>`.
- Added: a CSP `<meta>` tag (missing in the source) — `default-src 'self'; script-src 'self'; style-src 'self'; img-src 'self' data:; object-src 'none'; base-uri 'self';`.
- Removed: `onclick="toggleMenu()"`, `onclick="alert('Signed out')"`, the `javascript:void(0)` URL, `style="display:none"`, and the whole inline `<script>` block (its two functions' behaviour was ported into `app.js`).
- Added: one `<script type="module" src="./app.js"></script>` replacing the inline script.
- `dashboard.css` is unchanged (byte-for-byte) — not included in this answer per the output rules ("files you do not give stay as they are").
- Static areas (`.topbar__logo`, sidebar, KPI section, table head, footer, and the three order rows' original markup/text) are otherwise untouched.

## Removed unsafe items

- `onclick="toggleMenu()"` (inline handler) → replaced by `vf.attach` delegate on `[data-action="menu"]`.
- `onclick="alert('Signed out')"` (inline handler) and the `javascript:void(0)` href → replaced by `vf.attach` delegate on `[data-action="sign-out"]` with `e.event.preventDefault()`.
- `onclick="showTab('week'|'month')"` (inline handlers) → replaced by `vf.attach` delegate on `[data-action="tab"]`.
- `style="display:none"` (inline style, blocked by the CSP) → replaced with the `hidden` attribute.
- The whole inline `<script>` block → moved to `app.js` as an ES module.

## Assumptions and VERIFY items

- No server/API exists for this task ("There is no server: keep the three published orders and the KPI values as they are"), so the order rows and KPI values are hardcoded local data in `app.js` / `index.html`, matching the source exactly. No `api.js` was added.
- "Search... by order number (1041 or #1041)" — implemented by stripping a leading `#` from the typed term before comparing to the numeric id; this also makes a bare `1041` and `#1041` behave identically, matching the requirement.
- The BEHAVIOUR section did not ask for i18n/message keys or a router/store for this single static page, so `AGENTS.md`'s SPA structure (`store.js`, `pages/*.js`, locale files) was not introduced — everything fits in one `app.js`, which matches the source's own single-page, no-build nature. If this dashboard grows into a multi-page SPA, it should be restructured per `AGENTS.md` section 2.
- Clicking "Profile" or "Settings" inside the user menu was left as plain anchors (`href="#"`, no handler) since the task's BEHAVIOUR list only specifies "menu" and "sign-out" hooks; they do not close the menu automatically (only Escape, outside-click and sign-out do, as specified).
- `aria-haspopup` and similar extra ARIA attributes were **not** added, since rule 1 of `prompt-html-to-vfunc.md` lists only `aria-expanded`, `aria-selected`, `aria-pressed`, `aria-current`, `data-state` as allowed additions beyond the task's own named hooks.
