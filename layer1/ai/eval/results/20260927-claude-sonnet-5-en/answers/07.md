### app.js

```javascript
import vf from './lib/vfunc.esm.js';

const CATEGORIES = [
  ['all', 'All'],
  ['keyboard', 'Keyboards'],
  ['mouse', 'Mice'],
  ['monitor', 'Monitors']
];

const SORTS = [
  ['name', 'Name'],
  ['price-asc', 'Price: low to high'],
  ['price-desc', 'Price: high to low']
];

const ORDERS = {
  name: (a, b) => a.name.localeCompare(b.name, 'en'),
  'price-asc': (a, b) => a.price - b.price,
  'price-desc': (a, b) => b.price - a.price
};

function shownItems(s) {
  const q = s.query.trim().toLowerCase();
  const order = ORDERS[s.sort];
  return s.items
    .filter((p) => s.category === 'all' || p.category === s.category)
    .filter((p) => !s.inStock || p.stock > 0)
    .filter((p) => !q || p.name.toLowerCase().includes(q))
    .sort(order);
}

let alive = true;

const app = vf.vfunc({
  tag: 'section',
  opts: { className: 'pf' },
  state: {
    items: [],
    status: 'loading',
    query: '',
    category: 'all',
    inStock: false,
    sort: 'name'
  },

  render(s) {
    if (s.status === 'loading') {
      return vf.html`<p class="pf__note" data-ref="status" role="status">Loading products&hellip;</p>`;
    }
    if (s.status === 'error') {
      return vf.html`<p class="pf__note pf__note--error" data-ref="status" role="alert">Could not load products.</p>`;
    }

    const shown = shownItems(s);

    return vf.html`
      <div class="pf__controls">
        <label class="pf__field">
          Search
          <input class="pf__input" type="search" data-action="search" data-ref="search" value="${s.query}">
        </label>
        <label class="pf__field">
          Category
          <select class="pf__input" data-action="category">
            ${CATEGORIES.map(([value, label]) => vf.html`<option value="${value}" ${s.category === value ? 'selected' : ''}>${label}</option>`)}
          </select>
        </label>
        <label class="pf__check">
          <input type="checkbox" data-action="in-stock" ${s.inStock ? 'checked' : ''}>
          In stock only
        </label>
        <label class="pf__field">
          Sort
          <select class="pf__input" data-action="sort">
            ${SORTS.map(([value, label]) => vf.html`<option value="${value}" ${s.sort === value ? 'selected' : ''}>${label}</option>`)}
          </select>
        </label>
      </div>
      <p class="pf__count" data-ref="count">${shown.length === 1 ? '1 product' : `${shown.length} products`}</p>
      ${shown.length === 0
        ? vf.html`<p class="pf__empty" data-ref="empty">No products found.</p>`
        : vf.html`<ul class="pf__list" data-ref="list">
            ${shown.map((p) => vf.html`<li class="pf__item${p.stock === 0 ? ' pf__item--out' : ''}" data-id="${p.id}" data-state="${p.stock === 0 ? 'out' : 'in'}">
              <span class="pf__name" data-ref="name">${p.name}</span>
              <span class="pf__price" data-ref="price">${vf.fmt.currency(p.price, 'USD')}</span>
              ${p.stock === 0 ? vf.html`<span class="pf__badge">Sold out</span>` : ''}
            </li>`)}
          </ul>`}
    `;
  },

  delegates: [
    {
      selector: '[data-action="search"]',
      eventType: 'input',
      onEvent: (e) => { e.sender.setState({ query: e.target.value }); }
    },
    {
      selector: '[data-action="category"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ category: e.target.value }); }
    },
    {
      selector: '[data-action="in-stock"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ inStock: e.target.checked }); }
    },
    {
      selector: '[data-action="sort"]',
      eventType: 'change',
      onEvent: (e) => { e.sender.setState({ sort: e.target.value }); }
    }
  ],

  onMount(inst) {
    fetch('./data/products.json')
      .then((r) => {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.json();
      })
      .then((data) => {
        if (alive) inst.setState({ items: data, status: 'ready' });
      })
      .catch(() => {
        if (alive) inst.setState({ status: 'error' });
      });
  },

  onDestroy() {
    alive = false;
  }
});

app.mount('#app');
```

### REPORT.md

```markdown
# Report — 07-react-port

## Files changed / added
- `index.html` — new: `#app` mount point, CSP meta tag (no `'unsafe-inline'`), one `<script type="module" src="app.js">`, links `lib/vfunc.tokens.css` then `styles.css`.
- `app.js` — new: single `vf.vfunc` component (no router/store needed for one screen), mounted to `#app`.
- `styles.css` — new: all colors/spacing/radius/font-size through `var(--vf-*)` tokens; no hex/rgb literals.
- `data/products.json` — left unchanged (content already matched what the task needed).

## Assumptions (approval given per conditions, decided myself)
1. **No SPA scaffold.** The task asks for "a page of its own": `index.html` + `app.js` + a CSS file. I did not add `store.js`/`api.js`/`pages/`/router, since there is only one screen and no navigation — that structure would be unused ceremony. `app.js` does the one `fetch` directly in `onMount`, matching the source's single `useEffect`.
2. **No i18n.** Per the task, texts stay hardcoded in English exactly as in the source (`Loading products…`, `Could not load products.`, `In stock only`, `No products found.`, `Sold out`, category/sort labels). No `vf.i18n.setup()` call, so no locale-not-ready race to guard against.
3. **`status` ref is only present while loading/error.** The React source returns *only* the status `<p>` in those two states and the full UI otherwise; I mirrored that (no hidden duplicate element) rather than always rendering `data-ref="status"` alongside the rest.
4. **Empty vs. list are mutually exclusive**, exactly like the source's ternary: `data-ref="list"` (a `<ul>`) exists only when there is at least one item; `data-ref="empty"` exists only when there are none. Both are hooked by `data-ref` per the table, so having only one of them present at a time satisfies "absent or hidden when there are products."
5. **Search re-renders on every keystroke** (`input` event → `setState({ query })`), relying on vfunc's documented focus/caret restoration by matching `data-ref="search"` on the re-rendered input (llms.txt: "focus and caret return to the element with the same id, data-ref or name"). This keeps behavior simple and correct rather than deferring the render, and matches the task's requirement that focus survive while typing.
6. **Money formatting** uses `vf.fmt.currency(p.price, 'USD')` instead of a hand-rolled `Intl.NumberFormat`, per the kit rule ("money go through vf.fmt … never hand-made formats"). I assumed its default locale produces the same `$49.00` shape as the source's `en-US` formatter, since the kit does not expose a locale parameter for `vf.fmt.currency`.
7. **CSP meta tag** is same-origin only (`'self'` for script/style/img/connect), since the project loads no CDN resources and does the one same-origin `fetch('./data/products.json')`.
8. **Border width (`1px`)** in `styles.css` is a literal, since no `--vf-*` token exists for border width in the documented token list; every color, spacing, radius and font-size value otherwise comes from a token.
9. Kept `alive` as a module-level closure flag (not a property on the instance, per the "never add your own properties to an instance" rule) to skip a stray `setState` if the component were destroyed before the fetch resolves; not currently reachable since nothing destroys `app`, but costs nothing and matches the source's cleanup intent.

## Kit items I could not follow / verify
- I could not run the page in a browser (no server, no tool access to execute code in this one-shot task), so console-error/warning freedom, cross-browser behavior, and the exact rendered `$49.00` string from `vf.fmt.currency` are **not verified** — they are inferred from the llms.txt reference only.
- `AGENTS.md` describes a full `store.js`/`api.js`/`pages/*.js` SPA layout; I deviated from it for the reason in assumption 1 above (single static screen, no routes, no shared cross-page state).

## Checklist (from prompt-migrate-from-react-vue.md)
- [x] No per-row components: one delegated listener set on the section root handles search/category/in-stock/sort; product rows are plain markup in the list, not per-row components.
- [x] Controlled inputs: search re-renders per keystroke by design (focus/caret preserved via `data-ref`), which is the documented vfunc pattern for this case; category/checkbox/sort only fire on `change`, not per keystroke.
- [x] No timers/subscriptions were introduced, so there is nothing to release beyond the `alive` guard set in `onDestroy`.
- [x] All markup built with `vf.html`; no `innerHTML` from data anywhere.
- [ ] "No console errors or warnings" — not verified (no browser run available in this session).
```
