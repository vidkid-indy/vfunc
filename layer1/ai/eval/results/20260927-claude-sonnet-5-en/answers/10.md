### app.js

```js
// Team board — the page script.
import vf from './lib/vfunc.esm.js';
import { store, setTheme, setLang, toggleTask, removeTask } from './store.js';

// Preferences: theme and language.
const prefs = vf.attach('#prefs', {
  render: () => {
    const p = store.get('prefs');
    return vf.html`
      <p class="prefs__row">Theme: <span data-ref="theme">${p.theme}</span>
        <button class="prefs__button" type="button" data-action="theme">Switch theme</button></p>
      <p class="prefs__row">Language: <span data-ref="lang">${p.lang}</span>
        <button class="prefs__button" type="button" data-action="lang" data-lang="en">English</button>
        <button class="prefs__button" type="button" data-action="lang" data-lang="ko">한국어</button></p>`;
  },
  delegates: [
    { selector: '[data-action="theme"]', eventType: 'click', onEvent: () => setTheme(store.get('prefs').theme === 'dark' ? 'light' : 'dark') },
    { selector: '[data-action="lang"]', eventType: 'click', onEvent: (e) => setLang(e.target.getAttribute('data-lang')) }
  ]
});

// Details: a disclosure.
vf.attach('#details-panel', {
  state: { open: false },
  delegates: [{
    selector: '[data-action="details"]',
    eventType: 'click',
    onEvent: (e) => {
      const open = !e.sender.state.open;
      e.sender.state.open = open;
      e.sender.ids.details.hidden = !open;
      e.target.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
  }],
  onMount: (inst) => inst.$node.querySelector('[data-action="details"]').setAttribute('aria-expanded', 'false')
});

// Tasks, open ones first.
const tasks = vf.attach('#tasks', {
  render: () => vf.html`
    <h2 class="panel__title">Tasks</h2>
    <ul class="tasks">${store.get('tasks').slice().sort((a, b) => a.done - b.done).map((t) => vf.html`
      <li class="task" data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}">
        <input type="checkbox" id="toggle-${t.id}" data-action="toggle" aria-label="Done" ${t.done ? 'checked' : ''}>
        <span class="task__title">${t.title}</span>
        <button class="task__delete" type="button" data-action="remove">Remove</button>
      </li>`)}</ul>`,
  delegates: [
    { selector: '[data-action="toggle"]', eventType: 'change', onEvent: (e) => toggleTask(e.target.closest('[data-id]').getAttribute('data-id')) },
    { selector: '[data-action="remove"]', eventType: 'click', onEvent: (e) => removeTask(e.target.closest('[data-id]').getAttribute('data-id')) }
  ]
});

// Notice: how many tasks are open (shown after the tasks list).
const notice = vf.attach('#notice', {
  render: () => {
    const open = store.get('tasks').filter((t) => !t.done).length;
    return vf.html`<p class="notice__text">Open tasks: ${open}</p>`;
  }
});

// Comments.
const comments = [];
const commentList = vf.attach('[data-ref="comment-list"]', {
  render: () => comments.map((c) => vf.html`<li class="comments__item">${c}</li>`)
});
vf.attach('#comments', {
  delegates: [{
    selector: '[data-action="comment"]',
    eventType: 'submit',
    onEvent: (e) => {
      e.event.preventDefault();
      const text = e.sender.refs.comment.value.trim();
      if (!text) return;
      comments.push(text);
      commentList.refresh();
      e.sender.refs.comment.value = '';
    }
  }]
});

// Footer: how often the clock has updated.
const footer = vf.attach('#footer', {
  state: { updates: 0 },
  render: (s) => vf.html`Clock updates: <span data-ref="updates">${s.updates}</span>`
});

// Clock, with a close button.
let clockTimer = null;
const clock = vf.vfunc({
  state: { now: new Date() },
  render: (s) => vf.html`
    <p class="clock">Time: <span data-ref="time">${vf.fmt.date(s.now, { timeStyle: 'medium' })}</span>
      <button class="clock__close" type="button" data-action="close-clock">Close clock</button></p>`,
  delegates: [{ selector: '[data-action="close-clock"]', eventType: 'click', onEvent: (e) => e.sender.destroy() }],
  onMount: (inst) => {
    clockTimer = setInterval(() => {
      inst.setState({ now: new Date() });
      footer.setState({ updates: footer.state.updates + 1 });
    }, 250);
  },
  onDestroy: () => {
    clearInterval(clockTimer);
    clockTimer = null;
  }
});
clock.mount('#clock-slot');

store.subscribe(() => {
  prefs.refresh();
  notice.refresh();
  tasks.refresh();
});
```

### store.js

```js
// Shared state and the only functions that change it.
import vf from './lib/vfunc.esm.js';

export const store = vf.store({
  prefs: { theme: 'light', lang: 'en' },
  tasks: [
    { id: 't1', title: 'Write the release notes', done: false },
    { id: 't2', title: 'Review the login page', done: false },
    { id: 't3', title: 'Update the icons', done: true }
  ]
});

export function setTheme(theme) {
  store.set((s) => ({ prefs: { theme: theme, lang: s.prefs.lang } }));
}

export function setLang(lang) {
  store.set((s) => ({ prefs: { theme: s.prefs.theme, lang: lang } }));
}

export function toggleTask(id) {
  store.set((s) => ({ tasks: s.tasks.map((t) => (t.id === id ? { id: t.id, title: t.title, done: !t.done } : t)) }));
}

export function removeTask(id) {
  store.set((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }));
}
```

### REPORT.md

```markdown
# Bug fix report — Team board

## Cause and fix per bug

1. **Duplicate `#notice`.** `render` for `vf.attach('#notice', …)` returned a whole `<section id="notice">…</section>`,
   so the parsed content was inserted *inside* the existing `#notice` element, producing a second element with the
   same id (exactly the console warning: "render returned the target element itself"). Fix: `render` now returns
   only the inside (`<p class="notice__text">Open tasks: ${open}</p>`); the existing `<section id="notice">` in
   `index.html` stays as the one and only element. Also moved the `<section id="notice">` block in `index.html` to
   after `<section id="tasks">` so the notice "follows the tasks" as required.

2. **`aria-expanded=""` instead of `"false"`.** Both the `onMount` initializer and the click handler set the
   attribute to `''` when closed instead of the string `'false'`. Fixed both to write `'false'`/`'true'` explicitly.

3. **Language switch resets theme.** `store.set` shallow-merges only top-level keys, so `setLang`'s
   `store.set({ prefs: { lang } })` replaced the whole `prefs` object, dropping `theme`. Fixed `setLang` to read the
   previous `theme` from state and merge it back in, the same pattern `setTheme` already used for `lang`.

4. **Clock timer not cleared.** `onMount` stored the interval id on `inst._timer`, but `onDestroy` read
   `inst.timer` (no underscore) — a typo, and also a violation of the rule against adding custom properties to an
   instance (names starting with `_` are reserved for the engine). Fixed by keeping the timer id in a module-level
   closure variable (`clockTimer`) instead of on the instance, set in `onMount` and cleared in `onDestroy`.

5. **Remove button does nothing.** The delegate selector was `'.task__remove'` (a class), but the designer renamed
   the class to `task__delete` and, per the rules, delegated selectors must never target classes anyway. Fixed the
   selector to `'[data-action="remove"]'`, which was already present on the button and untouched by the class
   rename.

6. **Comment XSS.** Comments were rendered with manual string concatenation into `innerHTML`
   (`'<li>' + c + '</li>'` joined and assigned), which runs any HTML/script a user types. Fixed by giving the
   `<ul data-ref="comment-list">` its own `vf.attach` with a `render` that maps each comment through `vf.html`
   (which escapes text), and having the submit handler call `commentList.refresh()` instead of touching
   `innerHTML` directly.

7. **Focus jumps after Space-toggling a task.** The checkbox had no `id`/`data-ref`/`name`, so vfunc's focus
   restore fell back to "same `data-action` at the same position" — but toggling re-sorts the list, so the
   position no longer holds the same task. Fixed by adding `id="toggle-${t.id}"` to each checkbox; this id is
   stable per task regardless of where the `<li>` moves after the re-sort, so focus restore now finds the correct
   checkbox by id.

## Assumptions

- Where the task said "add a new attribute if a fix needs one," I added `id="toggle-${t.id}"` to the task
  checkboxes (bug 7) since no existing hook was suitable for stable per-row focus restore.
- For the comment list (bug 6), I split `#comments` into two `vf.attach` instances (the panel with the form
  delegate, and the `<ul data-ref="comment-list">` with its own small `render`) rather than adding `render` to the
  whole `#comments` panel, per the "attach `render` to the smallest changing element" guidance — this avoids
  rebuilding the static form/label on every new comment.
- Moved `<section id="notice">` below `<section id="tasks">` in `index.html` (no id/data-action/data-ref/text
  changed) because the task's "expected behaviour" explicitly requires the notice to follow the tasks; this is a
  markup reorder, not a hook rename.
- `style.css` needed no changes for any of the seven bugs, so it is not repeated above (unchanged files stay as
  given, per the output rules).

## Checklist

- [x] Every dynamic value goes through `vf.html`/`vf.tpl`; no `innerHTML` from data — fixed the one remaining
      offender (comment list).
- [x] Selectors use `data-action`/`data-ref`/`id` only; no colors or sizes in JS — fixed the `.task__remove` class
      selector.
- [x] New CSS files too use only `var(--vf-*)` tokens — not applicable, no CSS changed.
- [x] New text uses message keys in every locale; dates/numbers use `vf.fmt` — no new user-visible text was added;
      the app has no locale files in this kit, so pre-existing hardcoded English/Korean button labels were left as
      found (out of scope for a bug-fix task that must keep texts exactly as they are).
- [x] Listeners, timers and subscriptions are released in `onDestroy` — fixed the clock timer leak.
- [ ] The page opens with no console errors or warnings — not verified (no browser/test runner available in this
      one-shot text-only session); reasoned through each of the seven reported symptoms/console messages and
      addressed the one explicit `[vfunc]` warning quoted in the report (bug 1).
- [x] No secrets, tokens or personal data in code, logs or commits — none introduced.
```
