### places-map.js

```js
// The app's own map component (layer 1, L1 app wrapper around Leaflet).
// Leaflet 1.9.4 (BSD-2-Clause) is loaded by index.html as the global `L`.
// Nothing here is added to `vf`; the factory lives on the app's own global `app`.
(function () {
  if (!window.app) window.app = {};

  function primaryColor() {
    // Step 9 (prompt-integrate-third-party.md): no color literal in JS, not
    // even as a fallback — read the design token instead.
    return getComputedStyle(document.documentElement).getPropertyValue('--vf-color-primary').trim();
  }

  // props: { places: [{ id, name, lat, lng }], onSelect?({ sender, event, data: { id } }) }
  window.app.vfPlacesMap = function (props) {
    props = props || {};
    var places = props.places || [];
    var onSelect = props.onSelect;

    // Step 3/4/7: the vendor object lives only behind a `data-vf-keep` host,
    // created in onMount and destroyed in onDestroy. Kept in closures, never
    // written onto the instance itself.
    var map = null;
    var markers = {};

    function highlight(id) {
      Object.keys(markers).forEach(function (key) {
        markers[key].setStyle({ fillOpacity: key === id ? 1 : 0.4 });
      });
    }

    var inst = vf.vfunc({
      // Trusted static markup written in this file (no data interpolated),
      // so a plain string is fine here (rule 20.1).
      innerHTML: '<div data-vf-keep="map" data-ref="mapHost"></div>',
      methods: {
        // Public API asked by the task: highlight one place's marker.
        select: function (id) {
          highlight(id);
        }
      },
      onMount: function (instance) {
        var host = instance.refs.mapHost;
        // No tile layer: the page has no network access for tiles.
        map = L.map(host, { attributionControl: false });

        var color = primaryColor();
        var latLngs = [];

        places.forEach(function (place) {
          var marker = L.circleMarker([place.lat, place.lng], {
            radius: 8,
            weight: 2,
            color: color,
            fillColor: color,
            fillOpacity: 0.4
          }).addTo(map);

          // Step 11: bindTooltip renders a string with innerHTML, so the name
          // (which may contain "<") is escaped first; it still displays as
          // plain text once the browser parses the escaped entities back.
          marker.bindTooltip(vf.esc(place.name));

          marker.on('click', function (e) {
            highlight(place.id);
            if (onSelect) {
              onSelect({ sender: instance, event: e, data: { id: place.id } });
            }
          });

          markers[place.id] = marker;
          latLngs.push([place.lat, place.lng]);
        });

        // Show every place: fit the view to the markers.
        if (latLngs.length === 1) {
          map.setView(latLngs[0], 15);
        } else if (latLngs.length > 1) {
          map.fitBounds(latLngs, { padding: [20, 20] });
        }
      },
      onDestroy: function () {
        if (map) {
          map.remove();
          map = null;
        }
        markers = {};
      }
    });

    return inst;
  };
})();
```

### app.js

```js
// Places — the page script. Leaflet is loaded as the global `L`.
const PLACES = [
  { id: 'p1', name: 'Seoul Station', lat: 37.5547, lng: 126.9707 },
  { id: 'p2', name: 'Cafe <b>Moon</b>', lat: 37.5665, lng: 126.978 },
  { id: 'p3', name: 'Namsan Tower', lat: 37.5512, lng: 126.9882 }
];

(function () {
  const placesById = {};
  PLACES.forEach((place) => {
    placesById[place.id] = place;
  });

  const currentEl = vf.$('[data-ref="current"]');
  let mapInstance = null;

  function showCurrent(id) {
    const place = id ? placesById[id] : null;
    // Plain text assignment: safe (no markup parsing) and matches the
    // required "Selected: <name> (as text)" behaviour.
    currentEl.textContent = place ? `Selected: ${place.name}` : 'No place selected';
  }

  function onMarkerSelect(e) {
    showCurrent(e.data.id);
  }

  function createMap() {
    mapInstance = window.app.vfPlacesMap({ places: PLACES, onSelect: onMarkerSelect });
    mapInstance.mount('#map-slot');
  }

  // The map starts visible (header button reads "Hide map").
  createMap();

  // One delegated listener for every row button (rule 4.3).
  vf.attach('#places', {
    delegates: [
      {
        selector: '[data-action="select"]',
        eventType: 'click',
        onEvent: (e) => {
          const id = e.target.getAttribute('data-id');
          if (mapInstance) mapInstance.select(id);
          showCurrent(id);
        }
      }
    ]
  });

  vf.attach('[data-action="toggle-map"]', {
    events: [
      {
        eventType: 'click',
        onEvent: (e) => {
          if (mapInstance) {
            mapInstance.destroy();
            mapInstance = null;
            e.sender.$node.textContent = 'Show map';
          } else {
            createMap();
            showCurrent(null);
            e.sender.$node.textContent = 'Hide map';
          }
        }
      }
    ]
  });
})();
```
