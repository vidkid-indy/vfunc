### app.js

```js
// My tasks — the page script. The to-do list lives in #todo.

let nextId = 1;
let focusInputRequested = false;

const FILTERS = ['all', 'open', 'done'];

function filterTasks(tasks, filter) {
  if (filter === 'open') return tasks.filter((t) => !t.done);
  if (filter === 'done') return tasks.filter((t) => t.done);
  return tasks;
}

function renderTask(t) {
  const checkboxId = 'todo-' + t.id;
  return vf.html`<li data-id="${t.id}" data-state="${t.done ? 'done' : 'open'}" class="todo__item">
    <input type="checkbox" data-action="toggle" id="${checkboxId}" ${t.done ? 'checked' : ''}>
    <label for="${checkboxId}" data-ref="text" class="todo__text">${t.text}</label>
    <button type="button" data-action="remove" class="todo__remove">Remove</button>
  </li>`;
}

function render(s) {
  const visible = filterTasks(s.tasks, s.filter);
  const openCount = s.tasks.filter((t) => !t.done).length;
  return vf.html`<form data-action="add" class="todo__form">
    <label for="todo-input" class="todo__label">New task</label>
    <input type="text" id="todo-input" data-ref="input" name="task" autocomplete="off" class="todo__input">
    <button type="submit" class="todo__add">Add</button>
  </form>
  <div class="todo__filters" role="group" aria-label="Filter">
    <button type="button" data-action="filter" data-filter="all" aria-pressed="${s.filter === 'all'}" class="todo__filter">All</button>
    <button type="button" data-action="filter" data-filter="open" aria-pressed="${s.filter === 'open'}" class="todo__filter">Open</button>
    <button type="button" data-action="filter" data-filter="done" aria-pressed="${s.filter === 'done'}" class="todo__filter">Done</button>
  </div>
  <p class="todo__count">Left: <span data-ref="left">${openCount}</span></p>
  <ul data-ref="list" class="todo__list">${visible.map(renderTask)}</ul>
  ${visible.length === 0 ? vf.html`<p data-ref="empty" class="todo__empty">Nothing here.</p>` : ''}`;
}

vf.attach('#todo', {
  state: { tasks: [], filter: 'all' },
  render,
  onUpdate: (inst) => {
    if (focusInputRequested) {
      focusInputRequested = false;
      if (inst.refs.input) inst.refs.input.focus();
    }
  },
  delegates: [
    {
      selector: '[data-action="add"]',
      eventType: 'submit',
      onEvent: (e) => {
        e.event.preventDefault();
        const input = e.sender.refs.input;
        const text = (input.value || '').trim();
        focusInputRequested = true;
        if (!text) {
          e.sender.scheduleRefresh();
          return;
        }
        e.sender.setState((s) => ({
          tasks: s.tasks.concat({ id: nextId++, text: text, done: false })
        }));
      }
    },
    {
      selector: '[data-action="toggle"]',
      eventType: 'change',
      onEvent: (e) => {
        const li = e.target.closest('[data-id]');
        if (!li) return;
        const id = Number(li.getAttribute('data-id'));
        e.sender.setState((s) => ({
          tasks: s.tasks.map((t) => (t.id === id ? { ...t, done: !t.done } : t))
        }));
      }
    },
    {
      selector: '[data-action="remove"]',
      eventType: 'click',
      onEvent: (e) => {
        const li = e.target.closest('[data-id]');
        if (!li) return;
        const id = Number(li.getAttribute('data-id'));
        e.sender.setState((s) => ({
          tasks: s.tasks.filter((t) => t.id !== id)
        }));
      }
    },
    {
      selector: '[data-action="filter"]',
      eventType: 'click',
      onEvent: (e) => {
        const filter = e.target.getAttribute('data-filter');
        if (FILTERS.indexOf(filter) === -1) return;
        e.sender.setState({ filter: filter });
      }
    }
  ]
});
```

```markdown
# Report — 02-todo

## Files changed
- `app.js` — full to-do list component (JS)
- `style.css` — added token-based rules for the form, filters, list and empty state (CSS)
- `index.html` — not changed (the existing `#todo` section, CSP meta tag and script tags already fit the task; nothing needed adding)

## Approach
- One `vf.attach('#todo', { render, delegates, onUpdate })` component. `render` returns only the inside of the existing `<section id="todo">` (per the kit note that `attach`'s `render` draws the inside of the element, keeping its own attributes such as `class="todo"` and `aria-label`).
- State: `{ tasks: [{ id, text, done }], filter: 'all' | 'open' | 'done' }`. `nextId` is a module-level counter used for the required unique `data-id`.
- Adding: the `<form data-action="add">` is bound with `eventType: 'submit'` (so Enter and the Add button both work through the browser's native form submission), `e.event.preventDefault()` stops the reload, text is read from `refs.input`, trimmed, and rejected if empty. The text goes through `vf.html`, so it is always shown as escaped text, never as markup.
- Focus after add: relying only on vfunc's built-in "same id/data-ref/name" focus restoration was not enough, because clicking the Add button leaves focus on the button before the re-render, and the restore rule would return focus there, not to the input. So a small closure flag (`focusInputRequested`) is set whenever `add` fires (whether or not text was accepted) and consumed in `onUpdate` to call `refs.input.focus()` after the new markup is in the DOM. Since the input has no `value` attribute in the template, the fresh render also leaves it empty.
- Toggle: `change` on `[data-action="toggle"]`, found via `e.target.closest('[data-id]')`. No manual focus handling is needed here — the checkbox keeps its `id="todo-<data-id>"` across renders, so the engine's own focus-by-id restoration keeps the focus on it after Space toggles it.
- Remove: `click` on `[data-action="remove"]`, same `closest('[data-id]')` lookup, filters the task out of state.
- Filter buttons: `click` on `[data-action="filter"]`, reads `data-filter`, guarded against unknown values, sets `state.filter`. `aria-pressed` is written as a real boolean into an `aria-*` attribute so it prints `"true"`/`"false"` per vfunc's escaping rules.
- Count: `data-ref="left"` always shows the total count of open tasks (not the filtered subset), as the task specifies.
- Empty state: `data-ref="empty"` is only emitted into the markup when the currently filtered list is empty (including on first load, since the list starts empty and the default filter is "all") — so it is fully absent otherwise, not just hidden.
- No colors, fonts or spacing appear in `app.js`; every new CSS rule reads a `var(--vf-*)` token already defined in `lib/vfunc.tokens.css`. `data-state="done"` on the `<li>` drives the strikethrough via CSS, not JS.
- Selectors are all `data-action` / `data-ref` / `id`; nothing is selected or delegated by class.

## Assumptions
- "Visible label" for the new-task input was read literally: a plain `<label for="todo-input">New task</label>` next to the input, not a visually-hidden one.
- The text element per task (`data-ref="text"`) is a `<label for="todo-<id>">`, so clicking the task text also toggles its checkbox. Nothing in the task forbids this, and it is a common, accessible list pattern; if a plain non-interactive `<span>` was intended instead, that is a one-line change.
- The task did not ask for i18n/message keys or a locale folder, and the existing `index.html`/`style.css` already hardcode English text ("My tasks"), so `app.js` also hardcodes its English strings directly rather than introducing `vf.i18n` machinery the task never asked for. This departs from AGENTS.md §6.1 ("text via message keys"); flagging it rather than silently following or silently ignoring it.
- Object spread (`{ ...t, done: !t.done }`) and arrow functions are used, since condition 4 says modern browsers only and section 8 (IE11/ES5) does not apply here.
- `data-id` values are numeric (from an incrementing counter); the task only requires uniqueness, not a particular type.

- [x] Selectors use `data-action` / `data-ref` / `id` only; no colors or sizes in JS — verified by reading `app.js`; no class selectors, no literal colors/sizes.
- [x] New CSS files use only `var(--vf-*)` tokens — every declaration in the added `style.css` rules uses a `--vf-*` token; no hex/`rgb()`/`px` literals were added.
- [ ] New text uses message keys in every locale; dates and numbers use `vf.fmt` — **not followed**, see "Assumptions" above (no locale infrastructure existed and the task did not request one). No dates or numbers beyond the plain open-task count, which the task explicitly asks to show as "a bare number" rather than through `vf.fmt`.
- [x] Listeners, timers and subscriptions are released in `onDestroy` — not applicable: no timers, no store subscriptions, and no outside (non-delegated) listeners were added, so there is nothing to release. `vf.attach`'s own `destroy()` already tears down the delegated listeners it owns.

The code follows the documented `vf.attach`/`vf.html`/delegates patterns and the kit's "common mistakes" list was checked against (table/tbody parsing doesn't apply here since it's a `<ul>`, no `.join('')` string-building, no raw color fallback, id/data-ref stability preserved for focus).
